6int dppd_snapshot_build_port_pairs(const struct dppd_config *cfg, struct dppd_forwarding_snapshot *snapshot)
7{
8 uint16_t i;
9
10 if (cfg == NULL || snapshot == NULL || cfg->nb_ports > DPPD_MAX_PORTS || (cfg->nb_ports & 1U) != 0)
11 return -1;
12
13 memset(snapshot, 0, sizeof(*snapshot));
14 snapshot->generation = 1;
15 snapshot->nb_peers = cfg->nb_ports;
16 for (i = 0; i < cfg->nb_ports; i += 2U) {
17 snapshot->peers[i].ingress_port = cfg->ports[i];
18 snapshot->peers[i].egress_port = cfg->ports[i + 1U];
19 snapshot->peers[i + 1U].ingress_port = cfg->ports[i + 1U];
20 snapshot->peers[i + 1U].egress_port = cfg->ports[i];

23}
