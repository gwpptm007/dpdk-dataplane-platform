8struct dppd_slot_key {
9 /** 联合查找键，先比较 id 再比较 generation；规则账本使用 (规则 ID, 0) */
10 uint64_t id, generation;
11 /** 槽位空闲时指向下一个空闲槽位，DPPD_SLOT_NONE 表示链尾；已用槽位不读取此字段 */
12 uint32_t next_free;
13};

15static bool less(const struct dppd_slot_key *key, uint64_t id, uint64_t generation)
16{
17 return key->id < id || (key->id == id && key->generation < generation);
18}

20/**
21 * 在有序槽位编号数组中二分，返回首个键不小于 (id, generation) 的位置，或 count
22 * 返回的是 order 的位置，不是对象槽位：position → order[position] → slot → keys[slot]
23 * 插入时在此位置腾出空位，查询时还须检查这里的键是否相等
24 */
25static uint32_t lower_bound(const struct dppd_slot_index *index, uint64_t id, uint64_t generation)
26{
27 uint32_t first = 0, last = index->count;
28 while (first < last) {
29 uint32_t middle = first + (last - first) / 2U;
30 if (less(&index->keys[index->order[middle]], id, generation))
31 first = middle + 1U;
32 else
33 last = middle;
34 }
35 return first;
36}

38void dppd_slot_index_clear(struct dppd_slot_index *index)
39{
40 if (index == NULL || index->storage == NULL)
41 return;
42 memset(index->keys, 0, (size_t)index->capacity * sizeof(*index->keys));
43 for (uint32_t i = 0; i < index->capacity; ++i)
44 index->keys[i].next_free = i + 1U;
45 index->keys[index->capacity - 1U].next_free = DPPD_SLOT_NONE;
46 index->count = 0;
47 index->free_head = 0;
48}

50int dppd_slot_index_init(struct dppd_slot_index *index, uint32_t capacity)
51{
52 if (index == NULL || capacity == 0)
53 return -EINVAL;
54 size_t width = sizeof(struct dppd_slot_key) + sizeof(uint32_t);
55 if ((uint64_t)capacity > SIZE_MAX / width)
56 return -EOVERFLOW;
57 void *storage = calloc(capacity, width);
58 if (storage == NULL)
59 return -ENOMEM;
60 struct dppd_slot_index next = {.storage = storage,
61 .keys = storage,
62 .order = (uint32_t *)((unsigned char *)storage + (size_t)capacity * sizeof(struct dppd_slot_key)),
63 .capacity = capacity};
64 dppd_slot_index_clear(&next);
65 *index = next;
66 return 0;
67}

77uint32_t dppd_slot_index_find(const struct dppd_slot_index *index, uint64_t id, uint64_t generation)
78{
79 if (index == NULL || index->storage == NULL || id == 0)
80 return DPPD_SLOT_NONE;
81 uint32_t position = lower_bound(index, id, generation);
82 if (position == index->count)
83 return DPPD_SLOT_NONE;
84 uint32_t slot = index->order[position];
85 return index->keys[slot].id == id && index->keys[slot].generation == generation ? slot : DPPD_SLOT_NONE;
86}

88uint32_t dppd_slot_index_upper_bound(const struct dppd_slot_index *index, uint64_t id, uint64_t generation)
89{
90 if (index == NULL || index->storage == NULL)
91 return 0;
92 uint32_t position = lower_bound(index, id, generation);
93 if (position < index->count) {
94 const struct dppd_slot_key *key = &index->keys[index->order[position]];
95 if (key->id == id && key->generation == generation)
96 ++position;
97 }
98 return position;
99}

101uint32_t dppd_slot_index_latest(const struct dppd_slot_index *index, uint64_t id)
102{
103 if (index == NULL || index->storage == NULL || id == 0)
104 return DPPD_SLOT_NONE;
105 uint32_t position = dppd_slot_index_upper_bound(index, id, UINT64_MAX);
106 if (position == 0)
107 return DPPD_SLOT_NONE;
108 uint32_t slot = index->order[position - 1U];
109 return index->keys[slot].id == id ? slot : DPPD_SLOT_NONE;
110}

117/**
118 * 为 (id, generation) 分配空闲槽位并加入有序索引，成功返回 0，通过 output 返回槽位
119 * 参数无效返回 -EINVAL，键已存在返回 -EEXIST，无空闲槽位返回 -ENOSPC，失败不修改索引或 output
120 * 内存已在 init 分配，本函数只维护索引；完整规则由调用方写入 records[slot]
121 *
122 * 以规则账本为例，以下键的 generation 均为 0
123 * 插入前：order = [0, 2, 3] → 对应 ID [10, 30, 40]，count = 3
124 * keys[0].id = 10，keys[2].id = 30，keys[3].id = 40
125 * free_head → 槽位 4 → 槽位 1 → NONE，假设这些槽位当前空闲
126 * 插入 ID 20：二分得到 position = 1，空闲链提供 slot = 4
127 * → free_head 改为 1，keys[4] 写入 (20, 0)
128 * → order 中原位置 1、2 的编号右移到位置 2、3，在位置 1 写入编号 4
129 * 插入后：order = [0, 4, 2, 3] → 对应 ID [10, 20, 30, 40]，count = 4
130 * → *output = 4 → 调用方把完整规则写入 records[4]
131 * 已有完整规则始终留在原槽位，排序只移动 order 中的整数编号
132 */
133int dppd_slot_index_insert(struct dppd_slot_index *index, uint64_t id, uint64_t generation, uint32_t *output)
134{
135 if (index == NULL || index->storage == NULL || output == NULL || id == 0)
136 return -EINVAL;
137 uint32_t position = lower_bound(index, id, generation);
138 if (position < index->count) {
139 const struct dppd_slot_key *found = &index->keys[index->order[position]];
140 if (found->id == id && found->generation == generation)
141 return -EEXIST;
142 }
143 if (index->free_head == DPPD_SLOT_NONE)
144 return -ENOSPC;
145 /** 从空闲链头取出槽位，先保存其后继为新链头，再覆盖该槽位的键 */
146 uint32_t slot = index->free_head;
147 index->free_head = index->keys[slot].next_free;
148 index->keys[slot] = (struct dppd_slot_key){.id = id, .generation = generation};
149 /** 插入位置之后的编号整体右移一格，源和目标区域可能重叠，因此使用 memmove */
150 memmove(&index->order[position + 1U], &index->order[position],
151 (size_t)(index->count - position) * sizeof(*index->order));
152 index->order[position] = slot;
153 ++index->count;
154 *output = slot;
155 return 0;
156}

158/**
159 * 删除 (id, generation) 的索引并归还槽位，成功返回 0，参数无效返回 -EINVAL，键不存在返回 -ENOENT
160 * 这里只处理 keys、order 和空闲链，规则账本调用方还须清空对应 records[slot]
161 *
162 * 接续 insert 中的例子：order = [0, 4, 2, 3]，对应 ID [10, 20, 30, 40]
163 * 删除 ID 20 → position = 1 → slot = order[1] = 4，原空闲链为 1 → NONE
164 * → 将位置 2、3 的编号左移到位置 1、2 → 有效 order = [0, 2, 3]
165 * → 清空 keys[4] 的键，令 keys[4].next_free = 1
166 * → free_head = 4，空闲链变成 4 → 1 → NONE，count 从 4 减为 3
167 * 数组末尾可能留有旧编号，但只有 order[0..count) 有效，查询不会读取尾部
168 *
169 * 随后新增 ID 25 → insert 先取空闲槽位 4 → order = [0, 4, 2, 3]
170 * 调用方将 ID 25 的完整规则写入 records[4]，ID 10、30、40 的完整记录始终留在原槽位
171 * 槽位复用不代表规则身份相同，外部仍按 ID 和版本识别规则
172 */
173int dppd_slot_index_erase(struct dppd_slot_index *index, uint64_t id, uint64_t generation)
174{
175 if (index == NULL || index->storage == NULL || id == 0)
176 return -EINVAL;
177 uint32_t position = lower_bound(index, id, generation);
178 if (position == index->count)
179 return -ENOENT;
180 uint32_t slot = index->order[position];
181 if (index->keys[slot].id != id || index->keys[slot].generation != generation)
182 return -ENOENT;
183 /** 左移删除位置之后的有效编号填补空位，重叠区域使用 memmove，完整对象不随之移动 */
184 memmove(&index->order[position], &index->order[position + 1U],
185 (size_t)(index->count - position - 1U) * sizeof(*index->order));
186 /** 先接住原空闲链，再把刚释放的槽位设为链头，下一次插入可直接复用 */
187 index->keys[slot] = (struct dppd_slot_key){.next_free = index->free_head};
188 index->free_head = slot;
189 --index->count;
190 return 0;
191}
