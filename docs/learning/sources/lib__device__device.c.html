25/**
26 * 保存本次启动环境的设备身份与描述符上限，固件查询不支持时保留未知状态
27 * 它只补充观测信息，不改变端口能力选择，不根据厂商或固件字符串猜测规则支持情况
28 */
29static int capture_identity(uint16_t port_id, const struct rte_eth_dev_info *info,
30 struct dppd_capability_identity *identity, const struct dppd_device_set *devices, uint64_t sequence)
31{
32 int rc;
33
34 memset(identity, 0, sizeof(*identity));
35 identity->device_name_known = rte_eth_dev_get_name_by_port(port_id, identity->device_name) == 0;
36 if (!identity->device_name_known)
37 identity->device_name[0] = '\0';
38 if (devices != NULL && (rc = reset_event_fence(devices, sequence, false, NULL)) != 0)
39 return rc;
40 snprintf(identity->dpdk_version, sizeof(identity->dpdk_version), "%s", rte_version());
41 rc = rte_eth_dev_fw_version_get(port_id, identity->firmware_version, sizeof(identity->firmware_version));
42 identity->firmware_known = rc == 0 && identity->firmware_version[0] != '\0';
43 identity->firmware_error = rc > 0 ? -ENOSPC : rc;
44 if (!identity->firmware_known)
45 identity->firmware_version[0] = '\0';
46 identity->rx_desc_min = info->rx_desc_lim.nb_min;
47 identity->rx_desc_max = info->rx_desc_lim.nb_max;
48 identity->rx_desc_align = info->rx_desc_lim.nb_align;
49 identity->tx_desc_min = info->tx_desc_lim.nb_min;
50 identity->tx_desc_max = info->tx_desc_lim.nb_max;
51 identity->tx_desc_align = info->tx_desc_lim.nb_align;
52 return devices == NULL ? 0 : reset_event_fence(devices, sequence, false, NULL);
53}

55/**
56 * 先封住被移除端口，再发布整个设备组的退出请求，工作线程只需读取原子状态
57 * 移除标记不会在当前进程中复位，它与可以恢复的普通链路 down 状态具有不同含义
58 */
59static void request_removal(struct dppd_device_set *devices, struct dppd_port *port)
60{
61 atomic_store_explicit(&port->removed, true, memory_order_release);
62 atomic_store_explicit(&port->link_state, DPPD_LINK_DOWN, memory_order_release);
63 atomic_store_explicit(&devices->removal_requested, true, memory_order_release);
64}

66/**
67 * DPDK 可能在后台线程调用此函数，因此这里只发布原子事件与收发门禁
68 * 不在回调中停止设备、等待工作线程或释放内存，实际清理由主线程按顺序完成
69 * 端口编号和数量在注册回调前初始化，并保持到全部回调成功注销之后
70 */
71static int handle_removal(uint16_t port_id, enum rte_eth_event_type event, void *context, void *ret_param)
72{
73 struct dppd_device_set *devices = context;
74
75 (void)ret_param;
76 if (devices == NULL ||
77 (event != RTE_ETH_EVENT_INTR_RMV && event != RTE_ETH_EVENT_INTR_RESET && event != RTE_ETH_EVENT_DESTROY))
78 return 0;
79 for (uint16_t i = 0; i < devices->nb_ports; ++i) {
80 if (devices->ports[i].port_id == port_id) {
81 bool tracking = atomic_load_explicit(&devices->event_tracking_enabled, memory_order_acquire);
82 if (!tracking) {
83 if (event == RTE_ETH_EVENT_INTR_RMV)
84 request_removal(devices, &devices->ports[i]);
85 break;
86 }
87 /** 先封住所有新收发，在途计数防止恢复线程误把尚未发布完的事件当作已处理 */
88 atomic_fetch_add_explicit(&devices->event_callbacks_in_flight, 1, memory_order_acq_rel);
89 atomic_store_explicit(&devices->event_io_quiesced, true, memory_order_release);
90 unsigned int bit = UINT32_C(1) << i;
91 atomic_uint *ports = event == RTE_ETH_EVENT_INTR_RESET ? &devices->reset_event_ports
92 : event == RTE_ETH_EVENT_INTR_RMV ? &devices->removal_event_ports
93 : &devices->destroy_event_ports;
94 atomic_fetch_or_explicit(ports, bit, memory_order_release);
95 if (event != RTE_ETH_EVENT_INTR_RESET)
96 request_removal(devices, &devices->ports[i]);
97 uint_fast64_t previous = atomic_load_explicit(&devices->event_sequence, memory_order_relaxed);
98 while (previous != UINT64_MAX &&
99 !atomic_compare_exchange_weak_explicit(&devices->event_sequence, &previous, previous + 1U,
100 memory_order_release, memory_order_relaxed))
101 ;
102 if (previous == UINT64_MAX)
103 atomic_store_explicit(&devices->event_sequence_overflow, true, memory_order_release);
104 atomic_fetch_sub_explicit(&devices->event_callbacks_in_flight, 1, memory_order_release);
105 break;
106 }
107 }
108 return 0;
109}

111int dppd_devices_enable_event_tracking(struct dppd_device_set *devices)
112{
113 if (devices == NULL || devices->nb_ports == 0 || devices->nb_ports > DPPD_MAX_PORTS)
114 return -EINVAL;
115 if (dppd_devices_removal_requested(devices))
116 return -ENODEV;
117 for (uint16_t i = 0; i < devices->nb_ports; ++i)
118 for (uint16_t j = 0; j < i; ++j)
119 if (devices->ports[i].port_id == devices->ports[j].port_id)
120 return -EEXIST;
121 if (!atomic_load_explicit(&devices->event_tracking_enabled, memory_order_acquire)) {
122 atomic_init(&devices->event_io_quiesced, false);
123 atomic_init(&devices->event_sequence_overflow, false);
124 atomic_init(&devices->event_callbacks_in_flight, 0);
125 atomic_init(&devices->event_sequence, 0);
126 atomic_init(&devices->resumed_event_sequence, 0);
127 atomic_init(&devices->reset_event_ports, 0);
128 atomic_init(&devices->removal_event_ports, 0);
129 atomic_init(&devices->destroy_event_ports, 0);
130 /** 已有 RMV 回调也可能并发运行，因此所有新原子字段先初始化再开放记录 */
131 atomic_store_explicit(&devices->event_tracking_enabled, true, memory_order_release);
132 }
133 static const enum rte_eth_event_type events[] = {RTE_ETH_EVENT_INTR_RMV, RTE_ETH_EVENT_INTR_RESET,
134 RTE_ETH_EVENT_DESTROY};
135 for (uint16_t i = 0; i < devices->nb_ports; ++i) {
136 struct dppd_port *port = &devices->ports[i];
137 bool *registered[] = {&port->removal_callback_registered, &port->reset_callback_registered,
138 &port->destroy_callback_registered};
139 for (unsigned int j = 0; j < sizeof(events) / sizeof(events[0]); ++j) {
140 if (*registered[j])
141 continue;
142 int rc = rte_eth_dev_callback_register(port->port_id, events[j], handle_removal, devices);
143 if (rc != 0) {
144 atomic_store_explicit(&devices->event_io_quiesced, true, memory_order_release);
145 return rc;
146 }
147 *registered[j] = true;
148 }
149 }
150 return dppd_devices_removal_requested(devices) ? -ENODEV : 0;
151}

153void dppd_devices_event_snapshot(const struct dppd_device_set *devices, struct dppd_device_event_snapshot *snapshot)
154{
155 if (snapshot == NULL)
156 return;
157 memset(snapshot, 0, sizeof(*snapshot));
158 if (devices == NULL)
159 return;
160 snapshot->enabled = atomic_load_explicit(&devices->event_tracking_enabled, memory_order_acquire);
161 snapshot->quiesced = dppd_devices_io_quiesced(devices);
162 if (!snapshot->enabled)
163 return;
164 /** 在途事件只作暂停依据，调用方应等 callbacks_in_flight 归零后重新读取完整证据 */
165 snapshot->callbacks_in_flight = atomic_load_explicit(&devices->event_callbacks_in_flight, memory_order_acquire);
166 uint64_t before = atomic_load_explicit(&devices->event_sequence, memory_order_acquire);
167 snapshot->reset_ports = atomic_load_explicit(&devices->reset_event_ports, memory_order_acquire);
168 snapshot->removal_ports = atomic_load_explicit(&devices->removal_event_ports, memory_order_acquire);
169 snapshot->destroy_ports = atomic_load_explicit(&devices->destroy_event_ports, memory_order_acquire);
170 snapshot->sequence = atomic_load_explicit(&devices->event_sequence, memory_order_acquire);
171 snapshot->overflowed = atomic_load_explicit(&devices->event_sequence_overflow, memory_order_acquire);
172 unsigned int active = atomic_load_explicit(&devices->event_callbacks_in_flight, memory_order_acquire);
173 if (active != 0 || before != snapshot->sequence)
174 snapshot->callbacks_in_flight = active != 0 ? active : 1U;
175 snapshot->quiesced |= dppd_devices_io_quiesced(devices);
176}

178int dppd_devices_resume_after_event(struct dppd_device_set *devices, uint64_t expected_sequence)
179{
180 if (devices == NULL || !atomic_load_explicit(&devices->event_tracking_enabled, memory_order_acquire))
181 return -EINVAL;
182 if (devices->nb_ports == 0 || devices->nb_ports > DPPD_MAX_PORTS)
183 return -EINVAL;
184 if (dppd_devices_removal_requested(devices) ||
185 atomic_load_explicit(&devices->removal_event_ports, memory_order_acquire) != 0 ||
186 atomic_load_explicit(&devices->destroy_event_ports, memory_order_acquire) != 0)
187 return -ENODEV;
188 for (uint16_t i = 0; i < devices->nb_ports; ++i)
189 if (!devices->ports[i].removal_callback_registered || !devices->ports[i].reset_callback_registered ||
190 !devices->ports[i].destroy_callback_registered)
191 return -EUCLEAN;
192 if (atomic_load_explicit(&devices->event_sequence_overflow, memory_order_acquire))
193 return -EOVERFLOW;
194 if (atomic_load_explicit(&devices->event_callbacks_in_flight, memory_order_acquire) != 0)
195 return -EBUSY;
196 if (atomic_load_explicit(&devices->event_sequence, memory_order_acquire) != expected_sequence)
197 return -ESTALE;
198 atomic_store_explicit(&devices->resumed_event_sequence, expected_sequence, memory_order_release);
199 atomic_store_explicit(&devices->event_io_quiesced, false, memory_order_release);
200 /** 回调可在任意两次原子操作间到达，解除后再次核对，序号差本身也会继续封住收发 */
201 if (atomic_load_explicit(&devices->event_callbacks_in_flight, memory_order_acquire) != 0 ||
202 atomic_load_explicit(&devices->event_sequence, memory_order_acquire) != expected_sequence ||
203 dppd_devices_removal_requested(devices) ||
204 atomic_load_explicit(&devices->event_sequence_overflow, memory_order_acquire)) {
205 atomic_store_explicit(&devices->event_io_quiesced, true, memory_order_release);
206 return atomic_load_explicit(&devices->event_sequence_overflow, memory_order_acquire) ? -EOVERFLOW : -EAGAIN;
207 }
208 return 0;
209}

211/** 稳定序号只能证明本进程尚未收到下一条事件，不能替代硬件身份和旧对象清理证据 */
212static int reset_event_fence(const struct dppd_device_set *devices, uint64_t expected, bool allow_reset,
213 uint64_t *observed)
214{
215 struct dppd_device_event_snapshot snapshot;
216
217 dppd_devices_event_snapshot(devices, &snapshot);
218 if (observed != NULL)
219 *observed = snapshot.sequence;
220 if (!snapshot.enabled || devices->nb_ports == 0 || devices->nb_ports > DPPD_MAX_PORTS)
221 return -EINVAL;
222 uint32_t selected = UINT32_MAX >> (32U - devices->nb_ports);
223 if ((snapshot.reset_ports | snapshot.removal_ports | snapshot.destroy_ports) & ~selected)
224 return -EUCLEAN;
225 if (snapshot.overflowed)
226 return -EOVERFLOW;
227 if (dppd_devices_removal_requested(devices) || snapshot.removal_ports != 0 || snapshot.destroy_ports != 0)
228 return -ENODEV;
229 for (uint16_t i = 0; i < devices->nb_ports; ++i) {
230 const struct dppd_port *port = &devices->ports[i];
231 if (atomic_load_explicit(&port->removed, memory_order_acquire))
232 return -ENODEV;
233 if (!port->removal_callback_registered || !port->reset_callback_registered ||
234 !port->destroy_callback_registered)
235 return -EUCLEAN;
236 }
237 if (snapshot.callbacks_in_flight != 0)
238 return -EBUSY;
239 if (snapshot.sequence != expected && (!allow_reset || snapshot.sequence < expected || snapshot.reset_ports == 0))
240 return -ESTALE;
241 return 0;
242}

262/**
263 * 按网口所在的 NUMA 节点创建 mbuf 池，结果放在 devices->pools[节点编号]
264 * 两个网口同属节点 0 → 共用 pools[0]；分属节点 0、1 → 分别创建 pools[0]、pools[1]
265 * 每个池含 cfg->mbufs_per_socket 个对象；不是按网口或队列各建一个池
266 * 成功返回 0，失败返回负错误码，已创建的池由调用方的设备清理流程处理
267 */
268static int create_socket_pools(struct dppd_device_set *devices, const struct dppd_config *cfg)
269{
270 uint16_t i;
271
272 if (cfg->mbuf_cache > RTE_MEMPOOL_CACHE_MAX_SIZE) {
273 fprintf(stderr, "[dppd] mbuf cache %u exceeds DPDK maximum %u\n", cfg->mbuf_cache, RTE_MEMPOOL_CACHE_MAX_SIZE);
274 return -EINVAL;
275 }
276
277 for (i = 0; i < cfg->nb_ports; ++i) {
278 const int socket_id = port_socket(cfg->ports[i]);
279 char name[64];
280
281 if (devices->pools[socket_id] != NULL)
282 continue;
283 snprintf(name, sizeof(name), "dppd_mbuf_s%d", socket_id);
284 /** priv_size = 0，不额外预留应用私有区；使用 DPDK 默认数据区大小，在选定 NUMA 节点分配
285 *
286 * | 参数 | 本次含义 |
287 |---|---|
288 | `name` | 池名称，例如 `dppd_mbuf_s0` |
289 | `mbufs_per_socket` | 池内对象数量，默认 32768 |
290 | `mbuf_cache` | 每个 lcore 的本地缓存规模，默认 256 |
291 | `0` | 不额外预留应用私有区 |
292 | `RTE_MBUF_DEFAULT_BUF_SIZE` | 使用 DPDK 默认数据区大小 |
293 | `socket_id` | 在指定 NUMA 节点分配 |
294 */
295 devices->pools[socket_id] = rte_pktmbuf_pool_create(name, cfg->mbufs_per_socket, cfg->mbuf_cache, 0,
296 RTE_MBUF_DEFAULT_BUF_SIZE, socket_id);
297 if (devices->pools[socket_id] == NULL) {
298 fprintf(stderr, "[dppd] cannot create mbuf pool on socket %d: %s\n", socket_id, rte_strerror(rte_errno));
299 return -rte_errno;
300 }
301 }
302 return 0;
303}

323/** 两份未知固件不能互相作证，缺少启动时身份时在第一次 stop 之前就拒绝复位 */
324static int saved_reset_identity(const struct dppd_port *port, const struct dppd_config *cfg)
325{
326 const struct dppd_capability_identity *identity = &port->identity;
327
328 if (memchr(identity->device_name, '\0', sizeof(identity->device_name)) == NULL ||
329 memchr(identity->firmware_version, '\0', sizeof(identity->firmware_version)) == NULL ||
330 memchr(identity->dpdk_version, '\0', sizeof(identity->dpdk_version)) == NULL)
331 return -EINVAL;
332 if (!identity->device_name_known || !identity->firmware_known || identity->device_name[0] == '\0' ||
333 identity->firmware_version[0] == '\0' || identity->dpdk_version[0] == '\0' ||
334 !rte_is_valid_assigned_ether_addr(&port->mac))
335 return -ENODATA;
336 if (identity->configured_queues != cfg->nb_queues || identity->configured_rx_desc == 0 ||
337 identity->configured_tx_desc == 0)
338 return -EXDEV;
339 if (port->capabilities.max_rx_queues == 0 || port->capabilities.max_tx_queues == 0 || identity->rx_desc_max == 0 ||
340 identity->rx_desc_align == 0 || identity->tx_desc_max == 0 || identity->tx_desc_align == 0)
341 return -ENODATA;
342 struct rte_eth_desc_lim rx = {.nb_min = identity->rx_desc_min,
343 .nb_max = identity->rx_desc_max,
344 .nb_align = identity->rx_desc_align};
345 struct rte_eth_desc_lim tx = {.nb_min = identity->tx_desc_min,
346 .nb_max = identity->tx_desc_max,
347 .nb_align = identity->tx_desc_align};
348 if (cfg->nb_queues > port->capabilities.max_rx_queues || cfg->nb_queues > port->capabilities.max_tx_queues ||
349 !descriptor_supported(identity->configured_rx_desc, &rx) ||
350 !descriptor_supported(identity->configured_tx_desc, &tx))
351 return -ENOTSUP;
352 return 0;
353}

355static int reset_identity_ready(const struct dppd_device_set *devices, const struct dppd_config *cfg,
356 uint16_t *failed_port)
357{
358 if (devices == NULL || cfg == NULL || cfg->nb_ports == 0 || cfg->nb_ports > DPPD_MAX_PORTS || cfg->nb_queues == 0 ||
359 cfg->nd6_count > DPPD_ND6_LOCAL_LIMIT || cfg->nb_ports != devices->nb_ports)
360 return -EINVAL;
361 int rc = dppd_topology_validate_identity(&devices->topology, &devices->topology, cfg->ports, cfg->nb_ports);
362 if (rc != 0)
363 return rc;
364 for (uint16_t i = 0; i < devices->nb_ports; ++i) {
365 const struct dppd_port *port = &devices->ports[i];
366 if (failed_port != NULL)
367 *failed_port = port->port_id;
368 if (dppd_topology_find(&devices->topology, port->port_id) == NULL)
369 return -ENOENT;
370 for (uint16_t j = 0; j < i; ++j)
371 if (devices->ports[j].port_id == port->port_id)
372 return -EUCLEAN;
373 if (!port->configured || !port->started)
374 return -EUCLEAN;
375 if (port->socket_id < 0 || port->socket_id >= RTE_MAX_NUMA_NODES || devices->pools[port->socket_id] == NULL)
376 return -ENODATA;
377 rc = saved_reset_identity(port, cfg);
378 if (rc != 0)
379 return rc;
380 }
381 return 0;
382}

389/**
390 * 读取网口能力，检查队列数量，选择可用的 RSS 和发送卸载功能，将配置写入 plan
391 * 此处只生成配置，不创建队列或启动网口；成功返回 0，失败返回负错误码
392 * 普通启动时 devices 为 NULL；复位重建时还要核对身份，保留原来的功能与描述符数量
393 */
394static int prepare_port_configuration(const struct dppd_port *port, const struct dppd_config *cfg,
395 struct port_configuration *plan, const struct dppd_device_set *devices, uint64_t sequence)
396{
397 memset(plan, 0, sizeof(*plan));
398 plan->preserve = devices != NULL;
399 int rc = rte_eth_dev_info_get(port->port_id, &plan->info);
400 if (rc != 0)
401 return rc;
402 if (devices != NULL && (rc = reset_event_fence(devices, sequence, false, NULL)) != 0)
403 return rc;
404 rc = capture_identity(port->port_id, &plan->info, &plan->identity, devices, sequence);
405 if (rc != 0)
406 return rc;
407 const struct rte_eth_dev_info *info = &plan->info;
408 plan->capabilities = (struct dppd_port_capabilities){.max_rx_queues = info->max_rx_queues,
409 .max_tx_queues = info->max_tx_queues,
410 .reta_size = info->reta_size,
411 .rss_offloads = info->flow_type_rss_offloads,
412 .rx_offloads = info->rx_offload_capa,
413 .tx_offloads = info->tx_offload_capa,
414 .device_capabilities = info->dev_capa};
415 bool removal_interrupt = info->dev_flags != NULL && (*info->dev_flags & RTE_ETH_DEV_INTR_RMV) != 0;
416 plan->conf.intr_conf.rmv = removal_interrupt;
417 if (cfg->nb_queues > info->max_rx_queues || cfg->nb_queues > info->max_tx_queues)
418 return -ENOTSUP;
419 /**
420 * RSS（接收端扩展）负责把符合配置的报文分配给 RX 队列，worker 只轮询自己的队列
421 * 典型硬件路径：报文字段 → RSS 哈希 → RETA 队列映射 → RX 队列 → 对应 worker
422 * 示例：流 A 的包 → 队列 0 → worker 0；流 B 的包 → 队列 1 → worker 1
423 * 配置不变且所选字段相同的包通常留在同一队列，不是按包轮流分配，也不保证各队列均匀
424 *
425 * requested_rss 是希望启用的类型，flow_type_rss_offloads 是设备支持的类型
426 * 按位与只启用两者共有的部分，例如请求 IP/UDP/TCP、设备只支持 IP/TCP，则不启用 UDP 类型
427 */
428 uint64_t requested_rss = RTE_ETH_RSS_IP | RTE_ETH_RSS_UDP | RTE_ETH_RSS_TCP;
429 plan->rss_hf = cfg->nb_queues > 1U ? requested_rss & info->flow_type_rss_offloads : 0;
430 plan->tx_offloads = info->tx_offload_capa & RTE_ETH_TX_OFFLOAD_MULTI_SEGS;
431 plan->rx_desc = DPPD_DEFAULT_RX_DESC;
432 plan->tx_desc = DPPD_DEFAULT_TX_DESC;
433 if (devices != NULL) {
434 const struct dppd_capability_identity *old = &port->identity, *current = &plan->identity;
435 if (memchr(current->device_name, '\0', sizeof(current->device_name)) == NULL ||
436 memchr(current->firmware_version, '\0', sizeof(current->firmware_version)) == NULL)
437 return -EINVAL;
438 if (!current->device_name_known || !current->firmware_known || current->device_name[0] == '\0' ||
439 current->firmware_version[0] == '\0')
440 return -ENODATA;
441 if (current->device_name_known != old->device_name_known || current->firmware_known != old->firmware_known ||
442 current->firmware_error != old->firmware_error || strcmp(current->device_name, old->device_name) != 0 ||
443 strcmp(current->firmware_version, old->firmware_version) != 0 ||
444 strcmp(current->dpdk_version, old->dpdk_version) != 0)
445 return -EXDEV;
446 if ((port->configured_rss_hf & ~info->flow_type_rss_offloads) != 0 ||
447 (port->configured_tx_offloads & ~info->tx_offload_capa) != 0 ||
448 (port->configured_removal_interrupt && !removal_interrupt) ||
449 !descriptor_supported(old->configured_rx_desc, &info->rx_desc_lim) ||
450 !descriptor_supported(old->configured_tx_desc, &info->tx_desc_lim))
451 return -ENOTSUP;
452 plan->rss_hf = port->configured_rss_hf;
453 plan->tx_offloads = port->configured_tx_offloads;
454 plan->conf.intr_conf.rmv = port->configured_removal_interrupt;
455 plan->rx_desc = old->configured_rx_desc;
456 plan->tx_desc = old->configured_tx_desc;
457 rc = rte_eth_macaddr_get(port->port_id, &plan->mac);
458 if (rc != 0)
459 return rc;
460 rc = reset_event_fence(devices, sequence, false, NULL);
461 if (rc != 0)
462 return rc;
463 if (!rte_is_same_ether_addr(&port->mac, &plan->mac))
464 return -EXDEV;
465 }
466 /** 多队列但没有可用的 RSS 类型时拒绝启动，不能只建多个队列就认为已实现分流 */
467 if (cfg->nb_queues > 1U && plan->rss_hf == 0)
468 return -ENOTSUP;
469 /**
470 * 单队列不启用 RSS；多队列设置 RSS 模式和启用类型，后续 dev_configure 才提交给 PMD
471 * 此处未指定 RSS key，也未编程设置 RETA，具体默认值及分流行为由 PMD 决定
472 *
473 * RSS 与 worker 通过同一个网口的 queue_id 接起来
474 * runtime.c / dppd_runtime_init
475 * → worker->queue_id = worker_index，worker->lcore_id = 选中的 lcore
476 * 本文件 / configure_port_queues
477 * → 为每个网口建立队列 0..nb_queues-1，并启用这里准备的 RSS 配置
478 * runtime.c / dppd_runtime_start
479 * → remote_launch(worker_main, worker, worker->lcore_id)，将同一份 worker 配置传给工作函数
480 * worker.c / process_ingress
481 * → rx_burst(当前网口, worker->queue_id, ...)，只收这个编号的队列
482 *
483 * 例如选核结果是 lcore 3、5：
484 * RSS → 网口 0 的 RX 队列 0 → worker 0（queue_id 0）→ 在 lcore 3 上处理
485 * RSS → 网口 0 的 RX 队列 1 → worker 1（queue_id 1）→ 在 lcore 5 上处理
486 * RSS 选择队列，runtime 指定谁来轮询该队列；queue_id 不等于 lcore_id
487 */
488 plan->conf.rxmode.mq_mode = cfg->nb_queues > 1U ? RTE_ETH_MQ_RX_RSS : RTE_ETH_MQ_RX_NONE;
489 plan->conf.rx_adv_conf.rss_conf.rss_hf = plan->rss_hf;
490 plan->conf.txmode.offloads = plan->tx_offloads;
491 return 0;
492}

494static void publish_port_configuration(struct dppd_port *port, const struct port_configuration *plan)
495{
496 port->capabilities = plan->capabilities;
497 port->identity = plan->identity;
498 port->configured_rss_hf = plan->rss_hf;
499 port->configured_tx_offloads = plan->tx_offloads;
500 port->configured_removal_interrupt = plan->conf.intr_conf.rmv;
501 port->mac = plan->mac;
502}

504/**
505 * 将 plan 中的网口配置交给 PMD，建立 cfg 指定数量的 RX/TX 队列
506 * pool 是该网口 RX 队列共用的 mbuf 池，调整后的描述符数量记入 plan->identity
507 *
508 * 提交端口配置 → 调整描述符数量 → 按 queue 编号分别建立 RX 和 TX 队列
509 * 本函数不启动网口，后续由 start_configured_port 完成
510 * 普通启动 devices 为 NULL；复位时 devices 和 sequence 用于检测设备事件
511 */
512static int configure_port_queues(struct dppd_port *port, const struct dppd_config *cfg, struct rte_mempool *pool,
513 struct port_configuration *plan, const struct dppd_device_set *devices, uint64_t sequence)
514{
515 struct rte_eth_rxconf rx_conf;
516 struct rte_eth_txconf tx_conf;
517 int rc;
518
519 /** 提交 RX/TX 队列数和 RSS 等端口配置，具体队列随后由 setup 建立
520 * 网口编号：0
521 RX 队列数：2
522 TX 队列数：2
523 端口配置：plan->conf，包含前面准备的 RSS 设置
524 */
525 rc = rte_eth_dev_configure(port->port_id, cfg->nb_queues, cfg->nb_queues, &plan->conf);
526 if (rc != 0)
527 return rc;
528 port->configured = true;
529 if (devices != NULL && (rc = reset_event_fence(devices, sequence, false, NULL)) != 0)
530 return rc;
531 /**
532 * 默认每个 RX/TX 队列各请求 1024 个描述符，再按设备的数量范围和对齐要求调整
533 * 队列数是 cfg->nb_queues，描述符数是各队列的资源规模，burst_size 是单次收包上限
534 */
535 uint16_t rx_desc = plan->rx_desc, tx_desc = plan->tx_desc;
536 rc = rte_eth_dev_adjust_nb_rx_tx_desc(port->port_id, &rx_desc, &tx_desc);
537 if (rc != 0)
538 return rc;
539 if (devices != NULL && (rc = reset_event_fence(devices, sequence, false, NULL)) != 0)
540 return rc;
541 if (plan->preserve && (rx_desc != plan->rx_desc || tx_desc != plan->tx_desc))
542 return -EXDEV;
543 plan->identity.configured_queues = cfg->nb_queues;
544 plan->identity.configured_rx_desc = rx_desc;
545 plan->identity.configured_tx_desc = tx_desc;
546 rx_conf = plan->info.default_rxconf;
547 rx_conf.offloads = plan->conf.rxmode.offloads;
548 tx_conf = plan->info.default_txconf;
549 tx_conf.offloads = plan->conf.txmode.offloads;
550 /**
551 * 例：当前网口按 queue 0、1 分别建立 RX 0/TX 0、RX 1/TX 1
552 * 各 RX 队列关联同一个传入的 pool，不为每个队列另建池
553 *
554 * queue = 0 → 建立网口 0 的 RX 0、TX 0
555 * queue = 1 → 建立网口 0 的 RX 1、TX 1
556 */
557 for (uint16_t queue = 0; queue < cfg->nb_queues; ++queue) {
558 rc = rte_eth_rx_queue_setup(port->port_id, queue, rx_desc, port->socket_id, &rx_conf, pool);
559 if (rc != 0)
560 return rc;
561 if (devices != NULL && (rc = reset_event_fence(devices, sequence, false, NULL)) != 0)
562 return rc;
563 /** TX 使用应用在发送时提交的 mbuf，因此这里无需绑定接收池 */
564 rc = rte_eth_tx_queue_setup(port->port_id, queue, tx_desc, port->socket_id, &tx_conf);
565 if (rc != 0)
566 return rc;
567 if (devices != NULL && (rc = reset_event_fence(devices, sequence, false, NULL)) != 0)
568 return rc;
569 }
570 return 0;
571}
572
573/**
574 * 启动已配置好队列的网口，按 cfg 设置接收模式，将网口 MAC 保存到 plan
575 *
576 * 启动网口 → 按需启用混杂或全组播接收 → 读取 MAC
577 * 普通启动 devices 为 NULL；复位时还要核对设备事件及原 MAC
578 * started 记录驱动启动成功，不代表物理链路已连通或 worker 已开始收包
579 * 成功返回 0，失败返回负错误码；后续设置失败时也保留 started，供上层关闭已启动网口
580 */
581static int start_configured_port(struct dppd_port *port, const struct dppd_config *cfg, struct port_configuration *plan,
582 const struct dppd_device_set *devices, uint64_t sequence)
583{
584 int rc = rte_eth_dev_start(port->port_id);
585 if (rc != 0)
586 return rc;
587 /** 后续接收模式或 MAC 读取失败时，清理路径仍需关闭这个已启动的网口 */
588 port->started = true;
589 if (devices != NULL && (rc = reset_event_fence(devices, sequence, false, NULL)) != 0)
590 return rc;
591
592 if (cfg->promiscuous) {
593 /** 混杂模式放宽目的 MAC 过滤，允许接收发给其他 MAC 的帧 */
594 rc = rte_eth_promiscuous_enable(port->port_id);
595 if (rc != 0)
596 return rc;
597 if (devices != NULL && (rc = reset_event_fence(devices, sequence, false, NULL)) != 0)
598 return rc;
599 }
600 /** 当前网口配置 ND6 时启用全组播接收，让 IPv6 邻居发现的组播报文进入接收路径 */
601 for (uint32_t i = 0; i < cfg->nd6_count; ++i)
602 if (cfg->nd6[i].port_id == port->port_id) {
603 // 开启指定网口的“全组播接收模式”
604 rc = rte_eth_allmulticast_enable(port->port_id);
605 if (rc != 0)
606 return rc;
607 if (devices != NULL && (rc = reset_event_fence(devices, sequence, false, NULL)) != 0)
608 return rc;
609 break;
610 }
611 /** MAC 先存入 plan，整套配置成功后由 publish_port_configuration 写入 port->mac */
612 rc = rte_eth_macaddr_get(port->port_id, &plan->mac);
613 if (rc != 0)
614 return rc;
615 if (devices != NULL && (rc = reset_event_fence(devices, sequence, false, NULL)) != 0)
616 return rc;
617 if (plan->preserve && !rte_is_same_ether_addr(&port->mac, &plan->mac))
618 return -EXDEV;
619 return 0;
620}

622static void print_port_configuration(const struct dppd_port *port, const struct dppd_config *cfg)
623{
624 printf("[dppd] port=%u peer=%u socket=%d queues=%u rss=0x%" PRIx64 " tx-offloads=0x%" PRIx64
625 " mac=%02x:%02x:%02x:%02x:%02x:%02x\n",
626 port->port_id, port->peer_port_id, port->socket_id, cfg->nb_queues, port->configured_rss_hf,
627 port->configured_tx_offloads, port->mac.addr_bytes[0], port->mac.addr_bytes[1], port->mac.addr_bytes[2],
628 port->mac.addr_bytes[3], port->mac.addr_bytes[4], port->mac.addr_bytes[5]);
629}

631/**
632 * 将单个网口配置并启动，pool 是该网口所属 NUMA 节点的 mbuf 池
633 *
634 * prepare_port_configuration → 查询能力，准备 RSS、卸载功能和描述符配置
635 * configure_port_queues → 提交端口配置，建立 RX/TX 队列
636 * start_configured_port → 启动网口，设置接收模式，读取 MAC
637 * 全部成功后才保存最终能力信息，返回 0；任一步失败就返回错误，由上层清理已完成的操作
638 */
639static int configure_port(struct dppd_port *port, const struct dppd_config *cfg, struct rte_mempool *pool)
640{
641 struct port_configuration plan;
642 int rc = prepare_port_configuration(port, cfg, &plan, NULL, 0);
643 if (rc == 0)
644 rc = configure_port_queues(port, cfg, pool, &plan, NULL, 0);
645 if (rc == 0)
646 rc = start_configured_port(port, cfg, &plan, NULL, 0); // 启动网口
647 /** 后续步骤失败时，configured/started 仍保留已完成的状态，上层据此关闭设备 */
648 if (rc != 0)
649 return rc;
650 publish_port_configuration(port, &plan);
651 print_port_configuration(port, cfg);
652 return 0;
653}

655static int reset_failure(struct dppd_device_set *devices, struct dppd_device_reset_result *result, int error)
656{
657 result->last_error = error;
658 if (devices != NULL && atomic_load_explicit(&devices->event_tracking_enabled, memory_order_acquire)) {
659 atomic_store_explicit(&devices->event_io_quiesced, true, memory_order_release);
660 result->sequence_after = atomic_load_explicit(&devices->event_sequence, memory_order_acquire);
661 }
662 return error;
663}
664
665int dppd_devices_reset_reconfigure(struct dppd_device_set *devices, const struct dppd_config *cfg,
666 uint64_t expected_sequence, struct dppd_device_reset_result *result)
667{
668 if (result == NULL)
669 return -EINVAL;
670 *result = (struct dppd_device_reset_result){.phase = DPPD_DEVICE_RESET_PRECHECK,
671 .port_id = UINT16_MAX,
672 .sequence_after = expected_sequence};
673 int rc = reset_identity_ready(devices, cfg, &result->port_id);
674 if (rc != 0)
675 return reset_failure(devices, result, rc);
676 rc = dppd_devices_pause_io(devices, expected_sequence);
677 if (rc != 0)
678 return reset_failure(devices, result, rc);
679 struct port_configuration plans[DPPD_MAX_PORTS];
680 /** 先核对全组证据，再执行任何 stop，不让第二口的缺失身份在第一口已停止后才暴露 */
681 struct dppd_topology candidate;
682 rc = dppd_topology_rediscover_verified(cfg, &devices->topology, &candidate);
683 if (rc != 0)
684 return reset_failure(devices, result, rc);
685 for (uint16_t i = 0; i < devices->nb_ports; ++i) {
686 const struct dppd_port *port = &devices->ports[i];
687 result->port_id = port->port_id;
688 rc = reset_event_fence(devices, expected_sequence, false, NULL);
689 if (rc != 0)
690 return reset_failure(devices, result, rc);
691 if (!rte_eth_dev_is_valid_port(port->port_id) || rte_eth_dev_is_removed(port->port_id))
692 return reset_failure(devices, result, -ENODEV);
693 if (port_socket(port->port_id) != port->socket_id)
694 return reset_failure(devices, result, -EXDEV);
695 rc = reset_event_fence(devices, expected_sequence, false, NULL);
696 if (rc == 0)
697 rc = prepare_port_configuration(port, cfg, &plans[i], devices, expected_sequence);
698 if (rc != 0)
699 return reset_failure(devices, result, rc);
700 }
701
702 result->phase = DPPD_DEVICE_RESET_STOP;
703 int stop_error = 0;
704 uint16_t stop_error_port = UINT16_MAX;
705 for (uint16_t i = 0; i < devices->nb_ports; ++i) {
706 struct dppd_port *port = &devices->ports[i];
707 result->port_id = port->port_id;
708 rc = reset_event_fence(devices, expected_sequence, false, NULL);
709 if (rc != 0)
710 return reset_failure(devices, result, rc);
711 rc = rte_eth_dev_stop(port->port_id);
712 if (rc == 0) {
713 port->started = false;
714 atomic_store_explicit(&port->link_state, DPPD_LINK_DOWN, memory_order_release);
715 result->stopped_ports |= UINT32_C(1) << i;
716 } else if (stop_error == 0) {
717 stop_error = rc;
718 stop_error_port = port->port_id;
719 }
720 rc = reset_event_fence(devices, expected_sequence, false, NULL);
721 if (rc != 0)
722 return reset_failure(devices, result, rc);
723 }
724 if (stop_error != 0) {
725 result->port_id = stop_error_port;
726 return reset_failure(devices, result, stop_error);
727 }
728
729 result->phase = DPPD_DEVICE_RESET_RESET;
730 uint64_t sequence = expected_sequence;
731 for (uint16_t i = 0; i < devices->nb_ports; ++i) {
732 struct dppd_port *port = &devices->ports[i];
733 result->port_id = port->port_id;
734 rc = reset_event_fence(devices, sequence, false, NULL);
735 if (rc != 0)
736 return reset_failure(devices, result, rc);
737 rc = rte_eth_dev_reset(port->port_id);
738 if (rc != 0)
739 return reset_failure(devices, result, rc);
740 port->configured = false;
741 result->reset_ports |= UINT32_C(1) << i;
742 /** 只有 reset 调用窗口内允许新 RESET 证据，记录序号但不确认事件已恢复 */
743 rc = reset_event_fence(devices, sequence, true, &sequence);
744 result->sequence_after = sequence;
745 if (rc != 0)
746 return reset_failure(devices, result, rc);
747 }
748
749 result->phase = DPPD_DEVICE_RESET_VERIFY;
750 result->port_id = UINT16_MAX;
751 rc = dppd_topology_rediscover_verified(cfg, &devices->topology, &candidate);
752 if (rc != 0)
753 return reset_failure(devices, result, rc);
754 for (uint16_t i = 0; i < devices->nb_ports; ++i) {
755 result->port_id = devices->ports[i].port_id;
756 rc = reset_event_fence(devices, sequence, false, NULL);
757 if (rc == 0)
758 rc = prepare_port_configuration(&devices->ports[i], cfg, &plans[i], devices, sequence);
759 if (rc != 0)
760 return reset_failure(devices, result, rc);
761 }
762
763 result->phase = DPPD_DEVICE_RESET_CONFIGURE;
764 for (uint16_t i = 0; i < devices->nb_ports; ++i) {
765 struct dppd_port *port = &devices->ports[i];
766 result->port_id = port->port_id;
767 rc = reset_event_fence(devices, sequence, false, NULL);
768 if (rc == 0)
769 rc = configure_port_queues(port, cfg, devices->pools[port->socket_id], &plans[i], devices, sequence);
770 if (port->configured)
771 result->configured_ports |= UINT32_C(1) << i;
772 if (rc != 0)
773 return reset_failure(devices, result, rc);
774 }
775
776 result->phase = DPPD_DEVICE_RESET_START;
777 for (uint16_t i = 0; i < devices->nb_ports; ++i) {
778 struct dppd_port *port = &devices->ports[i];
779 result->port_id = port->port_id;
780 rc = reset_event_fence(devices, sequence, false, NULL);
781 if (rc == 0)
782 rc = start_configured_port(port, cfg, &plans[i], devices, sequence);
783 if (port->started)
784 result->started_ports |= UINT32_C(1) << i;
785 if (rc != 0)
786 return reset_failure(devices, result, rc);
787 }
788 rc = reset_event_fence(devices, sequence, false, NULL);
789 if (rc != 0)
790 return reset_failure(devices, result, rc);
791 for (uint16_t i = 0; i < devices->nb_ports; ++i)
792 publish_port_configuration(&devices->ports[i], &plans[i]);
793 result->phase = DPPD_DEVICE_RESET_READY;
794 result->port_id = UINT16_MAX;
795 result->sequence_after = sequence;
796 return 0;
797}

799/**
800 * 将 cfg 指定的 DPDK 网口配置为可收发状态，端口信息和内存池保存在 devices 中
801 * 前提是 EAL 已完成设备探测，cfg->ports 中的编号必须对应有效 ethdev
802 *
803 * 初始化顺序
804 * 1. 检查网口编号，记录端口配对、NUMA 节点和初始状态
805 * 2. 按 NUMA 节点创建 mbuf 池
806 * 3. 为每个网口登记移除回调，配置 RX/TX 队列并启动端口
807 * 4. 查询链路状态，收集设备拓扑供控制面使用
808 *
809 * 成功返回 0；资源准备失败时调用设备清理流程，并返回原错误码
810 * 这里启动网口，worker 收包循环仍由后续 dppd_runtime_start 启动
811 */
812int dppd_devices_init(struct dppd_device_set *devices, const struct dppd_config *cfg)
813{
814 uint16_t i;
815 int rc;
816
817 if (devices == NULL || cfg == NULL)
818 return -EINVAL;
819 memset(devices, 0, sizeof(*devices));
820 atomic_init(&devices->removal_requested, false);
821 atomic_init(&devices->event_tracking_enabled, false);
822
823 for (i = 0; i < cfg->nb_ports; ++i) {
824 if (!rte_eth_dev_is_valid_port(cfg->ports[i])) {
825 fprintf(stderr, "[dppd] invalid ethdev port %u\n", cfg->ports[i]);
826 return -ENODEV;
827 }
828 }
829
830 /**
831 * 一次性初始化所有端口元数据，再开始注册回调
832 * 注册过程中就可能收到移除事件，回调不能读取尚未初始化的其他端口状态
833 */
834 devices->nb_ports = cfg->nb_ports;
835 for (i = 0; i < cfg->nb_ports; ++i) {
836 struct dppd_port *port = &devices->ports[i];
837 const uint16_t pair_base = (uint16_t)(i & ~1U);
838
839 port->port_id = cfg->ports[i];
840 atomic_init(&port->removed, false);
841 atomic_init(&port->link_state, DPPD_LINK_UNKNOWN);
842 port->peer_port_id =
843 cfg->routing_path[0] == '\0' ? cfg->ports[pair_base + (i % 2U == 0 ? 1U : 0U)] : UINT16_MAX;
844 port->socket_id = port_socket(port->port_id);
845 }
846
847 /** 按 NUMA 节点准备 mbuf 池，后面的网口循环按 socket_id 取用 */
848 rc = create_socket_pools(devices, cfg);
849 if (rc != 0)
850 goto fail;
851
852 for (i = 0; i < cfg->nb_ports; ++i) {
853 struct dppd_port *port = &devices->ports[i];
854 rc = rte_eth_dev_callback_register(port->port_id, RTE_ETH_EVENT_INTR_RMV, handle_removal, devices);
855 if (rc != 0) {
856 fprintf(stderr, "[dppd] port=%u removal callback registration failed: %d\n", port->port_id, rc);
857 goto fail;
858 }
859 port->removal_callback_registered = true;
860 /** 注册调用返回前也可能已触发移除，此时直接清理，不再配置或启动端口 */
861 if (dppd_devices_removal_requested(devices)) {
862 rc = -ENODEV;
863 goto fail;
864 }
865 /**
866 * 按当前网口的 NUMA 节点取池，交给其 RX 队列
867 * 例：网口 0、1 的 socket_id 都为 0，两轮均传 pools[0]，各自队列仍独立
868 */
869 rc = configure_port(port, cfg, devices->pools[port->socket_id]);
870 if (rc != 0) {
871 fprintf(stderr, "[dppd] failed to configure port %u: %s (%d)\n", port->port_id, rte_strerror(-rc), rc);
872 goto fail;
873 }
874 }
875
876 rc = dppd_devices_poll_links(devices);
877 if (rc != 0)
878 goto fail;
879 rc = dppd_topology_discover(cfg, &devices->topology);
880 if (rc != 0)
881 goto fail;
882 dppd_topology_dump(&devices->topology);
883 return 0;
884
885fail:
886 /** 任一网口准备失败都进入统一清理，已经启动的其他网口也在清理范围内 */
887 (void)dppd_devices_stop(devices);
888 return rc;
889}

891/**
892 * 检查已启动网口是否仍有效，查询链路并保存到 port->link_state，供 worker 发送前读取
893 * 初次不支持链路查询的网口记为 UNSUPPORTED，后续仍检查移除状态，但跳过链路 API
894 * 返回 0 表示本轮检查完成，不要求所有链路都为 UP；设备失效或查询错误返回负错误码
895 */
896int dppd_devices_poll_links(struct dppd_device_set *devices)
897{
898 if (devices == NULL || devices->nb_ports > DPPD_MAX_PORTS)
899 return -EINVAL;
900 bool tracking = atomic_load_explicit(&devices->event_tracking_enabled, memory_order_acquire);
901 uint64_t sequence = tracking ? atomic_load_explicit(&devices->event_sequence, memory_order_acquire) : 0;
902 int result;
903 if (tracking && (result = reset_event_fence(devices, sequence, false, NULL)) != 0)
904 return result;
905 for (uint16_t i = 0; i < devices->nb_ports; ++i) {
906 struct dppd_port *port = &devices->ports[i];
907 struct rte_eth_link link = {0};
908 unsigned int previous, next;
909 int rc;
910
911 if (!port->started)
912 continue;
913 if (tracking && (rc = reset_event_fence(devices, sequence, false, NULL)) != 0)
914 return rc;
915 /**
916 * 先检查设备身份和移除状态，再判断是否跳过链路查询
917 * 即使 PMD 不支持 link API，也必须独立发现设备失效，不能永远跳过移除检测
918 */
919 bool removed = atomic_load_explicit(&port->removed, memory_order_acquire);
920 bool valid = removed ? false : rte_eth_dev_is_valid_port(port->port_id) != 0;
921 if (tracking && (rc = reset_event_fence(devices, sequence, false, NULL)) != 0)
922 return rc;
923 if (valid)
924 removed = rte_eth_dev_is_removed(port->port_id) != 0;
925 if (tracking && (rc = reset_event_fence(devices, sequence, false, NULL)) != 0)
926 return rc;
927 if (removed || !valid) {
928 /** 主动探测和异步 RMV 使用相同证据，避免主线程只看到退出请求却漏掉句柄隔离 */
929 if (tracking)
930 (void)handle_removal(port->port_id, RTE_ETH_EVENT_INTR_RMV, devices, NULL);
931 else
932 request_removal(devices, port);
933 fprintf(stderr, "[dppd] port=%u device removal detected\n", port->port_id);
934 return -ENODEV;
935 }
936 previous = atomic_load_explicit(&port->link_state, memory_order_acquire);
937 if (previous == DPPD_LINK_UNSUPPORTED)
938 continue;
939 /** 读取当前链路结果写入 link，不在这里等待链路变为 UP */
940 rc = rte_eth_link_get_nowait(port->port_id, &link);
941 /** 主动暂停时仍可查询链路，但查询中出现新事件就停止，不访问下一口旧实例 */
942 if (tracking && (result = reset_event_fence(devices, sequence, false, NULL)) != 0)
943 return result;
944 if (rc == -ENOTSUP && previous == DPPD_LINK_UNKNOWN) {
945 next = DPPD_LINK_UNSUPPORTED;
946 } else if (rc != 0) {
947 /** 查询失败先将状态置为 DOWN，阻止 worker 继续向此口发送，再返回原错误 */
948 atomic_store_explicit(&port->link_state, DPPD_LINK_DOWN, memory_order_release);
949 fprintf(stderr, "[dppd] port=%u link query failed: %d\n", port->port_id, rc);
950 return rc;
951 } else {
952 next = link.link_status == RTE_ETH_LINK_UP ? DPPD_LINK_UP : DPPD_LINK_DOWN;
953 }
954 /** 主线程更新状态供 worker 读取，只在状态变化时记录日志 */
955 atomic_store_explicit(&port->link_state, next, memory_order_release);
956 if (previous != next)
957 fprintf(stderr, "[dppd] port=%u link=%s\n", port->port_id, dppd_link_state_name(next));
958 }
959 if (tracking && (result = reset_event_fence(devices, sequence, false, NULL)) != 0)
960 return result;
961 /** 覆盖遍历期间收到事件或尚未启动的端口已移除的情况，避免错误返回正常状态 */
962 return dppd_devices_removal_requested(devices) ? -ENODEV : 0;
963}

965/** DESTROY 后端口编号可以分给新设备，必须保留资源而不是向这个编号发送 stop 或 close */
966static int verify_teardown_identity(const struct dppd_device_set *devices, uint16_t index)
967{
968 if (!atomic_load_explicit(&devices->event_tracking_enabled, memory_order_acquire))
969 return 0;
970 if ((atomic_load_explicit(&devices->destroy_event_ports, memory_order_acquire) & (UINT32_C(1) << index)) != 0)
971 return -ENODEV;
972 const struct dppd_port *port = &devices->ports[index];
973 const struct dppd_endpoint *endpoint = dppd_topology_find(&devices->topology, port->port_id);
974 const struct dppd_capability_identity *identity = &port->identity;
975 if (endpoint == NULL || !identity->device_name_known)
976 return -ENODATA;
977 if (memchr(endpoint->device_name, '\0', sizeof(endpoint->device_name)) == NULL ||
978 memchr(endpoint->driver_name, '\0', sizeof(endpoint->driver_name)) == NULL ||
979 memchr(identity->device_name, '\0', sizeof(identity->device_name)) == NULL)
980 return -EINVAL;
981 if (endpoint->device_name[0] == '\0' || endpoint->driver_name[0] == '\0' ||
982 strcmp(endpoint->driver_name, "unknown") == 0 || identity->device_name[0] == '\0')
983 return -ENODATA;
984 if (!rte_eth_dev_is_valid_port(port->port_id))
985 return -ENODEV;
986 char device_name[DPPD_CAPABILITY_NAME_SIZE] = {0};
987 int rc = rte_eth_dev_get_name_by_port(port->port_id, device_name);
988 if (rc != 0)
989 return rc;
990 if (memchr(device_name, '\0', sizeof(device_name)) == NULL)
991 return -EINVAL;
992 if (strcmp(device_name, identity->device_name) != 0)
993 return -EXDEV;
994 struct rte_eth_dev_info info = {0};
995 rc = rte_eth_dev_info_get(port->port_id, &info);
996 if (rc != 0)
997 return rc;
998 const char *bus_name = NULL;
999 if (info.device != NULL) {
1000#if RTE_VERSION >= RTE_VERSION_NUM(22, 11, 0, 0)
1001 bus_name = rte_dev_name(info.device);
1002#else
1003 bus_name = info.device->name;
1004#endif
1005 }
1006 if (bus_name == NULL || info.driver_name == NULL)
1007 return -ENODATA;
1008 if (strnlen(bus_name, sizeof(endpoint->device_name)) >= sizeof(endpoint->device_name) ||
1009 strnlen(info.driver_name, sizeof(endpoint->driver_name)) >= sizeof(endpoint->driver_name))
1010 return -ENAMETOOLONG;
1011 return strcmp(bus_name, endpoint->device_name) == 0 && strcmp(info.driver_name, endpoint->driver_name) == 0
1012 ? 0
1013 : -EXDEV;
1014}

1016int dppd_devices_stop(struct dppd_device_set *devices)
1017{
1018 uint16_t i;
1019 int socket_id;
1020 int result = 0;
1021 bool all_closed = true;
1022
1023 if (devices == NULL)
1024 return -EINVAL;
1025 if (devices->nb_ports > DPPD_MAX_PORTS)
1026 return -EINVAL;
1027 /**
1028 * 必须先注销所有端口的回调，再关闭任何端口或修改回调借用的元数据
1029 * 按端口单独注销并检查结果，不能假定一次批量调用能准确报告每个在途回调
1030 */
1031 for (i = 0; i < devices->nb_ports; ++i) {
1032 struct dppd_port *port = &devices->ports[i];
1033 const enum rte_eth_event_type events[] = {RTE_ETH_EVENT_INTR_RESET, RTE_ETH_EVENT_DESTROY,
1034 RTE_ETH_EVENT_INTR_RMV};
1035 bool *registered[] = {&port->reset_callback_registered, &port->destroy_callback_registered,
1036 &port->removal_callback_registered};
1037 for (unsigned int j = 0; j < sizeof(events) / sizeof(events[0]); ++j) {
1038 if (!*registered[j])
1039 continue;
1040 int rc;
1041 /** EAGAIN 表示回调仍在执行，短暂休眠后重试，避免忙等并保证上下文仍然有效 */
1042 do {
1043 rc = rte_eth_dev_callback_unregister(port->port_id, events[j], handle_removal, devices);
1044 if (rc == -EAGAIN) {
1045 struct timespec remaining = {.tv_nsec = 1000000L};
1046 while (nanosleep(&remaining, &remaining) != 0 && errno == EINTR)
1047 ;
1048 }
1049 } while (rc == -EAGAIN);
1050 if (rc != 0) {
1051 /** 其他注销错误不能当作成功，保留设备和缓冲池供调用方报告故障 */
1052 const char *name = events[j] == RTE_ETH_EVENT_INTR_RMV ? "removal" : "device event";
1053 fprintf(stderr, "[dppd] port=%u %s callback unregister failed: %d\n", port->port_id, name, rc);
1054 return rc;
1055 }
1056 *registered[j] = false;
1057 }
1058 }
1059 for (i = 0; i < devices->nb_ports; ++i) {
1060 struct dppd_port *port = &devices->ports[i];
1061 int rc;
1062
1063 if (port->started || port->configured) {
1064 rc = verify_teardown_identity(devices, i);
1065 if (rc != 0) {
1066 if (result == 0)
1067 result = rc;
1068 all_closed = false;
1069 continue;
1070 }
1071 }
1072 if (port->started) {
1073 rc = rte_eth_dev_stop(port->port_id);
1074 if (rc != 0) {
1075 fprintf(stderr, "[dppd] port=%u device stop failed: %d\n", port->port_id, rc);
1076 if (result == 0)
1077 result = rc;
1078 } else {
1079 port->started = false;
1080 atomic_store_explicit(&port->link_state, DPPD_LINK_DOWN, memory_order_release);
1081 }
1082 }
1083 if (port->configured) {
1084 rc = verify_teardown_identity(devices, i);
1085 if (rc != 0) {
1086 if (result == 0)
1087 result = rc;
1088 all_closed = false;
1089 continue;
1090 }
1091 /** stop 失败后仍尝试 close，成功关闭后才有机会安全归还该端口使用的资源 */
1092 rc = rte_eth_dev_close(port->port_id);
1093 if (rc != 0) {
1094 fprintf(stderr, "[dppd] port=%u device close failed: %d; retaining mbuf pools\n", port->port_id, rc);
1095 if (result == 0)
1096 result = rc;
1097 all_closed = false;
1098 continue;
1099 }
1100 }
1101 port->started = false;
1102 port->configured = false;
1103 }
1104 /** 多个端口可能共用同一缓冲池，任一 close 失败就不能释放任何共享池 */
1105 if (!all_closed)
1106 return result;
1107 for (socket_id = 0; socket_id < RTE_MAX_NUMA_NODES; ++socket_id) {
1108 if (devices->pools[socket_id] != NULL) {
1109 const struct rte_mempool *pool = devices->pools[socket_id];
1110 const unsigned int available = rte_mempool_avail_count(pool);
1111
1112 printf("[dppd] mbuf pool socket=%d available=%u capacity=%u in-use=%u\n", socket_id, available, pool->size,
1113 pool->size - available);
1114 rte_mempool_free(devices->pools[socket_id]);
1115 devices->pools[socket_id] = NULL;
1116 }
1117 }
1118 devices->nb_ports = 0;
1119 return result;
1120}
