35/** 十进制编号必须非负且完整，逐位检查避免 sscanf 接受负数、空白或整数溢出 */
36static int parse_number(const char **cursor, uint32_t *value)
37{
38 const char *text = *cursor;
39 uint32_t result = 0;
40
41 if (*text < '0' || *text > '9')
42 return -EINVAL;
43 do {
44 uint32_t digit = (uint32_t)(*text - '0');
45 if (result > (UINT32_MAX - digit) / 10U)
46 return -ERANGE;
47 result = result * 10U + digit;
48 ++text;
49 } while (*text >= '0' && *text <= '9');
50 *cursor = text;
51 *value = result;
52 return 0;
53}

55int dppd_topology_resolve_role(struct dppd_endpoint *endpoint, const char *driver_name, bool is_representor,
56 const char *phys_port_name)
57{
58 struct dppd_endpoint parsed = {0};
59 char name[sizeof(endpoint->phys_port_name)];
60 const char *cursor = name;
61 size_t length = phys_port_name == NULL ? 0 : strnlen(phys_port_name, sizeof(name));
62 int rc = -EINVAL;
63
64 if (endpoint == NULL)
65 return -EINVAL;
66 /** 先保存输入，允许调用方直接传入 endpoint 自己保存的原始名称 */
67 if (phys_port_name != NULL && length < sizeof(name))
68 memcpy(name, phys_port_name, length + 1U);
69 clear_role(endpoint, -EINVAL);
70 endpoint->phys_port_name[0] = '\0';
71 if (driver_name == NULL || phys_port_name == NULL)
72 return -EINVAL;
73 if (length >= sizeof(name)) {
74 endpoint->role_error = -ENAMETOOLONG;
75 return -ENAMETOOLONG;
76 }
77 memcpy(endpoint->phys_port_name, name, length + 1U);
78 if (strcmp(driver_name, "mlx5_pci") != 0 && strcmp(driver_name, "net_mlx5") != 0) {
79 endpoint->role_error = -ENOTSUP;
80 return -ENOTSUP;
81 }
82 if (*cursor == 'c') {
83 ++cursor;
84 rc = parse_number(&cursor, &parsed.controller_id);
85 if (rc != 0)
86 goto failed;
87 parsed.has_controller = true;
88 }
89 if (strncmp(cursor, "pf", 2) == 0) {
90 cursor += 2;
91 rc = parse_number(&cursor, &parsed.pf_id);
92 if (rc != 0)
93 goto failed;
94 parsed.has_pf = true;
95 parsed.role = DPPD_ENDPOINT_ROLE_PF;
96 if (strncmp(cursor, "vf", 2) == 0 || strncmp(cursor, "sf", 2) == 0) {
97 parsed.role = *cursor == 'v' ? DPPD_ENDPOINT_ROLE_VF : DPPD_ENDPOINT_ROLE_SF;
98 cursor += 2;
99 rc = parse_number(&cursor, &parsed.function_id);
100 if (rc != 0)
101 goto failed;
102 parsed.has_function = true;
103 }
104 } else if (*cursor == 'p') {
105 ++cursor;
106 rc = parse_number(&cursor, &parsed.physical_port_id);
107 if (rc != 0)
108 goto failed;
109 parsed.has_physical_port = true;
110 parsed.role = DPPD_ENDPOINT_ROLE_UPLINK;
111 } else {
112 rc = -EINVAL;
113 goto failed;
114 }
115 if (*cursor != '\0') {
116 rc = -EINVAL;
117 goto failed;
118 }
119 /** mlx5 的物理上联口不带 representor 标志，PF、VF 和 SF 名称必须带该标志 */
120 if ((parsed.role != DPPD_ENDPOINT_ROLE_UPLINK) != is_representor) {
121 rc = -EPROTO;
122 goto failed;
123 }
124 endpoint->role = parsed.role;
125 endpoint->role_source = DPPD_ENDPOINT_ROLE_SOURCE_MLX5_PHYS_PORT_NAME;
126 endpoint->role_error = 0;
127 endpoint->has_controller = parsed.has_controller;
128 endpoint->has_pf = parsed.has_pf;
129 endpoint->has_function = parsed.has_function;
130 endpoint->has_physical_port = parsed.has_physical_port;
131 endpoint->controller_id = parsed.controller_id;
132 endpoint->pf_id = parsed.pf_id;
133 endpoint->function_id = parsed.function_id;
134 endpoint->physical_port_id = parsed.physical_port_id;
135 return 0;
136
137failed:
138 clear_role(endpoint, rc);
139 return rc;
140}

142/** 内核返回的字符串必须带完整结尾，不能截断后继续认领身份 */
143static int copy_attribute(char *output, size_t capacity, const struct rtattr *attribute)
144{
145 size_t length = RTA_PAYLOAD(attribute);
146 const char *value = RTA_DATA(attribute);
147
148 if (length == 0 || length > capacity || value[length - 1U] != '\0')
149 return -EOVERFLOW;
150 if (memchr(value, '\0', length - 1U) != NULL)
151 return -EPROTO;
152 memcpy(output, value, length);
153 return 0;
154}

156/**
157 * 按 if_index 查询当前网络命名空间中的接口，填写接口编号、接口名和可选物理端口名
158 * 组装 RTM_GETLINK 请求 → 发送给内核 → 校验回复并提取属性 → 关闭 socket
159 * 成功返回 0，失败返回负错误码，失败时输出可能已部分填写，由调用方丢弃
160 * sysfs 挂载可能仍指向宿主，先查询当前网络命名空间，再由 read_identity 交叉核对
161 */
162static int read_netlink_identity(uint32_t if_index, struct dppd_topology_identity *identity)
163{
164 /**
165 * 请求布局：[Netlink 消息头 header][接口查询条件 link]
166 * RTM_GETLINK 表示查询，ifi_index 指定接口；seq 用于对应请求和回复
167 */
168 struct {
169 struct nlmsghdr header;
170 struct ifinfomsg link;
171 } request = {.header = {.nlmsg_len = NLMSG_LENGTH(sizeof(struct ifinfomsg)),
172 .nlmsg_type = RTM_GETLINK,
173 .nlmsg_flags = NLM_F_REQUEST,
174 .nlmsg_seq = 1},
175 .link = {.ifi_family = AF_UNSPEC, .ifi_index = (int)if_index}};
176 struct sockaddr_nl kernel = {.nl_family = AF_NETLINK};
177 struct timeval timeout = {.tv_sec = 0, .tv_usec = 200000};
178 /** 按消息头要求对齐接收缓冲；vector 指向缓冲，message 同时接收发送方地址 */
179 _Alignas(struct nlmsghdr) char buffer[16384];
180 struct iovec vector = {.iov_base = buffer, .iov_len = sizeof(buffer)};
181 struct msghdr message = {.msg_name = &kernel, .msg_namelen = sizeof(kernel), .msg_iov = &vector, .msg_iovlen = 1};
182 struct nlmsghdr *header;
183 ssize_t received;
184 int remaining, fd, rc = -ENODATA;
185 bool has_interface = false, has_phys_port = false;
186
187 if (if_index == 0 || if_index > INT_MAX)
188 return -ENODEV;
189 /** NETLINK_ROUTE 查询内核网络信息，消息不经业务网卡发出；接收最多等待 200 ms */
190 fd = socket(AF_NETLINK, SOCK_RAW | SOCK_CLOEXEC, NETLINK_ROUTE);
191 if (fd < 0)
192 return -errno;
193 if (setsockopt(fd, SOL_SOCKET, SO_RCVTIMEO, &timeout, sizeof(timeout)) != 0) {
194 rc = -errno;
195 goto done;
196 }
197 if (sendto(fd, &request, request.header.nlmsg_len, 0, (struct sockaddr *)&kernel, sizeof(kernel)) !=
198 (ssize_t)request.header.nlmsg_len) {
199 rc = errno == 0 ? -EIO : -errno;
200 goto done;
201 }
202 received = recvmsg(fd, &message, 0);
203 if (received < 0) {
204 rc = -errno;
205 goto done;
206 }
207 /** 拒绝被截断的回复，并检查发送方为内核单播地址 */
208 if ((message.msg_flags & MSG_TRUNC) != 0 || kernel.nl_pid != 0 || kernel.nl_groups != 0) {
209 rc = -EOVERFLOW;
210 goto done;
211 }
212 if (message.msg_namelen < sizeof(kernel) || kernel.nl_family != AF_NETLINK) {
213 rc = -EPROTO;
214 goto done;
215 }
216 remaining = (int)received;
217 /** 一次接收可包含多条消息，按各自长度和对齐位置遍历，只处理本次请求的回复 */
218 for (header = (struct nlmsghdr *)buffer; NLMSG_OK(header, remaining); header = NLMSG_NEXT(header, remaining)) {
219 struct ifinfomsg *link;
220 struct rtattr *attribute;
221 int attributes;
222
223 if (header->nlmsg_seq != request.header.nlmsg_seq)
224 continue;
225 if (header->nlmsg_type == NLMSG_ERROR) {
226 if (header->nlmsg_len < NLMSG_LENGTH(sizeof(struct nlmsgerr)))
227 rc = -EPROTO;
228 else
229 rc = ((struct nlmsgerr *)NLMSG_DATA(header))->error;
230 if (rc >= 0)
231 rc = -ENODATA;
232 goto done;
233 }
234 /** RTM_NEWLINK 在这里承载接口信息，收到它并不表示本函数创建了接口 */
235 if (header->nlmsg_type != RTM_NEWLINK || header->nlmsg_len < NLMSG_LENGTH(sizeof(*link)))
236 continue;
237 link = NLMSG_DATA(header);
238 if (link->ifi_index != (int)if_index)
239 continue;
240 identity->if_index = if_index;
241 attributes = IFLA_PAYLOAD(header);
242 /** 属性按类型、长度和值组织，只提取接口名和物理端口名，其他属性跳过 */
243 for (attribute = IFLA_RTA(link); RTA_OK(attribute, attributes); attribute = RTA_NEXT(attribute, attributes)) {
244 if (attribute->rta_type == IFLA_IFNAME) {
245 if (has_interface) {
246 rc = -EPROTO;
247 goto done;
248 }
249 has_interface = true;
250 rc = copy_attribute(identity->interface_name, sizeof(identity->interface_name), attribute);
251 } else if (attribute->rta_type == IFLA_PHYS_PORT_NAME) {
252 if (has_phys_port) {
253 rc = -EPROTO;
254 goto done;
255 }
256 has_phys_port = true;
257 rc = copy_attribute(identity->phys_port_name, sizeof(identity->phys_port_name), attribute);
258 } else
259 continue;
260 if (rc != 0)
261 goto done;
262 }
263 /** 物理端口名是可选证据，通用 PCI 接口缺少它时仍须继续核对接口和设备 */
264 rc = attributes != 0 ? -EPROTO : identity->interface_name[0] == '\0' ? -ENODATA : 0;
265 goto done;
266 }
267
268 /** socket 创建成功后统一从这里关闭，查询失败也不会遗漏释放 */
269done:
270 close(fd);
271 return rc;
272}

274static bool valid_interface_name(const char *name)
275{
276 if (*name == '\0' || strcmp(name, ".") == 0 || strcmp(name, "..") == 0)
277 return false;
278 for (; *name != '\0'; ++name)
279 if (!((*name >= 'a' && *name <= 'z') || (*name >= 'A' && *name <= 'Z') || (*name >= '0' && *name <= '9') ||
280 *name == '_' || *name == '-' || *name == '.'))
281 return false;
282 return true;
283}

285static bool valid_pci_name(const char *name)
286{
287 unsigned int domain, bus, slot, function;
288 char canonical[16];
289 int end = 0;
290
291 if (name == NULL || strlen(name) != 12 ||
292 sscanf(name, "%4x:%2x:%2x.%1x%n", &domain, &bus, &slot, &function, &end) != 4 || end != 12 || slot > 31U ||
293 function > 7U)
294 return false;
295 snprintf(canonical, sizeof(canonical), "%04x:%02x:%02x.%x", domain, bus, slot, function);
296 return strcmp(name, canonical) == 0;
297}

299/** 读取上限包含结尾检查，多读到一个字节就报错，不把长字符串截成合法前缀 */
300static int read_sysfs_text(const char *path, char *output, size_t capacity)
301{
302 size_t used = 0;
303 int rc = 0;
304 int fd = open(path, O_RDONLY | O_CLOEXEC | O_NOFOLLOW);
305
306 if (fd < 0)
307 return -errno;
308 while (used < capacity) {
309 ssize_t length = read(fd, output + used, capacity - used);
310 if (length < 0 && errno == EINTR)
311 continue;
312 if (length < 0) {
313 rc = -errno;
314 break;
315 }
316 if (length == 0)
317 break;
318 used += (size_t)length;
319 }
320 close(fd);
321 if (rc != 0) {
322 /** 已读到半份内容后才报缺失或不支持，不能把它当成可选字段完全不存在 */
323 if (used != 0 && (rc == -ENOENT || rc == -EOPNOTSUPP || rc == -ENODATA))
324 return -EIO;
325 return rc;
326 }
327 if (used >= capacity)
328 return -EOVERFLOW;
329 if (used > 0 && output[used - 1U] == '\n')
330 --used;
331 output[used] = '\0';
332 if (memchr(output, '\0', used) != NULL)
333 return -EPROTO;
334 if (used == 0)
335 return -ENODATA;
336 return 0;
337}

339/** 双方没有物理端口名时允许通用身份成立，权限或格式错误不能被吞成“没有名称” */
340static int read_optional_phys_port(const char *path, char *output, size_t capacity)
341{
342 int rc = read_sysfs_text(path, output, capacity);
343
344 if (rc == -ENOENT || rc == -EOPNOTSUPP || rc == -ENODATA) {
345 output[0] = '\0';
346 return 0;
347 }
348 return rc;
349}

351/**
352 * 核对 if_index 对应的 Linux 接口是否属于 device_name 指定的 PCI 设备
353 * Netlink 查询接口 → sysfs 核对编号、设备和物理端口名 → 再查接口是否变化
354 * 全部通过后写入 identity 并返回 0，失败返回负错误码，不写入输出
355 */
356static int read_identity(uint32_t if_index, const char *device_name, struct dppd_topology_identity *identity,
357 void *opaque)
358{
359 struct dppd_topology_identity observed = {0}, current = {0};
360 char path[PATH_MAX], target[PATH_MAX], text[64];
361 const char *basename;
362 const char *cursor;
363 uint32_t observed_index;
364 int rc;
365
366 (void)opaque;
367 if (!valid_pci_name(device_name))
368 return -ENOTSUP;
369 /** Netlink 向当前网络命名空间的内核查询接口，observed 保存首次结果 */
370 rc = read_netlink_identity(if_index, &observed);
371 if (rc != 0)
372 return rc;
373 if (!valid_interface_name(observed.interface_name))
374 return -EINVAL;
375 /** /sys/class/net/enp3s0/ifindex 例如编号 5 查到 enp3s0，还需确认 sysfs 中 enp3s0 的 ifindex 也是 5 */
376 snprintf(path, sizeof(path), "/sys/class/net/%s/ifindex", observed.interface_name);
377 rc = read_sysfs_text(path, text, sizeof(text));
378 if (rc != 0)
379 return rc;
380 cursor = text;
381 if (parse_number(&cursor, &observed_index) != 0 || *cursor != '\0' || observed_index != if_index)
382 return -EXDEV;
383 /** device 是设备链接，解析实际路径后取末段 PCI 名称，与 PMD 提供的设备名比较 */
384 snprintf(path, sizeof(path), "/sys/class/net/%s/device", observed.interface_name);
385 if (realpath(path, target) == NULL)
386 return -errno;
387 basename = strrchr(target, '/');
388 if (basename == NULL || strcmp(basename + 1, device_name) != 0)
389 return -EXDEV;
390 /** 物理端口名也要与 Netlink 一致；两边都没有名称时允许继续，具体角色由上层判断 */
391 snprintf(path, sizeof(path), "/sys/class/net/%s/phys_port_name", observed.interface_name);
392 rc = read_optional_phys_port(path, text, sizeof(text));
393 if (rc != 0)
394 return rc;
395 if (strcmp(text, observed.phys_port_name) != 0)
396 return -EXDEV;
397 /** 读取过程中接口可能改名或编号被复用，再查询并比较；发现变化返回 -EAGAIN */
398 rc = read_netlink_identity(if_index, &current);
399 if (rc != 0)
400 return rc;
401 if (strcmp(current.interface_name, observed.interface_name) != 0 ||
402 strcmp(current.phys_port_name, observed.phys_port_name) != 0 ||
403 if_nametoindex(observed.interface_name) != if_index)
404 return -EAGAIN;
405 /** 前面的步骤都只修改局部变量，成功后才一次性复制给调用方 */
406 memcpy(observed.device_name, device_name, strlen(device_name) + 1U);
407 *identity = observed;
408 return 0;
409}

422/**
423 * 为当前网口补充设备名称、已核对的 Linux 接口名和可确认的端口角色，结果写入 endpoint
424 * 保存 PMD 观察 → 检查 PCI 名称和接口编号 → reader 核对接口身份
425 * → 校验读取器结果并保存名称 → 对支持的 mlx5 网口继续解析角色
426 * 正式入口传入的 reader 是 read_identity，opaque 原样传给读取器
427 * 无法确认时保留 UNKNOWN 并填写 role_error，本函数返回不会中止外层的基础网口收集
428 */
429static void discover_identity(struct dppd_endpoint *endpoint, const struct rte_eth_dev_info *info,
430 dppd_topology_identity_reader reader, void *opaque)
431{
432 struct dppd_topology_identity identity = {0};
433 const char *device_name = get_device_name(info);
434 int rc;
435
436 endpoint->if_index = info->if_index;
437 /** 先清掉旧角色及编号，后面只用本次核对通过的信息确认角色 */
438 clear_role(endpoint, -ENOTSUP);
439 /** 可信接口字段只从完整成功的读取器发布，错误时不保留容易被误认领的原始名称 */
440 endpoint->interface_name[0] = endpoint->phys_port_name[0] = '\0';
441 if (device_name != NULL && strnlen(device_name, sizeof(endpoint->device_name)) < sizeof(endpoint->device_name))
442 memcpy(endpoint->device_name, device_name, strlen(device_name) + 1U);
443 bool mlx5 = strcmp(endpoint->driver_name, "mlx5_pci") == 0 || strcmp(endpoint->driver_name, "net_mlx5") == 0;
444 if (device_name == NULL || device_name[0] == '\0') {
445 endpoint->role_error = mlx5 || info->if_index != 0 ? -ENODATA : -ENOTSUP;
446 return;
447 }
448 /** 当前正式读取器按 PCI 设备与内核接口核对身份，非 PCI 名称保持未识别状态 */
449 if (!valid_pci_name(device_name))
450 return;
451 if (info->if_index == 0 || info->if_index > INT_MAX) {
452 endpoint->role_error = -ENODEV;
453 return;
454 }
455 if (reader == NULL) {
456 endpoint->role_error = -ENODATA;
457 return;
458 }
459 /** 正式入口中实际调用 read_identity，identity 接收核对后的接口信息 */
460 rc = reader(info->if_index, device_name, &identity, opaque);
461 if (rc != 0) {
462 endpoint->role_error = rc < 0 ? rc : -EIO;
463 return;
464 }
465 if (identity.if_index != info->if_index || identity.if_index == 0 ||
466 memchr(identity.device_name, '\0', sizeof(identity.device_name)) == NULL ||
467 memchr(identity.interface_name, '\0', sizeof(identity.interface_name)) == NULL ||
468 memchr(identity.phys_port_name, '\0', sizeof(identity.phys_port_name)) == NULL ||
469 !valid_interface_name(identity.interface_name) || strcmp(identity.device_name, device_name) != 0) {
470 endpoint->role_error = -EXDEV;
471 return;
472 }
473 memcpy(endpoint->interface_name, identity.interface_name, sizeof(endpoint->interface_name));
474 memcpy(endpoint->phys_port_name, identity.phys_port_name, sizeof(endpoint->phys_port_name));
475 /** 接口核对成功后保存名称；角色解析目前仅支持 mlx5，其他驱动仍保留 UNKNOWN */
476 if (!mlx5)
477 return;
478 if (identity.phys_port_name[0] == '\0') {
479 endpoint->role_error = -ENODATA;
480 return;
481 }
482 dppd_topology_resolve_role(endpoint, endpoint->driver_name, endpoint->kind == DPPD_ENDPOINT_REPRESENTOR,
483 identity.phys_port_name);
484}

498/**
499 * 按 cfg 的网口列表读取 PMD 信息，生成网口身份与关联记录，reader 用于进一步核对接口身份
500 * 清空输出 → 逐口填入临时 discovered → 基础信息全部读取成功后整体写入 topology
501 * 身份或角色无法确认时保留 UNKNOWN 和 role_error；参数、PMD 查询或名称复制错误返回负错误码
502 * 成功返回 0；topology 非空时，失败不会输出只发现一半的网口列表
503 */
504int dppd_topology_discover_with_reader(const struct dppd_config *cfg, struct dppd_topology *topology,
505 dppd_topology_identity_reader reader, void *opaque)
506{
507 /** 临时结果与调用方的输出分开，发现中途失败时不发布部分结果 */
508 struct dppd_topology discovered = {0};
509
510 if (topology == NULL)
511 return -EINVAL;
512 memset(topology, 0, sizeof(*topology));
513 if (cfg == NULL || cfg->nb_ports > DPPD_MAX_PORTS)
514 return -EINVAL;
515 for (uint16_t i = 0; i < cfg->nb_ports; ++i)
516 for (uint16_t j = 0; j < i; ++j)
517 if (cfg->ports[i] == cfg->ports[j])
518 return -EEXIST;
519
520 for (uint16_t i = 0; i < cfg->nb_ports; ++i) {
521 struct rte_eth_dev_info info = {0};
522 /** 本轮填写下一个空条目，完成当前网口后才增加 nb_endpoints */
523 struct dppd_endpoint *endpoint = &discovered.endpoints[discovered.nb_endpoints];
524 int rc = rte_eth_dev_info_get(cfg->ports[i], &info);
525
526 if (rc != 0)
527 return rc;
528 endpoint->ethdev_port_id = cfg->ports[i];
529 endpoint->kind = DPPD_ENDPOINT_ETHDEV;
530 if (info.dev_flags != NULL && ((*info.dev_flags & RTE_ETH_DEV_REPRESENTOR) != 0))
531 endpoint->kind = DPPD_ENDPOINT_REPRESENTOR;
532 endpoint->has_switch_domain = info.switch_info.domain_id != RTE_ETH_DEV_SWITCH_DOMAIN_ID_INVALID;
533 endpoint->switch_domain_id = info.switch_info.domain_id;
534 endpoint->switch_port_id = info.switch_info.port_id;
535 rc = copy_pmd_name(endpoint->switch_name, sizeof(endpoint->switch_name), info.switch_info.name, "-");
536 if (rc == 0)
537 rc = copy_pmd_name(endpoint->driver_name, sizeof(endpoint->driver_name), info.driver_name, "unknown");
538 if (rc != 0)
539 return rc;
540 const char *device_name = get_device_name(&info);
541 if (device_name != NULL && strnlen(device_name, sizeof(endpoint->device_name)) == sizeof(endpoint->device_name))
542 return -ENAMETOOLONG;
543 /** reader 核对接口身份，无法识别的角色及原因保存在当前条目中，不凭端口编号猜测 */
544 discover_identity(endpoint, &info, reader, opaque);
545 discovered.nb_endpoints++;
546 }
547 /** 所有 PMD 信息读取成功后才发布，调用方不会拿到只发现一半的拓扑 */
548 *topology = discovered;
549 return 0;
550}

552/** 使用正式身份读取函数 read_identity 发现 cfg 选定的网口，结果写入 topology，返回发现结果 */
553int dppd_topology_discover(const struct dppd_config *cfg, struct dppd_topology *topology)
554{
555 // read_identity 函数指针
556 return dppd_topology_discover_with_reader(cfg, topology, read_identity, NULL);
557}

559static bool role_is_valid(enum dppd_endpoint_role role)
560{
561 switch (role) {
562 case DPPD_ENDPOINT_ROLE_UNKNOWN:
563 case DPPD_ENDPOINT_ROLE_UPLINK:
564 case DPPD_ENDPOINT_ROLE_PF:
565 case DPPD_ENDPOINT_ROLE_VF:
566 case DPPD_ENDPOINT_ROLE_SF:
567 return true;
568 }
569 return false;
570}
571
572/** 不读取未终止字符串，也不让两份同样缺失的设备身份互相作证 */
573static int identity_shape(const struct dppd_endpoint *endpoint)
574{
575 if ((endpoint->kind != DPPD_ENDPOINT_ETHDEV && endpoint->kind != DPPD_ENDPOINT_REPRESENTOR) ||
576 !role_is_valid(endpoint->role) ||
577 (endpoint->role_source != DPPD_ENDPOINT_ROLE_SOURCE_UNKNOWN &&
578 endpoint->role_source != DPPD_ENDPOINT_ROLE_SOURCE_MLX5_PHYS_PORT_NAME) ||
579 (endpoint->has_switch_domain && endpoint->switch_domain_id == UINT16_MAX))
580 return -EINVAL;
581 if (memchr(endpoint->switch_name, '\0', sizeof(endpoint->switch_name)) == NULL ||
582 memchr(endpoint->driver_name, '\0', sizeof(endpoint->driver_name)) == NULL ||
583 memchr(endpoint->interface_name, '\0', sizeof(endpoint->interface_name)) == NULL ||
584 memchr(endpoint->device_name, '\0', sizeof(endpoint->device_name)) == NULL ||
585 memchr(endpoint->phys_port_name, '\0', sizeof(endpoint->phys_port_name)) == NULL)
586 return -EINVAL;
587 if (endpoint->driver_name[0] == '\0' || strcmp(endpoint->driver_name, "unknown") == 0 ||
588 endpoint->device_name[0] == '\0')
589 return -ENODATA;
590 return 0;
591}
592
593static int identity_port_set(const struct dppd_topology *topology, const uint16_t *ports, uint16_t count)
594{
595 if (topology == NULL || topology->nb_endpoints > DPPD_MAX_PORTS)
596 return -EINVAL;
597 for (uint16_t i = 0; i < topology->nb_endpoints; ++i) {
598 for (uint16_t j = 0; j < i; ++j)
599 if (topology->endpoints[i].ethdev_port_id == topology->endpoints[j].ethdev_port_id)
600 return -EUCLEAN;
601 int rc = identity_shape(&topology->endpoints[i]);
602 if (rc != 0)
603 return rc;
604 }
605 for (uint16_t i = 0; i < count; ++i)
606 if (dppd_topology_find(topology, ports[i]) == NULL)
607 return -ENOENT;
608 return topology->nb_endpoints == count ? 0 : -EXDEV;
609}

611/** 有效标志本身也属于身份，标志为未知时不使用残留编号猜测同一端点 */
612static bool same_identity(const struct dppd_endpoint *left, const struct dppd_endpoint *right)
613{
614 return left->ethdev_port_id == right->ethdev_port_id && left->kind == right->kind &&
615 left->has_switch_domain == right->has_switch_domain &&
616 (!left->has_switch_domain ||
617 (left->switch_domain_id == right->switch_domain_id && left->switch_port_id == right->switch_port_id)) &&
618 left->role == right->role && left->role_source == right->role_source && left->role_error == right->role_error &&
619 left->if_index == right->if_index && left->has_controller == right->has_controller &&
620 (!left->has_controller || left->controller_id == right->controller_id) && left->has_pf == right->has_pf &&
621 (!left->has_pf || left->pf_id == right->pf_id) && left->has_function == right->has_function &&
622 (!left->has_function || left->function_id == right->function_id) &&
623 left->has_physical_port == right->has_physical_port &&
624 (!left->has_physical_port || left->physical_port_id == right->physical_port_id) &&
625 strcmp(left->switch_name, right->switch_name) == 0 && strcmp(left->driver_name, right->driver_name) == 0 &&
626 strcmp(left->interface_name, right->interface_name) == 0 &&
627 strcmp(left->device_name, right->device_name) == 0 && strcmp(left->phys_port_name, right->phys_port_name) == 0;
628}
629
630int dppd_topology_validate_identity(const struct dppd_topology *expected, const struct dppd_topology *candidate,
631 const uint16_t *ports, uint16_t count)
632{
633 if (ports == NULL || count == 0 || count > DPPD_MAX_PORTS)
634 return -EINVAL;
635 for (uint16_t i = 0; i < count; ++i) {
636 if (ports[i] == UINT16_MAX)
637 return -EINVAL;
638 for (uint16_t j = 0; j < i; ++j)
639 if (ports[i] == ports[j])
640 return -EEXIST;
641 }
642 int rc = identity_port_set(expected, ports, count);
643 if (rc == 0)
644 rc = identity_port_set(candidate, ports, count);
645 if (rc != 0)
646 return rc;
647 for (uint16_t i = 0; i < count; ++i)
648 if (!same_identity(dppd_topology_find(expected, ports[i]), dppd_topology_find(candidate, ports[i])))
649 return -EXDEV;
650 return 0;
651}

653int dppd_topology_rediscover_verified_with_reader(const struct dppd_config *cfg, const struct dppd_topology *expected,
654 struct dppd_topology *output, dppd_topology_identity_reader reader, void *opaque)
655{
656 if (cfg == NULL || expected == NULL || output == NULL)
657 return -EINVAL;
658 int rc = dppd_topology_validate_identity(expected, expected, cfg->ports, cfg->nb_ports);
659 if (rc != 0)
660 return rc;
661 struct dppd_topology candidate;
662 rc = dppd_topology_discover_with_reader(cfg, &candidate, reader, opaque);
663 if (rc == 0)
664 rc = dppd_topology_validate_identity(expected, &candidate, cfg->ports, cfg->nb_ports);
665 if (rc == 0)
666 *output = candidate;
667 return rc;
668}

670int dppd_topology_rediscover_verified(const struct dppd_config *cfg, const struct dppd_topology *expected,
671 struct dppd_topology *output)
672{
673 return dppd_topology_rediscover_verified_with_reader(cfg, expected, output, read_identity, NULL);
674}

676const struct dppd_endpoint *dppd_topology_find(const struct dppd_topology *topology, uint16_t ethdev_port_id)
677{
678 if (topology == NULL || topology->nb_endpoints > DPPD_MAX_PORTS)
679 return NULL;
680 for (uint16_t i = 0; i < topology->nb_endpoints; ++i) {
681 if (topology->endpoints[i].ethdev_port_id == ethdev_port_id)
682 return &topology->endpoints[i];
683 }
684 return NULL;
685}

687void dppd_topology_dump(const struct dppd_topology *topology)
688{
689 if (topology == NULL || topology->nb_endpoints > DPPD_MAX_PORTS)
690 return;
691 for (uint16_t i = 0; i < topology->nb_endpoints; ++i) {
692 const struct dppd_endpoint *endpoint = &topology->endpoints[i];
693 printf("[dppd] endpoint port=%u kind=%s role=%s role-source=%s role-error=%d driver=%s switch-domain=%s",
694 endpoint->ethdev_port_id, endpoint->kind == DPPD_ENDPOINT_REPRESENTOR ? "representor" : "ethdev",
695 dppd_endpoint_role_name(endpoint->role), dppd_endpoint_role_source_name(endpoint->role_source),
696 endpoint->role_error, endpoint->driver_name, endpoint->has_switch_domain ? "yes" : "no");
697 if (endpoint->has_switch_domain)
698 printf(" domain=%u switch-port=%u switch=%s", endpoint->switch_domain_id, endpoint->switch_port_id,
699 endpoint->switch_name);
700 if (endpoint->if_index != 0)
701 printf(" ifindex=%u interface=%s phys-port-name=%s device=%s", endpoint->if_index, endpoint->interface_name,
702 endpoint->phys_port_name, endpoint->device_name);
703 printf("\n");
704 }
