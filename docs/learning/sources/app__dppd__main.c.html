42/** 从已落盘的端口身份读取观察范围，不再按可复用的 DPDK 名称推断内核接口 */
43static int observe_recovery_port(struct recovery_context *context, struct dppd_recovery_inspection *inspection)
44{
45 for (uint32_t index = 0; index < context->guard->record.count; ++index)
46 if (context->guard->record.ports[index].port_id == context->pending_port)
47 return dppd_recovery_inspect(&context->guard->record.ports[index], inspection);
48 return -ENOENT;
49}

51/** 不能只凭可重用的端口号记录身份，无法取得设备名称时拒绝这次硬件安装 */
52static int prepare_recovery(void *context, uint16_t port_id, const struct dppd_rule *rule, uint64_t *attempt,
53 uint8_t cookie[DPPD_TAP_COOKIE_SIZE])
54{
55 struct recovery_context *recovery = context;
56 const struct dppd_port *port = dppd_devices_find(recovery->devices, port_id);
57 const struct dppd_endpoint *endpoint = dppd_topology_find(&recovery->devices->topology, port_id);
58 int rc;
59
60 if (port == NULL || endpoint == NULL || !port->identity.device_name_known)
61 return -ENODEV;
62 rc = dppd_scope_check(&managed_scope);
63 if (rc != 0)
64 return rc;
65 if (recovery->monitor->error != 0)
66 return recovery->monitor->error;
67 uint64_t shared_id = 0, shared_generation = 0;
68 for (uint16_t index = 0; index < rule->nb_actions; ++index)
69 if (rule->actions[index].type == DPPD_ACTION_SHARED_COUNT) {
70 shared_id = rule->actions[index].conf.shared_count.id;
71 shared_generation = rule->actions[index].conf.shared_count.generation;
72 }
73 /** TAP 所有权模式依赖可读回的独立动作，不能给间接共享动作伪造同样的证明 */
74 if (shared_id != 0 && (recovery->native_owner || recovery->monitor->fd >= 0))
75 return -ENOTSUP;
76 memset(cookie, 0, DPPD_TAP_COOKIE_SIZE);
77 memset(recovery->pending_cookie, 0, sizeof(recovery->pending_cookie));
78 /** 此标识仅用于无副作用的能力检查，真正创建的随机标识稍后生成并落盘 */
79 if (recovery->native_owner) {
80 const uint8_t probe[DPPD_TAP_COOKIE_SIZE] = {1};
81 struct dppd_flow_error error;
82 rc = dppd_flow_validate_owned(port_id, rule, probe, &error);
83 if (rc != 0) {
84 fprintf(stderr, "[dppd] TAP owner validation failed: %s (%d)\n", error.message, rc);
85 return rc;
86 }
87 }
88 /** TAP 的 DPDK 名称和内核接口名不同，必须由驱动给出真实索引后再向内核核对 */
89 if (strcmp(endpoint->driver_name, "net_tap") == 0) {
90 struct rte_eth_dev_info info;
91 struct dppd_recovery_identity identity;
92 rc = rte_eth_dev_info_get(port_id, &info);
93 if (rc == 0)
94 rc = dppd_recovery_tap_identity(info.if_index, &identity);
95 if (rc != 0)
96 return recovery->monitor->fd >= 0 ? (recovery->monitor->error = rc) : rc;
97 if (rc == 0 && recovery->native_owner)
98 rc = dppd_tap_content_from_rule(rule, info.max_rx_queues, &recovery->pending_content);
99 if (rc == 0)
100 rc = dppd_monitor_client_identity(recovery->monitor, port_id, &identity);
101 if (rc == 0)
102 rc = dppd_monitor_client_check(recovery->monitor);
103 if (rc != 0)
104 return rc;
105 rc = dppd_recovery_guard_prepare_identity(recovery->guard, port_id, port->identity.device_name,
106 endpoint->driver_name, rule->id, rule->generation, &identity);
107 } else if (recovery->monitor->fd >= 0)
108 return recovery->monitor->error = -ENOTSUP;
109 else
110 rc = dppd_recovery_guard_prepare(recovery->guard, port_id, port->identity.device_name, endpoint->driver_name,
111 rule->id, rule->generation);
112 if (rc != 0)
113 return rc;
114 if (recovery->native_owner)
115 rc = dppd_recovery_guard_begin_owned(recovery->guard, port_id, rule->id, rule->generation, attempt, cookie,
116 &recovery->pending_content);
117 else if (shared_id != 0)
118 rc = dppd_recovery_guard_begin_flow(recovery->guard, port_id, rule->id, rule->generation, rule->domain,
119 shared_id, shared_generation, attempt);
120 else
121 rc = dppd_recovery_guard_begin(recovery->guard, port_id, rule->id, rule->generation, attempt);
122 if (rc != 0)
123 return rc;
124 recovery->pending_port = port_id;
125 memcpy(recovery->pending_cookie, cookie, sizeof(recovery->pending_cookie));
126 /** 观察是附加线索，失败仍允许有意图保护的创建，之后明确记录无法观察 */
127 recovery->before_error = observe_recovery_port(recovery, &recovery->before);
128 /** 写盘与内核查询可能耗时，进入驱动前再取一次新回复，不能复用启动时的确认 */
129 rc = dppd_scope_check(&managed_scope);
130 return rc != 0 ? rc : dppd_monitor_client_check(recovery->monitor);
131}

133/** 记录失败只报告一次并保持故障锁存，避免清理阶段反复刷屏或把不确定记录覆盖为干净 */
134static void recovery_record_error(struct recovery_context *context, int error)
135{
136 context->guard->faulted = true;
137 if (!context->record_error_reported) {
138 fprintf(stderr, "[dppd] recovery attempt record failed: %d; external reconciliation required\n", error);
139 context->record_error_reported = true;
140 }
141}

143/** 共享计数器先持久化独立对象意图，不把 indirect 指针写入日志或伪装成 TAP filter */
144static int prepare_shared_recovery(void *context, const struct dppd_shared_count_spec *spec, uint64_t *attempt)
145{
146 struct recovery_context *recovery = context;
147 const struct dppd_port *port = dppd_devices_find(recovery->devices, spec->install_port_id);
148 const struct dppd_endpoint *endpoint = dppd_topology_find(&recovery->devices->topology, spec->install_port_id);
149 if (port == NULL || endpoint == NULL || !port->identity.device_name_known)
150 return -ENODEV;
151 if (recovery->native_owner || recovery->monitor->fd >= 0)
152 return -ENOTSUP;
153 int rc = dppd_scope_check(&managed_scope);
154 if (rc != 0)
155 return rc;
156 if (recovery->monitor->error != 0)
157 return recovery->monitor->error;
158 struct dppd_recovery_identity identity, *captured = NULL;
159 if (strcmp(endpoint->driver_name, "net_tap") == 0) {
160 struct rte_eth_dev_info info;
161 rc = rte_eth_dev_info_get(spec->install_port_id, &info);
162 if (rc == 0)
163 rc = dppd_recovery_tap_identity(info.if_index, &identity);
164 if (rc != 0)
165 return rc;
166 captured = &identity;
167 }
168 rc = dppd_recovery_guard_prepare_shared(recovery->guard, spec, port->identity.device_name, endpoint->driver_name,
169 captured);
170 if (rc == 0)
171 rc = dppd_recovery_guard_begin_shared(recovery->guard, spec, attempt);
172 /** 日志同步期间设备或受控作用域也可能改变，进入 PMD 前再核对一次 */
173 if (rc == 0)
174 rc = dppd_scope_check(&managed_scope);
175 return rc;
176}

178/** 只记录真实返回值，普通 flow 的 TC 新增观察不能用来定位 indirect 对象 */
179static int finish_shared_recovery_create(void *context, uint64_t attempt, int create_error)
180{
181 struct recovery_context *recovery = context;
182 int rc = dppd_recovery_guard_created(recovery->guard, attempt, create_error, DPPD_EVIDENCE_NONE, 0, NULL);
183 if (rc != 0)
184 recovery_record_error(recovery, rc);
185 if (rc == 0 && create_error == 0)
186 rc = dppd_scope_check(&managed_scope);
187 return rc;
188}

190/** 对比创建前后的只读列表，保留原始驱动结果，候选坐标只表示观察到一次新增 */
191static int finish_recovery_create(void *context, uint64_t attempt, int create_error)
192{
193 struct recovery_context *recovery = context;
194 struct dppd_recovery_inspection after = {0};
195 struct dppd_recovery_filter candidate;
196 enum dppd_recovery_evidence evidence = DPPD_EVIDENCE_UNAVAILABLE;
197 int after_error = recovery->native_owner || recovery->before_error == 0 ? observe_recovery_port(recovery, &after)
198 : recovery->before_error;
199 int observed = recovery->before_error != 0 ? recovery->before_error : after_error;
200 if (observed == 0)
201 evidence = dppd_recovery_compare(&recovery->before, &after, &candidate);
202 int rc = dppd_recovery_guard_created(recovery->guard, attempt, create_error, evidence, observed,
203 evidence == DPPD_EVIDENCE_SINGLE_ADDITION ? &candidate : NULL);
204 if (rc != 0)
205 recovery_record_error(recovery, rc);
206 /** 创建期间监护也可能退出，失败时保留真实 handle 交给事务回滚 */
207 if (rc == 0 && create_error == 0)
208 rc = dppd_scope_check(&managed_scope);
209 if (rc == 0 && create_error == 0)
210 rc = dppd_monitor_client_check(recovery->monitor);
211 /**
212 * 驱动报告成功后还要读回唯一标识，防止内核忽略属性时把未标记规则当作受保护
213 * 原始创建结果已经保存，返回错误会触发事务使用真实 handle 回滚
214 */
215 if (rc == 0 && create_error == 0 && recovery->native_owner) {
216 int matches = after_error != 0 ? after_error : dppd_recovery_cookie_matches(&after, recovery->pending_cookie);
217 if (matches != 1) {
218 fprintf(stderr, "[dppd] TAP owner readback not unique: %d; rolling back\n", matches);
219 return matches < 0 ? matches : -EUCLEAN;
220 }
221 /** 唯一标识还不足以接受创建结果，内容或范围无法核实时用真实 handle 回滚 */
222 for (uint32_t i = 0; i < after.count; ++i) {
223 const struct dppd_recovery_filter *filter = &after.filters[i];
224 if (!filter->cookie_valid || memcmp(filter->cookie, recovery->pending_cookie, DPPD_TAP_COOKIE_SIZE) != 0)
225 continue;
226 int content = dppd_recovery_verify(&recovery->pending_content, filter, false);
227 int scope = dppd_recovery_verify(&recovery->pending_content, filter, true);
228 if (content != 0 || scope != 0) {
229 fprintf(stderr, "[dppd] TAP rule readback failed: content=%d scope=%d; rolling back\n", content, scope);
230 return -EUCLEAN;
231 }
232 }
233 }
234 return rc;
235}

253static void sleep_control_loop(void)
254{
255 struct timespec remaining = {
256 .tv_sec = 0,
257 .tv_nsec = 100000000L,
258 };
259
260 /* 被非退出信号打断时睡完剩余时间；退出信号到达后立即返回主循环 */
261 while (nanosleep(&remaining, &remaining) != 0 && errno == EINTR && !stop_signal)
262 ;
263}

265/** EAL 探测之后、项目配置队列和重放之前核对全部端口，缺少任何一个都不接受会话 */
266static int bind_monitor(const struct dppd_config *config, struct dppd_recovery_guard *guard,
267 struct dppd_monitor_client *client)
268{
269 struct dppd_recovery_record ports = {.count = config->nb_ports};
270 char error[128];
271 if (dppd_config_validate(config, error, sizeof(error)) != 0)
272 return -EINVAL;
273 for (uint32_t index = 0; index < ports.count; ++index) {
274 struct rte_eth_dev_info info;
275 uint16_t port = config->ports[index];
276 int rc = rte_eth_dev_info_get(port, &info);
277 if (rc != 0)
278 return rc;
279 if (info.driver_name == NULL || strcmp(info.driver_name, "net_tap") != 0)
280 return -ENOTSUP;
281 ports.ports[index].port_id = port;
282 rc = dppd_recovery_tap_identity(info.if_index, &ports.ports[index].identity);
283 if (rc != 0)
284 return rc;
285 }
286 return dppd_monitor_client_open(client, config->recovery_monitor, guard, &ports);
287}

289/** 启动时绑定稳定端口映射，恢复配置不能按可复用的 DPDK 编号猜测设备或 TAP */
290static int setup_ipv4_routing(struct dppd_runtime *runtime, struct dppd_control_service *control)
291{
292 if (runtime->config.routing_path[0] == '\0')
293 return 0;
294 struct dppd_routing_binding bindings[DPPD_MAX_PORTS] = {0};
295 for (uint16_t i = 0; i < runtime->devices.nb_ports; ++i) {
296 const struct dppd_port *port = &runtime->devices.ports[i];
297 const struct dppd_endpoint *endpoint = dppd_topology_find(&runtime->devices.topology, port->port_id);
298 if (endpoint == NULL || !port->identity.device_name_known || !rte_is_valid_assigned_ether_addr(&port->mac))
299 return -ENODEV;
300 bindings[i].port_id = port->port_id;
301 strcpy(bindings[i].device, port->identity.device_name);
302 strcpy(bindings[i].driver, endpoint->driver_name);
303 if (strcmp(endpoint->driver_name, "net_tap") == 0) {
304 struct rte_eth_dev_info information;
305 int rc = rte_eth_dev_info_get(port->port_id, &information);
306 if (rc != 0)
307 return rc;
308 if (if_indextoname(information.if_index, bindings[i].ifname) == NULL)
309 return -errno;
310 }
311 }
312 int rc = dppd_routing_open(&control->routing, runtime->config.routing_path, bindings, runtime->devices.nb_ports);
313 if (rc == 0) {
314 dppd_runtime_set_ipv4_router(runtime, dppd_routing_router(control->routing));
315 printf("[dppd] IPv4 routing ready: path=%s\n", runtime->config.routing_path);
316 }
317 return rc;
318}

320int main(int argc, char **argv)
321{
322 /* 受控入口必须先封住进程派生和权限变化，再让 EAL 创建线程或访问 TAP */
323 int scope_error = dppd_scope_open(&managed_scope, DPPD_SCOPE_BUSINESS);
324 if (scope_error != 0) {
325 fprintf(stderr, "[dppd] managed scope refused: %d\n", scope_error);
326 return EXIT_FAILURE;
327 }
328 struct dppd_config config;
329 struct dppd_control_service control;
330 struct dppd_management_server management;
331 struct dppd_runtime runtime;
332 struct dppd_recovery_guard recovery_guard = {.fd = -1};
333 struct dppd_monitor_client monitor_client = {.fd = -1};
334 struct recovery_context recovery_context = {.guard = &recovery_guard,
335 .devices = &runtime.devices,
336 .monitor = &monitor_client};
337 uint64_t started_at;
338 uint64_t next_stats;
339 uint64_t timer_hz;
340 int parse_result;
341 int eal_consumed;
342 int loop_error = 0;
343 int rc = EXIT_FAILURE;
344 bool recovery_isolation = false;
345
346 /* EAL 必须先消费 -l/-a/--vdev 等参数，后续只解析 -- 后的应用参数 */
347 eal_consumed = rte_eal_init(argc, argv);
348 if (eal_consumed < 0) {
349 fprintf(stderr, "[dppd] EAL initialization failed\n");
350 return EXIT_FAILURE;
351 }
352
353 argc -= eal_consumed;
354 argv += eal_consumed;
355 parse_result = dppd_config_parse(argc, argv, &config);
356 if (parse_result == DPPD_CONFIG_HELP) {
357 dppd_config_print_usage(argv[0]);
358 rc = EXIT_SUCCESS;
359 goto cleanup_eal;
360 }
361 if (parse_result != DPPD_CONFIG_OK) {
362 fprintf(stderr, "[dppd] invalid application arguments\n");
363 dppd_config_print_usage(argv[0]);
364 goto cleanup_eal;
365 }
366
367 if (config.flow_state_path[0] != '\0') {
368 char error[256];
369 if (dppd_config_validate(&config, error, sizeof(error)) != 0) {
370 fprintf(stderr, "[dppd] managed configuration refused: %s\n", error);
371 goto cleanup_eal;
372 }
373 rc = dppd_managed_run(&config, &managed_scope);
374 goto cleanup_eal;
375 }
376 const char *other_storage[] = {config.state_path, config.recovery_path, config.recovery_monitor,
377 config.control_socket};
378 int routing_paths_error =
379 dppd_routing_check_paths(config.routing_path, other_storage, sizeof(other_storage) / sizeof(other_storage[0]));
380 if (routing_paths_error != 0) {
381 fprintf(stderr, "[dppd] IPv4 storage paths refused: %d\n", routing_paths_error);
382 goto cleanup_eal;
383 }
384 dppd_config_dump(&config);
385 /**
386 * EAL 已完成参数和设备探测，但本项目尚未配置队列、启动 worker 或重放规则
387 * 旧进程的待核对标记必须在这些步骤之前拦住启动，空 backend 不能证明设备已干净
388 */
389 if (config.recovery_path[0] != '\0') {
390 int guard_rc = dppd_recovery_guard_open(&recovery_guard, config.recovery_path, config.state_path);
391 if (guard_rc != 0) {
392 fprintf(stderr, "[dppd] recovery guard open failed: %d\n", guard_rc);
393 goto cleanup_eal;
394 }
395 if (recovery_guard.record.count != 0) {
396 fprintf(stderr,
397 "[dppd] prior hardware activity requires external reconciliation: "
398 "revision=%" PRIu64 " ports=%u; inspect with dppd-recovery show PATH\n",
399 recovery_guard.record.revision, recovery_guard.record.count);
400 goto cleanup_eal;
401 }
402 if (dppd_recovery_record_requires_monitor(&recovery_guard.record) && config.recovery_monitor[0] == '\0') {
403 fprintf(stderr, "[dppd] this recovery record requires --recovery-monitor\n");
404 goto cleanup_eal;
405 }
406 if (config.recovery_monitor[0] != '\0') {
407 int monitor_rc = bind_monitor(&config, &recovery_guard, &monitor_client);
408 if (monitor_rc != 0) {
409 fprintf(stderr, "[dppd] recovery monitor handshake failed: %d\n", monitor_rc);
410 goto cleanup_eal;
411 }
412 printf("[dppd] recovery monitor bound; fresh replies required for hardware installation\n");
413 }
414 }
415 /*
416 * 初始化依赖顺序：先发现并启动 ethdev，得到拓扑；control 引用该拓扑；
417 * management 再引用 control。goto 清理标签严格按相反顺序释放
418 */
419 if (dppd_runtime_init(&runtime, &config) != 0)
420 goto cleanup_eal;
421 if (dppd_control_init(&control, &runtime.devices.topology, config.rule_capacity, NULL) != 0) {
422 fprintf(stderr, "[dppd] control service initialization failed\n");
423 goto cleanup_runtime;
424 }
425 /** 快照重放前就启用事件门禁，不能等到 worker 启动后才开始保护驱动句柄 */
426 int device_binding = dppd_runtime_device_bind(&runtime, &control);
427 if (device_binding != 0) {
428 fprintf(stderr, "[dppd] device event tracking refused: %d\n", device_binding);
429 goto cleanup_control;
430 }
431 /*
432 * worker 只借用 control 持有的 backend；退出时 main 的清理顺序会先停止并 wait 所有
433 * worker（使其注销 QSBR reader），再调用 control_fini 释放 snapshot/QSBR。这里不能
434 * 绑定临时对象，也不能在 runtime_start 后替换该地址
435 */
436 dppd_runtime_set_software_backend(&runtime, &control.software);
437 control.conntrack = runtime.conntrack;
438 control.nd6 = runtime.nd6;
439 int routing_error = setup_ipv4_routing(&runtime, &control);
440 if (routing_error != 0) {
441 fprintf(stderr, "[dppd] IPv4 routing restore refused: %d\n", routing_error);
442 goto cleanup_control;
443 }
444 /** 启动重放也会安装硬件规则，所以必须在读取并重放快照之前绑定保护 */
445 if (recovery_guard.fd >= 0) {
446 recovery_context.native_owner = config.tap_owner_cookie;
447 control.rte_flow.before_create = prepare_recovery;
448 control.rte_flow.before_create_context = &recovery_context;
449 control.rte_flow.after_create = finish_recovery_create;
450 control.rte_flow.after_remove = finish_recovery_remove;
451 control.rte_flow.before_shared_create = prepare_shared_recovery;
452 control.rte_flow.after_shared_create = finish_shared_recovery_create;
453 control.rte_flow.after_shared_remove = finish_recovery_remove;
454 }
455 if (config.state_path[0] != '\0') {
456 int restore_rc = dppd_control_persistence_restore(&control, config.state_path);
457
458 if (restore_rc != 0) {
459 if (restore_rc != -EUCLEAN || control.recovery_state != DPPD_CONTROL_RECOVERY_RECONCILIATION_REQUIRED) {
460 fprintf(stderr, "[dppd] rule snapshot restore failed: path=%s error=%s (%d)\n", config.state_path,
461 restore_rc < 0 && restore_rc >= -4095 ? strerror(-restore_rc) : "unknown", restore_rc);
462 goto cleanup_control;
463 }
464 /*
465 * transaction 回滚未完成：绝不能启动 worker 或开放普通规则接口，但保留
466 * 当前进程的 backend handle，供受限的 reconcile-retry 再次删除
467 */
468 recovery_isolation = true;
469 fprintf(stderr,
470 "[dppd] snapshot rollback incomplete; entering recovery isolation "
471 "for path=%s\n",
472 config.state_path);
473 }
474 if (!recovery_isolation)
475 printf("[dppd] rule snapshot ready: path=%s generation=%" PRIu64 " rules=%u\n", config.state_path,
476 dppd_rule_repository_generation(&control.rules), dppd_rule_repository_count(&control.rules));
477 }
478 if (dppd_management_start(&management, &control, &runtime.devices, &runtime, config.control_socket) != 0) {
479 fprintf(stderr, "[dppd] management socket initialization failed: %s\n", config.control_socket);
480 goto cleanup_control;
481 }
482 signal(SIGINT, handle_signal);
483 signal(SIGTERM, handle_signal);
484 /** 先发布本次重放或隔离的完成态，诊断线程只读取副本，不访问规则或驱动 */
485 if (dppd_telemetry_register(&runtime, &control) != 0) {
486 fprintf(stderr, "[dppd] telemetry command registration failed\n");
487 goto cleanup_management;
488 }
489 if (dppd_runtime_device_poll(&runtime) != 0)
490 goto cleanup_telemetry;
491 recovery_isolation |= control.recovery_state != DPPD_CONTROL_RECOVERY_READY;
492 if (recovery_isolation) {
493 /**
494 * 隔离保留只读诊断，不启动 ethdev worker，也不输出正常转发统计
495 * 清理重试结束后更新诊断副本，成功进入 RESTART_REQUIRED 后仍必须退出重启
496 */
497 while (!stop_signal) {
498 loop_error = dppd_runtime_device_poll(&runtime);
499 if (loop_error != 0)
500 break;
501 loop_error = dppd_monitor_client_check(&monitor_client);
502 if (loop_error != 0) {
503 fprintf(stderr, "[dppd] recovery monitor unavailable during isolation: %d\n", loop_error);
504 break;
505 }
506 /** 隔离仅停止软件转发，设备仍需监控，移除时应退出并清理而不是继续等待 */
507 loop_error = dppd_devices_io_quiesced(&runtime.devices) ? 0 : dppd_devices_poll_links(&runtime.devices);
508 if (loop_error != 0) {
509 fprintf(stderr, "[dppd] device monitoring failed during recovery isolation\n");
510 break;
511 }
512 if (runtime.conntrack != NULL)
513 dppd_conntrack_gc(runtime.conntrack, dppd_monotonic_ms(), 128);
514 loop_error = dppd_management_poll(&management);
515 if (loop_error != 0) {
516 fprintf(stderr, "[dppd] recovery management processing failed: %d\n", loop_error);
517 break;
518 }
519 if (dppd_telemetry_publish_rules(&control) != 0)
520 fprintf(stderr, "[dppd] rule telemetry snapshot unavailable\n");
521 if (control.recovery_state == DPPD_CONTROL_RECOVERY_RESTART_REQUIRED) {
522 fprintf(stderr, "[dppd] residual flows cleared; exiting for clean snapshot replay\n");
523 loop_error = -EUCLEAN;
524 break;
525 }
526 sleep_control_loop();
527 }
528 goto cleanup_telemetry;
529 }
530
531 loop_error = dppd_monitor_client_check(&monitor_client);
532 if (loop_error != 0) {
533 fprintf(stderr, "[dppd] recovery monitor unavailable before worker startup: %d\n", loop_error);
534 goto cleanup_telemetry;
535 }
536 if (dppd_runtime_start(&runtime) != 0)
537 goto cleanup_telemetry;
538
539 /* 使用 DPDK 单调 timer cycle，避免系统时间调整影响 duration 与统计周期 */
540 timer_hz = rte_get_timer_hz();
541 started_at = rte_get_timer_cycles();
542 next_stats = started_at + (timer_hz * config.stats_period_ms) / 1000U;
543 while (!stop_signal) {
544 const uint64_t now = rte_get_timer_cycles();
545
546 /** 事件优先于链路、AGE 和管理处理，先等待 worker 退出，再保留受限诊断入口 */
547 loop_error = dppd_runtime_device_poll(&runtime);
548 if (loop_error != 0)
549 break;
550 if (control.recovery_state != DPPD_CONTROL_RECOVERY_READY)
551 recovery_isolation = true;
552 else if (runtime.workers_started && runtime.device_recovery.status.state == DPPD_DEVICE_RECOVERY_RUNNING)
553 recovery_isolation = false;
554
555 loop_error = dppd_scope_check(&managed_scope);
556 if (loop_error != 0) {
557 fprintf(stderr, "[dppd] managed scope lost; stopping workers: %d\n", loop_error);
558 break;
559 }
560 loop_error = dppd_monitor_client_check(&monitor_client);
561 if (loop_error != 0) {
562 fprintf(stderr, "[dppd] recovery monitor unavailable; stopping workers: %d\n", loop_error);
563 break;
564 }
565
566 /**
567 * 进入恢复隔离后，不再按普通运行时长退出，也不继续周期性输出转发统计
568 * 这样用户仍有机会查看残留对象并发起清理，避免定时退出丢失本进程的 handle
569 */
570 if (!recovery_isolation && config.duration_s != 0 && now - started_at >= timer_hz * config.duration_s)
571 break;
572 loop_error = dppd_devices_io_quiesced(&runtime.devices) ? 0 : dppd_devices_poll_links(&runtime.devices);
573 if (loop_error != 0) {
574 fprintf(stderr, "[dppd] device monitoring failed; stopping workers\n");
575 break;
576 }
577 /**
578 * 线程可能在完成启动请求后才遇到规则读者注册失败
579 * 检测到异常结束就停止其他线程并退出，避免管理进程存活却缺少转发队列
580 */
581 loop_error = dppd_runtime_check_workers(&runtime);
582 if (loop_error != 0) {
583 fprintf(stderr, "[dppd] worker monitoring failed; stopping workers\n");
584 break;
585 }
586 if (!recovery_isolation && now >= next_stats) {
587 dppd_runtime_stats_dump(&runtime);
588 next_stats = now + (timer_hz * config.stats_period_ms) / 1000U;
589 }
590 /*
591 * 管理请求只在主线程串行执行，repository 与硬件事务无需额外加锁；
592 * poll 内部有连接预算，保证本循环仍能周期性观察 stop_signal
593 */
594 if (runtime.conntrack != NULL)
595 dppd_conntrack_gc(runtime.conntrack, dppd_monotonic_ms(), 128);
596 loop_error = dppd_management_poll(&management);
597 if (loop_error != 0) {
598 fprintf(stderr, "[dppd] management request processing failed: %d\n", loop_error);
599 break;
600 }
601 loop_error = dppd_runtime_device_poll(&runtime);
602 if (loop_error != 0)
603 break;
604 /** AGE 查询失败只保留错误并重试，不把缺少驱动数据当作已经过期 */
605 (void)dppd_control_age_poll(&control, dppd_monotonic_ms() * UINT64_C(1000000), DPPD_AGE_POLL_BUDGET);
606 /** 整轮请求处理完再发布，不让独立查询线程看到整批更新的中间账本 */
607 if (dppd_telemetry_publish_rules(&control) != 0)
608 fprintf(stderr, "[dppd] rule telemetry snapshot unavailable\n");
609 if (control.recovery_state != DPPD_CONTROL_RECOVERY_READY) {
610 if (!recovery_isolation) {
611 /**
612 * 启动之后发生的补偿失败，也要执行与启动恢复失败相同的隔离约束
613 * 先发送停止请求，再等待已经启动的工作线程结束，之后才处理清理重试
614 * 停止软件转发不代表硬件残留规则停止工作，硬件对象仍要单独清理
615 */
616 dppd_runtime_request_stop(&runtime);
617 if (dppd_runtime_wait(&runtime) != 0) {
618 loop_error = -EIO;
619 break;
620 }
621 recovery_isolation = true;
622 fprintf(stderr, "[dppd] rule recovery required; workers stopped\n");
623 }
624 if (control.recovery_state == DPPD_CONTROL_RECOVERY_RESTART_REQUIRED) {
625 /**
626 * 残留对象已清理，不在原进程中直接恢复正常服务
627 * 用失败码退出，让操作者或服务管理器重启并从旧快照重新建立完整状态
628 */
629 loop_error = -EUCLEAN;
630 break;
631 }
632 }
633 sleep_control_loop();
634 }
635
636 /* 主循环结束后立即封住入口，防止客户端在 worker 停止窗口继续排队 */
637 dppd_management_stop(&management);
638 dppd_runtime_request_stop(&runtime);
639 if (dppd_runtime_wait(&runtime) != 0)
640 loop_error = -EIO;
641 dppd_runtime_stats_dump(&runtime);
642 /** 隔离期间即使由用户发信号结束，也不能把这次故障退出报告为正常运行成功 */
643 if (loop_error == 0 && !recovery_isolation)
644 rc = EXIT_SUCCESS;
645
646cleanup_telemetry:
647 /* 标签按“只清理已成功初始化的层”组织，适用于启动失败和正常退出 */
648 dppd_telemetry_unregister_runtime();
649cleanup_management:
650 /* 先关闭入口，再回收已安装 flow，避免退出阶段接收新事务 */
651 dppd_management_stop(&management);
652cleanup_control:
653 /** 退出窗口也可能收到复位事件，未知句柄不能落入普通 fini 的逐项删除 */
654 (void)dppd_runtime_device_poll(&runtime);
655 if (dppd_control_fini(&control) != 0 && rc == EXIT_SUCCESS)
656 rc = EXIT_FAILURE;
657 /** backend 真正释放才证明已删除所有本进程 handle，失败时保留磁盘标记等待下次核对 */
658 if (recovery_guard.fd >= 0 && control.rte_flow.objects == NULL && dppd_recovery_guard_clean(&recovery_guard) != 0) {
659 fprintf(stderr, "[dppd] recovery guard could not record completed cleanup\n");
660 rc = EXIT_FAILURE;
661 }
662cleanup_runtime:
663 /** 清理失败或退出期间才收到移除通知，都必须保留失败退出码 */
664 if (dppd_runtime_destroy(&runtime) != 0 || dppd_devices_io_quiesced(&runtime.devices))
665 rc = EXIT_FAILURE;
666cleanup_eal:
667 dppd_monitor_client_close(&monitor_client);
668 /** 启动被旧标记阻断时同样只解锁，绝不能自动清除上一进程的线索 */
669 dppd_recovery_guard_close(&recovery_guard);
670 if (rte_eal_cleanup() != 0 && rc == EXIT_SUCCESS)
671 rc = EXIT_FAILURE;
672 return rc;
673}
