15/**
16 * 初始化数据面运行环境，为后续 worker 收包准备资源
17 * cfg 提供网口、队列数和 mbuf 数量等配置，初始化结果保存在 runtime 中
18 *
19 * 初始化顺序
20 * 1. 检查并保存配置
21 * 2. 分配 worker 使用的 lcore 和队列编号，清零状态与统计
22 * 3. 创建 mbuf 池，配置 RX/TX 队列并启动网口
23 * 4. 准备邻居发现，按配置启用连接跟踪和抓包服务
24 *
25 * 此时网口已启动，worker 要等 dppd_runtime_start 调用后才开始收包
26 * 成功返回 0，失败返回负错误码
27 */
28int dppd_runtime_init(struct dppd_runtime *runtime, const struct dppd_config *cfg)
29{
30 char error[256];
31 unsigned int lcore_id;
32 struct dppd_numa_cpu candidates[RTE_MAX_LCORE], selected[DPPD_MAX_WORKERS];
33 int port_sockets[DPPD_MAX_PORTS];
34 uint32_t candidate_count = 0;
35 int rc;
36
37 if (runtime == NULL || cfg == NULL)
38 return -EINVAL;
39 if (dppd_config_validate(cfg, error, sizeof(error)) != 0) {
40 fprintf(stderr, "[dppd] invalid configuration: %s\n", error);
41 return -EINVAL;
42 }
43
44 memset(runtime, 0, sizeof(*runtime));
45 runtime->config = *cfg;
46 /** worker 进度超时只用于 flow_state_path 启用的管理模式，普通模式将其关闭 */
47 if (cfg->flow_state_path[0] == '\0')
48 runtime->config.worker_timeout_ms = 0;
49 atomic_init(&runtime->stop_requested, false);
50
51 /**
52 * candidates 收集 EAL 已启用的非主 lcore 及其 NUMA 节点，不使用未启用的 CPU
53 * port_sockets 记录各网口所在的 NUMA 节点；设备返回未知节点时保留负值
54 * NUMA 节点用于选择靠近网口的核心，不是 RX/TX 队列编号
55 */
56 RTE_LCORE_FOREACH_WORKER (lcore_id) {
57 candidates[candidate_count++] = (struct dppd_numa_cpu){lcore_id, (int)rte_lcore_to_socket_id(lcore_id)};
58 }
59 for (uint16_t i = 0; i < cfg->nb_ports; ++i)
60 port_sockets[i] = rte_eth_dev_socket_id(cfg->ports[i]);
61 rc = dppd_numa_select(candidates, candidate_count, port_sockets, cfg->nb_ports, cfg->nb_queues, selected);
62 if (rc != 0) {
63 fprintf(stderr, "[dppd] queues=%u cannot select worker lcores from %u enabled candidates: %d\n", cfg->nb_queues,
64 candidate_count, rc);
65 return rc;
66 }
67 /**
68 * EAL 已准备线程及其 CPU 亲和性，本循环只记录 worker 的队列与执行 lcore，不重新创建或绑核
69 * 例：-l 0-2 --main-lcore 0，应用配置 --queues 2
70 * 主线程：lcore 0，对应逻辑 CPU 0，负责初始化、管理和监控
71 * worker 0：queue_id 0，选中 lcore 1，对应逻辑 CPU 1，处理各网口队列 0
72 * worker 1：queue_id 1，选中 lcore 2，对应逻辑 CPU 2，处理各网口队列 1
73 * lcore_id 来自 selected，其他配置下可能不连续；queue_id 连续分配，也用作 QSBR reader 编号
74 */
75 for (uint16_t worker_index = 0; worker_index < cfg->nb_queues; ++worker_index) {
76 struct dppd_worker *worker = &runtime->workers[worker_index];
77 worker->runtime = runtime;
78 worker->lcore_id = selected[worker_index].lcore_id;
79 worker->socket_id = selected[worker_index].socket_id;
80 worker->queue_id = worker_index;
81 atomic_init(&worker->state, DPPD_WORKER_IDLE);
82 atomic_init(&worker->heartbeat_ms, 0);
83 dppd_stats_init(&worker->stats);
84 for (uint16_t port_index = 0; port_index < cfg->nb_ports; ++port_index)
85 dppd_stats_init(&worker->port_stats[port_index]);
86 }
87 runtime->nb_workers = cfg->nb_queues;
88
89 rc = cfg->routing_path[0] == '\0' ? dppd_snapshot_build_port_pairs(cfg, &runtime->snapshot) : 0;
90 if (rc != 0)
91 return rc;
92 rc = dppd_devices_init(&runtime->devices, cfg);
93 if (rc != 0)
94 return rc;
95
96 uint8_t macs[DPPD_MAX_PORTS][6] = {{0}};
97 for (uint16_t i = 0; i < runtime->devices.nb_ports; ++i)
98 memcpy(macs[i], runtime->devices.ports[i].mac.addr_bytes, 6);
99 rc = dppd_nd6_create(&runtime->nd6, cfg, (const uint8_t(*)[6])macs, dppd_monotonic_ms());
100 if (rc != 0) {
101 dppd_devices_stop(&runtime->devices);
102 return rc;
103 }
104 /**
105 * 启用连接跟踪时创建一张会话表，由 runtime 持有，所有 worker 共用
106 * 启动只创建空表，报文通过 stateful_decide 处理后才可能产生会话
107 * 不启用时 conntrack 保持 NULL，process_batch 直接走普通 routed_decide
108 */
109 if (cfg->conntrack_capacity != 0) {
110 rc = dppd_conntrack_create_dual(&runtime->conntrack, cfg->conntrack_capacity, cfg->conntrack_timeout_s,
111 &cfg->snat, &cfg->snat6, cfg->dnat, cfg->dnat_count);
112 if (rc != 0) {
113 dppd_nd6_destroy(runtime->nd6);
114 runtime->nd6 = NULL;
115 dppd_devices_stop(&runtime->devices);
116 return rc;
117 }
118 }
119
120 if (cfg->enable_pdump) {
121 rc = rte_pdump_init();
122 if (rc != 0) {
123 fprintf(stderr, "[dppd] cannot initialize pdump service\n");
124 dppd_nd6_destroy(runtime->nd6);
125 runtime->nd6 = NULL;
126 dppd_conntrack_destroy(runtime->conntrack);
127 runtime->conntrack = NULL;
128 dppd_devices_stop(&runtime->devices);
129 return -EIO;
130 }
131 runtime->pdump_initialized = true;
132 }
133 return 0;
134}

149/**
150 * 将已准备好的 worker 工作函数提交到各自的 lcore，开始执行收包任务
151 *
152 * 清除停止标记 → 逐个设为 STARTING → remote_launch(worker_main, worker, lcore_id)
153 * → 提交成功，记录 launched，主线程继续启动下一个 worker
154 * → 提交失败，请求已启动的 worker 停止，并等待它们返回
155 *
156 * 全部提交成功返回 0，失败返回负错误码
157 * 返回 0 不代表所有 worker 已完成准备，实际可运行状态由 worker 自己发布为 RUNNING
158 */
159int dppd_runtime_start(struct dppd_runtime *runtime)
160{
161 uint16_t i;
162 int rc;
163
164 if (runtime == NULL || runtime->workers_started)
165 return -EINVAL;
166 /** 普通停止后可以重新启动线程，但设备移除后必须重建整个运行实例 */
167 if (dppd_devices_io_quiesced(&runtime->devices))
168 return dppd_devices_removal_requested(&runtime->devices) ? -ENODEV : -EUCLEAN;
169
170 atomic_store_explicit(&runtime->stop_requested, false, memory_order_release);
171 /** 依次启动各队列的 worker；任一提交失败时，下面的错误分支停止并等待已启动者 */
172 for (i = 0; i < runtime->nb_workers; ++i) {
173 struct dppd_worker *worker = &runtime->workers[i];
174
175 /** 必须在提交线程之前发布 STARTING，避免覆盖线程已经发布的 RUNNING 状态 */
176 atomic_store_explicit(&worker->heartbeat_ms, dppd_monotonic_ms(), memory_order_release);
177 atomic_store_explicit(&worker->state, DPPD_WORKER_STARTING, memory_order_release);
178 /**
179 * 把函数地址、worker 地址和目标 lcore 交给 EAL，在对应的已有工作线程上执行
180 * EAL 初始化时已准备线程和 CPU 亲和性，这里让线程开始执行业务任务
181 * 例如 worker->lcore_id = 3：lcore 3 调用 dppd_worker_main(worker)
182 * worker 地址原样成为工作函数的 arg，不复制结构体；主线程不等待整个收包循环结束
183 *
184 * | 参数 | 含义 |
185 |---|---|
186 | `dppd_worker_main` | 要执行的函数地址 |
187 | `worker` | 传给工作函数的参数地址 |
188 | `worker->lcore_id` | 在哪个 DPDK lcore 上执行 |
189 */
190 rc = rte_eal_remote_launch(dppd_worker_main, worker, worker->lcore_id);
191 if (rc != 0) {
192 atomic_store_explicit(&worker->state, DPPD_WORKER_FAILED, memory_order_release);
193 fprintf(stderr, "[dppd] cannot launch queue %u on lcore %u: %d\n", worker->queue_id, worker->lcore_id, rc);
194 dppd_runtime_request_stop(runtime);
195 int waited = dppd_runtime_wait(runtime);
196 return waited == -ETIMEDOUT ? waited : rc;
197 }
198 worker->launched = true;
199 /** 已有启动成功的任务，后面即使再启动失败，清理时也必须先等待这些任务退出 */
200 runtime->workers_started = true;
201 }
202 runtime->workers_started = true;
203 return 0;
204}

206/**
207 * 此检查不等待线程，也不调用网卡接口，因此可放在主线程的周期循环中
208 * STARTING 是正常短暂阶段；在没有停止或移除请求时结束的线程必须按故障处理
209 */
210int dppd_runtime_check_workers(const struct dppd_runtime *runtime)
211{
212 if (runtime == NULL)
213 return -EINVAL;
214 if (!runtime->workers_started || atomic_load_explicit(&runtime->stop_requested, memory_order_acquire) ||
215 dppd_devices_io_quiesced(&runtime->devices))
216 return 0;
217 for (uint16_t i = 0; i < runtime->nb_workers; ++i) {
218 const unsigned int state = atomic_load_explicit(&runtime->workers[i].state, memory_order_acquire);
219
220 if (state != DPPD_WORKER_STARTING && state != DPPD_WORKER_RUNNING) {
221 /**
222 * 初次检查后可能才收到停止或移除请求，线程会先看到请求并发布结束状态
223 * 读到结束状态后再次检查原因，避免把按通知结束的线程误报为独立故障
224 */
225 if (atomic_load_explicit(&runtime->stop_requested, memory_order_acquire) ||
226 dppd_devices_io_quiesced(&runtime->devices))
227 return 0;
228 return -EIO;
229 }
230 if (dppd_runtime_worker_stalled(runtime, i, dppd_monotonic_ms()))
231 return -ETIMEDOUT;
232 }
233 return 0;
234}

236bool dppd_runtime_worker_stalled(const struct dppd_runtime *runtime, uint16_t worker, uint64_t now_ms)
237{
238 if (runtime == NULL || worker >= runtime->nb_workers || !runtime->workers_started ||
239 runtime->config.worker_timeout_ms == 0 ||
240 atomic_load_explicit(&runtime->stop_requested, memory_order_acquire) ||
241 dppd_devices_io_quiesced(&runtime->devices))
242 return false;
243 uint64_t heartbeat = atomic_load_explicit(&runtime->workers[worker].heartbeat_ms, memory_order_acquire);
244 /** 读者采样之后 worker 可能更新心跳，略晚于 now 的值仍表示线程正在前进 */
245 return heartbeat == 0 || (now_ms >= heartbeat && now_ms - heartbeat > runtime->config.worker_timeout_ms);
246}

255/**
256 * 等待已提交启动任务的 worker 返回，收集工作函数结果并更新运行状态
257 * launched 为 false 的条目无需等待；已返回的任务清除 launched，全部结束后清除 workers_started
258 * 成功返回 0，任务失败返回其错误；启用进度超时的模式可能返回 -ETIMEDOUT，此时仍不能释放线程使用的资源
259 */
260int dppd_runtime_wait(struct dppd_runtime *runtime)
261{
262 uint16_t i;
263 int result = 0;
264
265 if (runtime == NULL)
266 return -EINVAL;
267 uint64_t started = dppd_monotonic_ms();
268 for (i = 0; i < runtime->nb_workers; ++i) {
269 struct dppd_worker *worker = &runtime->workers[i];
270 int rc;
271
272 if (!worker->launched)
273 continue;
274 /** 管理模式停机也有时限，已请求停止的卡死线程不能绕过主循环心跳检查后永久等待 */
275 while (runtime->config.worker_timeout_ms != 0 && rte_eal_get_lcore_state(worker->lcore_id) == RUNNING) {
276 if (dppd_monotonic_ms() - started > runtime->config.worker_timeout_ms)
277 return -ETIMEDOUT;
278 struct timespec pause = {.tv_nsec = 1000000L};
279 nanosleep(&pause, NULL);
280 }
281 rc = rte_eal_wait_lcore(worker->lcore_id);
282 worker->launched = false;
283 /** 等待结果是最终依据，保留失败状态供清理前的诊断读取 */
284 atomic_store_explicit(&worker->state, rc == 0 ? DPPD_WORKER_STOPPED : DPPD_WORKER_FAILED, memory_order_release);
285 if (rc != 0 && result == 0)
286 result = rc;
287 }
288 runtime->workers_started = false;
289 return result;
290}

292void dppd_runtime_stats_read(const struct dppd_runtime *runtime, struct dppd_stats_values *total)
293{
294 uint16_t i;
295
296 if (runtime == NULL || total == NULL)
297 return;
298 memset(total, 0, sizeof(*total));
299 for (i = 0; i < runtime->nb_workers; ++i) {
300 struct dppd_stats_values values;
301
302 dppd_stats_read(&runtime->workers[i].stats, &values);
303 dppd_stats_accumulate(total, &values);
304 }
305}

307int dppd_runtime_stats_query(const struct dppd_runtime *runtime, uint16_t port_id, uint16_t queue_id,
308 struct dppd_stats_values *total)
309{
310 uint16_t i, j;
311 bool has_port = port_id == DPPD_STATS_ALL;
312 bool has_queue = queue_id == DPPD_STATS_ALL;
313
314 if (runtime == NULL || total == NULL)
315 return -EINVAL;
316 memset(total, 0, sizeof(*total));
317 for (j = 0; j < runtime->devices.nb_ports; ++j)
318 has_port |= runtime->devices.ports[j].port_id == port_id;
319 for (i = 0; i < runtime->nb_workers; ++i)
320 has_queue |= runtime->workers[i].queue_id == queue_id;
321 if (!has_port || !has_queue)
322 return -ENOENT;
323 for (i = 0; i < runtime->nb_workers; ++i) {
324 const struct dppd_worker *worker = &runtime->workers[i];
325
326 if (queue_id != DPPD_STATS_ALL && worker->queue_id != queue_id)
327 continue;
328 for (j = 0; j < runtime->devices.nb_ports; ++j) {
329 struct dppd_stats_values values;
330
331 if (port_id != DPPD_STATS_ALL && runtime->devices.ports[j].port_id != port_id)
332 continue;
333 dppd_stats_read(&worker->port_stats[j], &values);
334 dppd_stats_accumulate(total, &values);
335 }
336 }
337 return 0;
338}

351/**
352 * 释放 runtime 拥有的资源，成功返回 0，失败返回负错误码
353 * 确认 worker 已退出 → 关闭抓包服务 → 释放邻居等待报文和连接会话 → 关闭设备、释放 mbuf 池
354 * 软件规则表和路由由调用方清理，runtime 只通过指针使用它们
355 */
356int dppd_runtime_destroy(struct dppd_runtime *runtime)
357{
358 if (runtime == NULL)
359 return -EINVAL;
360 /** 尚有 worker 任务时先通知并等待，确认全部返回后才允许继续释放资源 */
361 if (runtime->workers_started) {
362 dppd_runtime_request_stop(runtime);
363 int waited = dppd_runtime_wait(runtime);
364 /** 未返回的线程仍借用设备和规则，拒绝清理，让正式入口走失败退出 */
365 if (runtime->workers_started)
366 return waited != 0 ? waited : -EBUSY;
367 }
368 if (runtime->pdump_initialized) {
369 (void)rte_pdump_uninit();
370 runtime->pdump_initialized = false;
371 }
372 dppd_nd6_destroy(runtime->nd6);
373 runtime->nd6 = NULL;
374 dppd_conntrack_destroy(runtime->conntrack);
375 runtime->conntrack = NULL;
376 /** 将端口关闭或回调注销错误交给进程入口，避免清理失败却报告正常退出 */
377 return dppd_devices_stop(&runtime->devices);
