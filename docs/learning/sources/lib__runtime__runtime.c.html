15/**
16 * 初始化数据面运行环境，为后续 worker 收包准备资源
17 * cfg 提供网口、队列数和 mbuf 数量等配置，初始化结果保存在 runtime 中
18 *
19 * 初始化顺序
20 * 1. 检查并保存配置
21 * 2. 分配 worker 使用的 lcore 和队列编号，清零状态与统计
22 * 3. 创建 mbuf 池，配置 RX/TX 队列并启动网口
23 * 4. 准备邻居发现，按配置启用连接跟踪和抓包服务
24 *
25 * 此时网口已启动，worker 要等 dppd_runtime_start 调用后才开始收包
26 * 成功返回 0，失败返回负错误码
27 */
28int dppd_runtime_init(struct dppd_runtime *runtime, const struct dppd_config *cfg)
29{
30 char error[256];
31 unsigned int lcore_id;
32 struct dppd_numa_cpu candidates[RTE_MAX_LCORE], selected[DPPD_MAX_WORKERS];
33 int port_sockets[DPPD_MAX_PORTS];
34 uint32_t candidate_count = 0;
35 int rc;
36
37 if (runtime == NULL || cfg == NULL)
38 return -EINVAL;
39 if (dppd_config_validate(cfg, error, sizeof(error)) != 0) {
40 fprintf(stderr, "[dppd] invalid configuration: %s\n", error);
41 return -EINVAL;
42 }
43
44 memset(runtime, 0, sizeof(*runtime));
45 runtime->config = *cfg;
46 /** worker 进度超时只用于 flow_state_path 启用的管理模式，普通模式将其关闭 */
47 if (cfg->flow_state_path[0] == '\0')
48 runtime->config.worker_timeout_ms = 0;
49 atomic_init(&runtime->stop_requested, false);
50
51 /**
52 * candidates 收集 EAL 已启用的非主 lcore 及其 NUMA 节点，不使用未启用的 CPU
53 * port_sockets 记录各网口所在的 NUMA 节点；设备返回未知节点时保留负值
54 * NUMA 节点用于选择靠近网口的核心，不是 RX/TX 队列编号
55 */
56 RTE_LCORE_FOREACH_WORKER (lcore_id) {
57 candidates[candidate_count++] = (struct dppd_numa_cpu){lcore_id, (int)rte_lcore_to_socket_id(lcore_id)};
58 }
59 for (uint16_t i = 0; i < cfg->nb_ports; ++i)
60 port_sockets[i] = rte_eth_dev_socket_id(cfg->ports[i]);
61 rc = dppd_numa_select(candidates, candidate_count, port_sockets, cfg->nb_ports, cfg->nb_queues, selected);
62 if (rc != 0) {
63 fprintf(stderr, "[dppd] queues=%u cannot select worker lcores from %u enabled candidates: %d\n", cfg->nb_queues,
64 candidate_count, rc);
65 return rc;
66 }
67 /**
68 * EAL 已准备线程及其 CPU 亲和性，本循环只记录 worker 的队列与执行 lcore，不重新创建或绑核
69 * 例：-l 0-2 --main-lcore 0，应用配置 --queues 2
70 * 主线程：lcore 0，对应逻辑 CPU 0，负责初始化、管理和监控
71 * worker 0：queue_id 0，选中 lcore 1，对应逻辑 CPU 1，处理各网口队列 0
72 * worker 1：queue_id 1，选中 lcore 2，对应逻辑 CPU 2，处理各网口队列 1
73 * lcore_id 来自 selected，其他配置下可能不连续；queue_id 连续分配，也用作 QSBR reader 编号
74 */
75 for (uint16_t worker_index = 0; worker_index < cfg->nb_queues; ++worker_index) {
76 struct dppd_worker *worker = &runtime->workers[worker_index];
77 worker->runtime = runtime;
78 worker->lcore_id = selected[worker_index].lcore_id;
79 worker->socket_id = selected[worker_index].socket_id;
80 worker->queue_id = worker_index;
81 atomic_init(&worker->state, DPPD_WORKER_IDLE);
82 atomic_init(&worker->heartbeat_ms, 0);
83 dppd_stats_init(&worker->stats);
84 for (uint16_t port_index = 0; port_index < cfg->nb_ports; ++port_index)
85 dppd_stats_init(&worker->port_stats[port_index]);
86 }
87 runtime->nb_workers = cfg->nb_queues;
88
89 rc = dppd_config_uses_routing(cfg) ? 0 : dppd_snapshot_build_port_pairs(cfg, &runtime->snapshot);
90 if (rc != 0)
91 return rc;
92 rc = dppd_devices_init(&runtime->devices, cfg);
93 if (rc != 0)
94 return rc;
95
96 /**
97 * 网口准备完成后，按 cfg->ports 的顺序收集 MAC，每行保存一个网口的 6 字节地址
98 * ND 创建函数按端口号找到对应行并复制 MAC，返回后不保留这个局部数组的地址
99 */
100 uint8_t macs[DPPD_MAX_PORTS][6] = {{0}};
101 for (uint16_t i = 0; i < runtime->devices.nb_ports; ++i)
102 memcpy(macs[i], runtime->devices.ports[i].mac.addr_bytes, 6);
103 /** 转为指向只读 6 字节数组的指针；&runtime->nd6 用于接收新建管理表的地址 */
104 rc = dppd_nd6_create(&runtime->nd6, cfg, (const uint8_t(*)[6])macs, dppd_monotonic_ms());
105 if (rc != 0) {
106 dppd_devices_stop(&runtime->devices);
107 return rc;
108 }
109 runtime->nd6_owned = runtime->nd6 != NULL;
110 /**
111 * 启用连接跟踪时创建一张会话表，由 runtime 持有，所有 worker 共用
112 * 启动只创建空表，报文通过 stateful_decide 处理后才可能产生会话
113 * 不启用时 conntrack 保持 NULL，process_batch 直接走普通 routed_decide
114 */
115 if (cfg->conntrack_capacity != 0) {
116 rc = dppd_conntrack_create_dual(&runtime->conntrack, cfg->conntrack_capacity, cfg->conntrack_timeout_s,
117 &cfg->snat, &cfg->snat6, cfg->dnat, cfg->dnat_count);
118 if (rc != 0) {
119 dppd_nd6_destroy(runtime->nd6);
120 runtime->nd6 = NULL;
121 runtime->nd6_owned = false;
122 dppd_devices_stop(&runtime->devices);
123 return rc;
124 }
125 runtime->conntrack_owned = true;
126 }
127
128 /**
129 * --enable-pdump 只初始化抓包 IPC 请求处理，具体端口和文件由独立抓包程序选择
130 * 本项目脚本调用 dpdk-dumpcap 发起捕获，此处不自动抓包或写文件
131 */
132 if (cfg->enable_pdump) {
133 rc = rte_pdump_init();
134 if (rc != 0) {
135 fprintf(stderr, "[dppd] cannot initialize pdump service\n");
136 /** worker 尚未启动，先清理已建的 ND、会话表并调用设备清理，再将失败返回 main */
137 dppd_nd6_destroy(runtime->nd6);
138 runtime->nd6 = NULL;
139 runtime->nd6_owned = false;
140 dppd_conntrack_destroy(runtime->conntrack);
141 runtime->conntrack = NULL;
142 runtime->conntrack_owned = false;
143 dppd_devices_stop(&runtime->devices);
144 return -EIO;
145 }
146 /** 记录本次初始化成功，runtime_destroy 据此调用对应的 uninit */
147 runtime->pdump_initialized = true;
148 }
149 /** 收发资源准备结束，返回 main 继续控制与管理入口初始化；worker 要等 runtime_start 提交 */
150 return 0;
151}

194int dppd_runtime_rollback_nd6(struct dppd_runtime *runtime, struct dppd_nd6_rx_preparation *preparation)
195{
196 uint16_t failed = 0;
197
198 if (runtime == NULL || preparation == NULL || preparation->count > DPPD_MAX_PORTS)
199 return -EINVAL;
200 if (runtime->workers_started)
201 return -EBUSY;
202 /** 即使一个端口补偿失败，也继续关闭其余本次新增设置，并保留无法确认的端口 */
203 for (uint16_t i = 0; i < preparation->count; ++i) {
204 uint16_t port_id = preparation->ports[i];
205 const struct dppd_port *port = dppd_devices_find(&runtime->devices, port_id);
206 int rc = port == NULL || !port->started || atomic_load_explicit(&port->removed, memory_order_acquire) ?
207 -ENODEV : rte_eth_allmulticast_disable(port_id);
208 if (rc != 0 || rte_eth_allmulticast_get(port_id) != 0)
209 preparation->ports[failed++] = port_id;
210 }
211 preparation->count = failed;
212 return failed == 0 ? 0 : -EUCLEAN;
213}

215int dppd_runtime_prepare_nd6(struct dppd_runtime *runtime, const struct dppd_nd6_policy *policy,
216 struct dppd_nd6_rx_preparation *preparation)
217{
218 uint16_t ports[DPPD_MAX_PORTS], count = 0;
219 bool enabled[DPPD_MAX_PORTS];
220 int rc;
221
222 if (runtime == NULL || preparation == NULL)
223 return -EINVAL;
224 memset(preparation, 0, sizeof(*preparation));
225 rc = dppd_nd6_policy_validate(policy);
226 if (rc != 0)
227 return rc;
228 if (runtime->workers_started)
229 return -EBUSY;
230 /** 关闭 ND 的普通管理转发不需要驱动支持组播开关，也不额外访问网卡 */
231 if (policy->local_count == 0)
232 return 0;
233 if (dppd_devices_io_quiesced(&runtime->devices))
234 return -ENODEV;
235 /** 先检查全部引用端口并读取原状态，避免后面的端口无效时已经改变前面的端口 */
236 for (uint32_t i = 0; i < policy->local_count; ++i) {
237 uint16_t port_id = policy->locals[i].port_id, index = 0;
238 while (index < count && ports[index] != port_id)
239 ++index;
240 if (index < count)
241 continue;
242 const struct dppd_port *port = dppd_devices_find(&runtime->devices, port_id);
243 if (port == NULL || !port->started || atomic_load_explicit(&port->removed, memory_order_acquire))
244 return -ENODEV;
245 if (count == DPPD_MAX_PORTS)
246 return -E2BIG;
247 ports[count++] = port_id;
248 }
249 for (uint16_t i = 0; i < count; ++i) {
250 rc = rte_eth_allmulticast_get(ports[i]);
251 if (rc < 0)
252 return rc;
253 if (rc > 1)
254 return -EIO;
255 enabled[i] = rc != 0;
256 }
257 for (uint16_t i = 0; i < count; ++i) {
258 if (enabled[i])
259 continue;
260 rc = rte_eth_allmulticast_enable(ports[i]);
261 int state = rte_eth_allmulticast_get(ports[i]);
262 /** 成功开启或状态不明时都登记补偿，明确拒绝且仍关闭则不调用不支持的 disable */
263 if (rc == 0 || state != 0)
264 preparation->ports[preparation->count++] = ports[i];
265 if (rc == 0 && state == 1)
266 continue;
267 if (rc == 0)
268 rc = state < 0 ? state : -EIO;
269 if (dppd_runtime_rollback_nd6(runtime, preparation) != 0)
270 return -EUCLEAN;
271 return rc;
272 }
273 return 0;
274}

276/**
277 * 将已准备好的 worker 工作函数提交到各自的 lcore，开始执行收包任务
278 *
279 * 清除停止标记 → 逐个设为 STARTING → remote_launch(worker_main, worker, lcore_id)
280 * → 提交成功，记录 launched，主线程继续启动下一个 worker
281 * → 提交失败，请求已启动的 worker 停止，并等待它们返回
282 *
283 * 全部提交成功返回 0，失败返回负错误码
284 * 返回 0 不代表所有 worker 已完成准备，实际可运行状态由 worker 自己发布为 RUNNING
285 */
286int dppd_runtime_start(struct dppd_runtime *runtime)
287{
288 uint16_t i;
289 int rc;
290
291 if (runtime == NULL || runtime->workers_started)
292 return -EINVAL;
293 /** 普通停止后可以重新提交收包任务，设备移除后必须重建整个运行实例 */
294 if (dppd_devices_io_quiesced(&runtime->devices))
295 return dppd_devices_removal_requested(&runtime->devices) ? -ENODEV : -EUCLEAN;
296
297 atomic_store_explicit(&runtime->stop_requested, false, memory_order_release);
298 /** 依次启动各队列的 worker；任一提交失败时，下面的错误分支停止并等待已启动者 */
299 for (i = 0; i < runtime->nb_workers; ++i) {
300 struct dppd_worker *worker = &runtime->workers[i];
301
302 /** 必须在提交线程之前发布 STARTING，避免覆盖线程已经发布的 RUNNING 状态 */
303 atomic_store_explicit(&worker->heartbeat_ms, dppd_monotonic_ms(), memory_order_release);
304 atomic_store_explicit(&worker->state, DPPD_WORKER_STARTING, memory_order_release);
305 /**
306 * 通知指定 lcore 的 EAL 工作线程执行收包函数，线程和 CPU 亲和性已由 EAL 准备
307 *
308 * dppd_worker_main → 要执行的函数
309 * worker → 原样传给工作函数的 arg，不复制结构体
310 * worker->lcore_id → 执行任务的 lcore
311 *
312 * 例：worker 0 选中 lcore 1，则该线程执行 dppd_worker_main(&runtime->workers[0])
313 * 主线程提交后继续启动下一个 worker，不等待收包循环结束
314 */
315 rc = rte_eal_remote_launch(dppd_worker_main, worker, worker->lcore_id);
316 if (rc != 0) {
317 atomic_store_explicit(&worker->state, DPPD_WORKER_FAILED, memory_order_release);
318 fprintf(stderr, "[dppd] cannot launch queue %u on lcore %u: %d\n", worker->queue_id, worker->lcore_id, rc);
319 dppd_runtime_request_stop(runtime);
320 int waited = dppd_runtime_wait(runtime);
321 return waited == -ETIMEDOUT ? waited : rc;
322 }
323 worker->launched = true;
324 /** 已有任务提交成功，清理时必须等待其退出；是否进入 RUNNING 由工作线程自行报告 */
325 runtime->workers_started = true;
326 }
327 runtime->workers_started = true;
328 return 0;
329}

331/**
332 * 此检查不等待线程，也不调用网卡接口，因此可放在主线程的周期循环中
333 * STARTING 是正常短暂阶段；在没有停止或移除请求时结束的线程必须按故障处理
334 */
335int dppd_runtime_check_workers(const struct dppd_runtime *runtime)
336{
337 if (runtime == NULL)
338 return -EINVAL;
339 if (!runtime->workers_started || atomic_load_explicit(&runtime->stop_requested, memory_order_acquire) ||
340 dppd_devices_io_quiesced(&runtime->devices))
341 return 0;
342 for (uint16_t i = 0; i < runtime->nb_workers; ++i) {
343 const unsigned int state = atomic_load_explicit(&runtime->workers[i].state, memory_order_acquire);
344
345 if (state != DPPD_WORKER_STARTING && state != DPPD_WORKER_RUNNING) {
346 /**
347 * 初次检查后可能才收到停止或移除请求，线程会先看到请求并发布结束状态
348 * 读到结束状态后再次检查原因，避免把按通知结束的线程误报为独立故障
349 */
350 if (atomic_load_explicit(&runtime->stop_requested, memory_order_acquire) ||
351 dppd_devices_io_quiesced(&runtime->devices))
352 return 0;
353 return -EIO;
354 }
355 if (dppd_runtime_worker_stalled(runtime, i, dppd_monotonic_ms()))
356 return -ETIMEDOUT;
357 }
358 return 0;
359}

361bool dppd_runtime_worker_stalled(const struct dppd_runtime *runtime, uint16_t worker, uint64_t now_ms)
362{
363 if (runtime == NULL || worker >= runtime->nb_workers || !runtime->workers_started ||
364 runtime->config.worker_timeout_ms == 0 ||
365 atomic_load_explicit(&runtime->stop_requested, memory_order_acquire) ||
366 dppd_devices_io_quiesced(&runtime->devices))
367 return false;
368 uint64_t heartbeat = atomic_load_explicit(&runtime->workers[worker].heartbeat_ms, memory_order_acquire);
369 /** 读者采样之后 worker 可能更新心跳，略晚于 now 的值仍表示线程正在前进 */
370 return heartbeat == 0 || (now_ms >= heartbeat && now_ms - heartbeat > runtime->config.worker_timeout_ms);
371}

380/**
381 * 等待已提交启动任务的 worker 返回，收集工作函数结果并更新运行状态
382 * launched 为 false 的条目无需等待；已返回的任务清除 launched，全部结束后清除 workers_started
383 * 成功返回 0，任务失败返回其错误；启用进度超时的模式可能返回 -ETIMEDOUT，此时仍不能释放线程使用的资源
384 */
385int dppd_runtime_wait(struct dppd_runtime *runtime)
386{
387 uint16_t i;
388 int result = 0;
389
390 if (runtime == NULL)
391 return -EINVAL;
392 uint64_t started = dppd_monotonic_ms();
393 for (i = 0; i < runtime->nb_workers; ++i) {
394 struct dppd_worker *worker = &runtime->workers[i];
395 int rc;
396
397 if (!worker->launched)
398 continue;
399 /** 管理模式停机也有时限，已请求停止的卡死线程不能绕过主循环心跳检查后永久等待 */
400 while (runtime->config.worker_timeout_ms != 0 && rte_eal_get_lcore_state(worker->lcore_id) == RUNNING) {
401 if (dppd_monotonic_ms() - started > runtime->config.worker_timeout_ms)
402 return -ETIMEDOUT;
403 struct timespec pause = {.tv_nsec = 1000000L};
404 nanosleep(&pause, NULL);
405 }
406 /** 等待该 lcore 上的 dppd_worker_main 返回，并取得其返回值；STOPPED 标记不能代替这一步 */
407 rc = rte_eal_wait_lcore(worker->lcore_id);
408 worker->launched = false;
409 /** 等待结果是最终依据，保留失败状态供清理前的诊断读取 */
410 atomic_store_explicit(&worker->state, rc == 0 ? DPPD_WORKER_STOPPED : DPPD_WORKER_FAILED, memory_order_release);
411 if (rc != 0 && result == 0)
412 result = rc;
413 }
414 runtime->workers_started = false;
415 return result;
416}

433int dppd_runtime_stats_query(const struct dppd_runtime *runtime, uint16_t port_id, uint16_t queue_id,
434 struct dppd_stats_values *total)
435{
436 uint16_t i, j;
437 bool has_port = port_id == DPPD_STATS_ALL;
438 bool has_queue = queue_id == DPPD_STATS_ALL;
439
440 if (runtime == NULL || total == NULL)
441 return -EINVAL;
442 memset(total, 0, sizeof(*total));
443 for (j = 0; j < runtime->devices.nb_ports; ++j)
444 has_port |= runtime->devices.ports[j].port_id == port_id;
445 for (i = 0; i < runtime->nb_workers; ++i)
446 has_queue |= runtime->workers[i].queue_id == queue_id;
447 if (!has_port || !has_queue)
448 return -ENOENT;
449 for (i = 0; i < runtime->nb_workers; ++i) {
450 const struct dppd_worker *worker = &runtime->workers[i];
451
452 if (queue_id != DPPD_STATS_ALL && worker->queue_id != queue_id)
453 continue;
454 for (j = 0; j < runtime->devices.nb_ports; ++j) {
455 struct dppd_stats_values values;
456
457 if (port_id != DPPD_STATS_ALL && runtime->devices.ports[j].port_id != port_id)
458 continue;
459 dppd_stats_read(&worker->port_stats[j], &values);
460 dppd_stats_accumulate(total, &values);
461 }
462 }
463 return 0;
464}

466void dppd_runtime_stats_dump(const struct dppd_runtime *runtime)
467{
468 struct dppd_stats_values total;
469
470 dppd_runtime_stats_read(runtime, &total);
471 printf("[dppd] rx=%" PRIu64 " (%" PRIu64 " bytes) tx=%" PRIu64 " (%" PRIu64 " bytes) malformed=%" PRIu64
472 " unsupported=%" PRIu64 " policy-drop=%" PRIu64 " tx-drop=%" PRIu64 "\n",
473 total.rx_packets, total.rx_bytes, total.tx_packets, total.tx_bytes, total.rx_malformed, total.rx_unsupported,
474 total.policy_drops, total.tx_drops);
475}
476
477/**
478 * 释放 runtime 拥有的资源，成功返回 0，失败返回负错误码
479 * 确认 worker 已退出 → 关闭抓包服务 → 释放邻居等待报文和连接会话 → 关闭设备、释放 mbuf 池
480 * 软件规则表和路由由调用方清理，runtime 只通过指针使用它们
481 */
482int dppd_runtime_destroy(struct dppd_runtime *runtime)
483{
484 if (runtime == NULL)
485 return -EINVAL;
486 /** 尚有 worker 任务时先通知并等待，确认全部返回后才允许继续释放资源 */
487 if (runtime->workers_started) {
488 dppd_runtime_request_stop(runtime);
489 int waited = dppd_runtime_wait(runtime);
490 /** 未返回的线程仍借用设备和规则，拒绝清理，让正式入口走失败退出 */
491 if (runtime->workers_started)
492 return waited != 0 ? waited : -EBUSY;
493 }
494 /** 工作线程退出后，仅对本次初始化成功的 pdump 注销抓包 IPC 请求处理 */
495 if (runtime->pdump_initialized) {
496 (void)rte_pdump_uninit();
497 runtime->pdump_initialized = false;
498 }
499 if (runtime->nd6_owned)
500 dppd_nd6_destroy(runtime->nd6);
501 runtime->nd6 = NULL;
502 runtime->nd6_owned = false;
503 if (runtime->conntrack_owned)
504 dppd_conntrack_destroy(runtime->conntrack);
505 runtime->conntrack = NULL;
506 runtime->conntrack_owned = false;
507 /** 将端口关闭或回调注销错误交给进程入口，避免清理失败却报告正常退出 */
508 return dppd_devices_stop(&runtime->devices);
509}
