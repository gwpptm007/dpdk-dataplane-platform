8static uint16_t checksum(const uint8_t *header, uint16_t length)
9{
10 uint32_t sum = 0;
11 for (uint16_t i = 0; i < length; i += 2)
12 sum += (uint32_t)header[i] << 8 | header[i + 1];
13 while (sum >> 16)
14 sum = (sum & 0xffffU) + (sum >> 16);
15 return (uint16_t)~sum;
16}

18static bool valid_mac(const uint8_t address[6])
19{
20 const uint8_t zero[6] = {0};
21 return (address[0] & 1U) == 0 && memcmp(address, zero, sizeof(zero)) != 0;
22}

24/** 检查整条链，后续分段共享时也不能直接改写本报文看到的头部或借用的缓冲区 */
25static bool needs_copy(const struct rte_mbuf *mbuf)
26{
27 for (const struct rte_mbuf *segment = mbuf; segment != NULL; segment = segment->next)
28 if (!RTE_MBUF_DIRECT(segment) || RTE_MBUF_HAS_EXTBUF(segment) || rte_mbuf_refcnt_read(segment) != 1)
29 return true;
30 return false;
31}

33/** 此前 read 已核对完整前缀可读，独占链在此期间不变，写入无需合并整包或分配头部空间 */
34static void write_header(struct rte_mbuf *mbuf, const uint8_t *header, uint16_t length)
35{
36 for (struct rte_mbuf *segment = mbuf; segment != NULL && length != 0; segment = segment->next) {
37 uint16_t size = segment->data_len < length ? segment->data_len : length;
38 memcpy(rte_pktmbuf_mtod(segment, uint8_t *), header, size);
39 header += size;
40 length -= size;
41 }
42}

44int dppd_ipv4_rewrite_flow(struct rte_mbuf **mbuf, const struct dppd_packet *packet,
45 const struct dppd_ipv4_forward *forward, const uint8_t source_mac[6], const struct dppd_flow_tuple *translation)
46{
47 if (mbuf == NULL || *mbuf == NULL || packet == NULL || forward == NULL || source_mac == NULL ||
48 !valid_mac(source_mac) || !valid_mac(forward->destination_mac))
49 return -EINVAL;
50 if (packet->l3_type != DPPD_L3_IPV4 || packet->l2_len < 14 || packet->l2_len > 22 || packet->l3_len < 20 ||
51 packet->l3_len > 60 || (packet->l3_len & 3U) != 0 || packet->packet_len != rte_pktmbuf_pkt_len(*mbuf))
52 return -EINVAL;
53 if (translation != NULL) {
54 int rc = dppd_ipv4_transport_validate(*mbuf, packet);
55 if (rc != 0)
56 return rc;
57 if (translation->protocol != packet->ip_protocol)
58 return -EINVAL;
59 }
60 uint8_t header[102];
61 uint16_t length = packet->l2_len + packet->l3_len;
62 if (translation != NULL)
63 length += packet->l4_type == DPPD_L4_TCP ? 20 : 8;
64 const void *source = rte_pktmbuf_read(*mbuf, 0, length, header);
65 if (source == NULL)
66 return -EINVAL;
67 if (source != header)
68 memcpy(header, source, length);
69 uint8_t *ipv4 = header + packet->l2_len;
70 uint16_t total = (uint16_t)((uint16_t)ipv4[2] << 8 | ipv4[3]);
71 if (header[packet->l2_len - 2] != 8 || header[packet->l2_len - 1] != 0 || (ipv4[0] >> 4) != 4 ||
72 (ipv4[0] & 15U) * 4U != packet->l3_len || total < packet->l3_len ||
73 total > packet->packet_len - packet->l2_len || memcmp(ipv4 + 16, &packet->ipv4_dst_be, 4) != 0)
74 return -EINVAL;
75 if (checksum(ipv4, packet->l3_len) != 0)
76 return -EBADMSG;
77 if (ipv4[8] <= 1)
78 return -ETIMEDOUT;
79 bool ended = false;
80 for (uint16_t i = 20; i < packet->l3_len; ++i) {
81 if (ipv4[i] == 0)
82 ended = true;
83 else if (ended || ipv4[i] != 1)
84 return -ENOTSUP;
85 }
86 /* 所有校验都已通过，先在局部头部完成改写，再取得独占报文副本 */
87 memcpy(header, forward->destination_mac, 6);
88 memcpy(header + 6, source_mac, 6);
89 if (translation != NULL) {
90 uint8_t *l4 = ipv4 + packet->l3_len;
91 uint16_t checksum_offset = packet->l4_type == DPPD_L4_TCP ? 16 : 6;
92 uint16_t old_checksum = (uint16_t)((uint16_t)l4[checksum_offset] << 8 | l4[checksum_offset + 1]);
93 uint8_t old[12], updated[12];
94 memcpy(old, ipv4 + 12, 8);
95 memcpy(old + 8, l4, 4);
96 memcpy(ipv4 + 12, &translation->source_be, 4);
97 memcpy(ipv4 + 16, &translation->destination_be, 4);
98 memcpy(l4, &translation->source_port_be, 2);
99 memcpy(l4 + 2, &translation->destination_port_be, 2);
100 memcpy(updated, ipv4 + 12, 8);
101 memcpy(updated + 8, l4, 4);
102 /** 按一补数增量更新地址与端口，UDP 原校验和为零时继续保留禁用校验的语义 */
103 if (packet->l4_type == DPPD_L4_TCP || old_checksum != 0) {
104 uint32_t sum = (uint16_t)~old_checksum;
105 for (unsigned int i = 0; i < sizeof(old); i += 2) {
106 sum += (uint16_t) ~((uint16_t)old[i] << 8 | old[i + 1]);
107 sum += (uint16_t)((uint16_t)updated[i] << 8 | updated[i + 1]);
108 }
109 while (sum >> 16)
110 sum = (sum & UINT16_MAX) + (sum >> 16);
111 uint16_t value = (uint16_t)~sum;
112 if (value == 0 && packet->l4_type == DPPD_L4_UDP)
113 value = UINT16_MAX;
114 l4[checksum_offset] = (uint8_t)(value >> 8);
115 l4[checksum_offset + 1] = (uint8_t)value;
116 }
117 }
118 ipv4[8]--;
119 ipv4[10] = ipv4[11] = 0;
120 uint16_t updated = checksum(ipv4, packet->l3_len);
121 ipv4[10] = (uint8_t)(updated >> 8);
122 ipv4[11] = (uint8_t)updated;
123 struct rte_mbuf *writable = *mbuf;
124 if (needs_copy(writable)) {
125 writable = rte_pktmbuf_copy(*mbuf, (*mbuf)->pool, 0, UINT32_MAX);
126 if (writable == NULL)
127 return -ENOMEM;
128 if (needs_copy(writable)) {
129 rte_pktmbuf_free(writable);
130 return -ENOTSUP;
131 }
132 }
133 write_header(writable, header, length);
134 /* IPv4 校验和由软件写好，不能继承旧报文的发送卸载请求 */
135 writable->ol_flags &= ~RTE_MBUF_F_TX_OFFLOAD_MASK;
136 writable->tx_offload = 0;
137 writable->l2_len = packet->l2_len;
138 writable->l3_len = packet->l3_len;
139 writable->l4_len = packet->l4_len;
140 if (writable != *mbuf) {
141 rte_pktmbuf_free(*mbuf);
142 *mbuf = writable;
143 }
144 return 0;
145}
