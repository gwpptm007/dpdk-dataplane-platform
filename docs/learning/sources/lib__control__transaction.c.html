6/** 根据规划结果选择软件或硬件的操作接口，让事务流程不用分别编写两套安装逻辑 */
7static const struct dppd_transaction_backend *backend_for(const struct dppd_transaction_backends *backends,
8 enum dppd_plan_backend backend)
9{
10 switch (backend) {
11 case DPPD_PLAN_BACKEND_SOFTWARE:
12 return &backends->software;
13 case DPPD_PLAN_BACKEND_RTE_FLOW:
14 return &backends->rte_flow;
15 }
16 return NULL;
17}

19/**
20 * 校验、准备、提交和回滚是一个后端必须提供的四项能力
21 * finalize 只用于释放额外的临时资源，没有这类资源的后端可以不提供它
22 */
23static int backend_valid(const struct dppd_transaction_backend *backend)
24{
25 return backend != NULL && backend->validate != NULL && backend->prepare != NULL && backend->commit != NULL &&
26 backend->rollback != NULL;
27}

29/**
30 * 按相反顺序撤销本次事务占用的资源，将首个清理错误记入 rollback_code
31 *
32 * 软件示例：R1 已发布，R2 提交失败但未发布，R3 只完成准备
33 * R3 → 释放准备记录
34 * R2 → 释放准备记录
35 * R1 → 删除已发布版本，再释放准备记录
36 *
37 * 某一条清理失败时，继续尝试其余条目
38 */
39static void rollback_items(struct dppd_transaction *transaction, const struct dppd_transaction_backends *backends,
40 uint32_t commit_attempts)
41{
42 uint32_t i = transaction->nb_items;
43
44 /** 按最后准备或提交的对象优先撤销，避免先释放了后续对象仍可能依赖的资源 */
45 transaction->state = DPPD_TRANSACTION_ROLLING_BACK;
46 while (i > 0) {
47 struct dppd_transaction_item *item;
48 const struct dppd_transaction_backend *backend;
49 bool commit_was_attempted;
50 int rc;
51
52 --i;
53 item = &transaction->items[i];
54 /**
55 * 只经过校验的条目没有占用资源，不需要回滚
56 * 失败条目若留下非零 token，说明仍有需要清理的资源，不能仅因为 FAILED 就跳过
57 */
58 if (item->state != DPPD_TRANSACTION_ITEM_PREPARED && item->state != DPPD_TRANSACTION_ITEM_COMMITTED &&
59 item->state != DPPD_TRANSACTION_ITEM_FAILED)
60 continue;
61 if (item->state == DPPD_TRANSACTION_ITEM_FAILED && item->backend_token == 0)
62 continue;
63 backend = backend_for(backends, item->plan.backend);
64 /** commit_attempts 包含提交失败的那条，例如 R2 失败时为 2，R1、R2 均算尝试过提交 */
65 commit_was_attempted = i < commit_attempts;
66 rc = backend->rollback(backend->context, item, item->backend_token, commit_was_attempted);
67 if (rc == 0)
68 item->state = DPPD_TRANSACTION_ITEM_ROLLED_BACK;
69 /** 保留第一个清理错误，同时继续尝试清理其余条目，便于上层判断是否必须隔离 */
70 else if (transaction->rollback_code == 0) {
71 transaction->rollback_code = rc;
72 transaction->rollback_item = i;
73 }
74 }
75 transaction->state = transaction->rollback_code == 0 ? DPPD_TRANSACTION_ROLLED_BACK : DPPD_TRANSACTION_FAILED;
76}

78/**
79 * 初始化规则安装事务：记录编号、关联条目，重置状态和错误记录
80 *
81 * transaction（NEW）
82 * └─ items → 调用方的条目（PENDING，保留 rule 和 plan）
83 *
84 * items 的内存由调用方保留到事务结束
85 * 成功返回 0，参数无效返回 -EINVAL
86 */
87int dppd_transaction_init(struct dppd_transaction *transaction, uint64_t transaction_id,
88 struct dppd_transaction_item *items, uint32_t nb_items)
89{
90 uint32_t i;
91
92 if (transaction == NULL || transaction_id == 0 || items == NULL || nb_items == 0)
93 return -EINVAL;
94 transaction->id = transaction_id;
95 transaction->state = DPPD_TRANSACTION_NEW;
96 transaction->items = items;
97 transaction->nb_items = nb_items;
98 transaction->failure_code = 0;
99 transaction->rollback_code = 0;
100 transaction->failure_stage = DPPD_RULE_STAGE_NONE;
101 transaction->failure_item = UINT32_MAX;
102 transaction->rollback_item = UINT32_MAX;
103 /** 清零 token 不会释放后端资源，复用条目前需先完成旧事务的回滚或收尾 */
104 for (i = 0; i < nb_items; ++i) {
105 items[i].state = DPPD_TRANSACTION_ITEM_PENDING;
106 items[i].backend_token = 0;
107 items[i].error_code = 0;
108 }
109 return 0;
110}

112/**
113 * 按条目的执行计划安装规则，失败时撤销已占用的资源
114 *
115 * 软件后端示例：dppd_transaction_run
116 * ├─ backend->validate(...) → software_backend.c：transaction_validate
117 * │ 检查软件后端能否处理这条规则
118 * └─ backend->prepare(...) → software_backend.c：transaction_prepare
119 * 分配临时记录，保存规则，把记录地址交回事务
120 *
121 * 全部 validate → 全部 prepare → 逐条 commit → COMMITTED
122 * 校验失败 → 直接返回错误
123 * 准备或提交失败 → 逆序回滚，再返回原始错误
124 *
125 * 返回 0 表示后端安装完成，控制层随后更新规则记录、收尾并保存配置
126 * 失败时另查 rollback_code：0 表示清理成功，非 0 表示仍有清理错误
127 */
128int dppd_transaction_run(struct dppd_transaction *transaction, const struct dppd_transaction_backends *backends)
129{
130 uint32_t i;
131 int rc;
132
133 if (transaction == NULL || backends == NULL || transaction->state != DPPD_TRANSACTION_NEW)
134 return -EINVAL;
135
136 transaction->state = DPPD_TRANSACTION_VALIDATING;
137 /**
138 * 校验后端能力，以及规则和计划的 ID、版本是否一致
139 * 此阶段不占用后端资源，失败时无需回滚
140 */
141 for (i = 0; i < transaction->nb_items; ++i) {
142 struct dppd_transaction_item *item = &transaction->items[i];
143 /** 本例 plan.backend = SOFTWARE → backends.software → software_backend.c 中的各阶段函数 */
144 const struct dppd_transaction_backend *backend = backend_for(backends, item->plan.backend);
145
146 if (item->rule.id == 0 || item->plan.rule_id != item->rule.id ||
147 item->plan.rule_generation != item->rule.generation)
148 rc = -EINVAL;
149 else if (!backend_valid(backend))
150 rc = -ENOTSUP;
151 else
152 // 从事务调度进入软件后端
153 // 通过函数指针，调用的是software_backend.c 中的 transaction_validate
154 rc = backend->validate(backend->context, item);
155 if (rc != 0) {
156 item->state = DPPD_TRANSACTION_ITEM_FAILED;
157 item->error_code = rc;
158 transaction->failure_code = rc;
159 transaction->failure_stage = DPPD_RULE_STAGE_VALIDATE;
160 transaction->failure_item = i;
161 transaction->state = DPPD_TRANSACTION_FAILED;
162 return rc;
163 }
164 item->state = DPPD_TRANSACTION_ITEM_VALIDATED;
165 }
166
167 transaction->state = DPPD_TRANSACTION_PREPARING;
168 /**
169 * 为所有条目准备资源，任一条失败都要归还此前已准备的资源
170 * 尚未调用 commit，因此回滚时传入的提交次数为 0
171 */
172 for (i = 0; i < transaction->nb_items; ++i) {
173 struct dppd_transaction_item *item = &transaction->items[i];
174 const struct dppd_transaction_backend *backend = backend_for(backends, item->plan.backend);
175
176 /**
177 * 软件后端的函数指针从初始化到调用：
178 * control.c：backends.software = dppd_software_transaction_backend(&service->software)
179 * → 接口保存 context = &service->software、prepare = transaction_prepare
180 * → backend_for 返回 &backends->software
181 * → 下面的调用进入 software_backend.c 的 transaction_prepare
182 *
183 * backend 指向操作接口，backend->context 指向实际的软件后端对象
184 * item 原样传入，&item->backend_token 让回调把资源凭据写回事务条目
185 * 选择硬件后端时，同一调用语句使用 rte_flow 接口中保存的函数地址
186 */
187 rc = backend->prepare(backend->context, item, &item->backend_token);
188 if (rc != 0) {
189 item->state = DPPD_TRANSACTION_ITEM_FAILED;
190 item->error_code = rc;
191 transaction->failure_code = rc;
192 transaction->failure_stage = DPPD_RULE_STAGE_PREPARE;
193 transaction->failure_item = i;
194 rollback_items(transaction, backends, 0);
195 return rc;
196 }
197 item->state = DPPD_TRANSACTION_ITEM_PREPARED;
198 }
199
200 transaction->state = DPPD_TRANSACTION_COMMITTING;
201 /**
202 * 使用 prepare 返回的 token 安装规则，提交失败的条目也可能需要回滚
203 * 软件后端每提交一条就发布一次快照，worker 可能先看到本批中的部分规则
204 */
205 for (i = 0; i < transaction->nb_items; ++i) {
206 struct dppd_transaction_item *item = &transaction->items[i];
207 const struct dppd_transaction_backend *backend = backend_for(backends, item->plan.backend);
208 // 返回事务函数，标记安装完成
209 rc = backend->commit(backend->context, item, item->backend_token);
210 if (rc != 0) {
211 item->state = DPPD_TRANSACTION_ITEM_FAILED;
212 item->error_code = rc;
213 transaction->failure_code = rc;
214 transaction->failure_stage = DPPD_RULE_STAGE_COMMIT;
215 transaction->failure_item = i;
216 rollback_items(transaction, backends, i + 1U);
217 return rc;
218 }
219 item->state = DPPD_TRANSACTION_ITEM_COMMITTED;
220 }
221
222 transaction->state = DPPD_TRANSACTION_COMMITTED;
223 return 0;
224}

226/**
227 * 撤销已提交的事务，例如新版本装好后，删除旧版本失败时撤回新版本
228 * 只接受 COMMITTED 状态；返回 0 表示全部撤销成功，负值表示清理失败
229 */
230int dppd_transaction_rollback_committed(struct dppd_transaction *transaction,
231 const struct dppd_transaction_backends *backends)
232{
233 if (transaction == NULL || backends == NULL || transaction->state != DPPD_TRANSACTION_COMMITTED)
234 return -EINVAL;
235 rollback_items(transaction, backends, transaction->nb_items);
236 return transaction->rollback_code;
237}

239/**
240 * 释放已提交事务的临时资源，保留已安装的规则
241 *
242 * 后端 finalize → 清零各条目的 token → 状态改为 FINALIZED
243 *
244 * 控制层确认不再回滚后调用；返回 0 表示收尾成功，负值表示状态或后端异常
245 */
246int dppd_transaction_finalize(struct dppd_transaction *transaction, const struct dppd_transaction_backends *backends)
247{
248 uint32_t i;
249
250 if (transaction == NULL || backends == NULL || transaction->state != DPPD_TRANSACTION_COMMITTED)
251 return -EINVAL;
252 for (i = 0; i < transaction->nb_items; ++i) {
253 struct dppd_transaction_item *item = &transaction->items[i];
254 const struct dppd_transaction_backend *backend = backend_for(backends, item->plan.backend);
255
256 if (item->state != DPPD_TRANSACTION_ITEM_COMMITTED || !backend_valid(backend))
257 return -EUCLEAN;
258 /** finalize 回调只释放临时资源，不返回错误；未提供回调时也要清零 token */
259 if (backend->finalize != NULL)
260 backend->finalize(backend->context, item, item->backend_token);
261 item->backend_token = 0;
262 }
263 transaction->state = DPPD_TRANSACTION_FINALIZED;
264 return 0;
