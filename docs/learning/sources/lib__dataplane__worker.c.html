14{
15 uint16_t i;
16
17 for (i = 0; i < count; ++i)

21/** 每次准备调用驱动前都重新检查，事件不能等到整轮端口扫描结束后才阻止下一次收发 */
22static bool worker_io_quiesced(const struct dppd_worker *worker)
23{
24 return atomic_load_explicit(&worker->runtime->stop_requested, memory_order_acquire) ||
25 dppd_devices_io_quiesced(&worker->runtime->devices);
26}

28/**
29 * 向指定出口的当前 worker 队列批量提交报文，count 是本出口实际待发送数量
30 * 提交前再次检查停止状态和出口可用性，失效则释放整批并计入入口丢弃
31 * 驱动接收的报文移交驱动管理，未接收的尾部报文由 worker 释放，不在此重试
32 * 成功提交和队列丢弃计入出口统计
33 */
34static void flush_egress(struct dppd_worker *worker, const struct dppd_port *egress, struct rte_mbuf **packets,
35 const uint32_t *lengths, uint16_t count, struct dppd_stats_values *ingress, struct dppd_stats_values *output)
36{
37 if (count == 0)
38 return;
39 if (worker_io_quiesced(worker) || !dppd_port_tx_available(egress)) {
40 ingress->policy_drops += count;
41 ingress->egress_drops += count;
42 free_packets(packets, count);
43 return;
44 }
45 /**
46 * sent 表示驱动接收了待发送数组中的前多少个包，不表示对端已经收到
47 * packets 只包含本出口的待发送包，之前已丢弃的包不在其中，不能按原始 RX 数组编号理解 sent
48 * 例：count 为 3、sent 为 2 → packets[0] 和 packets[1] 交给驱动，packets[2] 仍归 worker
49 */
50 uint16_t sent = rte_eth_tx_burst(egress->port_id, worker->queue_id, packets, count);
51 output->tx_packets += sent;
52 /** 使用入队前保存的 lengths 统计，避免访问已移交驱动的 mbuf */
53 for (uint16_t i = 0; i < sent; ++i)
54 output->tx_bytes += lengths[i];
55 /**
56 * packets + sent 指向第一个未被接收的包，只释放剩余 count - sent 个，不在这里重试
57 * 例：3 个包只接收 2 个 → 从 &packets[2] 开始释放 1 个，前两个不能再次访问或释放
58 */
59 if (sent < count) {
60 output->tx_drops += count - sent;
61 output->tx_queue_drops += count - sent;
62 free_packets(packets + sent, (uint16_t)(count - sent));
63 }
64}

66/** 每个工作线程只写自己的 TX 队列，ND 不借主线程并发写同一个硬件队列 */
67static int send_nd6(void *context, uint16_t port_id, const uint8_t *wire, uint16_t length)
68{
69 struct dppd_worker *worker = context;
70 const struct dppd_port *port = dppd_devices_find(&worker->runtime->devices, port_id);
71 if (port == NULL || !dppd_port_tx_available(port) || worker_io_quiesced(worker))
72 return -ENODEV;
73 struct rte_mbuf *mbuf = rte_pktmbuf_alloc(worker->runtime->devices.pools[port->socket_id]);
74 if (mbuf == NULL)
75 return -ENOMEM;
76 void *bytes = rte_pktmbuf_append(mbuf, length);
77 if (bytes == NULL) {
78 rte_pktmbuf_free(mbuf);
79 return -ENOMEM;
80 }
81 memcpy(bytes, wire, length);
82 /** 分配与复制期间也可能收到事件，尚未交给驱动的报文仍由本线程归还 */
83 if (worker_io_quiesced(worker) || !dppd_port_tx_available(port)) {
84 rte_pktmbuf_free(mbuf);
85 return -ENODEV;
86 }
87 if (rte_eth_tx_burst(port_id, worker->queue_id, &mbuf, 1) != 1) {
88 rte_pktmbuf_free(mbuf);
89 return -EAGAIN;
90 }
91 struct dppd_stats_values output = {.tx_packets = 1, .tx_bytes = length, .nd6_tx_packets = 1};
92 dppd_stats_add(&worker->stats, &output);
93 dppd_stats_add(&worker->port_stats[port - worker->runtime->devices.ports], &output);
94 return 0;
95}

97/**
98 * 执行软件规则与出口选择，将动作、原因和出口写入 decision
99 * IPv6 缺静态邻居且已启用 ND 时，用出口和下一跳查询动态邻居表，补齐 MAC 或标记等待
100 * 转发前检查出口 IPv6 地址是否就绪，将决定交回 process_batch 或 stateful_decide
101 */
102static void routed_decide(struct dppd_runtime *runtime, uint16_t ingress, int parsed, const struct dppd_packet *packet,
103 struct dppd_pipeline_decision *decision, struct dppd_software_count_batch *counts)
104{
105 dppd_pipeline_decide_counted(&runtime->snapshot, runtime->software_backend, runtime->ipv4_router, ingress, parsed,
106 packet, decision, counts);
107 if (runtime->nd6 == NULL || packet->l3_type != DPPD_L3_IPV6)
108 return;
109 /**
110 * 静态邻居未命中时，pipeline 返回 DROP / NO_NEIGHBOR，但已留下出口和下一跳
111 * 上方检查保证此处是 IPv6 且已启用 ND（IPv6 邻居发现）
112 *
113 * 沿用示例：
114 * decision->ipv4.port_id = 1，next_hop6 = fe80::1，destination_mac 仍为全 0
115 * 用网口 1 和 fe80::1 查询 ND 动态邻居表
116 * 若找到可用 MAC，dppd_nd6_resolve 直接写入 decision->ipv4.destination_mac
117 * 静态表没有邻居，并不代表动态表也没有，因此这里还需再查一次
118 */
119 if (decision->drop_reason == DPPD_DROP_NO_NEIGHBOR) {
120 int rc = dppd_nd6_resolve(runtime->nd6, decision->ipv4.port_id, decision->ipv4.next_hop6, packet,
121 dppd_monotonic_ms(), decision->ipv4.destination_mac);
122
123 /**
124 * rc = 0 → MAC 已写入，改为 FORWARD，填写出口并设置 routed
125 * rc = -EAGAIN → MAC 尚未取得，改为 ND_PENDING，让 process_batch 尝试暂存报文
126 * 其他错误 → 保持 DROP / NO_NEIGHBOR，交回调用方处理失败
127 * ND_PENDING 时 action 仍是 DROP，但 process_batch 会先处理等待入队，再判断丢弃
128 */
129 if (rc == 0) {
130 decision->action = DPPD_PACKET_FORWARD;
131 decision->drop_reason = DPPD_DROP_NONE;
132 decision->egress_port = decision->ipv4.port_id;
133 decision->routed = true;
134 } else if (rc == -EAGAIN)
135 decision->drop_reason = DPPD_DROP_ND_PENDING;
136 }
137 /** 取得下一跳 MAC 后，仍须确认出口 IPv6 地址已就绪，否则改为 DROP / ND_NOT_READY */
138 if (decision->action == DPPD_PACKET_FORWARD && !dppd_nd6_port_ready(runtime->nd6, decision->egress_port)) {
139 decision->action = DPPD_PACKET_DROP;
140 decision->drop_reason = DPPD_DROP_ND_NOT_READY;
141 }
142}

144static enum dppd_drop_reason stateful_error(int error)
145{
146 switch (error) {
147 case -ENOSPC:
148 return DPPD_DROP_CT_FULL;
149 case -EADDRNOTAVAIL:
150 return DPPD_DROP_NAT_POOL;
151 case -ENOENT:
152 return DPPD_DROP_NAT_MISS;
153 case -EXDEV:
154 return DPPD_DROP_CT_ZONE;
155 case -ENOTSUP:
156 return DPPD_DROP_CT_UNSUPPORTED;
157 case -EBADMSG:
158 return DPPD_DROP_CT_CHECKSUM;
159 case -ETIMEDOUT:
160 return DPPD_DROP_TTL;
161 case -ENOMEM:
162 return DPPD_DROP_REWRITE;
163 default:
164 return DPPD_DROP_CT_INVALID;
165 }
166}

168/**
169 * 结合连接状态生成处理决定，完成报文校验、会话/NAT 准备及策略匹配
170 * 流程：准备候选会话 → 构造 policy → routed_decide → 改写报文 → 提交或取消会话
171 * policy 使用会话得到的逻辑地址和端口匹配规则，packet 保留原包字段供改写校验
172 * decision 输出动作与出口，mbuf 可能被改写函数替换，实际发送与丢弃仍由 process_batch 执行
173 * 放行且改写成功才提交会话，失败取消候选并归还临时 NAT 端口，不刷新会话超时
174 */
175static void stateful_decide(struct dppd_worker *worker, const struct dppd_port *ingress, struct rte_mbuf **mbuf,
176 int parsed, const struct dppd_packet *packet, struct dppd_pipeline_decision *decision,
177 struct dppd_software_count_batch *counts)
178{
179 struct dppd_runtime *runtime = worker->runtime;
180 struct dppd_connection_attempt attempt = {0};
181 struct dppd_packet policy = *packet;
182 int rc = 0;
183 /** 先校验 TCP/UDP 报文，再准备候选会话，并将连接状态交给后面的策略匹配 */
184 if (parsed != DPPD_PARSE_MALFORMED) {
185 if (packet->l4_type == DPPD_L4_TCP || packet->l4_type == DPPD_L4_UDP)
186 rc = packet->l3_type == DPPD_L3_IPV6 ? dppd_ipv6_transport_validate(*mbuf, packet)
187 : dppd_ipv4_transport_validate(*mbuf, packet);
188 if (rc == 0)
189 rc = dppd_conntrack_begin(runtime->conntrack, ingress->port_id, packet, dppd_monotonic_ms(), &attempt);
190 policy.connection_state = rc == 0 ? attempt.state : DPPD_CT_INVALID;
191 if (rc == 0) {
192 memcpy(policy.ipv6_src, attempt.policy.source6, 16);
193 memcpy(policy.ipv6_dst, attempt.policy.destination6, 16);
194 policy.ipv4_src_be = attempt.policy.source_be;
195 policy.ipv4_dst_be = attempt.policy.destination_be;
196 policy.l4_src_port_be = attempt.policy.source_port_be;
197 policy.l4_dst_port_be = attempt.policy.destination_port_be;
198 }
199 }
200 if (parsed != DPPD_PARSE_MALFORMED && rc != 0) {
201 /** INVALID 规则仍可记录拒绝计数，但允许动作不能把校验错误或无会话回包放行 */
202 struct dppd_software_decision policy_result = {0};
203 dppd_software_backend_decide_counted(runtime->software_backend, ingress->port_id, &policy, &policy_result,
204 counts);
205 memset(decision, 0, sizeof(*decision));
206 decision->drop_reason = policy_result.drop ? DPPD_DROP_POLICY : stateful_error(rc);
207 dppd_conntrack_finish(&attempt, 0, false);
208 return;
209 }
210 /** 用带连接状态和逻辑地址的 policy，复用通用软件规则与出口选择 */
211 routed_decide(runtime, ingress->port_id, parsed, &policy, decision, counts);
212 if (decision->action != DPPD_PACKET_FORWARD) {
213 dppd_conntrack_finish(&attempt, 0, false);
214 return;
215 }
216 if (rc == 0) {
217 const struct dppd_port *egress = dppd_devices_find(&runtime->devices, decision->egress_port);
218 if (egress == NULL || !dppd_port_tx_available(egress)) {
219 decision->action = DPPD_PACKET_DROP;
220 decision->drop_reason = DPPD_DROP_EGRESS;
221 dppd_conntrack_finish(&attempt, 0, false);
222 return;
223 }
224 /** 出口可用后改写实际报文，调整 MAC、TTL/HopLimit，并按需应用 NAT 和更新校验和 */
225 rc = packet->l3_type == DPPD_L3_IPV6 ? dppd_ipv6_rewrite_flow(mbuf, packet, &decision->ipv4,
226 egress->mac.addr_bytes, attempt.nat ? &attempt.rewrite : NULL)
227 : dppd_ipv4_rewrite_flow(mbuf, packet, &decision->ipv4,
228 egress->mac.addr_bytes, attempt.nat ? &attempt.rewrite : NULL);
229 }
230 /** finish 提交或取消候选并释放会话表锁，此时尚未调用网卡发送 */
231 int commit = dppd_conntrack_finish(&attempt, decision->egress_port, rc == 0);
232 if (rc == 0)
233 rc = commit;
234 if (rc != 0) {
235 decision->action = DPPD_PACKET_DROP;
236 decision->drop_reason = stateful_error(rc);
237 }
238}

240/**
241 * 处理同一入口的一批报文，逐包决定去向，再按出口批量发送并提交统计
242 *
243 * worker：当前处理线程，提供运行配置、收发队列编号和统计对象
244 * ingress：这批报文的原入口网口，ND 重试也沿用原入口
245 * rx / received：待处理的 mbuf 指针数组及实际包数，received 不超过 DPPD_MAX_BURST
246 * retry：普通收包传 false，ND 等待包取出重试传 true，避免重复累计 RX、隧道观察和规则 COUNT
247 * expires：ND 等待的毫秒截止时间，首次传 0，由 hold 设置期限；重试沿用原期限，不延长等待
248 *
249 * 处理路径：
250 * 逐包解析 → 处理本机 ND 控制报文、检查 IPv6 入口地址是否就绪
251 * 未启用连接跟踪 → routed_decide 匹配软件规则，选择路由或配对出口
252 * 启用连接跟踪 → stateful_decide 处理会话，并完成决策及报文改写
253 * 等待下一跳 MAC → 尝试交给 ND 等待队列；拒绝或处理失败 → 释放当前包
254 * 允许转发 → 检查出口，按需完成普通路由改写、MARK 和多段报文合并，加入对应出口批次
255 * 逐包处理结束 → 提交规则 COUNT → flush_egress 提交发送 → 累加入口、出口和 worker 统计
256 *
257 * 普通三层转发修改源/目的 MAC、递减 TTL 或 Hop Limit，IPv4 还更新头部校验和
258 * 连接跟踪路径已完成改写时不重复执行；NAT 按配置另行改写 IP、端口和相关校验和
259 *
260 * 本函数接管 rx 中的报文，改写时可能替换 rx[i]，返回后调用方不能再访问或释放这些 mbuf
261 * ND 入队成功：等待队列持有；TX 提交成功：驱动持有；其余已处理报文：本线程释放
262 * 无返回值；received 为 0 时直接结束，停止收发时释放整批，不等待凑满或重试 TX 未接收的包
263 * 函数返回前提交本批 COUNT，QSBR 静默点由 dppd_worker_main 在一轮端口扫描结束后报告
264 */
265static void process_batch(struct dppd_worker *worker, const struct dppd_port *ingress, struct rte_mbuf **rx,
266 uint16_t received, bool retry, uint64_t expires)
267{
268 struct dppd_runtime *runtime = worker->runtime;
269 /** 按出口暂存待发送的 mbuf 指针、长度和数量，逐包处理结束后统一批量发送 */
270 struct rte_mbuf *tx[DPPD_MAX_PORTS][DPPD_MAX_BURST];
271 uint32_t tx_lengths[DPPD_MAX_PORTS][DPPD_MAX_BURST];
272 uint16_t counts[DPPD_MAX_PORTS] = {0};
273 struct dppd_stats_values delta = {0}, outputs[DPPD_MAX_PORTS] = {0};
274 struct dppd_software_count_batch rule_counts = {0};
275 if (worker_io_quiesced(worker)) {
276 /** RX 返回的报文已归线程所有，事件不能让这批 mbuf 因提前返回而泄漏 */
277 free_packets(rx, received);
278 return;
279 }
280 if (received == 0)
281 return;
282 delta.rx_packets = retry ? 0 : received;
283 const struct dppd_port *paired =
284 runtime->ipv4_router == NULL ? dppd_devices_find(&runtime->devices, ingress->peer_port_id) : NULL;
285 for (uint16_t i = 0; i < received; ++i) {
286 /** packet 保存解析字段，decision 保存处理决定，rx[i] 指向原始报文 */
287 struct dppd_pipeline_decision decision;
288 struct dppd_packet packet = {0};
289 uint32_t length = rte_pktmbuf_pkt_len(rx[i]);
290 if (!retry)
291 delta.rx_bytes += length;
292 /** 逐包进入 packet_mbuf.c 提取报文头字段，burst 只批量收发，不省略逐包解析 */
293 int parsed = dppd_packet_parse_mbuf(rx[i], &packet);
294 packet.neighbor_retry = retry;
295 /** 隧道统计是外层中转观察，既不替换 ACL 字段，也不因内层错误改变已有放行结果 */
296 if (!retry) {
297 delta.tunnel_vxlan_packets += packet.tunnel.type == DPPD_TUNNEL_VXLAN;
298 delta.tunnel_geneve_packets += packet.tunnel.type == DPPD_TUNNEL_GENEVE;
299 delta.tunnel_gre_packets += packet.tunnel.type == DPPD_TUNNEL_GRE;
300 delta.tunnel_malformed += packet.tunnel.status == DPPD_PARSE_MALFORMED;
301 delta.tunnel_unsupported += packet.tunnel.status == DPPD_PARSE_UNSUPPORTED;
302 }
303 if (parsed == DPPD_PARSE_MALFORMED)
304 delta.rx_malformed++;
305 else if (parsed == DPPD_PARSE_UNSUPPORTED)
306 delta.rx_unsupported++;
307 if (runtime->nd6 != NULL && parsed != DPPD_PARSE_MALFORMED &&
308 dppd_nd6_input(runtime->nd6, ingress->port_id, rx[i], &packet, dppd_monotonic_ms(), send_nd6, worker)) {
309 delta.nd6_rx_packets++;
310 rte_pktmbuf_free(rx[i]);
311 continue;
312 }
313 if (packet.l3_type == DPPD_L3_IPV6 && !dppd_nd6_port_ready(runtime->nd6, ingress->port_id)) {
314 delta.nd6_not_ready_drops++;
315 delta.policy_drops++;
316 rte_pktmbuf_free(rx[i]);
317 continue;
318 }
319 /**
320 * 未启用连接跟踪
321 * → routed_decide → decision
322 *
323 * 启用连接跟踪
324 * → stateful_decide
325 * 校验、准备候选会话
326 * → routed_decide
327 * → 检查出口、改写实际报文
328 * → 成功提交会话，失败取消候选
329 */
330 if (runtime->conntrack != NULL)
331 stateful_decide(worker, ingress, &rx[i], parsed, &packet, &decision, &rule_counts);
332 else
333 routed_decide(runtime, ingress->port_id, parsed, &packet, &decision, &rule_counts);
334
335 /**
336 * ND_PENDING 表示下一跳 MAC 尚未取得，先尝试把当前 mbuf 交给 ND 等待队列
337 * 入队成功 → 队列接管 mbuf，continue 处理下一个包，此处不能释放或发送它
338 * 入队失败 → mbuf 仍归当前线程，继续下面的丢弃分支并释放
339 */
340 if (decision.drop_reason == DPPD_DROP_ND_PENDING) {
341 struct dppd_nd6_pending pending = {.mbuf = rx[i],
342 .ingress = ingress->port_id,
343 .queue = worker->queue_id,
344 .expires_ms = expires};
345 if (dppd_nd6_hold(runtime->nd6, decision.ipv4.port_id, decision.ipv4.next_hop6, pending,
346 dppd_monotonic_ms()) == 0) {
347 delta.nd6_queued++;
348 continue;
349 }
350 delta.nd6_queue_drops++;
351 }
352 /**
353 * 决定函数只填写结果，worker 在这里执行丢弃并按原因更新统计
354 * 释放当前报文后继续处理本批下一个包，不影响其他报文
355 */
356 if (decision.action != DPPD_PACKET_FORWARD) {
357 if (decision.drop_reason != DPPD_DROP_MALFORMED)
358 delta.policy_drops++;
359 if (decision.drop_reason == DPPD_DROP_ND_NOT_READY)
360 delta.nd6_not_ready_drops++;
361 else if (decision.drop_reason == DPPD_DROP_POLICY)
362 delta.rule_drops++;
363 else if (decision.drop_reason == DPPD_DROP_NO_ROUTE)
364 delta.no_route_drops++;
365 else if (decision.drop_reason == DPPD_DROP_NO_NEIGHBOR)
366 delta.no_neighbor_drops++;
367 else if (decision.drop_reason == DPPD_DROP_IPV4_UNSUPPORTED)
368 delta.ipv4_unsupported_drops++;
369 else if (decision.drop_reason == DPPD_DROP_IPV6_UNSUPPORTED)
370 delta.ipv6_unsupported_drops++;
371 else if (decision.drop_reason == DPPD_DROP_CT_INVALID)
372 delta.ct_invalid_drops++;
373 else if (decision.drop_reason == DPPD_DROP_CT_FULL)
374 delta.ct_full_drops++;
375 else if (decision.drop_reason == DPPD_DROP_CT_UNSUPPORTED)
376 delta.ct_unsupported_drops++;
377 else if (decision.drop_reason == DPPD_DROP_CT_ZONE)
378 delta.ct_zone_drops++;
379 else if (decision.drop_reason == DPPD_DROP_CT_CHECKSUM)
380 delta.ct_checksum_drops++;
381 else if (decision.drop_reason == DPPD_DROP_NAT_POOL)
382 delta.nat_pool_drops++;
383 else if (decision.drop_reason == DPPD_DROP_NAT_MISS)
384 delta.nat_miss_drops++;
385 else if (decision.drop_reason == DPPD_DROP_TTL) {
386 if (packet.l3_type == DPPD_L3_IPV6)
387 delta.ipv6_hop_drops++;
388 else
389 delta.ipv4_ttl_drops++;
390 } else if (decision.drop_reason == DPPD_DROP_REWRITE) {
391 if (packet.l3_type == DPPD_L3_IPV6)
392 delta.ipv6_rewrite_drops++;
393 else
394 delta.ipv4_rewrite_drops++;
395 } else if (decision.drop_reason == DPPD_DROP_EGRESS)
396 delta.egress_drops++;
397 /** 归还当前 mbuf，再进入逐包循环的下一轮 */
398 rte_pktmbuf_free(rx[i]);
399 continue;
400 }
401 /**
402 * 根据处理决定取得实际出口对象，三层模式按出口编号查找，端口配对模式复用 paired
403 * FORWARD 只表示策略允许，发送前仍须确认出口存在、编号一致且可发送
404 */
405 const struct dppd_port *egress =
406 runtime->ipv4_router != NULL ? dppd_devices_find(&runtime->devices, decision.egress_port) : paired;
407 if (egress == NULL || egress->port_id != decision.egress_port || !dppd_port_tx_available(egress)) {
408 delta.policy_drops++;
409 delta.egress_drops++;
410 rte_pktmbuf_free(rx[i]);
411 continue;
412 }
413 /**
414 * 三层转发在这里改写报文头，启用连接跟踪时已由 stateful_decide 改写，不能再次递减跳数
415 * IPv6 普通转发：目的 MAC 改为下一跳 MAC，源 MAC 改为出口 MAC，Hop Limit 减 1
416 *
417 * &rx[i] 允许改写函数在需要复制报文时替换 mbuf 指针，后续发送使用替换后的指针
418 * &packet 提供已解析的头部信息，decision.ipv4 是 IPv4/IPv6 共用的转发结果
419 * 改写失败则释放当前包；成功后继续加入出口发送批次，此处尚未交给网卡
420 */
421 if (decision.routed && runtime->conntrack == NULL) {
422 bool ipv6 = packet.l3_type == DPPD_L3_IPV6;
423 int rc = ipv6 ? dppd_ipv6_rewrite(&rx[i], &packet, &decision.ipv4, egress->mac.addr_bytes)
424 : dppd_ipv4_rewrite(&rx[i], &packet, &decision.ipv4, egress->mac.addr_bytes);
425 if (rc != 0) {
426 delta.policy_drops++;
427 if (ipv6) {
428 if (rc == -ETIMEDOUT)
429 delta.ipv6_hop_drops++;
430 else if (rc == -EBADMSG)
431 delta.ipv6_checksum_drops++;
432 else if (rc == -ENOTSUP)
433 delta.ipv6_unsupported_drops++;
434 else
435 delta.ipv6_rewrite_drops++;
436 } else if (rc == -ETIMEDOUT)
437 delta.ipv4_ttl_drops++;
438 else if (rc == -EBADMSG)
439 delta.ipv4_checksum_drops++;
440 else if (rc == -ENOTSUP)
441 delta.ipv4_unsupported_drops++;
442 else
443 delta.ipv4_rewrite_drops++;
444 rte_pktmbuf_free(rx[i]);
445 continue;
446 }
447 }
448 /** 将规则 MARK 写入 mbuf 的附加信息并设置有效标志，不向报文字节中插入标记 */
449 if (decision.has_mark) {
450 rx[i]->hash.fdir.hi = decision.mark_id;
451 rx[i]->ol_flags |= RTE_MBUF_F_RX_FDIR_ID;
452 }
453 /**
454 * 以出口在 devices.ports 中的数组下标分组，指针相减得到元素间距
455 * 例：ports[0].port_id = 1、ports[1].port_id = 5，出口为网口 5 时 index 为 1
456 * tx[index]、tx_lengths[index]、counts[index] 和 outputs[index] 都属于这个出口
457 */
458 uint16_t index = (uint16_t)(egress - runtime->devices.ports);
459 /**
460 * 多段报文且出口未启用多段发送时，尝试将全部数据合并到第一段
461 * 第一段空间不足等原因导致合并失败时，统计并释放当前包，继续处理下一个
462 */
463 if (rx[i]->nb_segs > 1U && (egress->configured_tx_offloads & RTE_ETH_TX_OFFLOAD_MULTI_SEGS) == 0 &&
464 rte_pktmbuf_linearize(rx[i]) != 0) {
465 outputs[index].tx_drops++;
466 outputs[index].tx_linearize_drops++;
467 rte_pktmbuf_free(rx[i]);
468 continue;
469 }
470 /**
471 * 将当前包追加到对应出口的待发送批次，只保存指针，此处尚未交给驱动
472 * 提前保存长度供发送统计使用，成功提交后不再读取已转移所有权的 mbuf
473 * 例：counts[index] 为 2，长度和指针都写入下标 2，然后 counts[index] 增为 3
474 * counts 只统计本次 process_batch 收集的包，本批处理结束就发送，不等待凑满
475 */
476 tx_lengths[index][counts[index]] = length;
477 tx[index][counts[index]++] = rx[i];
478 }
479 /** 接收批次内所有路径都会汇合到这里，先提交 COUNT 再发送或报告 QSBR 静默 */
480 dppd_software_count_batch_flush(&rule_counts);
481 /** 逐包处理已完成，进入本文件的 flush_egress，按出口提交各自的待发送批次 */
482 for (uint16_t index = 0; index < runtime->devices.nb_ports; ++index)
483 flush_egress(worker, &runtime->devices.ports[index], tx[index], tx_lengths[index], counts[index], &delta,
484 &outputs[index]);
485 /**
486 * 此时 delta 保存本批入口接收和处理中的丢弃统计，outputs 按出口保存发送及发送失败统计
487 * 先将 delta 记入入口网口，再将各 outputs 记入对应出口，并合并到 delta 供 worker 汇总
488 * 例：入口 0 收到 3 包，均送往出口 1，其中 2 包提交成功、1 包因发送队列未接收而丢弃
489 * 入口 0：RX 加 3；出口 1：TX 加 2、tx_queue_drops 加 1；worker 汇总包含这三项
490 */
491 dppd_stats_add(&worker->port_stats[ingress - runtime->devices.ports], &delta);
492 for (uint16_t index = 0; index < runtime->devices.nb_ports; ++index) {
493 dppd_stats_add(&worker->port_stats[index], &outputs[index]);
494 dppd_stats_accumulate(&delta, &outputs[index]);
495 }
496 /** 每批只更新一次工作线程汇总，避免入口丢弃已累计而出口字节还没累计的额外窗口 */
497 dppd_stats_add(&worker->stats, &delta);
498}

500/**
501 * 从入口端口的当前 worker 队列收包，再交给本文件的 process_batch 处理
502 */
503static void process_ingress(struct dppd_worker *worker, const struct dppd_port *ingress)
504{
505 if (worker_io_quiesced(worker) || atomic_load_explicit(&ingress->removed, memory_order_acquire))
506 return;
507 struct rte_mbuf *packets[DPPD_MAX_BURST];
508 /** 最多接收 burst_size 个包，不等待凑满 */
509 uint16_t count = rte_eth_rx_burst(ingress->port_id, worker->queue_id, packets, worker->runtime->config.burst_size);
510 process_batch(worker, ingress, packets, count, false, 0);
511}
512
513/**
514 * 从 ND 等待数组取回本队列的报文，每次最多处理 DPPD_MAX_BURST 个
515 *
516 * dppd_nd6_take 返回 true → 当前线程重新持有 pending.mbuf
517 * 报文失效或原入口已不存在 → 释放 mbuf，更新丢弃统计
518 * 否则 → process_batch 重新解析，并按最新软件规则、会话和路由决定去向
519 *
520 * retry 传 true，避免重复累计 RX 和规则 COUNT；expires_ms 沿用首次入队的截止时间
521 * 重新处理后仍缺 MAC 的报文可以再次入队，不能依靠反复入队延长等待期限
522 */
523static void process_waiting(struct dppd_worker *worker)
524{
525 struct dppd_nd6_pending pending;
526 for (unsigned int i = 0;
527 i < DPPD_MAX_BURST && dppd_nd6_take(worker->runtime->nd6, worker->queue_id, dppd_monotonic_ms(), &pending);
528 ++i) {
529 const struct dppd_port *ingress = dppd_devices_find(&worker->runtime->devices, pending.ingress);
530 /** 取回不代表可以发送，失效包或已失去原入口的包在此归还 mbuf */
531 if (pending.expired || ingress == NULL) {
532 rte_pktmbuf_free(pending.mbuf);
533 struct dppd_stats_values dropped = {.policy_drops = 1, .nd6_queue_drops = 1};
534 dppd_stats_add(&worker->stats, &dropped);
535 if (ingress != NULL)
536 dppd_stats_add(&worker->port_stats[ingress - worker->runtime->devices.ports], &dropped);
537 } else
538 process_batch(worker, ingress, &pending.mbuf, 1, true, pending.expires_ms);
539 }
540}

542/** 工作线程先登记为规则读者，再循环处理端口；收到停止请求后注销并返回 */
543int dppd_worker_main(void *arg)
544{
545 struct dppd_worker *worker = arg;
546 struct dppd_runtime *runtime;
547
548 if (worker == NULL || worker->runtime == NULL)
549 return -1;
550 runtime = worker->runtime;
551
552 /**
553 * queue_id 是启动前分配的连续队列编号，同时作为 QSBR 的读者编号
554 * 首次读取软件规则之前必须注册并上线，否则回收器不知道本线程正在使用旧表
555 * 本实现统一在一轮端口扫描结束后报告安全点，此时各次规则匹配均已返回
556 */
557 if (runtime->software_backend != NULL &&
558 dppd_software_backend_worker_register(runtime->software_backend, worker->queue_id) != 0) {
559 /** 尚未开始收发就注册失败，也要通知主线程，不能留下貌似已启动的空线程 */
560 atomic_store_explicit(&worker->state, DPPD_WORKER_FAILED, memory_order_release);
561 return -1;
562 }
563
564 if (runtime->ipv4_router != NULL && dppd_ipv4_router_reader_register(runtime->ipv4_router, worker->queue_id) != 0) {
565 dppd_software_backend_worker_unregister(runtime->software_backend, worker->queue_id);
566 atomic_store_explicit(&worker->state, DPPD_WORKER_FAILED, memory_order_release);
567 return -1;
568 }
569 /** 完成全部读者注册后才宣布线程可以收发，健康查询据此统计实际运行的队列 */
570 atomic_store_explicit(&worker->heartbeat_ms, dppd_monotonic_ms(), memory_order_release);
571 atomic_store_explicit(&worker->state, DPPD_WORKER_RUNNING, memory_order_release);
572 uint64_t heartbeat_cycle = rte_get_timer_cycles();
573 uint64_t heartbeat_interval = rte_get_timer_hz() / 100U;
574
575 /** 设备事件直接通知所有工作线程退出，不依赖主线程先设置普通停止标记 */
576 while (!worker_io_quiesced(worker)) {
577 uint16_t i;
578
579 if (runtime->nd6 != NULL) {
580 if (worker->queue_id == 0)
581 dppd_nd6_tick(runtime->nd6, dppd_monotonic_ms(), send_nd6, worker);
582 process_waiting(worker);
583 }
584 /** 依次处理各网口的本 worker 队列，一个 process_batch 返回后继续下一个网口 */
585 for (i = 0; i < runtime->devices.nb_ports; ++i)
586 process_ingress(worker, &runtime->devices.ports[i]);
587 /**
588 * 一轮扫描结束后报告本线程已不再使用之前借用的规则和路由表，供回收器判断旧表能否释放
589 * 静默点是读者进度报告，不是线程休眠；即使没有收包也要报告，然后继续下一轮
590 * 例：控制面将规则 A 换成 B，仍在使用 A 的线程未报告安全点前，回收器必须保留 A
591 * 本轮规则匹配和 COUNT 提交均已结束，此处报告进度；回收条件满足后由控制面释放旧表
592 */
593 dppd_software_backend_worker_quiescent(runtime->software_backend, worker->queue_id);
594 dppd_ipv4_router_reader_quiescent(runtime->ipv4_router, worker->queue_id);
595 /** 快路径只读廉价计时器，每十毫秒才更新时间，空流量也持续证明线程前进 */
596 uint64_t cycle = rte_get_timer_cycles();
597 if (cycle - heartbeat_cycle >= heartbeat_interval) {
598 atomic_store_explicit(&worker->heartbeat_ms, dppd_monotonic_ms(), memory_order_release);
599 heartbeat_cycle = cycle;
600 }
601 }
602 /** 停止读取规则后注销；主线程还需等待本线程返回，之后才能销毁共享后端 */
603 dppd_ipv4_router_reader_unregister(runtime->ipv4_router, worker->queue_id);
604 dppd_software_backend_worker_unregister(runtime->software_backend, worker->queue_id);
605 /** 先注销借用的规则读者，再发布已结束状态，资源释放仍须由主线程等待返回 */
606 atomic_store_explicit(&worker->state, DPPD_WORKER_STOPPED, memory_order_release);
607 return 0;
