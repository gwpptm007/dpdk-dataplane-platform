20/** 路由配置的管理对象，持有供 worker 查询的 router，以及路由文件和独立锁 */
21struct dppd_routing_service {
22 /** 内存中的路由与静态邻居表，runtime 借用此地址 */
23 struct dppd_ipv4_router *router;
24 /**
25 * lock_fd 持有独立 .lock 文件，snapshot_fd 保留当前已核对的路由文件
26 * directory_fd 用于保存文件后的目录同步，last_error 记录最近一次保存结果
27 */
28 int lock_fd, snapshot_fd, directory_fd, last_error;
29 /** 保存未完整成功时置位，普通配置替换暂停，等待显式同步修复 */
30 bool dirty;
31 /** 本次启动的网口与设备对应记录数量 */
32 uint16_t count;
33 /** 最近一次确认保存成功的配置版本 */
34 uint64_t persisted_generation;
35 /** 路由文件、独立锁文件及父目录路径，例如 /tmp/routes.bin、同名 .lock 和 /tmp */
36 char path[DPPD_STATE_PATH_CAPACITY], lock_path[DPPD_STATE_PATH_CAPACITY], parent[DPPD_STATE_PATH_CAPACITY];
37 /** 已复制并按端口编号排序的网口对应记录，不依赖调用方局部数组 */
38 struct dppd_routing_binding bindings[DPPD_MAX_PORTS];
39 /** 当前文件的有效长度，区分原 IPv4 格式与双栈格式 */
40 uint32_t expected_size;
41 /** 上次确认的完整文件内容，后续保存前用它检测文件是否被改动 */
42 uint8_t expected[DPPD_ROUTING_DUAL_FILE_SIZE];
43};

45static void put(uint8_t *target, uint64_t value, unsigned int width)
46{
47 for (unsigned int i = 0; i < width; ++i)
48 target[i] = (uint8_t)(value >> (8U * i));
49}

51static uint64_t get(const uint8_t *bytes, unsigned int width)
52{
53 uint64_t value = 0;
54 for (unsigned int i = 0; i < width; ++i)
55 value |= (uint64_t)bytes[i] << (8U * i);
56 return value;
57}

59static uint32_t checksum(const uint8_t *bytes, uint32_t size)
60{
61 uint32_t value = UINT32_MAX;
62 for (size_t i = 0; i < size; ++i) {
63 value ^= i >= 32 && i < 36 ? 0 : bytes[i];
64 for (unsigned int bit = 0; bit < 8; ++bit)
65 value = (value >> 1) ^ (0xedb88320U & (0U - (value & 1U)));
66 }
67 return ~value;
68}

70/** 固定头部、端口映射、路由和邻居分别编码，IP 字节直接复制，其他整数明确用小端 */
71static uint32_t encode(const struct dppd_routing_service *service, const struct dppd_ipv4_configuration *configuration,
72 uint8_t bytes[DPPD_ROUTING_DUAL_FILE_SIZE])
73{
74 /** 纯 IPv4 文件继续使用旧版，首次出现 IPv6 后保留双栈格式，避免故障修复时降级 */
75 uint32_t size = service->expected_size == DPPD_ROUTING_DUAL_FILE_SIZE || configuration->route6_count != 0 ||
76 configuration->neighbor6_count != 0
77 ? DPPD_ROUTING_DUAL_FILE_SIZE
78 : DPPD_ROUTING_FILE_SIZE;
79 memset(bytes, 0, size);
80 memcpy(bytes, "DPPIPV41", 8);
81 put(bytes + 8, size == DPPD_ROUTING_FILE_SIZE ? 1 : 2, 4);
82 put(bytes + 12, size, 4);
83 put(bytes + 16, configuration->generation, 8);
84 put(bytes + 24, configuration->route_count, 4);
85 put(bytes + 28, configuration->neighbor_count, 4);
86 put(bytes + 36, service->count, 4);
87 for (uint16_t i = 0; i < service->count; ++i) {
88 uint8_t *entry = bytes + HEADER_SIZE + i * BINDING_SIZE;
89 const struct dppd_routing_binding *binding = &service->bindings[i];
90 put(entry, binding->port_id, 2);
91 memcpy(entry + 8, binding->device, strlen(binding->device));
92 memcpy(entry + 136, binding->driver, strlen(binding->driver));
93 memcpy(entry + 200, binding->ifname, strlen(binding->ifname));
94 }
95 for (uint32_t i = 0; i < configuration->route_count; ++i) {
96 uint8_t *entry = bytes + ROUTE_OFFSET + 16U * i;
97 const struct dppd_ipv4_route *route = &configuration->routes[i];
98 memcpy(entry, &route->prefix_be, 4);
99 memcpy(entry + 4, &route->gateway_be, 4);
100 put(entry + 8, route->port_id, 2);
101 entry[10] = route->depth;
102 }
103 for (uint32_t i = 0; i < configuration->neighbor_count; ++i) {
104 uint8_t *entry = bytes + NEIGHBOR_OFFSET + 16U * i;
105 const struct dppd_ipv4_neighbor *neighbor = &configuration->neighbors[i];
106 memcpy(entry, &neighbor->address_be, 4);
107 put(entry + 4, neighbor->port_id, 2);
108 memcpy(entry + 6, neighbor->mac, 6);
109 }
110 if (size == DPPD_ROUTING_DUAL_FILE_SIZE) {
111 put(bytes + 40, configuration->route6_count, 4);
112 put(bytes + 44, configuration->neighbor6_count, 4);
113 for (uint32_t i = 0; i < configuration->route6_count; ++i) {
114 uint8_t *entry = bytes + ROUTE6_OFFSET + 40U * i;
115 const struct dppd_ipv6_route *route = &configuration->routes6[i];
116 memcpy(entry, route->prefix, 16);
117 memcpy(entry + 16, route->gateway, 16);
118 put(entry + 32, route->port_id, 2);
119 entry[34] = route->depth;
120 }
121 for (uint32_t i = 0; i < configuration->neighbor6_count; ++i) {
122 uint8_t *entry = bytes + NEIGHBOR6_OFFSET + 24U * i;
123 const struct dppd_ipv6_neighbor *neighbor = &configuration->neighbors6[i];
124 memcpy(entry, neighbor->address, 16);
125 put(entry + 16, neighbor->port_id, 2);
126 memcpy(entry + 18, neighbor->mac, 6);
127 }
128 }
129 put(bytes + 32, checksum(bytes, size), 4);
130 return size;
131}

133static int decode(struct dppd_routing_service *service, const uint8_t *bytes,
134 struct dppd_ipv4_configuration *configuration)
135{
136 uint32_t size = service->expected_size;
137 bool dual = size == DPPD_ROUTING_DUAL_FILE_SIZE;
138 if (memcmp(bytes, "DPPIPV41", 8) != 0 || get(bytes + 8, 4) != (dual ? 2U : 1U) || get(bytes + 12, 4) != size ||
139 get(bytes + 32, 4) != checksum(bytes, size) || get(bytes + 24, 4) > DPPD_IPV4_ROUTE_LIMIT ||
140 get(bytes + 28, 4) > DPPD_IPV4_NEIGHBOR_LIMIT ||
141 (dual && (get(bytes + 40, 4) > DPPD_IPV4_ROUTE_LIMIT || get(bytes + 44, 4) > DPPD_IPV4_NEIGHBOR_LIMIT)))
142 return -EBADMSG;
143 memset(configuration, 0, sizeof(*configuration));
144 configuration->generation = get(bytes + 16, 8);
145 configuration->route_count = (uint32_t)get(bytes + 24, 4);
146 configuration->neighbor_count = (uint32_t)get(bytes + 28, 4);
147 for (uint32_t i = 0; i < configuration->route_count; ++i) {
148 const uint8_t *entry = bytes + ROUTE_OFFSET + 16U * i;
149 struct dppd_ipv4_route *route = &configuration->routes[i];
150 memcpy(&route->prefix_be, entry, 4);
151 memcpy(&route->gateway_be, entry + 4, 4);
152 route->port_id = (uint16_t)get(entry + 8, 2);
153 route->depth = entry[10];
154 }
155 for (uint32_t i = 0; i < configuration->neighbor_count; ++i) {
156 const uint8_t *entry = bytes + NEIGHBOR_OFFSET + 16U * i;
157 struct dppd_ipv4_neighbor *neighbor = &configuration->neighbors[i];
158 memcpy(&neighbor->address_be, entry, 4);
159 neighbor->port_id = (uint16_t)get(entry + 4, 2);
160 memcpy(neighbor->mac, entry + 6, 6);
161 }
162 if (dual) {
163 configuration->route6_count = (uint32_t)get(bytes + 40, 4);
164 configuration->neighbor6_count = (uint32_t)get(bytes + 44, 4);
165 for (uint32_t i = 0; i < configuration->route6_count; ++i) {
166 const uint8_t *entry = bytes + ROUTE6_OFFSET + 40U * i;
167 struct dppd_ipv6_route *route = &configuration->routes6[i];
168 memcpy(route->prefix, entry, 16);
169 memcpy(route->gateway, entry + 16, 16);
170 route->port_id = (uint16_t)get(entry + 32, 2);
171 route->depth = entry[34];
172 }
173 for (uint32_t i = 0; i < configuration->neighbor6_count; ++i) {
174 const uint8_t *entry = bytes + NEIGHBOR6_OFFSET + 24U * i;
175 struct dppd_ipv6_neighbor *neighbor = &configuration->neighbors6[i];
176 memcpy(neighbor->address, entry, 16);
177 neighbor->port_id = (uint16_t)get(entry + 16, 2);
178 memcpy(neighbor->mac, entry + 18, 6);
179 }
180 }
181 uint8_t canonical[DPPD_ROUTING_DUAL_FILE_SIZE];
182 encode(service, configuration, canonical);
183 /* 完整比较同时约束端口身份、保留位、字符串尾部和未使用槽位 */
184 return memcmp(bytes, canonical, size) == 0 ? 0 : -EXDEV;
185}

187static int file_identity(int fd, const char *path, off_t size)
188{
189 struct stat held, named;
190 if (fstat(fd, &held) != 0 || lstat(path, &named) != 0)
191 return -errno;
192 if (!S_ISREG(named.st_mode) || held.st_dev != named.st_dev || held.st_ino != named.st_ino || held.st_nlink != 1 ||
193 held.st_uid != geteuid() || (held.st_mode & 07777) != 0600 || held.st_size != size)
194 return -ESTALE;
195 return 0;
196}

198static int read_exact(int fd, uint8_t *bytes, size_t size)
199{
200 size_t done = 0;
201 while (done < size) {
202 ssize_t read_size = pread(fd, bytes + done, size - done, (off_t)done);
203 if (read_size < 0 && errno == EINTR)
204 continue;
205 if (read_size <= 0)
206 return read_size < 0 ? -errno : -EBADMSG;
207 done += (size_t)read_size;
208 }
209 return 0;
210}

212/** 写入前再次核对同一目录、锁文件及上次完整快照，不能盲目覆盖被替换或损坏的文件 */
213static int unchanged(struct dppd_routing_service *service)
214{
215 struct stat held, named;
216 if (fstat(service->directory_fd, &held) != 0 || lstat(service->parent, &named) != 0)
217 return -errno;
218 if (!S_ISDIR(named.st_mode) || held.st_dev != named.st_dev || held.st_ino != named.st_ino)
219 return -ESTALE;
220 int rc = file_identity(service->lock_fd, service->lock_path, 0);
221 if (rc == 0 && service->snapshot_fd >= 0) {
222 rc = file_identity(service->snapshot_fd, service->path, service->expected_size);
223 uint8_t bytes[DPPD_ROUTING_DUAL_FILE_SIZE];
224 if (rc == 0)
225 rc = read_exact(service->snapshot_fd, bytes, service->expected_size);
226 if (rc == 0 && memcmp(bytes, service->expected, service->expected_size) != 0)
227 rc = -EBADMSG;
228 }
229 return rc;
230}

232/** 文件同步、原子替换和目录同步均完成才成功，任何错误都禁止随后直接发布候选 */
233static int save(void *context, const struct dppd_ipv4_configuration *configuration)
234{
235 struct dppd_routing_service *service = context;
236 char temporary[DPPD_STATE_PATH_CAPACITY];
237 int fd = -1, rc = unchanged(service);
238 if (rc != 0)
239 goto fail;
240 if (snprintf(temporary, sizeof(temporary), "%s.XXXXXX", service->path) >= (int)sizeof(temporary)) {
241 rc = -ENAMETOOLONG;
242 goto fail;
243 }
244 fd = mkostemp(temporary, O_CLOEXEC);
245 if (fd < 0) {
246 rc = -errno;
247 goto fail;
248 }
249 uint8_t bytes[DPPD_ROUTING_DUAL_FILE_SIZE];
250 uint32_t size = encode(service, configuration, bytes);
251 size_t done = 0;
252 while (done < size) {
253 ssize_t written = pwrite(fd, bytes + done, size - done, (off_t)done);
254 if (written < 0 && errno == EINTR)
255 continue;
256 if (written <= 0) {
257 rc = written < 0 ? -errno : -EIO;
258 goto fail;
259 }
260 done += (size_t)written;
261 }
262 if (fsync(fd) != 0) {
263 rc = -errno;
264 goto fail;
265 }
266 rc = unchanged(service);
267 if (rc != 0)
268 goto fail;
269 int renamed = service->snapshot_fd < 0 ? renameat2(AT_FDCWD, temporary, AT_FDCWD, service->path, RENAME_NOREPLACE)
270 : rename(temporary, service->path);
271 if (renamed != 0) {
272 rc = -errno;
273 goto fail;
274 }
275 if (service->snapshot_fd >= 0)
276 close(service->snapshot_fd);
277 service->snapshot_fd = fd;
278 service->expected_size = size;
279 memcpy(service->expected, bytes, size);
280 fd = -1;
281 if (fsync(service->directory_fd) != 0) {
282 rc = -errno;
283 goto fail;
284 }
285 service->persisted_generation = configuration->generation;
286 service->last_error = 0;
287 service->dirty = false;
288 return 0;
289fail:
290 if (fd >= 0) {
291 close(fd);
292 unlink(temporary);
293 }
294 service->last_error = rc;
295 service->dirty = true;
296 return rc;
297}

305static int canonical_target(const char *path, char target[DPPD_STATE_PATH_CAPACITY])
306{
307 char parent[DPPD_STATE_PATH_CAPACITY], name[DPPD_STATE_PATH_CAPACITY], resolved[DPPD_STATE_PATH_CAPACITY];
308 if (path == NULL || path[0] == '\0' || strlen(path) >= sizeof(parent))
309 return -EINVAL;
310 strcpy(parent, path);
311 char *slash = strrchr(parent, '/');
312 const char *base = slash != NULL ? slash + 1 : path;
313 if (*base == '\0' || strcmp(base, ".") == 0 || strcmp(base, "..") == 0)
314 return -EINVAL;
315 strcpy(name, base);
316 if (slash != NULL)
317 slash[slash == parent ? 1 : 0] = '\0';
318 else
319 strcpy(parent, ".");
320 if (realpath(parent, resolved) == NULL)
321 return -errno;
322 return snprintf(target, DPPD_STATE_PATH_CAPACITY, "%s%s%s", resolved, strcmp(resolved, "/") == 0 ? "" : "/",
323 name) >= (int)DPPD_STATE_PATH_CAPACITY
324 ? -ENAMETOOLONG
325 : 0;
326}

328int dppd_routing_check_paths(const char *path, const char *const *others, size_t count)
329{
330 if (path == NULL || *path == '\0')
331 return 0;
332 if (count != 0 && others == NULL)
333 return -EINVAL;
334 char canonical[DPPD_STATE_PATH_CAPACITY], lock[DPPD_STATE_PATH_CAPACITY], other[DPPD_STATE_PATH_CAPACITY];
335 int rc = canonical_target(path, canonical);
336 if (rc != 0)
337 return rc;
338 if (snprintf(lock, sizeof(lock), "%s.lock", canonical) >= (int)sizeof(lock))
339 return -ENAMETOOLONG;
340 for (size_t i = 0; i < count; ++i) {
341 if (others[i] == NULL || *others[i] == '\0')
342 continue;
343 rc = canonical_target(others[i], other);
344 if (rc != 0)
345 return rc;
346 if (strcmp(canonical, other) == 0 || strcmp(lock, other) == 0)
347 return -EINVAL;
348 }
349 return 0;
350}

352/**
353 * 规范化路由文件路径，生成同名 .lock 路径，并打开父目录供后续同步
354 * 例如 /tmp/routes.bin → /tmp/routes.bin.lock；文件可以不存在，父目录必须存在
355 * 路径和目录句柄写入 service，成功返回 0，失败返回负错误码
356 */
357static int prepare_paths(struct dppd_routing_service *service, const char *path)
358{
359 int rc = canonical_target(path, service->path);
360 if (rc != 0)
361 return rc;
362 if (snprintf(service->lock_path, sizeof(service->lock_path), "%s.lock", service->path) >=
363 (int)sizeof(service->lock_path))
364 return -ENAMETOOLONG;
365 strcpy(service->parent, service->path);
366 char *slash = strrchr(service->parent, '/');
367 slash[slash == service->parent ? 1 : 0] = '\0';
368 service->directory_fd = open(service->parent, O_RDONLY | O_DIRECTORY | O_NOFOLLOW | O_CLOEXEC);
369 return service->directory_fd >= 0 ? 0 : -errno;
370}

372/**
373 * 打开项目路由配置，创建供 runtime 借用的路由器，并持有该配置的独立文件锁
374 * 复制本次网口对应记录 → 取得 .lock 排他锁
375 * 文件已存在 → 核对格式、设备对应关系和路由内容，恢复配置
376 * 文件不存在 → 创建版本 1 的空路由和静态邻居表，并保存
377 * 全部准备并复查成功后，将服务写入 *result 并返回 0；失败调用清理并返回负错误码
378 */
379int dppd_routing_open(struct dppd_routing_service **result, const char *path,
380 const struct dppd_routing_binding *bindings, uint16_t count)
381{
382 if (result == NULL || *result != NULL || bindings == NULL || count == 0 || count > DPPD_MAX_PORTS)
383 return -EINVAL;
384 struct dppd_routing_service *service = calloc(1, sizeof(*service));
385 if (service == NULL)
386 return -ENOMEM;
387 /** 避免失败清理时误关标准输入（fd 0） */
388 service->lock_fd = service->snapshot_fd = service->directory_fd = -1;
389 int rc = -EINVAL;
390 service->count = count;
391 uint16_t ports[DPPD_MAX_PORTS];
392 for (uint16_t i = 0; i < count; ++i) {
393 const struct dppd_routing_binding *binding = &bindings[i];
394 if (binding->port_id == UINT16_MAX || binding->device[0] == '\0' || binding->driver[0] == '\0' ||
395 memchr(binding->device, 0, sizeof(binding->device)) == NULL ||
396 memchr(binding->driver, 0, sizeof(binding->driver)) == NULL ||
397 memchr(binding->ifname, 0, sizeof(binding->ifname)) == NULL)
398 goto fail;
399 service->bindings[i].port_id = binding->port_id;
400 strcpy(service->bindings[i].device, binding->device);
401 strcpy(service->bindings[i].driver, binding->driver);
402 strcpy(service->bindings[i].ifname, binding->ifname);
403 }
404 /** 固定网口记录的存储顺序，便于恢复时核对 */
405 qsort(service->bindings, count, sizeof(service->bindings[0]), compare_binding);
406 for (uint16_t i = 0; i < count; ++i) {
407 if (i != 0 && service->bindings[i].port_id == service->bindings[i - 1].port_id)
408 goto fail;
409 ports[i] = service->bindings[i].port_id;
410 }
411 rc = prepare_paths(service, path);
412 if (rc != 0)
413 goto fail;
414 service->lock_fd = open(service->lock_path, O_RDWR | O_CREAT | O_NOFOLLOW | O_NONBLOCK | O_CLOEXEC, 0600);
415 if (service->lock_fd < 0) {
416 rc = -errno;
417 goto fail;
418 }
419 rc = file_identity(service->lock_fd, service->lock_path, 0);
420 if (rc != 0)
421 goto fail;
422 /** 路由文件会被替换，用独立 .lock 文件保持互斥 */
423 if (flock(service->lock_fd, LOCK_EX | LOCK_NB) != 0) {
424 rc = errno == EWOULDBLOCK ? -EBUSY : -errno;
425 goto fail;
426 }
427 /** 仅文件不存在时创建空表，旧文件损坏必须报错 */
428 struct dppd_ipv4_configuration initial = {.generation = 1};
429 service->snapshot_fd = open(service->path, O_RDONLY | O_NOFOLLOW | O_NONBLOCK | O_CLOEXEC);
430 if (service->snapshot_fd >= 0) {
431 struct stat metadata;
432 if (fstat(service->snapshot_fd, &metadata) != 0) {
433 rc = -errno;
434 goto fail;
435 }
436 if (metadata.st_size != DPPD_ROUTING_FILE_SIZE && metadata.st_size != DPPD_ROUTING_DUAL_FILE_SIZE) {
437 rc = -EBADMSG;
438 goto fail;
439 }
440 service->expected_size = (uint32_t)metadata.st_size;
441 rc = file_identity(service->snapshot_fd, service->path, service->expected_size);
442 if (rc == 0)
443 rc = read_exact(service->snapshot_fd, service->expected, service->expected_size);
444 if (rc == 0)
445 rc = decode(service, service->expected, &initial);
446 } else if (errno != ENOENT)
447 rc = -errno;
448 if (rc == 0)
449 rc = dppd_ipv4_router_create(&service->router, ports, count, &initial);
450 if (rc != 0)
451 goto fail;
452 /** 用路由器整理后的配置复核原文件的内容和顺序 */
453 struct dppd_ipv4_configuration canonical;
454 rc = dppd_ipv4_router_get(service->router, &canonical);
455 if (rc == 0 && service->snapshot_fd >= 0) {
456 uint8_t bytes[DPPD_ROUTING_DUAL_FILE_SIZE];
457 encode(service, &canonical, bytes);
458 if (memcmp(bytes, service->expected, service->expected_size) != 0)
459 rc = -EBADMSG;
460 if (rc == 0)
461 rc = unchanged(service);
462 if (rc == 0 && fsync(service->directory_fd) != 0)
463 rc = -errno;
464 } else if (rc == 0)
465 rc = save(service, &canonical);
466 if (rc != 0)
467 goto fail;
468 service->persisted_generation = canonical.generation;
469 *result = service;
470 return 0;
471fail:
472 dppd_routing_close(service);
473 return rc;
474}

476/** 销毁路由器并关闭数据文件、独立锁和目录句柄；路由器销毁失败则返回错误并保留服务 */
477int dppd_routing_close(struct dppd_routing_service *service)
478{
479 if (service == NULL)
480 return 0;
481 int rc = dppd_ipv4_router_destroy(service->router);
482 if (rc != 0)
483 return rc;
484 if (service->snapshot_fd >= 0)
485 close(service->snapshot_fd);
486 if (service->lock_fd >= 0)
487 close(service->lock_fd);
488 if (service->directory_fd >= 0)
489 close(service->directory_fd);
490 free(service);
491 return 0;
492}

500int dppd_routing_get(struct dppd_routing_service *service, struct dppd_ipv4_configuration *configuration)
501{
502 return service != NULL ? dppd_ipv4_router_get(service->router, configuration) : -ENODEV;
503}
504
505int dppd_routing_replace(struct dppd_routing_service *service, uint64_t expected,
506 const struct dppd_ipv4_configuration *configuration)
507{
508 if (service == NULL)
509 return -ENODEV;
510 if (service->dirty)
511 return -EUCLEAN;
512 int rc = dppd_ipv4_router_replace(service->router, expected, configuration, save, service);
513 return service->dirty ? -EUCLEAN : rc;
514}
515
516int dppd_routing_flush(struct dppd_routing_service *service)
517{
518 if (service == NULL)
519 return -ENODEV;
520 struct dppd_ipv4_configuration configuration;
521 int rc = dppd_routing_get(service, &configuration);
522 return rc == 0 ? save(service, &configuration) : rc;
523}

525void dppd_routing_status(struct dppd_routing_service *service, struct dppd_routing_status *status)
526{
527 if (status == NULL)
528 return;
529 memset(status, 0, sizeof(*status));
530 if (service == NULL)
531 return;
532 struct dppd_ipv4_configuration configuration;
533 if (dppd_routing_get(service, &configuration) != 0)
534 return;
535 status->enabled = true;
536 status->dirty = service->dirty;
537 status->last_error = service->last_error;
538 status->generation = configuration.generation;
539 status->persisted_generation = service->persisted_generation;
540 status->route_count = configuration.route_count;
541 status->neighbor_count = configuration.neighbor_count;
542 status->route6_count = configuration.route6_count;
543 status->neighbor6_count = configuration.neighbor6_count;
544}
