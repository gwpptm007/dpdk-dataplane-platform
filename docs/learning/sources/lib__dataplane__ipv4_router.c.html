19/** IPv4/IPv6 共用的二叉前缀树节点，每层按地址的一位选择分支 */
20struct route_node {
21 /** 下一位为 0 或 1 时的子节点下标，0 表示无子节点；保存实际下标，不加一 */
22 uint32_t child[2];
23 /** 对应地址族路由数组的下标加一，0 表示此前缀未配置路由 */
24 uint32_t route;
25};

27/** 一次发布的双栈路由和邻居配置及其查询树，发布后查询内容保持不变 */
28struct routing_snapshot {
29 /** 当前版本的 IPv4/IPv6 路由和静态邻居，邻居已按出口和 IP 排序供二分查询 */
30 struct dppd_ipv4_configuration configuration;
31 /** 旧版本等待回收时的链表后继，由控制面维护 */
32 struct routing_snapshot *retired_next;
33 /** 此版本退役时取得的 QSBR 宽限期标记，用于确认旧版本读者已离开 */
34 uint64_t retired_token;
35 /** nodes 中已使用的节点数，建树时也用作下一个可分配节点的下标 */
36 uint32_t node_count;
37 /** IPv6 根节点在 nodes 中的下标，IPv4 根节点固定为 nodes[0] */
38 uint32_t root6;
39 /** 随快照一并分配的变长节点数组，两种地址族共用存储但分别从各自根节点查询 */
40 struct route_node nodes[];
41};

43/** 双栈路由器，统一管理当前快照、控制面更新和旧版本的安全回收 */
44struct dppd_ipv4_router {
45 /** 当前查询快照的原子指针，控制面完整构建新版本后替换，报文线程读取时不加写锁 */
46 _Atomic(struct routing_snapshot *) active;
47 /** 已被替换、仍可能被读者使用的旧快照链表，等待安全回收 */
48 struct routing_snapshot *retired;
49 /** QSBR 读者静默状态，用于判断旧快照是否已不再被任何读者使用 */
50 struct rte_rcu_qsbr *qsbr;
51 /** 串行化控制面的配置查询、替换及回收，不参与逐包路由查询 */
52 pthread_mutex_t writer;
53 /** 已注册读者的位图，第 reader 位为 1 表示已注册；不是读者数量 */
54 atomic_uint_fast64_t readers;
55 /** retired 链表中的快照数量，用于限制等待回收的版本积压 */
56 uint32_t retired_count;
57 /** ports 中有效网口 ID 的数量，创建后保持不变 */
58 uint16_t port_count;
59 /** 路由器允许使用的实际网口 ID，校验路由出口和邻居所属网口时查询 */
60 uint16_t ports[DPPD_MAX_PORTS];
61};

63static bool port_present(const struct dppd_ipv4_router *router, uint16_t port)
64{
65 for (uint16_t i = 0; i < router->port_count; ++i)
66 if (router->ports[i] == port)
67 return true;
68 return false;

71/** 该阶段只做 IPv4 单播转发，零网段、回环、组播和保留高地址不能由默认路由放行 */
72static bool unicast_address(uint32_t address_be)
73{
74 uint32_t address = ntohl(address_be);
75 return (address >> 24) != 0 && (address >> 24) != 127 && address < 0xe0000000U;
76}

78static int compare_route(const void *left, const void *right)
79{
80 const struct dppd_ipv4_route *a = left, *b = right;
81 uint32_t first = ntohl(a->prefix_be), second = ntohl(b->prefix_be);
82 if (first != second)
83 return first < second ? -1 : 1;
84 return (int)a->depth - b->depth;
85}

87static int compare_neighbor(const void *left, const void *right)
88{
89 const struct dppd_ipv4_neighbor *a = left, *b = right;
90 if (a->port_id != b->port_id)
91 return a->port_id < b->port_id ? -1 : 1;
92 uint32_t first = ntohl(a->address_be), second = ntohl(b->address_be);
93 return first == second ? 0 : first < second ? -1 : 1;
94}

96static int compare_route6(const void *left, const void *right)
97{
98 const struct dppd_ipv6_route *a = left, *b = right;
99 int address = memcmp(a->prefix, b->prefix, 16);
100 return address != 0 ? address : (int)a->depth - b->depth;
101}

103/** 先比较出口，再比较 16 字节 IPv6 地址，均相等返回 0，供邻居排序和二分查询共用 */
104static int compare_neighbor6(const void *left, const void *right)
105{
106 const struct dppd_ipv6_neighbor *a = left, *b = right;
107 return a->port_id != b->port_id ? (int)a->port_id - b->port_id : memcmp(a->address, b->address, 16);
108}

110/** 每个地址族有独立根节点，IPv6 按线缆字节逐位建树，非整字节前缀也保持精确 */
111static int compile6(const struct dppd_ipv4_router *router, struct routing_snapshot *snapshot)
112{
113 struct dppd_ipv4_configuration *configuration = &snapshot->configuration;
114 qsort(configuration->routes6, configuration->route6_count, sizeof(*configuration->routes6), compare_route6);
115 qsort(configuration->neighbors6, configuration->neighbor6_count, sizeof(*configuration->neighbors6),
116 compare_neighbor6);
117 const uint8_t zero[16] = {0};
118 snapshot->root6 = snapshot->node_count++;
119 for (uint32_t i = 0; i < configuration->route6_count; ++i) {
120 const struct dppd_ipv6_route *route = &configuration->routes6[i];
121 if (route->depth > 128 || route->reserved != 0 || !port_present(router, route->port_id) ||
122 (memcmp(route->gateway, zero, 16) != 0 && !dppd_ipv6_unicast(route->gateway, true)))
123 return -EINVAL;
124 for (unsigned int bit = route->depth; bit < 128; ++bit)
125 if ((route->prefix[bit / 8] & (0x80U >> (bit % 8))) != 0)
126 return -EINVAL;
127 if (i != 0 && compare_route6(route - 1, route) == 0)
128 return -EEXIST;
129 uint32_t node = snapshot->root6;
130 for (unsigned int bit = 0; bit < route->depth; ++bit) {
131 unsigned int branch = (route->prefix[bit / 8] >> (7U - bit % 8)) & 1U;
132 if (snapshot->nodes[node].child[branch] == 0)
133 snapshot->nodes[node].child[branch] = snapshot->node_count++;
134 node = snapshot->nodes[node].child[branch];
135 }
136 snapshot->nodes[node].route = i + 1;
137 }
138 for (uint32_t i = 0; i < configuration->neighbor6_count; ++i) {
139 const struct dppd_ipv6_neighbor *neighbor = &configuration->neighbors6[i];
140 const uint8_t empty_mac[6] = {0};
141 if (!port_present(router, neighbor->port_id) || !dppd_ipv6_unicast(neighbor->address, true) ||
142 (neighbor->mac[0] & 1U) != 0 || memcmp(neighbor->mac, empty_mac, 6) == 0)
143 return -EINVAL;
144 if (i != 0 && compare_neighbor6(neighbor - 1, neighbor) == 0)
145 return -EEXIST;
146 }
147 return 0;
148}

150/** 控制面先构造并验证全部内容，不能把半张路由表或无效邻居发布给工作线程 */
151static int compile(const struct dppd_ipv4_router *router, const struct dppd_ipv4_configuration *input,
152 struct routing_snapshot **result)
153{
154 if (input->route_count > DPPD_IPV4_ROUTE_LIMIT || input->neighbor_count > DPPD_IPV4_NEIGHBOR_LIMIT ||
155 input->route6_count > DPPD_IPV4_ROUTE_LIMIT || input->neighbor6_count > DPPD_IPV4_NEIGHBOR_LIMIT)
156 return -E2BIG;
157 size_t capacity = (size_t)input->route_count * 32 + (size_t)input->route6_count * 128 + 2;
158 struct routing_snapshot *snapshot = calloc(1, sizeof(*snapshot) + capacity * sizeof(struct route_node));
159 if (snapshot == NULL)
160 return -ENOMEM;
161 struct dppd_ipv4_configuration *configuration = &snapshot->configuration;
162 configuration->generation = input->generation;
163 configuration->route_count = input->route_count;
164 configuration->neighbor_count = input->neighbor_count;
165 memcpy(configuration->routes, input->routes, input->route_count * sizeof(input->routes[0]));
166 memcpy(configuration->neighbors, input->neighbors, input->neighbor_count * sizeof(input->neighbors[0]));
167 configuration->route6_count = input->route6_count;
168 configuration->neighbor6_count = input->neighbor6_count;
169 memcpy(configuration->routes6, input->routes6, input->route6_count * sizeof(*input->routes6));
170 memcpy(configuration->neighbors6, input->neighbors6, input->neighbor6_count * sizeof(*input->neighbors6));
171 qsort(configuration->routes, configuration->route_count, sizeof(configuration->routes[0]), compare_route);
172 qsort(configuration->neighbors, configuration->neighbor_count, sizeof(configuration->neighbors[0]),
173 compare_neighbor);
174 int rc = -EINVAL;
175 snapshot->node_count = 1;
176 for (uint32_t i = 0; i < configuration->route_count; ++i) {
177 const struct dppd_ipv4_route *route = &configuration->routes[i];
178 if (route->depth > 32 || route->reserved != 0 || !port_present(router, route->port_id) ||
179 (route->gateway_be != 0 && !unicast_address(route->gateway_be)))
180 goto fail;
181 uint32_t address = ntohl(route->prefix_be);
182 uint32_t mask = route->depth == 0 ? 0 : UINT32_MAX << (32U - route->depth);
183 if ((address & mask) != address)
184 goto fail;
185 if (i != 0 && compare_route(route - 1, route) == 0) {
186 rc = -EEXIST;
187 goto fail;
188 }
189 uint32_t node = 0;
190 for (uint8_t bit = 0; bit < route->depth; ++bit) {
191 unsigned int branch = (address >> (31U - bit)) & 1U;
192 if (snapshot->nodes[node].child[branch] == 0)
193 snapshot->nodes[node].child[branch] = snapshot->node_count++;
194 node = snapshot->nodes[node].child[branch];
195 }
196 snapshot->nodes[node].route = i + 1;
197 }
198 for (uint32_t i = 0; i < configuration->neighbor_count; ++i) {
199 const struct dppd_ipv4_neighbor *neighbor = &configuration->neighbors[i];
200 const uint8_t zero[6] = {0};
201 if (!port_present(router, neighbor->port_id) || !unicast_address(neighbor->address_be) ||
202 (neighbor->mac[0] & 1U) != 0 || memcmp(neighbor->mac, zero, sizeof(zero)) == 0)
203 goto fail;
204 if (i != 0 && compare_neighbor(neighbor - 1, neighbor) == 0) {
205 rc = -EEXIST;
206 goto fail;
207 }
208 }
209 rc = compile6(router, snapshot);
210 if (rc != 0)
211 goto fail;
212 *result = snapshot;
213 return 0;
214fail:
215 free(snapshot);
216 return rc;
217}

219/** 不阻塞数据面或等待慢读者，超过退役上限时让下一次配置更新显式重试 */
220static void reclaim(struct dppd_ipv4_router *router)
221{
222 struct routing_snapshot **link = &router->retired;
223 while (*link != NULL) {
224 struct routing_snapshot *snapshot = *link;
225 if (!rte_rcu_qsbr_check(router->qsbr, snapshot->retired_token, false)) {
226 link = &snapshot->retired_next;
227 continue;
228 }
229 *link = snapshot->retired_next;
230 router->retired_count--;
231 free(snapshot);
232 }
233}

235int dppd_ipv4_router_create(struct dppd_ipv4_router **result, const uint16_t *ports, uint16_t count,
236 const struct dppd_ipv4_configuration *initial)
237{
238 if (result == NULL || *result != NULL || ports == NULL || count == 0 || count > DPPD_MAX_PORTS ||
239 (initial != NULL && initial->generation == 0))
240 return -EINVAL;
241 struct dppd_ipv4_router *router = calloc(1, sizeof(*router));
242 if (router == NULL)
243 return -ENOMEM;
244 int rc = -EINVAL;
245 router->port_count = count;
246 memcpy(router->ports, ports, count * sizeof(*ports));
247 for (uint16_t i = 0; i < count; ++i) {
248 if (ports[i] == UINT16_MAX)
249 goto fail;
250 for (uint16_t j = 0; j < i; ++j)
251 if (ports[i] == ports[j])
252 goto fail;
253 }
254 struct dppd_ipv4_configuration empty = {.generation = 1};
255 struct routing_snapshot *snapshot = NULL;
256 rc = compile(router, initial != NULL ? initial : &empty, &snapshot);
257 if (rc != 0)
258 goto fail;
259 size_t size = rte_rcu_qsbr_get_memsize(DPPD_MAX_WORKERS);
260 size = RTE_ALIGN_CEIL(size, RTE_CACHE_LINE_SIZE);
261 router->qsbr = aligned_alloc(RTE_CACHE_LINE_SIZE, size);
262 if (router->qsbr == NULL || rte_rcu_qsbr_init(router->qsbr, DPPD_MAX_WORKERS) != 0) {
263 rc = -ENOMEM;
264 free(snapshot);
265 goto fail;
266 }
267 int lock_error = pthread_mutex_init(&router->writer, NULL);
268 if (lock_error != 0) {
269 rc = -lock_error;
270 free(snapshot);
271 goto fail;
272 }
273 atomic_init(&router->readers, 0);
274 atomic_init(&router->active, snapshot);
275 *result = router;
276 return 0;
277fail:
278 free(router->qsbr);
279 free(router);
280 return rc;
281}

283int dppd_ipv4_router_destroy(struct dppd_ipv4_router *router)
284{
285 if (router == NULL)
286 return 0;
287 if (atomic_load_explicit(&router->readers, memory_order_acquire) != 0)
288 return -EBUSY;
289 free(atomic_load_explicit(&router->active, memory_order_relaxed));
290 struct routing_snapshot *snapshot = router->retired;
291 while (snapshot != NULL) {
292 struct routing_snapshot *next = snapshot->retired_next;
293 free(snapshot);
294 snapshot = next;
295 }
296 pthread_mutex_destroy(&router->writer);
297 free(router->qsbr);
298 free(router);
299 return 0;
300}

302/**
303 * 查询 IPv6 最长前缀路由及下一跳邻居，生成单包转发结果
304 * 目的 IP → 最长前缀路由 → 出口和下一跳 → 目的 MAC
305 *
306 * 入参：
307 * router 双栈路由器
308 * destination 16 字节目的 IPv6 地址，按报文字节顺序保存，只读
309 *
310 * 出参：
311 * forward 配置版本、出口、下一跳 IPv6 地址和目的 MAC
312 *
313 * 返回：
314 * 0 查询成功，结果完整
315 * -EINVAL 参数指针为空
316 * -EAFNOSUPPORT 目的地址不在支持范围内
317 * -ENOENT 没有匹配路由
318 * -EHOSTUNREACH 缺少静态邻居，仍输出版本、出口和下一跳，供 ND 解析
319 *
320 * 调用线程须已注册为 QSBR 读者，并在查询期间保持在线
321 */
322int dppd_ipv6_router_lookup(struct dppd_ipv4_router *router, const uint8_t destination[16], struct dppd_ipv4_forward *forward)
323{
324 if (router == NULL || destination == NULL || forward == NULL)
325 return -EINVAL;
326 memset(forward, 0, sizeof(*forward));
327
328 /** 拒绝未指定、回环、组播、IPv4 映射和链路本地目的地址 */
329 if (!dppd_ipv6_unicast(destination, false))
330 return -EAFNOSUPPORT;
331
332 /**
333 * active：原子变量，保存 struct routing_snapshot * 类型的快照地址
334 * &router->active：要读取的原子变量的地址
335 * atomic_load_explicit：<stdatomic.h> 的原子读取操作，此处返回 struct routing_snapshot *
336 * snapshot：以 const 指针接收，通过它只读访问快照，不复制整张表
337 *
338 * acquire 与发布端 release 配对，读到新指针时，发布前构建的内容也已可见
339 * QSBR 保护本次持有的快照，防止旧版本在查询期间被释放
340 * router->active → snapshot → configuration（路由和邻居）、nodes（前缀树）
341 */
342 const struct routing_snapshot *snapshot = atomic_load_explicit(&router->active, memory_order_acquire);
343 const struct dppd_ipv4_configuration *configuration = &snapshot->configuration;
344
345 /**
346 * 根节点可保存 /0 默认路由，selected 始终保留路径上最近命中的路由编号
347 * node 保存当前树节点的数组下标；selected 保存路由下标加一，0 表示尚未匹配
348 * 按地址从高位到低位沿 child[0/1] 查询，越深的命中前缀越长
349 * bit / 8 选字节，当前读取地址的第几个字节
350 * 7 - bit % 8 选该字节中的位 把这个字节中的目标位移到最低位，需要右移多少位
351 * & 1U 只保留最低位，得到 0 或 1
352 *
353 * 示例：2001:db8:1::1234 的首字节为 0x20，二进制为 0010 0000
354 * bit = 0、1、2 时依次取到 0、0、1，每次在当前节点选择 child[0]、child[0]、child[1]
355 * bit = 8 时转到 destination[1]，继续从该字节最高位读取
356 */
357 uint32_t node = snapshot->root6, selected = snapshot->nodes[node].route;
358 for (unsigned int bit = 0; bit < 128; ++bit) {
359 node = snapshot->nodes[node].child[(destination[bit / 8] >> (7U - bit % 8)) & 1U];
360 if (node == 0)
361 break;
362 if (snapshot->nodes[node].route != 0)
363 selected = snapshot->nodes[node].route;
364 }
365 if (selected == 0)
366 return -ENOENT;
367
368 /**
369 * 路由编号减一得到数组下标，例如 selected = 3 → routes6[2]
370 * 下一跳选择：
371 * 网关全 0 → 直连，使用报文目的地址
372 * 网关非 0 → 经网关，使用配置的网关地址
373 * 示例：目的 2001:db8:1::1234、网关 fe80::1 → key.address 保存 fe80::1
374 */
375 const struct dppd_ipv6_route *route = &configuration->routes6[selected - 1];
376 struct dppd_ipv6_neighbor key = {
377 .port_id = route->port_id
378 };
379 // 把出口填进去，再确定下一跳 IP
380 const uint8_t zero[16] = {0};
381 /**
382 * 网关地址的 16 个字节是否全为零
383 * 全为零：表示直连，下一跳就是报文目的地址
384 * 非零：下一跳使用路由里配置的网关地址
385 * | 路由配置的网关 | 最终邻居查询条件 |
386 * | `::`，即全零 | 网口 `1` + `2001:db8:1::1234` |
387 * | `fe80::1` | 网口 `1` + `fe80::1` |
388 */
389 memcpy(key.address, memcmp(route->gateway, zero, 16) == 0 ? destination : route->gateway, 16);
390
391 /**
392 * bsearch 在已排序的 neighbors6 中查找，命中返回邻居条目地址，否则返回 NULL
393 * neighbor6_count 为有效条目数，sizeof(key) 为每项大小，按 compare_neighbor6 比较
394 * 比较顺序为出口 → 16 字节 IPv6 地址，两者都相等才命中，与建表排序方式一致
395 * 示例：key 为出口 1、fe80::1，命中后可从 neighbor->mac 取得该下一跳的 MAC
396 */
397
398 /**
399 * | 参数 | 作用 |
400 |---|---|
401 | `&key` | 要查的条件：出口和下一跳 IP |
402 | `neighbors6` | 静态邻居数组 |
403 | `neighbor6_count` | 数组中有效条目的数量 |
404 | `sizeof(key)` | 每条邻居记录占多少字节 |
405 | `compare_neighbor6` | 判断查询条件与条目是否相等 |
406 */
407 const struct dppd_ipv6_neighbor *neighbor =
408 bsearch(&key, configuration->neighbors6, configuration->neighbor6_count, sizeof(key), compare_neighbor6);
409
410 /**
411 * forward 指向 pipeline.c 中的 decision->ipv4，下面直接写入调用方的结果区
412 * 示例：路由出口为 1、下一跳为 fe80::1，但静态邻居表中没有对应记录
413 * generation = 本次配置版本
414 * port_id = 1
415 * next_hop6 = fe80::1
416 * destination_mac = 全 0，尚未取得下一跳 MAC
417 *
418 * 返回 -EHOSTUNREACH → pipeline 写入 DROP / NO_NEIGHBOR → 返回 worker.c/routed_decide
419 * routed_decide 若已启用 ND，再用出口 1 和 fe80::1 查动态邻居表
420 * 因此缺少静态 MAC 时也必须写出出口和下一跳，不能直接丢掉这些查询结果
421 */
422 forward->generation = configuration->generation;
423 forward->port_id = route->port_id;
424 memcpy(forward->next_hop6, key.address, 16);
425 if (neighbor == NULL)
426 return -EHOSTUNREACH;
427
428 memcpy(forward->destination_mac, neighbor->mac, 6);
429 return 0;
430}

432int dppd_ipv4_router_get(struct dppd_ipv4_router *router, struct dppd_ipv4_configuration *configuration)
433{
434 if (router == NULL || configuration == NULL)
435 return -EINVAL;
436 pthread_mutex_lock(&router->writer);
437 *configuration = atomic_load_explicit(&router->active, memory_order_relaxed)->configuration;
438 reclaim(router);
439 pthread_mutex_unlock(&router->writer);
440 return 0;
441}

443int dppd_ipv4_router_replace(struct dppd_ipv4_router *router, uint64_t expected,
444 const struct dppd_ipv4_configuration *configuration,
445 int (*durable)(void *context, const struct dppd_ipv4_configuration *candidate), void *context)
446{
447 if (router == NULL || configuration == NULL || expected == 0)
448 return -EINVAL;
449 pthread_mutex_lock(&router->writer);
450 struct routing_snapshot *old = atomic_load_explicit(&router->active, memory_order_relaxed), *next = NULL;
451 int rc = old->configuration.generation != expected ? -ESTALE : expected == UINT64_MAX ? -EOVERFLOW : 0;
452 reclaim(router);
453 if (rc == 0 && router->retired_count >= DPPD_IPV4_RETIRED_LIMIT)
454 rc = -EAGAIN;
455 if (rc == 0)
456 rc = compile(router, configuration, &next);
457 if (rc == 0) {
458 next->configuration.generation = expected + 1;
459 if (durable != NULL)
460 rc = durable(context, &next->configuration);
461 }
462 if (rc == 0) {
463 atomic_store_explicit(&router->active, next, memory_order_release);
464 old->retired_token = rte_rcu_qsbr_start(router->qsbr);
465 old->retired_next = router->retired;
466 router->retired = old;
467 router->retired_count++;
468 reclaim(router);
469 } else
470 free(next);
471 pthread_mutex_unlock(&router->writer);
472 return rc;
473}
474
475int dppd_ipv4_router_reader_register(struct dppd_ipv4_router *router, unsigned int reader)
476{
477 if (router == NULL || reader >= DPPD_MAX_WORKERS)
478 return -EINVAL;
479 uint64_t bit = UINT64_C(1) << reader;
480 if ((atomic_fetch_or_explicit(&router->readers, bit, memory_order_acq_rel) & bit) != 0)
481 return -EALREADY;
482 if (rte_rcu_qsbr_thread_register(router->qsbr, reader) != 0) {
483 atomic_fetch_and_explicit(&router->readers, ~bit, memory_order_release);
484 return -EINVAL;
485 }
486 rte_rcu_qsbr_thread_online(router->qsbr, reader);
487 return 0;
488}
489
490void dppd_ipv4_router_reader_quiescent(struct dppd_ipv4_router *router, unsigned int reader)
491{
492 if (router != NULL && reader < DPPD_MAX_WORKERS)
493 rte_rcu_qsbr_quiescent(router->qsbr, reader);
494}
495
496void dppd_ipv4_router_reader_unregister(struct dppd_ipv4_router *router, unsigned int reader)
497{
498 if (router == NULL || reader >= DPPD_MAX_WORKERS)
499 return;
500 rte_rcu_qsbr_thread_offline(router->qsbr, reader);
501 rte_rcu_qsbr_thread_unregister(router->qsbr, reader);
502 atomic_fetch_and_explicit(&router->readers, ~(UINT64_C(1) << reader), memory_order_release);
503}

505/**
506 * 按 IPv4 目的地址执行最长前缀匹配，再按出口和下一跳 IP 查询静态邻居
507 * 成功将配置版本、出口、下一跳 IP 和目的 MAC 写入 forward，返回 0
508 * 无路由返回 -ENOENT，无邻居返回 -EHOSTUNREACH，无效参数或地址返回相应负错误码
509 */
510int dppd_ipv4_router_lookup(struct dppd_ipv4_router *router, uint32_t destination_be, struct dppd_ipv4_forward *forward)
511{
512 if (router == NULL || forward == NULL)
513 return -EINVAL;
514 memset(forward, 0, sizeof(*forward));
515 if (!unicast_address(destination_be))
516 return -EAFNOSUPPORT;
517 const struct routing_snapshot *snapshot = atomic_load_explicit(&router->active, memory_order_acquire);
518 const struct dppd_ipv4_configuration *configuration = &snapshot->configuration;
519 uint32_t address = ntohl(destination_be), node = 0, selected = snapshot->nodes[0].route;
520 for (unsigned int bit = 0; bit < 32; ++bit) {
521 node = snapshot->nodes[node].child[(address >> (31U - bit)) & 1U];
522 if (node == 0)
523 break;
524 if (snapshot->nodes[node].route != 0)
525 selected = snapshot->nodes[node].route;
526 }
527 if (selected == 0)
528 return -ENOENT;
529 const struct dppd_ipv4_route *route = &configuration->routes[selected - 1];
530 struct dppd_ipv4_neighbor key = {.port_id = route->port_id,
531 .address_be = route->gateway_be != 0 ? route->gateway_be : destination_be};
532 const struct dppd_ipv4_neighbor *neighbor =
533 bsearch(&key, configuration->neighbors, configuration->neighbor_count, sizeof(key), compare_neighbor);
534 if (neighbor == NULL)
535 return -EHOSTUNREACH;
536 forward->generation = configuration->generation;
537 forward->port_id = route->port_id;
538 forward->next_hop_be = key.address_be;
539 memcpy(forward->destination_mac, neighbor->mac, sizeof(neighbor->mac));
540 return 0;
541}
