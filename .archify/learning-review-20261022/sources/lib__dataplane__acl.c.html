8/** 由配置规则编译出的单条 ACL 匹配项，保存筛选条件和排序信息，动作通过 tag 查找 */
9struct acl_row {
10 /** 规则 ID，group 和 priority 相同时用于排序 */
11 uint64_t id;
12 /** 规则组和组内优先级，匹配同一入口时依次比较，数值小的优先 */
13 uint32_t group, priority;
14 /** 规则对象下标加一，命中后交给软件后端查找动作，也用于最终排序 */
15 uint32_t tag;
16 /** IPv4 源地址及掩码，保留网络字节序，src 已预先与 src_mask 按位与 */
17 uint32_t src, src_mask;
18 /** IPv4 目的地址及掩码，保留网络字节序，dst 已预先与 dst_mask 按位与 */
19 uint32_t dst, dst_mask;
20 /** 规则适用的入口网口 ID，用于定位本入口的规则段 */
21 uint16_t ingress;
22 /**
23 * 源/目的端口条件，含义由 range 决定
24 * false：first 为目标值，last 为掩码，均为网络字节序
25 * true：first 为下界，last 为上界，均为主机字节序，包含两个端点
26 */
27 uint16_t src_first, src_last, dst_first, dst_last;
28 /** 传输层分类，使用 DPPD_L4_UDP/TCP；0 表示不限制传输层类型和端口 */
29 uint8_t l4;
30 /** IP 协议号及掩码，如 TCP 为 6、UDP 为 17；掩码为 0 表示不限制 */
31 uint8_t protocol, protocol_mask;
32 /** 允许的连接状态位集合，0 表示不限制，非零时须与报文状态有交集 */
33 uint8_t connection_states;
34 /** 是否启用对应地址族的匹配，为真时同时校验报文类型和地址条件 */
35 bool ipv4, ipv6;
36 /** 端口是否采用范围匹配，为假时采用掩码匹配 */
37 bool range;
38 /** IPv6 源/目的地址及各自掩码，每个地址或掩码占 16 字节 */
39 struct dppd_match_ipv6 address6;
40};

60/** 相同 ID 的两个版本短暂并存时延续原槽位次序，避免改变已有单规则事务的语义 */
61static bool before(const struct acl_row *a, const struct acl_row *b)
62{
63 if (a->ingress != b->ingress)
64 return a->ingress < b->ingress;
65 if (a->group != b->group)
66 return a->group < b->group;
67 if (a->priority != b->priority)
68 return a->priority < b->priority;
69 if (a->id != b->id)
70 return a->id < b->id;
71 return a->tag < b->tag;
72}

74int dppd_acl_add(struct dppd_acl *acl, const struct dppd_rule *rule, uint32_t tag)
75{
76 char error[128];
77 if (acl == NULL || rule == NULL || tag == 0 || rule->domain != DPPD_RULE_DOMAIN_INGRESS ||
78 dppd_rule_validate(rule, error, sizeof(error)) != 0)
79 return -EINVAL;
80 if (acl->count == acl->capacity)
81 return -ENOSPC;
82 struct acl_row row = {.id = rule->id,
83 .group = rule->group,
84 .priority = rule->priority,
85 .tag = tag,
86 .ingress = rule->install_port_id};
87 for (uint16_t i = 0; i < rule->nb_matches; ++i) {
88 const struct dppd_match *match = &rule->matches[i];
89 switch (match->type) {
90 case DPPD_MATCH_ETH:
91 break;
92 case DPPD_MATCH_IPV4:
93 row.ipv4 = true;
94 row.src_mask = match->spec.ipv4.src_mask_be;
95 row.dst_mask = match->spec.ipv4.dst_mask_be;
96 row.src = match->spec.ipv4.src_be & row.src_mask;
97 row.dst = match->spec.ipv4.dst_be & row.dst_mask;
98 break;
99 case DPPD_MATCH_IP_PROTOCOL:
100 row.protocol = match->spec.protocol.value;
101 row.protocol_mask = match->spec.protocol.mask;
102 break;
103 case DPPD_MATCH_IPV6:
104 row.ipv6 = true;
105 row.address6 = match->spec.ipv6;
106 break;
107 case DPPD_MATCH_CONNECTION_STATE:
108 row.connection_states = (uint8_t)match->spec.connection_states;
109 break;
110 case DPPD_MATCH_UDP:
111 case DPPD_MATCH_TCP:
112 row.l4 = match->type == DPPD_MATCH_UDP ? DPPD_L4_UDP : DPPD_L4_TCP;
113 row.src_first = match->spec.l4.src_be;
114 row.src_last = match->spec.l4.src_mask_be;
115 row.dst_first = match->spec.l4.dst_be;
116 row.dst_last = match->spec.l4.dst_mask_be;
117 break;
118 case DPPD_MATCH_UDP_RANGE:
119 case DPPD_MATCH_TCP_RANGE:
120 row.l4 = match->type == DPPD_MATCH_UDP_RANGE ? DPPD_L4_UDP : DPPD_L4_TCP;
121 row.range = true;
122 row.src_first = ntohs(match->spec.range.src_first_be);
123 row.src_last = ntohs(match->spec.range.src_last_be);
124 row.dst_first = ntohs(match->spec.range.dst_first_be);
125 row.dst_last = ntohs(match->spec.range.dst_last_be);
126 break;
127 default:
128 return -ENOTSUP;
129 }
130 }
131 uint32_t position = 0;
132 while (position < acl->count && before(&acl->rows[position], &row))
133 ++position;
134 memmove(acl->rows + position + 1, acl->rows + position, (acl->count - position) * sizeof(row));
135 acl->rows[position] = row;
136 acl->count++;
137 return 0;
138}

140/**
141 * 检查当前报文是否满足一条 ACL 规则，入口已由调用方筛选
142 * 依次检查连接状态、IP 类型/地址/协议及传输层条件，任一不符返回 false
143 * 满足全部条件返回 true，仅表示规则命中，不代表允许转发
144 * row: 待匹配的 ACL 规则行
145 * packet: 待匹配的报文
146 * 返回: 若报文满足规则返回 true，否则返回 false
147 * 返回 false 表示这条规则不匹配，外面继续检查下一条规则；并不表示丢包
148 */
149static bool matches(const struct acl_row *row, const struct dppd_packet *packet)
150{
151 /** 状态条件为 0 时不限，否则报文状态必须属于规则允许的状态集合 */
152 if (row->connection_states != 0 && (row->connection_states & packet->connection_state) == 0)
153 return false;
154 /**
155 * 规则地址已在 dppd_acl_add 中执行 地址 & 掩码，报文地址用同一掩码后比较
156 * 掩码全 0 → 地址不限；全 1 → 完整地址相等；其余仅比较掩码指定的位
157 * 地址不限仍须满足 IPv4 类型，地址和掩码均按网络字节序直接做位运算
158 */
159 if (row->ipv4 &&
160 (packet->l3_type != DPPD_L3_IPV4 || (packet->ipv4_src_be & row->src_mask) != row->src ||
161 (packet->ipv4_dst_be & row->dst_mask) != row->dst))
162 return false;
163 /** IP 协议号按掩码比较，掩码为 0 时不限，为 0xff 时要求协议号完全相等 */
164 if ((packet->ip_protocol & row->protocol_mask) != (row->protocol & row->protocol_mask))
165 return false;
166 /** 仅配置 IPv6 地址条件时进入，按字节比较源/目的地址中掩码指定的位 */
167 if (row->ipv6) {
168 if (packet->l3_type != DPPD_L3_IPV6)
169 return false;
170 for (unsigned int i = 0; i < 16; ++i)
171 if ((packet->ipv6_src[i] & row->address6.src_mask[i]) !=
172 (row->address6.src[i] & row->address6.src_mask[i]) ||
173 (packet->ipv6_dst[i] & row->address6.dst_mask[i]) != (row->address6.dst[i] & row->address6.dst_mask[i]))
174 return false;
175 }
176 /**
177 * 前面的条件已通过，未配置传输层条件即可命中，否则先核对 UDP/TCP 类型
178 * 分片分类为 FRAGMENT，无法通过 UDP/TCP 类型检查，不会用空端口误匹配
179 */
180 if (row->l4 == 0)
181 return true;
182 if (packet->l4_type != row->l4)
183 return false;
184 /**
185 * 掩码模式：first 保存目标端口，last 保存掩码，均保留网络字节序
186 * 掩码为 0 时该端口不限，为 0xffff 时要求端口完全相等，源和目的条件须同时满足
187 * R1 示例：源端口掩码为 0，目的端口值为 9000 且掩码为 0xffff
188 */
189 if (!row->range)
190 return (packet->l4_src_port_be & row->src_last) == (row->src_first & row->src_last) &&
191 (packet->l4_dst_port_be & row->dst_last) == (row->dst_first & row->dst_last);
192 /** 范围模式：转为主机字节序后与预存上下界比较，源和目的范围均包含两个端点 */
193 uint16_t source = ntohs(packet->l4_src_port_be), destination = ntohs(packet->l4_dst_port_be);
194 return source >= row->src_first && source <= row->src_last && destination >= row->dst_first &&
195 destination <= row->dst_last;
196}

198/**
199 * 在 ACL 表中匹配入口和报文字段，支持掩码、端口范围及连接状态等条件
200 * 二分定位入口对应的规则段，再按优先顺序检查，返回首条命中规则的 tag
201 * 未命中或参数无效返回 0，仅查找候选，不执行规则动作
202 */
203uint32_t dppd_acl_classify(const struct dppd_acl *acl, uint16_t ingress, const struct dppd_packet *packet)
204{
205 if (acl == NULL || packet == NULL)
206 return 0;
207 /**
208 * 写入时先按 ingress 排序，同入口规则连续存放，内部按规则优先顺序排列
209 * 在 [first, last) 内二分，找到首个 ingress >= 目标入口的位置
210 * 入口示例：[0, 0, 1, 1, 1, 3]，查入口 1 → first = 2
211 */
212 uint32_t first = 0, last = acl->count;
213 while (first < last) {
214 uint32_t middle = first + (last - first) / 2;
215 if (acl->rows[middle].ingress < ingress)
216 first = middle + 1;
217 else
218 last = middle;
219 }
220 /** 只遍历目标入口的规则段，首条命中即返回，不再检查后续规则 */
221 for (; first < acl->count && acl->rows[first].ingress == ingress; ++first)
222 if (matches(&acl->rows[first], packet))
223 return acl->rows[first].tag;
224 return 0;
