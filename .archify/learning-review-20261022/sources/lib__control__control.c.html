47static void observe_transaction(struct dppd_control_service *service, const struct dppd_transaction *transaction)
48{
49 if (transaction->failure_code != 0 && transaction->failure_item < transaction->nb_items) {
50 const struct dppd_transaction_item *item = &transaction->items[transaction->failure_item];
51
52 observe_rule(service, transaction->failure_stage, &item->rule, item->plan.install_port_id, &item->plan);
53 service->observation.context.transaction_id = transaction->id;
54 dppd_rule_observation_fault(&service->observation, transaction->failure_code);
55 }

57 uint64_t id = transaction->rollback_item < transaction->nb_items
58 ? transaction->items[transaction->rollback_item].rule.id
59 : 0;
60
61 dppd_rule_observation_compensation(&service->observation, transaction->rollback_code, id);
62 }

85/**
86 * 把当前完整规则账本写入快照，未配置保存路径时直接成功返回
87 * 保存发生在规则生效之后，写磁盘失败不能被解释为已经回滚规则
88 * 此时设置 dirty 并返回 EUCLEAN，后续修改必须先补写成功，避免磁盘与内存越差越远
89 * 后端规则已生效 + 内存账本已更新 → 尝试写文件
90 * 未配置路径 → 跳过保存，返回 0
91 * 保存成功 → 清除 dirty，更新 persisted_generation，返回 0
92 * 保存失败 → 保留已生效规则和内存账本，设置 dirty，返回 -EUCLEAN
93 */
94static int persist_current_repository(struct dppd_control_service *service)
95{
96 struct dppd_rule_failure_event previous = service->observation.context;
97 int rc;
98
99 if (service->persistence_path == NULL)
100 return 0;
101 observe_stage(service, DPPD_RULE_STAGE_PERSIST);
102 if (service->observation.context.rule_count > 1)
103 observe_batch(service, DPPD_RULE_STAGE_PERSIST);
104 rc = save_current_snapshot(service);
105 if (rc != 0) {
106 dppd_rule_observation_fault(&service->observation, rc);
107 service->persistence_dirty = true;
108 service->persistence_last_error = rc;
109 return -EUCLEAN;
110 }
111 service->persistence_dirty = false;
112 service->persistence_last_error = 0;
113 service->persisted_generation = dppd_rule_repository_generation(&service->rules);
114 service->observation.context = previous;
115 return 0;
116}

118/** 新修改开始前检查上次保存是否失败，必要时先补写当前已经生效的完整状态 */
119static int persistence_write_preflight(struct dppd_control_service *service)
120{
121 /**
122 * clean 状态不重复保存，dirty 状态先修复上次未完成的保存
123 * 修复失败就拒绝本次修改，这里不会通过丢弃内存中的规则来消除 dirty
124 */
125 return service->persistence_dirty ? persist_current_repository(service) : 0;
126}

142int dppd_control_shared_count_create(struct dppd_control_service *service, const struct dppd_shared_count_spec *spec,
143 struct dppd_shared_count_info *result)
144{
145 if (service == NULL || spec == NULL || result == NULL || spec->id == 0 || spec->id == UINT64_MAX ||
146 spec->generation != 0 || spec->install_port_id == UINT16_MAX || spec->domain < DPPD_RULE_DOMAIN_INGRESS ||
147 spec->domain > DPPD_RULE_DOMAIN_TRANSFER)
148 return -EINVAL;
149 int rc = recovery_write_preflight(service);
150 if (rc != 0)
151 return rc;
152 rc = persistence_write_preflight(service);
153 if (rc != 0)
154 return rc;
155 if (service->shared_count_revision >= UINT64_MAX - 1U)

157 const struct dppd_endpoint *endpoint = dppd_topology_find(service->topology, spec->install_port_id);
158 if (endpoint == NULL)
159 return -ENOENT;

162 return -ENOTSUP;
163 struct dppd_shared_count_spec candidate = *spec;
164 candidate.generation = service->shared_count_revision + 1U;

167 if (rc != 0)

169 service->shared_count_revision = candidate.generation;

172 return rc;

253 }

258/**
259 * 找出一条已发布规则究竟安装在硬件还是软件中，规则账本本身不记录后端类型
260 * PREFER 只表达“优先硬件”的意图，驱动校验失败后实际规则可能已经落到软件后端
261 * 因此要在两个对象仓库中按同一个 ID 和版本查找，不能只根据策略字段猜测
262 * 正常情况恰好找到一个；两个都有或两个都没有，都说明控制层约定已经被破坏
263 */
264static int actual_backend_for(const struct dppd_control_service *service, uint64_t rule_id, uint64_t generation,
265 enum dppd_plan_backend *backend_kind)
266{
267 bool has_hardware;
268 bool has_software;
269
270 if (service == NULL || backend_kind == NULL)
271 return -EINVAL;
272 has_hardware = dppd_rte_flow_backend_find_version(&service->rte_flow, rule_id, generation) != NULL;
273 has_software = dppd_software_backend_contains_version(&service->software, rule_id, generation);
274 if (has_hardware == has_software)
275 return -EUCLEAN;
276 *backend_kind = has_hardware ? DPPD_PLAN_BACKEND_RTE_FLOW : DPPD_PLAN_BACKEND_SOFTWARE;
277 return 0;
278}

303/**
304 * 恢复批量删除或更新过程中已经删掉的那一部分旧对象
305 * 账本尚未变化，所以要保留原 ID、版本、端口和后端，不能调用普通 apply 分配一个新版本
306 * 这里通过创建事务重建对象，count 为零表示还没有删掉任何旧对象，无需补偿
307 * 恢复失败时把错误交回上层，由上层决定隔离，不能假装账本中的规则已经重新存在
308 */
309static int restore_actual_rules(struct dppd_control_service *service, struct dppd_transaction_item *items,
310 uint32_t count)
311{
312 struct dppd_transaction transaction = {0};
313 struct dppd_transaction_backends backends;
314 uint32_t i;
315 int rc;
316
317 if (count == 0)
318 return 0;
319 memset(&backends, 0, sizeof(backends));
320 backends.software = dppd_software_transaction_backend(&service->software);
321 backends.rte_flow = dppd_rte_flow_transaction_backend(&service->rte_flow);
322 for (i = 0; i < count; ++i) {
323 items[i].plan.rule_id = items[i].rule.id;
324 items[i].plan.rule_generation = items[i].rule.generation;
325 items[i].plan.install_port_id = items[i].rule.install_port_id;
326 }
327 rc = dppd_transaction_init(&transaction, service->next_transaction_id++, items, count);
328 if (rc == 0)
329 rc = dppd_transaction_run(&transaction, &backends);
330 if (rc == 0)
331 rc = dppd_transaction_finalize(&transaction, &backends);
332 if (rc != 0) {
333 uint64_t id = transaction.failure_item < count ? items[transaction.failure_item].rule.id : items[0].rule.id;
334
335 dppd_rule_observation_compensation(&service->observation, rc, id);
336 }
337 return rc;
338}

340int dppd_control_init(struct dppd_control_service *service, const struct dppd_topology *topology,
341 uint32_t rule_capacity, const struct dppd_flow_api *flow_api)
342{
343 int rc;
344
345 if (service == NULL || topology == NULL || rule_capacity == 0 || rule_capacity == UINT32_MAX ||
346 topology->nb_endpoints > DPPD_MAX_PORTS)
347 return -EINVAL;
348 memset(service, 0, sizeof(*service));
349 rc = dppd_rule_repository_init(&service->rules, rule_capacity);
350 if (rc != 0)
351 return rc;
352 /**
353 * 后端比规则账本多一个槽位，确保满表时仍可进行一次单规则的先建后删
354 * 批量更新需要同时容纳多条新版本，仅多一个槽位并不保证满表时也能完成批量更新
355 */
356 rc = dppd_rte_flow_backend_init(&service->rte_flow, rule_capacity + 1U, flow_api);
357 if (rc != 0) {
358 dppd_rule_repository_destroy(&service->rules);
359 return rc;
360 }
361 /** 启动时登记拓扑内的全部端口，画像查询即使尚无规则也能显示空的校验记录 */
362 for (uint16_t i = 0; i < topology->nb_endpoints; ++i) {
363 rc = dppd_flow_probe_cache_register(service->rte_flow.probes, topology->endpoints[i].ethdev_port_id);
364 if (rc != 0) {
365 (void)dppd_rte_flow_backend_fini(&service->rte_flow);
366 dppd_rule_repository_destroy(&service->rules);
367 return rc;
368 }
369 }
370 rc = dppd_software_backend_init(&service->software, rule_capacity + 1U);
371 if (rc != 0) {
372 (void)dppd_rte_flow_backend_fini(&service->rte_flow);
373 dppd_rule_repository_destroy(&service->rules);
374 return rc;
375 }
376 service->topology = topology;
377 service->next_transaction_id = 1;
378 return 0;
379}

381/**
382 * 先检查完整规则和端口拓扑，再做无安装副作用的驱动探测
383 * 硬件探测会忽略用户的后端偏好，因此 software 规则也能询问相同内容的驱动支持情况
384 */
385int dppd_control_probe_rule(struct dppd_control_service *service, uint16_t install_port_id,
386 const struct dppd_rule *rule, bool refresh, struct dppd_flow_probe_result *result)
387{
388 struct dppd_planner_context context = {0};
389 struct dppd_execution_plan plan;
390 struct dppd_rule candidate;
391 char error[128];
392 int rc;
393
394 if (service == NULL || rule == NULL || result == NULL)
395 return -EINVAL;
396 memset(result, 0, sizeof(*result));
397 if (recovery_write_preflight(service) != 0)
398 return -EUCLEAN;
399 if (rule->id == 0 || dppd_rule_validate(rule, error, sizeof(error)) != 0)
400 return -EINVAL;
401 candidate = *rule;
402 candidate.install_port_id = install_port_id;
403 candidate.generation = 0;
404 candidate.fallback = DPPD_FALLBACK_REQUIRE_HARDWARE;
405 context.topology = service->topology;
406 context.ipv4_routing = service->routing != NULL;
407 context.connection_tracking = service->conntrack != NULL;
408 context.neighbor_discovery = service->nd6 != NULL;
409 context.install_port_id = install_port_id;
410 context.hardware_available = true;
411 rc = dppd_plan_rule(&context, &candidate, &plan);
412 if (rc != 0)
413 return rc;
414 rc = dppd_rte_flow_backend_driver_ready(&service->rte_flow);
415 if (rc != 0)
416 return rc;
417 rc = dppd_flow_probe_cache_query(service->rte_flow.probes, install_port_id, &candidate, refresh,
418 service->rte_flow.api.validate, result);
419 if (dppd_rte_flow_backend_driver_ready(&service->rte_flow) != 0) {
420 memset(result, 0, sizeof(*result));
421 return -EUCLEAN;
422 }
423 if (rc == 0)
424 result->software_equivalent = dppd_software_backend_rule_supported(&candidate);
425 return rc;
426}

429{
430 int persistence_rc = 0;
431 int rc;
432
433 if (service == NULL)
434 return -EINVAL;
435 if (service->routing != NULL) {
436 struct dppd_routing_status status;
437 dppd_routing_status(service->routing, &status);
438 rc = dppd_routing_close(service->routing);
439 if (rc != 0)
440 return rc;
441 service->routing = NULL;
442 if (status.dirty)
443 persistence_rc = -EUCLEAN;
444 }
445 if (service->recovery_state == DPPD_CONTROL_RECOVERY_READY && service->persistence_dirty) {
446 int flush_error = control_flush_impl(service);
447 if (persistence_rc == 0)
448 persistence_rc = flush_error;
449 }
450 rc = dppd_rte_flow_backend_fini(&service->rte_flow);
451 if (rc != 0)
452 return rc;
453 dppd_software_backend_fini(&service->software);
454 dppd_control_reset_plan_destroy(service);
455 dppd_rule_repository_destroy(&service->rules);
456 free(service->persistence_path);
457 memset(service, 0, sizeof(*service));
458 return persistence_rc;
459}

461int dppd_control_persistence_attach(struct dppd_control_service *service, const char *path)
462{
463 char *copy;
464 uint64_t current_generation;
465
466 if (service == NULL || path == NULL || path[0] == '\0' || service->rules.records == NULL)
467 return -EINVAL;
468 if (service->persistence_path != NULL)
469 return -EALREADY;
470 if (recovery_write_preflight(service) != 0)
471 return -EUCLEAN;
472 copy = strdup(path);
473 if (copy == NULL)
474 return -ENOMEM;
475 service->persistence_path = copy;
476 current_generation = dppd_rule_repository_generation(&service->rules);
477 /*
478 * attach 本身不读取也不写入文件，因此不能凭空声称非零 generation 已落盘
479 * 如果调用方刚完成 snapshot 恢复，保守地标记 dirty 仍然安全：显式 flush 或下次
480 * mutation 的 preflight 会重新保存一次当前完整状态，再允许新的变更继续
481 */
482 service->persisted_generation = 0;
483 service->persistence_dirty = current_generation != 0 || service->shared_count_revision != 0;
484 service->persistence_last_error = 0;
485 return 0;
486}

490 struct dppd_persisted_snapshot snapshot;
491 struct dppd_transaction_item *items = NULL;
492 struct dppd_transaction transaction;
493 struct dppd_transaction_backends backends;
494 char *path_copy = NULL;
495 uint32_t i;

497 int rc;
498 bool shared_touched = false, published = false;
499
500 if (service == NULL || service->topology == NULL || path == NULL || path[0] == '\0' ||

503 if (recovery_write_preflight(service) != 0)
504 return -EUCLEAN;
505 if (service->persistence_path != NULL || dppd_rule_repository_count(&service->rules) != 0 ||
506 dppd_rule_repository_generation(&service->rules) != 0 || dppd_rte_flow_backend_count(&service->rte_flow) != 0 ||
507 dppd_software_backend_count(&service->software) != 0 || service->shared_count_revision != 0 ||
508 dppd_rte_flow_backend_shared_count_count(&service->rte_flow) != 0)
509 return -EBUSY;
510
511 observe_stage(service, DPPD_RULE_STAGE_LOAD);
512 rc = dppd_persistence_load(path, &snapshot);
513 if (rc == -ENOENT) {
514 /*
515 * 显式配置了 state path 就要求启动时建立可持久化的空基线。若目录或权限
516 * 不可用，daemon 直接启动失败，不能悄悄退化成仅内存模式
517 */
518 observe_stage(service, DPPD_RULE_STAGE_PREFLIGHT);
519 rc = dppd_control_persistence_attach(service, path);
520 if (rc != 0)
521 return rc;
522 return control_flush_impl(service);
523 }
524 if (rc != 0)
525 return rc;
526 observe_stage(service, DPPD_RULE_STAGE_PREFLIGHT);
527 service->observation.context.rule_count = snapshot.count;
528 if (snapshot.count > service->rules.capacity) {
529 rc = -ENOSPC;
530 goto cleanup;
531 }
532
533 /* 在接触硬件前完成所有本地内存分配，避免提交后因 ENOMEM 进入模糊状态 */
534 observe_stage(service, DPPD_RULE_STAGE_PREPARE);
535 path_copy = strdup(path);
536 if (path_copy == NULL) {
537 rc = -ENOMEM;
538 goto cleanup;
539 }
540 if (snapshot.count != 0) {
541 items = calloc(snapshot.count, sizeof(*items));
542 if (items == NULL) {
543 rc = -ENOMEM;
544 goto cleanup;
545 }
546 }
547
548 /** 先重建共享对象，后面的规则才能按精确身份解析引用，任一步失败都不能发布半张表 */
549 for (i = 0; i < snapshot.shared_count; ++i) {
550 struct dppd_flow_error error;
551 const struct dppd_endpoint *endpoint =
552 dppd_topology_find(service->topology, snapshot.shared_counts[i].install_port_id);
553 if (endpoint == NULL) {
554 rc = -ENOENT;
555 goto cleanup;
556 }
557 if (snapshot.shared_counts[i].domain == DPPD_RULE_DOMAIN_TRANSFER &&
558 (!endpoint->has_switch_domain || endpoint->switch_domain_id == UINT16_MAX)) {
559 rc = -ENOTSUP;
560 goto cleanup;
561 }
562 shared_touched = true;
563 rc = dppd_rte_flow_backend_shared_count_create(&service->rte_flow, &snapshot.shared_counts[i], &error);
564 if (rc != 0)
565 goto cleanup;
566 }
567 for (i = 0; i < snapshot.count; ++i) {
568 struct dppd_planner_context planner_context;
569
570 memset(&planner_context, 0, sizeof(planner_context));
571 planner_context.topology = service->topology;
572 planner_context.ipv4_routing = service->routing != NULL;
573 planner_context.connection_tracking = service->conntrack != NULL;
574 planner_context.neighbor_discovery = service->nd6 != NULL;
575 planner_context.install_port_id = snapshot.rules[i].install_port_id;
576 planner_context.hardware_available = true;
577 planner_context.software_equivalent = dppd_software_backend_rule_supported(&snapshot.rules[i]);
578 items[i].rule = snapshot.rules[i];
579 observe_rule(service, DPPD_RULE_STAGE_PLAN, &items[i].rule, planner_context.install_port_id, NULL);
580 rc = dppd_plan_rule(&planner_context, &items[i].rule, &items[i].plan);
581 if (rc != 0)
582 goto cleanup;
583 if (items[i].plan.backend == DPPD_PLAN_BACKEND_RTE_FLOW &&
584 items[i].rule.fallback == DPPD_FALLBACK_PREFER_HARDWARE && planner_context.software_equivalent) {
585 struct dppd_flow_error flow_error;
586
587 /*
588 * 恢复事务必须在进入 prepare 前完成每条可回退规则的 backend 选择
589 * 这里额外 validate 一次仅作无副作用的能力探测；失败后整个批次仍可
590 * 与其他硬件规则一起原子提交，不会因为 net_ring 等 PMD 而拒绝启动
591 */
592 rc = dppd_rte_flow_backend_validate_rule(&service->rte_flow, items[i].plan.install_port_id, &items[i].rule,
593 &flow_error);
594 if (rc != 0) {
595 planner_context.hardware_available = false;
596 rc = dppd_plan_rule(&planner_context, &items[i].rule, &items[i].plan);
597 if (rc != 0)
598 goto cleanup;
599 service->observation.fallback_rules++;
600 }
601 }
602 }
603
604 memset(&transaction, 0, sizeof(transaction));
605 memset(&backends, 0, sizeof(backends));
606 backends.software = dppd_software_transaction_backend(&service->software);
607 backends.rte_flow = dppd_rte_flow_transaction_backend(&service->rte_flow);
608 if (snapshot.count != 0) {
609 rc = dppd_transaction_init(&transaction, service->next_transaction_id++, items, snapshot.count);
610 if (rc != 0)
611 goto cleanup;
612 /* 全量 validate/prepare/commit；任一失败由 transaction 自动逆序回滚 */
613 rc = dppd_transaction_run(&transaction, &backends);
614 if (rc != 0) {
615 observe_transaction(service, &transaction);
616 if (transaction.rollback_code != 0) {
617 /*
618 * transaction 已尽力回滚，但 backend 仍持有可定位对象。保留 service
619 * 和 handle，交给恢复隔离模式中的显式 retry；不能继续向下发布规则
620 */
621 service->recovery_state = DPPD_CONTROL_RECOVERY_RECONCILIATION_REQUIRED;
622 service->recovery_last_error = transaction.rollback_code;
623 rc = -EUCLEAN;
624 }
625 goto cleanup;
626 }
627 }
628
629 observe_batch(service, DPPD_RULE_STAGE_PUBLISH);
630 rc = dppd_rule_repository_restore(&service->rules, snapshot.rules, snapshot.count, snapshot.repository_generation);
631 if (rc != 0) {
632 dppd_rule_observation_fault(&service->observation, rc);
633 /*
634 * restore 已全量预检且不分配，正常不应失败；仍保留防御性硬件回滚，避免
635 * 未来 repository 实现变化时留下“硬件已装、desired 未发布”的对象
636 */
637 if (snapshot.count != 0) {
638 rollback_rc = dppd_transaction_rollback_committed(&transaction, &backends);
639 if (rollback_rc != 0)
640 rc = -EUCLEAN;
641 observe_transaction(service, &transaction);
642 }
643 goto cleanup;
644 }
645 published = true;
646 service->shared_count_revision = snapshot.shared_revision;
647 service->observation.applied =
648 snapshot.count != 0 || snapshot.repository_generation != 0 || snapshot.shared_revision != 0;
649 if (snapshot.count != 0) {
650 rc = dppd_transaction_finalize(&transaction, &backends);
651 if (rc != 0)
652 goto cleanup;
653 }
654
655 /* 文件正是本次恢复来源，因此可以确认磁盘与内存 generation 完全一致 */
656 service->persistence_path = path_copy;
657 path_copy = NULL;
658 service->persisted_generation = snapshot.repository_generation;
659 service->persistence_dirty = false;
660 service->persistence_last_error = 0;
661 rc = 0;
662
663cleanup:
664 /** 规则回滚成功后才能撤销共享对象，残留引用或删除失败必须继续隔离并保留句柄 */
665 if (rc != 0 && shared_touched && !published && service->recovery_state == DPPD_CONTROL_RECOVERY_READY) {
666 uint32_t residual;
667 int cleanup_rc = dppd_rte_flow_backend_reconcile(&service->rte_flow, &residual);
668 if (cleanup_rc != 0 || residual != 0)
669 rc = isolate_control(service, cleanup_rc != 0 ? cleanup_rc : -EUCLEAN);
670 }
671 if (rc != 0 && published)
672 rc = isolate_control(service, rc);
673 free(path_copy);
674 free(items);
675 dppd_persisted_snapshot_destroy(&snapshot);
676 return rc;
677}

679static int control_flush_impl(struct dppd_control_service *service)
680{
681 int rc;
682
683 if (service == NULL || service->persistence_path == NULL)
684 return -EINVAL;
685 if (recovery_write_preflight(service) != 0)
686 return -EUCLEAN;
687 observe_stage(service, DPPD_RULE_STAGE_PERSIST);
688 rc = save_current_snapshot(service);
689 if (rc != 0) {
690 dppd_rule_observation_fault(&service->observation, rc);

693 return rc;
694 }
695 service->persistence_dirty = false;
696 service->persistence_last_error = 0;
697 service->persisted_generation = dppd_rule_repository_generation(&service->rules);
698 return 0;
699}

730static int control_reconcile_impl(struct dppd_control_service *service)
731{
732 uint32_t residual_objects;
733 int rc;
734
735 if (service == NULL)
736 return -EINVAL;
737 /** 模拟复位保留的是完整恢复计划，不能用普通清理入口把它变成等待进程重启 */
738 if (service->reset_plan != NULL)
739 return -EBUSY;
740 if (service->recovery_state == DPPD_CONTROL_RECOVERY_READY ||
741 service->recovery_state == DPPD_CONTROL_RECOVERY_RESTART_REQUIRED)
742 return -EALREADY;
743
744 observe_stage(service, DPPD_RULE_STAGE_RECONCILE);
745 rc = dppd_rte_flow_backend_reconcile(&service->rte_flow, &residual_objects);
746 if (rc != 0 || residual_objects != 0) {
747 dppd_rule_observation_fault(&service->observation, rc != 0 ? rc : -EUCLEAN);
748 /*
749 * remove 失败时仍保留对象和其 handle，下一次 retry 可以继续调用同一个 PMD
750 * destroy。API 返回 EUCLEAN，具体底层 errno 通过 status.last_error 提供
751 */
752 service->recovery_last_error = rc != 0 ? rc : -EUCLEAN;
753 return -EUCLEAN;
754 }
755 service->recovery_last_error = 0;
756 service->recovery_state = DPPD_CONTROL_RECOVERY_RESTART_REQUIRED;
757 return 0;
758}

760/**
761 * 创建或更新一条规则，依次完成版本检查、后端安装、内存账本更新和配置保存
762 * rule 提供规则内容，install_port_id 指定安装端口，expected_generation 用于检查版本冲突
763 * result 返回执行计划、事务编号及规则状态和版本；返回 0 表示本次请求成功，负值表示错误
764 *
765 * 新增软件规则的成功路径，假设规划结果选中软件后端：
766 * candidate → item.rule，执行计划 → item.plan
767 * → dppd_transaction_run → 软件 prepare / commit → active 已发布新规则
768 * → dppd_rule_repository_apply → service->rules 登记规则内容和版本
769 * → dppd_transaction_finalize → 释放 pending 准备记录，保留已安装规则
770 * → persist_current_repository → 配置了路径时保存当前账本
771 *
772 * worker 查软件后端的 active 快照；控制面用 service->rules 查询、管理和保存规则
773 * 后端安装成功不等于账本已更新或文件已保存，保存失败也不会撤回已生效规则
774 */
775static int control_apply_impl(struct dppd_control_service *service, uint16_t install_port_id,
776 const struct dppd_rule *rule, uint64_t expected_generation, struct dppd_control_apply_result *result)
777{
778 struct dppd_rule existing;
779 struct dppd_rule candidate;
780 struct dppd_planner_context planner_context;
781 struct dppd_transaction_item item;
782 struct dppd_transaction transaction;
783 struct dppd_transaction_backends backends;
784 struct dppd_rule_apply_result repository_result;
785 bool updating = false;
786 int rc;
787
788 if (service == NULL || service->topology == NULL || rule == NULL || rule->id == 0 || result == NULL)
789 return -EINVAL;
790 memset(result, 0, sizeof(*result));
791 rc = recovery_write_preflight(service);
792 if (rc != 0)
793 return rc;
794 rc = persistence_write_preflight(service);
795 if (rc != 0)
796 return rc;
797
798 /* install_port_id 是持久化 desired state 的一部分，由 API 参数统一覆盖客户端值 */
799 candidate = *rule;
800 candidate.install_port_id = install_port_id;
801 rc = dppd_rule_repository_get(&service->rules, rule->id, &existing);
802 if (rc == 0) {
803 if (dppd_rule_equal(&existing, &candidate)) {
804 result->status = DPPD_RULE_UNCHANGED;
805 result->generation = existing.generation;
806 service->observation.unchanged = true;
807 return 0;
808 }
809 if (expected_generation != DPPD_RULE_GENERATION_ANY && expected_generation != existing.generation)
810 return -ESTALE;
811 updating = true;
812 } else if (rc != -ENOENT) {
813 return rc;
814 } else {
815 if (expected_generation != DPPD_RULE_GENERATION_ANY && expected_generation != 0)
816 return -ESTALE;
817 if (dppd_rule_repository_count(&service->rules) >= service->rules.capacity)
818 return -ENOSPC;
819 }
820
821 /** 版本耗尽时先拒绝，不能等实际对象安装后才发现账本无法接收新版本 */
822 if (dppd_rule_repository_generation(&service->rules) >= UINT64_MAX - 1U)
823 return -EOVERFLOW;
824 candidate.generation = dppd_rule_repository_generation(&service->rules) + 1U;
825 planner_context.topology = service->topology;
826 planner_context.ipv4_routing = service->routing != NULL;
827 planner_context.connection_tracking = service->conntrack != NULL;
828 planner_context.neighbor_discovery = service->nd6 != NULL;
829 planner_context.install_port_id = install_port_id;
830 planner_context.hardware_available = true;
831 planner_context.software_equivalent = dppd_software_backend_rule_supported(&candidate);
832 observe_rule(service, DPPD_RULE_STAGE_PLAN, &candidate, install_port_id, NULL);
833 rc = dppd_plan_rule(&planner_context, &candidate, &result->plan);
834 if (rc != 0)
835 return rc;
836
837 memset(&item, 0, sizeof(item));
838 item.rule = candidate;
839 item.plan = result->plan;
840 memset(&backends, 0, sizeof(backends));
841 backends.software = dppd_software_transaction_backend(&service->software);
842 backends.rte_flow = dppd_rte_flow_transaction_backend(&service->rte_flow);
843 result->transaction_id = service->next_transaction_id++;
844 service->observation.context.transaction_id = result->transaction_id;
845 rc = dppd_transaction_init(&transaction, result->transaction_id, &item, 1);
846 if (rc != 0)
847 return rc;
848 rc = dppd_transaction_run(&transaction, &backends);
849 if (rc != 0 && candidate.fallback == DPPD_FALLBACK_PREFER_HARDWARE &&
850 transaction.state == DPPD_TRANSACTION_FAILED && item.state == DPPD_TRANSACTION_ITEM_FAILED &&
851 item.backend_token == 0 && dppd_software_backend_rule_supported(&candidate)) {
852 /*
853 * 只在 validate 尚未预留/创建对象时回退，绝不把 create 或 rollback 的
854 * 不确定失败伪装为 software 成功。这样 REQUIRE_HARDWARE 仍是硬边界
855 * PREFER_HARDWARE 也只在已证明可等价执行时才改变 backend
856 */
857 planner_context.hardware_available = false;
858 planner_context.software_equivalent = true;
859 rc = dppd_plan_rule(&planner_context, &candidate, &result->plan);
860 if (rc != 0)
861 return rc;
862 service->observation.fallback_rules++;
863 item.plan = result->plan;
864 rc = dppd_transaction_init(&transaction, result->transaction_id, &item, 1);
865 if (rc != 0)
866 return rc;
867 rc = dppd_transaction_run(&transaction, &backends);
868 }
869 if (rc != 0) {
870 observe_transaction(service, &transaction);
871 if (transaction.rollback_code != 0)
872 return isolate_control(service, transaction.rollback_code);
873 return rc;
874 }
875
876 if (updating) {
877 int rollback_rc;
878
879 /*
880 * 新 generation 已创建后才删除旧对象；旧对象删除失败时撤销新对象
881 * repository 仍指向旧 generation，因此控制面不会发布半更新状态
882 */
883 observe_rule(service, DPPD_RULE_STAGE_REMOVE, &existing, existing.install_port_id, NULL);
884 rc = remove_actual_rule(service, existing.id, existing.generation);
885 if (rc != 0) {
886 dppd_rule_observation_fault(&service->observation, rc);
887 rollback_rc = remove_actual_rule(service, candidate.id, candidate.generation);
888 dppd_rule_observation_compensation(&service->observation, rollback_rc, candidate.id);
889 return rollback_rc == 0 ? rc : isolate_control(service, rollback_rc);
890 }
891 }
892
893 /**
894 * 到这里后端已安装新版本，更新操作也已删除旧版本，现在登记控制面的内存账本
895 * 软件 active 快照 → 供 worker 匹配报文
896 * service->rules → 记录受管理规则的内容和版本，供控制面查询及持久化
897 * repository_apply 本身不安装规则；此处若失败则账本与实际后端不一致，进入恢复隔离
898 */
899 observe_rule(service, DPPD_RULE_STAGE_PUBLISH, &candidate, install_port_id, &result->plan);
900 rc = dppd_rule_repository_apply(&service->rules, &candidate, updating ? existing.generation : expected_generation,
901 &repository_result);
902 if (rc != 0) {
903 /* 单控制线程下不应到达这里；到达即表示 actual/desired 已失配 */
904 dppd_rule_observation_fault(&service->observation, rc);
905 return isolate_control(service, rc);
906 }
907 service->observation.applied = true;
908 result->status = repository_result.status;
909 result->generation = repository_result.generation;
910 rc = dppd_transaction_finalize(&transaction, &backends);
911 if (rc != 0) {
912 dppd_rule_observation_fault(&service->observation, rc);
913 return isolate_control(service, rc);
914 }
915 return persist_current_repository(service);
916}
917
918/**
919 * 一次创建一组原先不存在的规则，避免逐条调用单规则接口造成部分创建成功
920 *
921 * 每条输入都必须使用新 ID 和 expected_generation=0，不混入更新或删除语义
922 * 先检查容量并逐条形成计划，必要时通过无副作用的驱动校验决定是否走软件后端
923 * 新对象安装由同一事务完成，事务成功后才按请求顺序写入规则账本
924 * 安装失败会尝试撤销本批对象，撤销也失败则进入恢复隔离，不能声称已经全部回滚
925 */
926static int control_create_batch_impl(struct dppd_control_service *service,
927 const struct dppd_control_batch_create_request *requests, uint32_t request_count,
928 struct dppd_control_apply_result *results)
929{
930 struct dppd_transaction_item *items = NULL;
931 struct dppd_transaction transaction;
932 struct dppd_transaction_backends backends;
933 uint64_t base_generation;
934 uint64_t transaction_id;
935 uint32_t i;
936 uint32_t published_count = 0;
937 int rc;
938
939 if (service == NULL || service->topology == NULL || requests == NULL || results == NULL || request_count == 0)
940 return -EINVAL;
941 rc = recovery_write_preflight(service);
942 if (rc != 0)
943 return rc;
944 rc = persistence_write_preflight(service);
945 if (rc != 0)
946 return rc;
947 if (request_count > service->rules.capacity - dppd_rule_repository_count(&service->rules))
948 return -ENOSPC;
949 /** 在探测或创建驱动对象前预留连续版本，不能分配 ANY 保留值或发生回绕 */
950 base_generation = dppd_rule_repository_generation(&service->rules);
951 if (base_generation >= UINT64_MAX - request_count)
952 return -EOVERFLOW;
953
954 observe_stage(service, DPPD_RULE_STAGE_PREPARE);
955 items = calloc(request_count, sizeof(*items));
956 if (items == NULL)
957 return -ENOMEM;
958 /**
959 * 逐条检查 ID、创建条件和规划结果，暂不安装对象或预留资源
960 * PREFER 条目可能在本循环调用驱动 validate，这只是能力探测，不是安装
961 * 后面的条目即使检查失败，也不应留下前面条目的实际规则
962 */
963 for (i = 0; i < request_count; ++i) {
964 struct dppd_planner_context planner_context;
965 struct dppd_rule existing;
966 uint32_t previous;
967
968 observe_rule(service, DPPD_RULE_STAGE_PREFLIGHT, &requests[i].rule, requests[i].install_port_id, NULL);
969 service->observation.context.generation = requests[i].expected_generation;
970 /**
971 * 新建必须明确声明旧版本为零，不接受 ANY 或具体旧版本
972 * 如果允许混入覆盖已有规则的请求，失败时还要恢复旧对象，不能复用纯创建的回滚逻辑
973 */
974 if (requests[i].rule.id == 0 || requests[i].expected_generation != 0) {
975 rc = -ESTALE;
976 goto cleanup;
977 }
978 for (previous = 0; previous < i; ++previous) {
979 if (requests[previous].rule.id == requests[i].rule.id) {
980 rc = -EEXIST;
981 goto cleanup;
982 }
983 }
984 rc = dppd_rule_repository_get(&service->rules, requests[i].rule.id, &existing);
985 if (rc != -ENOENT) {
986 rc = rc == 0 ? -EEXIST : rc;
987 goto cleanup;
988 }
989
990 memset(&results[i], 0, sizeof(results[i]));
991 items[i].rule = requests[i].rule;
992 /** 外层安装端口是本次请求的明确目标，覆盖规则内部字段以保持接口含义一致 */
993 items[i].rule.install_port_id = requests[i].install_port_id;
994 items[i].rule.generation = base_generation + i + 1U;
995 memset(&planner_context, 0, sizeof(planner_context));
996 planner_context.topology = service->topology;
997 planner_context.ipv4_routing = service->routing != NULL;
998 planner_context.connection_tracking = service->conntrack != NULL;
999 planner_context.neighbor_discovery = service->nd6 != NULL;
1000 planner_context.install_port_id = requests[i].install_port_id;
1001 planner_context.hardware_available = true;
1002 planner_context.software_equivalent = dppd_software_backend_rule_supported(&items[i].rule);
1003 observe_rule(service, DPPD_RULE_STAGE_PLAN, &items[i].rule, planner_context.install_port_id, NULL);
1004 rc = dppd_plan_rule(&planner_context, &items[i].rule, &items[i].plan);
1005 if (rc != 0)
1006 goto cleanup;
1007
1008 /**
1009 * 在事务准备资源之前决定最终后端，这样一批规则可以同时包含硬件和软件条目
1010 * 例如 net_ring 的校验不支持 flow，但软件可表达相同规则时，只把这一条改为软件
1011 * 创建或回滚阶段的错误不允许走这个分支，防止掩盖已经产生的安装副作用
1012 */
1013 if (items[i].plan.backend == DPPD_PLAN_BACKEND_RTE_FLOW &&
1014 items[i].rule.fallback == DPPD_FALLBACK_PREFER_HARDWARE && planner_context.software_equivalent) {
1015 struct dppd_flow_error flow_error;
1016
1017 rc = dppd_rte_flow_backend_validate_rule(&service->rte_flow, items[i].plan.install_port_id, &items[i].rule,
1018 &flow_error);
1019 if (rc != 0) {
1020 planner_context.hardware_available = false;
1021 rc = dppd_plan_rule(&planner_context, &items[i].rule, &items[i].plan);
1022 if (rc != 0)
1023 goto cleanup;
1024 service->observation.fallback_rules++;
1025 }
1026 }
1027 }
1028
1029 memset(&backends, 0, sizeof(backends));
1030 backends.software = dppd_software_transaction_backend(&service->software);
1031 backends.rte_flow = dppd_rte_flow_transaction_backend(&service->rte_flow);
1032 transaction_id = service->next_transaction_id++;
1033 service->observation.context.transaction_id = transaction_id;
1034 rc = dppd_transaction_init(&transaction, transaction_id, items, request_count);
1035 if (rc != 0)
1036 goto cleanup;
1037 /** 先让本批所有实际对象安装完成，再在同一控制线程中连续发布对应的账本记录 */
1038 rc = dppd_transaction_run(&transaction, &backends);
1039 if (rc != 0) {
1040 observe_transaction(service, &transaction);
1041 /** 撤销也失败时可能仍有残留对象，记录恢复错误并封锁后续普通写入 */
1042 if (transaction.rollback_code != 0) {
1043 service->recovery_state = DPPD_CONTROL_RECOVERY_RECONCILIATION_REQUIRED;
1044 service->recovery_last_error = transaction.rollback_code;
1045 rc = -EUCLEAN;
1046 }
1047 goto cleanup;
1048 }
1049
1050 /**
1051 * 全批检查和串行执行保证这里应当只产生 CREATED，且版本与预分配值一致
1052 * 如果仍出现异常，撤回已经写入的本批记录并尝试回滚实际对象，统一返回 EUCLEAN
1053 * 结果数组只有整个函数成功时才可作为成功回执，不能只看其中已填好的前几项
1054 */
1055 for (i = 0; i < request_count; ++i) {
1056 struct dppd_rule_apply_result repository_result;
1057
1058 observe_rule(service, DPPD_RULE_STAGE_PUBLISH, &items[i].rule, items[i].plan.install_port_id, &items[i].plan);
1059 rc = dppd_rule_repository_apply(&service->rules, &items[i].rule, 0, &repository_result);
1060 if (rc != 0 || repository_result.status != DPPD_RULE_CREATED ||
1061 repository_result.generation != items[i].rule.generation) {
1062 int rollback_rc;
1063
1064 dppd_rule_observation_fault(&service->observation, rc != 0 ? rc : -EUCLEAN);
1065 /**
1066 * published_count 只记录当前批次已经写入账本的条数，逆序移除这部分记录
1067 * 删除记录仍会推进全局版本，所以撤回记录不等于把历史版本号恢复到原值
1068 * 任一账本撤回或实际对象回滚失败，都要进入恢复隔离
1069 */
1070 while (published_count > 0) {
1071 bool removed;
1072 uint64_t ignored_generation;
1073
1074 --published_count;
1075 if (dppd_rule_repository_remove(&service->rules, items[published_count].rule.id,
1076 items[published_count].rule.generation, &removed, &ignored_generation) != 0 ||
1077 !removed) {
1078 service->recovery_state = DPPD_CONTROL_RECOVERY_RECONCILIATION_REQUIRED;
1079 service->recovery_last_error = -EUCLEAN;
1080 }
1081 }
1082 rollback_rc = dppd_transaction_rollback_committed(&transaction, &backends);
1083 observe_transaction(service, &transaction);
1084 if (rollback_rc != 0) {
1085 service->recovery_state = DPPD_CONTROL_RECOVERY_RECONCILIATION_REQUIRED;
1086 service->recovery_last_error = rollback_rc;
1087 }
1088 rc = -EUCLEAN;
1089 goto cleanup;
1090 }
1091 published_count++;
1092 results[i].status = repository_result.status;
1093 results[i].generation = repository_result.generation;
1094 results[i].transaction_id = transaction_id;
1095 results[i].plan = items[i].plan;
1096 }
1097 service->observation.applied = true;
1098 /** 实际规则和账本均已写好，不再需要本次回滚凭据，先结束事务临时资源再保存快照 */
1099 rc = dppd_transaction_finalize(&transaction, &backends);
1100 if (rc != 0) {
1101 rc = -EUCLEAN;
1102 goto cleanup;
1103 }
1104 rc = persist_current_repository(service);
1105
1106cleanup:
1107 free(items);
1108 return rc;
1109}

1111/**
1112 * 一次删除一组已经存在且版本精确匹配的规则
1113 * 先保存原规则和实际后端，再删除全部实际对象，最后才从账本中连续移除记录
1114 * 如果实际删除到一半失败，账本仍保留全部旧记录，可按原版本重建已经删掉的对象
1115 * 通用事务接口描述的是对象创建，因此这里使用显式删除循环，并用创建事务完成补偿
1116 */
1117static int control_remove_batch_impl(struct dppd_control_service *service,
1118 const struct dppd_control_batch_remove_request *requests, uint32_t request_count,
1119 struct dppd_control_batch_remove_result *results)
1120{
1121 struct dppd_transaction_item *items;
1122 uint32_t removed_actual = 0;
1123 uint32_t i;
1124 int rc;
1125
1126 if (service == NULL || requests == NULL || results == NULL || request_count == 0)
1127 return -EINVAL;
1128 rc = recovery_write_preflight(service);
1129 if (rc != 0)
1130 return rc;
1131 rc = persistence_write_preflight(service);
1132 if (rc != 0)
1133 return rc;
1134 /** 整批删除会逐条推进版本，在删除任何实际对象前确认不会使用 ANY 保留值 */
1135 if (dppd_rule_repository_generation(&service->rules) >= UINT64_MAX - request_count)
1136 return -EOVERFLOW;
1137 observe_stage(service, DPPD_RULE_STAGE_PREPARE);
1138 items = calloc(request_count, sizeof(*items));
1139 if (items == NULL)
1140 return -ENOMEM;
1141
1142 /**
1143 * 删除任何实际对象前先检查整批：旧版本必须精确、ID 不重复、对应对象只存在于一个后端
1144 * 同时把规则和后端归属保存下来，后续即使对象已删除也知道应该怎样重建
1145 */
1146 for (i = 0; i < request_count; ++i) {
1147 uint32_t previous;
1148 struct dppd_rule target = {.id = requests[i].rule_id, .generation = requests[i].expected_generation};
1149
1150 observe_rule(service, DPPD_RULE_STAGE_PREFLIGHT, &target, 0, NULL);
1151 service->observation.context.port_known = false;
1152 if (requests[i].rule_id == 0 || requests[i].expected_generation == 0 ||
1153 requests[i].expected_generation == DPPD_RULE_GENERATION_ANY) {
1154 rc = -ESTALE;
1155 goto cleanup;
1156 }
1157 for (previous = 0; previous < i; ++previous) {
1158 if (requests[previous].rule_id == requests[i].rule_id) {
1159 rc = -EEXIST;
1160 goto cleanup;
1161 }
1162 }
1163 rc = dppd_rule_repository_get(&service->rules, requests[i].rule_id, &items[i].rule);
1164 if (rc != 0)
1165 goto cleanup;
1166 if (items[i].rule.generation != requests[i].expected_generation) {
1167 rc = -ESTALE;
1168 goto cleanup;
1169 }
1170 rc = actual_backend_for(service, items[i].rule.id, items[i].rule.generation, &items[i].plan.backend);
1171 if (rc != 0)
1172 goto cleanup;
1173 items[i].plan.rule_id = items[i].rule.id;
1174 items[i].plan.rule_generation = items[i].rule.generation;
1175 items[i].plan.install_port_id = items[i].rule.install_port_id;
1176 }
1177
1178 /** removed_actual 只在删除成功后增加，失败时只恢复前面确实已经删掉的对象 */
1179 for (i = 0; i < request_count; ++i) {
1180 observe_rule(service, DPPD_RULE_STAGE_REMOVE, &items[i].rule, items[i].rule.install_port_id, &items[i].plan);
1181 rc = remove_actual_rule(service, items[i].rule.id, items[i].rule.generation);
1182 if (rc != 0) {
1183 int restore_rc;
1184
1185 dppd_rule_observation_fault(&service->observation, rc);
1186 restore_rc = restore_actual_rules(service, items, removed_actual);
1187
1188 /**
1189 * 账本还没修改，补偿成功后实际对象又与旧账本一致，可以返回最初的删除错误
1190 * 补偿也失败时已经不能证明一致性，改为返回 EUCLEAN 并记录具体恢复错误
1191 */
1192 if (restore_rc != 0) {
1193 service->recovery_state = DPPD_CONTROL_RECOVERY_RECONCILIATION_REQUIRED;
1194 service->recovery_last_error = restore_rc;
1195 rc = -EUCLEAN;
1196 }
1197 goto cleanup;
1198 }
1199 removed_actual++;
1200 }
1201
1202 /**
1203 * 实际对象全部删除后才从账本移除记录，每次删除都会推进全局版本
1204 * 精确预检和串行调用保证正常情况下不会失败；若仍失败，不能简单重新创建来冒充旧版本
1205 * 因为部分账本记录也可能已经消失，此时只能进入隔离，由恢复流程统一处理
1206 */
1207 for (i = 0; i < request_count; ++i) {
1208 bool removed;
1209 uint64_t generation;
1210
1211 observe_rule(service, DPPD_RULE_STAGE_PUBLISH, &items[i].rule, items[i].rule.install_port_id, &items[i].plan);
1212 rc = dppd_rule_repository_remove(&service->rules, items[i].rule.id, items[i].rule.generation, &removed,
1213 &generation);
1214 if (rc != 0 || !removed) {
1215 dppd_rule_observation_fault(&service->observation, rc != 0 ? rc : -EUCLEAN);
1216 service->recovery_state = DPPD_CONTROL_RECOVERY_RECONCILIATION_REQUIRED;
1217 service->recovery_last_error = rc != 0 ? rc : -EUCLEAN;
1218 rc = -EUCLEAN;
1219 goto cleanup;
1220 }
1221 results[i].rule_id = items[i].rule.id;
1222 results[i].generation = generation;
1223 }
1224 service->observation.applied = true;
1225 rc = persist_current_repository(service);
1226
1227cleanup:
1228 free(items);
1229 return rc;
1230}

1232/**
1233 * 把一组已有规则作为同一笔更新处理，避免调用方看到只更新了一半的规则清单
1234 *
1235 * 可以把 repository 理解为“系统希望保留的规则账本”，把 actual 对象理解为
1236 * “硬件或软件中已经安装的规则”。只有新规则全部装好、旧规则全部删完之后
1237 * 才能整批改写账本。此前任何一步失败，都优先撤销新规则并恢复已删除的旧规则
1238 *
1239 * 本函数由管理主线程串行调用，旧对象与新计划全为软件时整批只发布一次快照
1240 * 涉及硬件时仍采用逐条安装与补偿流程，不保证所有报文在同一时刻切换规则
1241 * 返回 EUCLEAN 时要进一步区分恢复隔离和保存失败，不能直接认定本次更新没有生效
1242 */
1243static int control_update_batch_impl(struct dppd_control_service *service,
1244 const struct dppd_control_batch_update_request *requests, uint32_t request_count,
1245 struct dppd_control_apply_result *results)
1246{
1247 /**
1248 * items 保存待安装的新版本，old 保存失败补偿所需的旧规则及原后端类型
1249 * candidates 是交给规则账本的完整新内容，expected 是用户读取过的精确旧版本
1250 * 这些数组使用相同下标，保证请求、安装计划和返回结果始终一一对应
1251 */
1252 struct dppd_transaction_item items[DPPD_CONTROL_BATCH_UPDATE_MAX] = {0};
1253 struct dppd_transaction_item old[DPPD_CONTROL_BATCH_UPDATE_MAX] = {0};
1254 struct dppd_rule candidates[DPPD_CONTROL_BATCH_UPDATE_MAX];
1255 uint64_t expected[DPPD_CONTROL_BATCH_UPDATE_MAX];
1256 bool may_fallback[DPPD_CONTROL_BATCH_UPDATE_MAX] = {false};
1257 bool all_old_software = true, atomic_software;
1258 struct dppd_transaction transaction;
1259 struct dppd_transaction_backends backends = {0};
1260 uint32_t hardware_needed = 0, software_needed = 0;
1261 uint32_t hardware_count, software_count;
1262 uint64_t base_generation;
1263 uint32_t i;
1264 int rc;
1265
1266 if (service == NULL || service->topology == NULL || requests == NULL || results == NULL || request_count < 2 ||
1267 request_count > DPPD_CONTROL_BATCH_UPDATE_MAX)
1268 return -EINVAL;
1269 memset(results, 0, request_count * sizeof(*results));
1270 /**
1271 * 已在恢复隔离中时禁止继续写入；上次保存失败时先把已生效状态重新写入磁盘
1272 * 两个检查都通过后才开始本次更新，防止在已有不一致上继续叠加新修改
1273 */
1274 rc = recovery_write_preflight(service);
1275 if (rc != 0)
1276 return rc;
1277 rc = persistence_write_preflight(service);
1278 if (rc != 0)
1279 return rc;
1280 base_generation = dppd_rule_repository_generation(&service->rules);
1281 /**
1282 * 每条规则需要一个连续新版本，版本号不能绕回零或占用 ANY 的特殊值
1283 * 事务编号也要保留有效空间，以便后续安装和失败补偿能够被明确识别
1284 */
1285 if (base_generation >= UINT64_MAX - request_count || service->next_transaction_id == 0 ||
1286 service->next_transaction_id == UINT64_MAX)
1287 return -EOVERFLOW;
1288
1289 /**
1290 * 第一轮只做本地检查，不调用网卡驱动 PMD
1291 * 例如第二条规则版本过期时，应立即拒绝整批，而不是先为第一条接触硬件
1292 * 0 和 ANY 都不代表用户明确确认过的旧版本，因此批量更新不接受这两个值
1293 */
1294 for (i = 0; i < request_count; ++i) {
1295 struct dppd_planner_context context = {0};
1296 char validation_error[128];
1297 uint32_t j;
1298
1299 observe_rule(service, DPPD_RULE_STAGE_PREFLIGHT, &requests[i].rule, requests[i].install_port_id, NULL);
1300 service->observation.context.generation = requests[i].expected_generation;
1301 if (requests[i].reserved != 0 || requests[i].rule.id == 0)
1302 return -EINVAL;
1303 if (requests[i].expected_generation == 0 || requests[i].expected_generation == DPPD_RULE_GENERATION_ANY)
1304 return -ESTALE;
1305 for (j = 0; j < i; ++j) {
1306 if (requests[j].rule.id == requests[i].rule.id)
1307 return -EEXIST;
1308 }
1309 rc = dppd_rule_repository_get(&service->rules, requests[i].rule.id, &old[i].rule);
1310 if (rc != 0)
1311 return rc;
1312 if (old[i].rule.generation != requests[i].expected_generation)
1313 return -ESTALE;
1314 /**
1315 * 旧规则必须恰好存在于一个实际后端中，不能仅根据“优先硬件”策略猜归属
1316 * 同时存在于两个后端或两个后端都不存在，说明账本与安装状态已经失配
1317 * 此时进入隔离，不能继续更新并掩盖原来的问题
1318 */
1319 rc = actual_backend_for(service, old[i].rule.id, old[i].rule.generation, &old[i].plan.backend);
1320 if (rc != 0)
1321 goto isolate;
1322
1323 expected[i] = old[i].rule.generation;
1324 if (old[i].plan.backend != DPPD_PLAN_BACKEND_SOFTWARE)
1325 all_old_software = false;
1326 candidates[i] = requests[i].rule;
1327 candidates[i].install_port_id = requests[i].install_port_id;
1328 candidates[i].generation = base_generation + i + 1U;
1329 items[i].rule = candidates[i];
1330 /**
1331 * 匹配条件和动作来自请求，必须先检查数组长度和规则结构
1332 * 后面的软件能力判断会遍历数组，先校验可以避免非法长度导致越界读取
1333 */
1334 if (dppd_rule_validate(&candidates[i], validation_error, sizeof(validation_error)) != 0)
1335 return -EINVAL;
1336 context.topology = service->topology;
1337 context.ipv4_routing = service->routing != NULL;
1338 context.connection_tracking = service->conntrack != NULL;
1339 context.neighbor_discovery = service->nd6 != NULL;
1340 context.install_port_id = requests[i].install_port_id;
1341 context.hardware_available = true;
1342 context.software_equivalent = dppd_software_backend_rule_supported(&candidates[i]);
1343 observe_rule(service, DPPD_RULE_STAGE_PLAN, &items[i].rule, context.install_port_id, NULL);
1344 rc = dppd_plan_rule(&context, &items[i].rule, &items[i].plan);
1345 if (rc != 0)
1346 return rc;
1347 may_fallback[i] = items[i].plan.backend == DPPD_PLAN_BACKEND_RTE_FLOW &&
1348 candidates[i].fallback == DPPD_FALLBACK_PREFER_HARDWARE && context.software_equivalent;
1349 if (items[i].plan.backend == DPPD_PLAN_BACKEND_RTE_FLOW)
1350 hardware_needed++;
1351 if (items[i].plan.backend == DPPD_PLAN_BACKEND_SOFTWARE || may_fallback[i])
1352 software_needed++;
1353 }
1354
1355 /**
1356 * 硬件及混合路径需要新旧版本暂时共存，因此预检临时空间
1357 * 旧对象全为软件且所有目标都可能走软件时，允许稍后确认能否整表替换
1358 * 如果最终规划为混合路径，仍须在安装之前复查软件临时空间
1359 * 容量不足不触发降级，硬件目标的本地容量仍在 PMD 校验之前保守检查
1360 */
1361 hardware_count = dppd_rte_flow_backend_count(&service->rte_flow);
1362 software_count = dppd_software_backend_count(&service->software);
1363 observe_batch(service, DPPD_RULE_STAGE_PREFLIGHT);
1364 if (hardware_count > service->rte_flow.capacity || software_count > service->software.capacity) {
1365 rc = -EUCLEAN;
1366 goto isolate;
1367 }
1368 if (hardware_needed > service->rte_flow.capacity - hardware_count ||
1369 (software_needed > service->software.capacity - software_count &&
1370 !(all_old_software && software_needed == request_count)))
1371 return -ENOSPC;
1372
1373 /**
1374 * 到这里整批输入和容量都已通过检查，才允许询问 PMD 是否支持这些规则
1375 * 只有尚未安装对象的 validate 失败，并且软件能表达相同语义时，才改走软件
1376 * 后面的创建、删除或回滚失败都不能再用软件回退来伪装成成功
1377 */
1378 for (i = 0; i < request_count; ++i) {
1379 if (may_fallback[i]) {
1380 struct dppd_flow_error error;
1381
1382 rc = dppd_rte_flow_backend_validate_rule(&service->rte_flow, items[i].plan.install_port_id, &items[i].rule,
1383 &error);
1384 if (rc != 0) {
1385 struct dppd_planner_context context = {0};
1386
1387 context.topology = service->topology;
1388 context.ipv4_routing = service->routing != NULL;
1389 context.connection_tracking = service->conntrack != NULL;
1390 context.neighbor_discovery = service->nd6 != NULL;
1391 context.install_port_id = requests[i].install_port_id;
1392 context.software_equivalent = true;
1393 rc = dppd_plan_rule(&context, &items[i].rule, &items[i].plan);
1394 if (rc != 0)
1395 return rc;
1396 service->observation.fallback_rules++;
1397 }
1398 }
1399 }
1400 /**
1401 * 确认最终计划是否允许整批软件替换，否则按实际软件目标数复查临时空间
1402 * 纯软件路径复用旧槽位，硬件或混合路径继续由事务逐条安装与补偿
1403 */
1404 atomic_software = all_old_software;
1405 software_needed = 0;
1406 for (i = 0; i < request_count; ++i) {
1407 if (items[i].plan.backend == DPPD_PLAN_BACKEND_SOFTWARE)
1408 software_needed++;
1409 else
1410 atomic_software = false;
1411 }
1412 if (!atomic_software && software_needed > service->software.capacity - software_count)
1413 return -ENOSPC;
1414 backends.software = dppd_software_transaction_backend(&service->software);
1415 backends.rte_flow = dppd_rte_flow_transaction_backend(&service->rte_flow);
1416 rc = dppd_transaction_init(&transaction, service->next_transaction_id++, items, request_count);
1417 if (rc != 0)
1418 return rc;
1419 if (atomic_software) {
1420 /** 整批软件更新在一个函数里分配并发布，无法定位成员时明确报告批次而不猜 ID */
1421 observe_stage(service, DPPD_RULE_STAGE_COMMIT);
1422 service->observation.context.rule_id = 0;
1423 service->observation.context.generation = 0;
1424 service->observation.context.port_known = false;
1425 service->observation.context.backend = DPPD_PLAN_BACKEND_SOFTWARE;
1426 service->observation.context.backend_known = true;
1427 service->observation.context.transaction_id = transaction.id;
1428 rc = dppd_software_backend_update_batch(&service->software, candidates, expected, request_count);
1429 if (rc != 0)
1430 return rc;
1431 observe_stage(service, DPPD_RULE_STAGE_PUBLISH);
1432 rc = dppd_rule_repository_update_batch(&service->rules, candidates, expected, request_count);
1433 if (rc != 0)
1434 goto isolate;
1435 goto committed;
1436 }
1437 rc = dppd_transaction_run(&transaction, &backends);
1438 if (rc != 0) {
1439 observe_transaction(service, &transaction);
1440 if (transaction.rollback_code == 0)
1441 return rc;
1442 rc = transaction.rollback_code;
1443 goto isolate;
1444 }
1445
1446 /**
1447 * 全部新对象安装成功后再逐条删除旧对象，这时规则账本仍完整保留旧版本
1448 * 删除下标 i 失败，意味着只有前 i 条旧对象已经删除，补偿只恢复这部分
1449 */
1450 for (i = 0; i < request_count; ++i) {
1451 observe_rule(service, DPPD_RULE_STAGE_REMOVE, &old[i].rule, old[i].rule.install_port_id, &old[i].plan);
1452 service->observation.context.transaction_id = transaction.id;
1453 rc = remove_actual_rule(service, old[i].rule.id, old[i].rule.generation);
1454 if (rc != 0) {
1455 int rollback_rc;
1456 int restore_rc;
1457
1458 dppd_rule_observation_fault(&service->observation, rc);
1459 /**
1460 * 先撤销本批全部新版本，再按原规则、原版本和原后端恢复已删除的旧对象
1461 * 即使撤新失败也继续尝试恢复旧对象，以尽量减少缺失的规则
1462 * 两种补偿都成功才返回原始删除错误，否则返回 EUCLEAN 并封锁后续更新
1463 */
1464 rollback_rc = dppd_transaction_rollback_committed(&transaction, &backends);
1465 observe_transaction(service, &transaction);
1466 restore_rc = restore_actual_rules(service, old, i);
1467 if (rollback_rc == 0 && restore_rc == 0)
1468 return rc;
1469 rc = rollback_rc != 0 ? rollback_rc : restore_rc;
1470 goto isolate;
1471 }
1472 }
1473
1474 /**
1475 * 新旧对象的切换全部成功后，最后整批改写账本
1476 * 仓库接口会先复查所有条件，再执行不分配内存的替换循环
1477 * 内容没有变化的规则也获得新版本，这样整批结果始终使用连续的版本号
1478 */
1479 observe_batch(service, DPPD_RULE_STAGE_PUBLISH);
1480 rc = dppd_rule_repository_update_batch(&service->rules, candidates, expected, request_count);
1481 if (rc != 0) {
1482 /**
1483 * 正常串行流程下不应失败；如果复查仍失败，旧实际对象已经被删除
1484 * 此时不能靠重新 apply 冒充恢复旧版本，只释放事务临时资源并进入隔离
1485 */
1486 (void)dppd_transaction_finalize(&transaction, &backends);
1487 goto isolate;
1488 }
1489 /** 账本已经完整发布，之后的临时资源收尾错误也必须如实报告为已经生效 */
1490 service->observation.applied = true;
1491 rc = dppd_transaction_finalize(&transaction, &backends);
1492 if (rc != 0)
1493 goto isolate;
1494committed:
1495 service->observation.applied = true;
1496 for (i = 0; i < request_count; ++i) {
1497 results[i].status = DPPD_RULE_UPDATED;
1498 results[i].generation = candidates[i].generation;
1499 results[i].transaction_id = transaction.id;
1500 results[i].plan = items[i].plan;
1501 }
1502 /**
1503 * 保存失败时整批更新已经在内存和实际后端生效，不能再声称已经回滚
1504 * 保存函数会设置 dirty 状态并返回 EUCLEAN，后续写入要先修复持久化状态
1505 */
1506 return persist_current_repository(service);
1507
1508isolate:
1509 /**
1510 * 隔离表示“已经无法证明账本和实际规则一致”，普通管理请求将被拒绝
1511 * 主循环随后停止软件工作线程，只保留恢复查询和清理重试入口
1512 * 底层错误另存于 recovery_last_error，便于区分创建、删除或补偿失败
1513 */
1514 dppd_rule_observation_fault(&service->observation, rc);
1515 service->recovery_state = DPPD_CONTROL_RECOVERY_RECONCILIATION_REQUIRED;
1516 service->recovery_last_error = rc;
1517 return -EUCLEAN;
1518}
1519
1520/**
1521 * 按 ID 和预期版本删除规则，removed 和 generation 返回账本删除结果及仓库版本
1522 * 查账本确定版本 → 删除实际后端对象 → 删除内存账本及索引 → 保存当前配置
1523 * 后端删除失败时保留账本；后端已删但账本更新失败时进入恢复隔离
1524 * 返回 0 表示成功，负值表示错误；保存失败不会撤销已完成的删除
1525 */
1526static int control_remove_impl(struct dppd_control_service *service, uint64_t rule_id, uint64_t expected_generation,
1527 bool *removed, uint64_t *generation)
1528{
1529 struct dppd_rule existing;
1530 int rc;
1531
1532 if (service == NULL || rule_id == 0 || removed == NULL || generation == NULL)
1533 return -EINVAL;
1534 rc = recovery_write_preflight(service);
1535 if (rc != 0)
1536 return rc;
1537 rc = persistence_write_preflight(service);
1538 if (rc != 0)
1539 return rc;
1540 rc = dppd_rule_repository_get(&service->rules, rule_id, &existing);
1541 if (rc == -ENOENT) {
1542 rc = dppd_rule_repository_remove(&service->rules, rule_id, expected_generation, removed, generation);
1543 service->observation.unchanged = rc == 0;
1544 return rc;
1545 }
1546 if (rc != 0)
1547 return rc;
1548 if (expected_generation != DPPD_RULE_GENERATION_ANY && expected_generation != existing.generation)
1549 return -ESTALE;
1550 if (dppd_rule_repository_generation(&service->rules) >= UINT64_MAX - 1U)
1551 return -EOVERFLOW;
1552 observe_rule(service, DPPD_RULE_STAGE_REMOVE, &existing, existing.install_port_id, NULL);
1553 rc = remove_actual_rule(service, rule_id, existing.generation);
1554 if (rc != 0)
1555 return rc;
1556 observe_stage(service, DPPD_RULE_STAGE_PUBLISH);
1557 rc = dppd_rule_repository_remove(&service->rules, rule_id, existing.generation, removed, generation);
1558 if (rc != 0 || !*removed) {
1559 dppd_rule_observation_fault(&service->observation, rc != 0 ? rc : -EUCLEAN);
1560 return isolate_control(service, rc != 0 ? rc : -EUCLEAN);
1561 }
1562 service->observation.applied = true;
1563 return persist_current_repository(service);
1564}

1566/**
1567 * 先以规则账本确定当前版本，再读取两个后端的安装记录，不能仅凭 prefer 策略猜测实际位置
1568 * 缺失或重复安装都属于账本与实际对象失配，保留原状态并报告错误，由已有恢复流程处理
1569 */
1570int dppd_control_rule_status(const struct dppd_control_service *service, uint64_t rule_id, uint64_t expected_generation,
1571 struct dppd_control_rule_status *result)
1572{
1573 struct dppd_rule_install_info hardware, software;
1574 const struct dppd_rule_install_info *installed;
1575 struct dppd_rule rule;
1576 bool has_count = false;
1577 uint16_t action;
1578 int hardware_rc, software_rc, rc;
1579
1580 if (service == NULL || rule_id == 0 || expected_generation == 0 || result == NULL)
1581 return -EINVAL;
1582 memset(result, 0, sizeof(*result));
1583 if (recovery_write_preflight(service) != 0)
1584 return -EUCLEAN;
1585 rc = dppd_rule_repository_get(&service->rules, rule_id, &rule);
1586 if (rc != 0)
1587 return rc;
1588 if (expected_generation != DPPD_RULE_GENERATION_ANY && expected_generation != rule.generation)
1589 return -ESTALE;
1590
1591 /** 这两个入口只读取安装记录，不借用会顺带回收旧快照的 contains 或 COUNT 查询 */
1592 hardware_rc = dppd_rte_flow_backend_install_info(&service->rte_flow, rule.id, rule.generation, &hardware);
1593 software_rc = dppd_software_backend_install_info(&service->software, rule.id, rule.generation, &software);
1594 if ((hardware_rc != 0 && hardware_rc != -ENOENT) || (software_rc != 0 && software_rc != -ENOENT) ||
1595 (hardware_rc == 0) == (software_rc == 0))
1596 return -EUCLEAN;
1597 installed = hardware_rc == 0 ? &hardware : &software;
1598 for (action = 0; action < rule.nb_actions; ++action) {
1599 if (rule.actions[action].type == DPPD_ACTION_COUNT)
1600 has_count = true;
1601 }
1602 if (installed->rule_id != rule.id || installed->generation != rule.generation ||
1603 installed->install_port_id != rule.install_port_id || installed->has_count != has_count ||
1604 installed->commit_rule_count == 0 || (hardware_rc == 0 && rule.fallback == DPPD_FALLBACK_SOFTWARE_ONLY) ||
1605 (software_rc == 0 && rule.fallback == DPPD_FALLBACK_REQUIRE_HARDWARE))
1606 return -EUCLEAN;
1607
1608 result->installation = *installed;
1609 result->backend = hardware_rc == 0 ? DPPD_PLAN_BACKEND_RTE_FLOW : DPPD_PLAN_BACKEND_SOFTWARE;
1610 result->fallback = rule.fallback;
1611 dppd_control_persistence_status(service, &result->persistence);
1612 return 0;
1613}

1615int dppd_control_query_count(struct dppd_control_service *service, uint64_t rule_id, uint64_t expected_generation,
1616 struct dppd_control_count_result *result)
1617{
1618 const struct dppd_flow_handle *handle;
1619 struct dppd_flow_error flow_error;
1620 struct dppd_rule rule;
1621 int rc;
1622
1623 if (service == NULL || rule_id == 0 || result == NULL)
1624 return -EINVAL;
1625 memset(result, 0, sizeof(*result));
1626 if (recovery_write_preflight(service) != 0)
1627 return -EUCLEAN;
1628
1629 /*
1630 * repository 是 desired state 的发布点。先读 repository 再定位同 generation
1631 * 的硬件 handle，确保查询结果不会来自尚未发布或已经退休的 flow
1632 */
1633 rc = dppd_rule_repository_get(&service->rules, rule_id, &rule);
1634 if (rc != 0)
1635 return rc;
1636 if (expected_generation != DPPD_RULE_GENERATION_ANY && expected_generation != rule.generation)
1637 return -ESTALE;
1638 if (dppd_software_backend_contains_version(&service->software, rule.id, rule.generation)) {
1639 rc = dppd_software_backend_query_count(&service->software, rule.id, rule.generation, &result->hits,
1640 &result->bytes);
1641 } else {
1642 handle = dppd_rte_flow_backend_find_version(&service->rte_flow, rule.id, rule.generation);
1643 if (handle == NULL)
1644 return -EUCLEAN;
1645 if (!handle->has_count)
1646 return -ENODATA;
1647 rc = dppd_rte_flow_backend_query_count(&service->rte_flow, rule.id, rule.generation, &result->hits,
1648 &result->bytes, &flow_error);
1649 }
1650 if (rc != 0)
1651 return rc;
1652 result->rule_id = rule.id;
1653 result->generation = rule.generation;
1654 return 0;
1655}

1685/**
1686 * 接收单条规则的创建或更新请求，执行安装、内存记录更新和保存，并汇总本次操作的观测结果
1687 * service 提供各模块，port_id 指定安装端口，rule 提供完整规则，expected 检查预期旧版本
1688 * result 返回执行计划、事务编号、操作状态和版本；返回 0 表示成功，负值表示错误
1689 *
1690 * 普通 RULE_APPLY 请求的完整入口路径：
1691 * 用户执行 dppctl 命令
1692 * │
1693 * │ 客户端进程：app/dppctl/main.c
1694 * ├─ main
1695 * │ → build_request 把命令参数填成 request
1696 * │ → exchange
1697 * │ → send 通过本机 Unix socket 发送 request
1698 * │
1699 * │ 服务端进程：dppd
1700 * └─ app/dppd/main.c 的主循环
1701 * → dppd_management_poll
1702 * → recv 接收 request
1703 * → dppd_management_handle
1704 * → 根据 request->operation 分发
1705 * → RULE_APPLY 分支
1706 * → dppd_control_apply
1707 * → control_apply_impl
1708 *
1709 * CLI 与 daemon 是两个进程，中间通过 socket 通信，不是跨进程直接调用函数
1710 * 处理结果经管理响应返回 dppctl，这条路径不经过网卡收包队列
1711 *
1712 * 报文入口：worker.c/process_batch → routed_decide → pipeline → 软件规则匹配 → 读取 active
1713 * 规则配置入口：本函数 → control_apply_impl → transaction_run → 软件 commit → publish_locked
1714 * 联系点是软件后端的 active：配置路径提前发布规则，报文路径读取已发布结果
1715 * 从报文匹配转来阅读本函数，是追查规则来源，不是 process_batch 调用了规则配置入口
1716 * 单条 apply 的计数涵盖新建、更新和幂等重放，保存失败时已生效规则仍可能保留
1717 */
1718int dppd_control_apply(struct dppd_control_service *service, uint16_t port_id, const struct dppd_rule *rule,
1719 uint64_t expected, struct dppd_control_apply_result *result)
1720{
1721 bool observed =
1722 begin_observed(service, DPPD_RULE_OPERATION_APPLY, rule != NULL ? rule->id : 0, expected, 1, true, port_id);
1723 int rc = control_apply_impl(service, port_id, rule, expected, result);
1724
1725 return finish_observed(service, observed, rc);
1726}
