22/* 统一初始化响应头和全部 padding，避免把栈中的未初始化数据发送给客户端 */
23static void response_header(const struct dppd_management_request *request, struct dppd_management_response *response)
24{
25 memset(response, 0, sizeof(*response));
26 response->version = DPPD_MANAGEMENT_VERSION;
27 response->operation = request->operation;
28 response->size = sizeof(*response);
29 response->request_id = request->request_id;
30}

32/*
33 * 把进程内 device/topology 两份状态合并为不含指针的协议快照
34 * device 记录队列和 offload 能力，topology 记录 representor/switch-domain 语义；
35 * 两者缺少任意一份都说明该 port 不属于当前 dppd 管理范围
36 */
37static int build_port_info(const struct dppd_device_set *devices, uint16_t port_id,
38 struct dppd_management_port_info *output)
39{
40 const struct dppd_endpoint *endpoint;
41 const struct dppd_port *port;
42
43 if (devices == NULL || output == NULL)
44 return -EINVAL;
45 port = dppd_devices_find(devices, port_id);
46 endpoint = dppd_topology_find(&devices->topology, port_id);
47 if (port == NULL || endpoint == NULL)
48 return -ENOENT;
49
50 /*
51 * response_header 已清零整个响应，此处仍只逐字段赋值，禁止 memcpy 内部结构：
52 * dppd_port 的布局含未来可变字段，不能成为管理协议 ABI
53 */
54 output->port_id = port->port_id;
55 output->peer_port_id = port->peer_port_id;
56 output->socket_id = port->socket_id;
57 output->endpoint_kind = (uint8_t)endpoint->kind;
58 output->configured = port->configured;
59 output->started = port->started;
60 output->has_switch_domain = endpoint->has_switch_domain;
61 output->switch_domain_id = endpoint->switch_domain_id;
62 output->switch_port_id = endpoint->switch_port_id;
63 output->max_rx_queues = port->capabilities.max_rx_queues;
64 output->max_tx_queues = port->capabilities.max_tx_queues;
65 output->reta_size = port->capabilities.reta_size;
66 memcpy(output->mac, port->mac.addr_bytes, sizeof(output->mac));
67 output->rss_offloads = port->capabilities.rss_offloads;
68 output->rx_offloads = port->capabilities.rx_offloads;
69 output->tx_offloads = port->capabilities.tx_offloads;
70 output->device_capabilities = port->capabilities.device_capabilities;
71 output->configured_rss_hf = port->configured_rss_hf;
72 output->configured_tx_offloads = port->configured_tx_offloads;
73 output->link_state = (uint8_t)atomic_load_explicit(&port->link_state, memory_order_acquire);
74 memcpy(output->switch_name, endpoint->switch_name, sizeof(output->switch_name));
75 memcpy(output->driver_name, endpoint->driver_name, sizeof(output->driver_name));
76 output->role = (uint8_t)endpoint->role;
77 output->role_source = (uint8_t)endpoint->role_source;
78 output->role_error = endpoint->role_error;
79 output->if_index = endpoint->if_index;
80 output->has_controller = endpoint->has_controller;
81 output->has_pf = endpoint->has_pf;
82 output->has_function = endpoint->has_function;
83 output->has_physical_port = endpoint->has_physical_port;
84 output->controller_id = endpoint->controller_id;
85 output->pf_id = endpoint->pf_id;
86 output->function_id = endpoint->function_id;
87 output->physical_port_id = endpoint->physical_port_id;
88 memcpy(output->interface_name, endpoint->interface_name, sizeof(output->interface_name));
89 memcpy(output->device_name, endpoint->device_name, sizeof(output->device_name));
90 memcpy(output->phys_port_name, endpoint->phys_port_name, sizeof(output->phys_port_name));
91 return 0;
92}

111/**
112 * 健康查询只读取已发布的状态，不调用 PMD，也不读取或修复快照文件
113 * 因此恢复隔离期间仍可安全响应，部署脚本能区分主线程存活与转发服务已就绪
114 * 链路和线程是独立原子采样，查询结果不承诺这些状态在未来持续不变
115 */
116static void build_health(const struct dppd_control_service *control, const struct dppd_runtime *runtime,
117 struct dppd_management_health *output)
118{
119 struct dppd_control_persistence_status persistence;
120
121 dppd_control_persistence_status(control, &persistence);
122 output->recovery_state = (uint8_t)control->recovery_state;
123 output->recovery_last_error = control->recovery_last_error;
124 output->persistence_enabled = persistence.enabled;
125 output->persistence_dirty = persistence.dirty;
126 output->persisted_generation = persistence.persisted_generation;
127 output->repository_generation = persistence.current_generation;
128 output->persistence_last_error = persistence.last_error;
129 output->rule_count = dppd_rule_repository_count(&control->rules);
130 if (control->recovery_state != DPPD_CONTROL_RECOVERY_READY)
131 output->blockers |= DPPD_HEALTH_RECOVERY_REQUIRED;
132 if (persistence.enabled && persistence.dirty)
133 output->blockers |= DPPD_HEALTH_PERSISTENCE_DIRTY;
134 dppd_routing_status(control->routing, &output->routing);
135 if (output->routing.dirty)
136 output->blockers |= DPPD_HEALTH_ROUTING_DIRTY;
137 if (runtime == NULL) {
138 output->blockers |= DPPD_HEALTH_NO_RUNTIME;
139 return;
140 }
141
142 struct dppd_nd6_status neighbors;
143 dppd_nd6_status(runtime->nd6, &neighbors);
144 for (uint32_t i = 0; i < neighbors.local_count; ++i)
145 if (neighbors.locals[i].state != DPPD_ND6_PREFERRED)
146 output->blockers |= DPPD_HEALTH_ND6_NOT_READY;
147 output->workers_expected = runtime->config.nb_queues;
148 output->ports_expected = runtime->config.nb_ports;
149 if (atomic_load_explicit(&runtime->stop_requested, memory_order_acquire))
150 output->blockers |= DPPD_HEALTH_STOP_REQUESTED;
151 for (uint16_t i = 0; i < runtime->nb_workers; ++i) {
152 const unsigned int state = atomic_load_explicit(&runtime->workers[i].state, memory_order_acquire);
153
154 output->workers_running += state == DPPD_WORKER_RUNNING;
155 output->workers_failed += state == DPPD_WORKER_FAILED;
156 }
157 /** 检查实际队列数和启动状态，避免配置了两条队列却只有一条线程运行也报告就绪 */
158 if (!runtime->workers_started || output->workers_expected == 0 || runtime->nb_workers != output->workers_expected ||
159 output->workers_running != output->workers_expected)
160 output->blockers |= DPPD_HEALTH_WORKERS_NOT_RUNNING;
161 if (output->workers_failed != 0)
162 output->blockers |= DPPD_HEALTH_WORKER_FAILED;
163
164 for (uint16_t i = 0; i < runtime->devices.nb_ports; ++i) {
165 const struct dppd_port *port = &runtime->devices.ports[i];
166 const unsigned int link = atomic_load_explicit(&port->link_state, memory_order_acquire);
167
168 output->ports_started += port->configured && port->started;
169 if (atomic_load_explicit(&port->removed, memory_order_acquire))
170 output->blockers |= DPPD_HEALTH_DEVICE_REMOVED;
171 switch (link) {
172 case DPPD_LINK_UP:
173 output->links_up++;
174 break;
175 case DPPD_LINK_DOWN:
176 output->links_down++;
177 output->blockers |= DPPD_HEALTH_LINK_DOWN;
178 break;
179 case DPPD_LINK_UNSUPPORTED:
180 /** 保留原有转发行为，并单独显示未验证的链路，不把它冒充为 up */
181 output->links_unsupported++;
182 break;
183 default:
184 output->links_unknown++;
185 output->blockers |= DPPD_HEALTH_LINK_UNKNOWN;
186 break;
187 }
188 }
189 if (output->ports_expected == 0 || runtime->devices.nb_ports != output->ports_expected ||
190 output->ports_started != output->ports_expected)
191 output->blockers |= DPPD_HEALTH_PORTS_NOT_STARTED;
192 /** 最后读取设备组标记，覆盖遍历端口期间收到移除通知的情况 */
193 if (dppd_devices_removal_requested(&runtime->devices))
194 output->blockers |= DPPD_HEALTH_DEVICE_REMOVED;
195 output->ready = output->blockers == 0;
196}

198/** IPv6 单项修改仍提交完整双栈副本，不能用一族的修改覆盖另一族的活跃配置 */
199static int routing_change6(struct dppd_routing_service *service, const struct dppd_management_request *request)
200{
201 struct dppd_routing_status status;
202 dppd_routing_status(service, &status);
203 if (!status.enabled || status.dirty)
204 return status.enabled ? -EUCLEAN : -ENODEV;
205 bool route =
206 request->operation == DPPD_MANAGEMENT_ROUTE6_SET || request->operation == DPPD_MANAGEMENT_ROUTE6_DELETE;
207 bool remove =
208 request->operation == DPPD_MANAGEMENT_ROUTE6_DELETE || request->operation == DPPD_MANAGEMENT_NEIGHBOR6_DELETE;
209 uint64_t expected =
210 route ? request->payload.route6.expected_generation : request->payload.neighbor6.expected_generation;
211 if (expected == 0 || expected != status.generation)
212 return expected == 0 ? -EINVAL : -ESTALE;
213 struct dppd_ipv4_configuration configuration;
214 int rc = dppd_routing_get(service, &configuration);
215 if (rc != 0)
216 return rc;
217 if (route) {
218 const struct dppd_ipv6_route *entry = &request->payload.route6.entry;
219 const uint8_t zero[16] = {0};
220 if (entry->depth > 128 || entry->reserved != 0 ||
221 (remove && (entry->port_id != 0 || memcmp(entry->gateway, zero, 16) != 0)))
222 return -EINVAL;
223 uint32_t i = 0;
224 while (i < configuration.route6_count &&
225 (configuration.routes6[i].depth != entry->depth ||
226 memcmp(configuration.routes6[i].prefix, entry->prefix, 16) != 0))
227 i++;
228 if (remove) {
229 if (i == configuration.route6_count)
230 return -ENOENT;
231 memmove(configuration.routes6 + i, configuration.routes6 + i + 1,
232 (configuration.route6_count - i - 1) * sizeof(*entry));
233 configuration.route6_count--;
234 } else {
235 if (i == DPPD_IPV4_ROUTE_LIMIT)
236 return -ENOSPC;
237 configuration.routes6[i] = *entry;
238 if (i == configuration.route6_count)
239 configuration.route6_count++;
240 }
241 } else {
242 const struct dppd_ipv6_neighbor *entry = &request->payload.neighbor6.entry;
243 const uint8_t zero[6] = {0};
244 if (remove && memcmp(entry->mac, zero, 6) != 0)
245 return -EINVAL;
246 uint32_t i = 0;
247 while (i < configuration.neighbor6_count &&
248 (configuration.neighbors6[i].port_id != entry->port_id ||
249 memcmp(configuration.neighbors6[i].address, entry->address, 16) != 0))
250 i++;
251 if (remove) {
252 if (i == configuration.neighbor6_count)
253 return -ENOENT;
254 memmove(configuration.neighbors6 + i, configuration.neighbors6 + i + 1,
255 (configuration.neighbor6_count - i - 1) * sizeof(*entry));
256 configuration.neighbor6_count--;
257 } else {
258 if (i == DPPD_IPV4_NEIGHBOR_LIMIT)
259 return -ENOSPC;
260 configuration.neighbors6[i] = *entry;
261 if (i == configuration.neighbor6_count)
262 configuration.neighbor6_count++;
263 }
264 }
265 return dppd_routing_replace(service, expected, &configuration);
266}

268/** 单条管理修改也在完整副本上执行，路由和邻居的读取与提交共用精确全局版本 */
269static int routing_change(struct dppd_routing_service *service, const struct dppd_management_request *request)
270{
271 struct dppd_routing_status status;
272 dppd_routing_status(service, &status);
273 if (!status.enabled)
274 return -ENODEV;
275 if (status.dirty)
276 return -EUCLEAN;
277 bool route = request->operation == DPPD_MANAGEMENT_ROUTE_SET || request->operation == DPPD_MANAGEMENT_ROUTE_DELETE;
278 bool remove =
279 request->operation == DPPD_MANAGEMENT_ROUTE_DELETE || request->operation == DPPD_MANAGEMENT_NEIGHBOR_DELETE;
280 uint64_t expected =
281 route ? request->payload.route.expected_generation : request->payload.neighbor.expected_generation;
282 if (expected == 0)
283 return -EINVAL;
284 if (expected != status.generation)
285 return -ESTALE;
286 struct dppd_ipv4_configuration configuration;
287 int rc = dppd_routing_get(service, &configuration);
288 if (rc != 0)
289 return rc;
290 if (route) {
291 const struct dppd_ipv4_route *entry = &request->payload.route.entry;
292 if (entry->reserved != 0 || entry->depth > 32 || (remove && (entry->gateway_be != 0 || entry->port_id != 0)))
293 return -EINVAL;
294 uint32_t i;
295 for (i = 0; i < configuration.route_count; ++i)
296 if (configuration.routes[i].prefix_be == entry->prefix_be && configuration.routes[i].depth == entry->depth)
297 break;
298 if (remove) {
299 if (i == configuration.route_count)
300 return -ENOENT;
301 memmove(configuration.routes + i, configuration.routes + i + 1,
302 (configuration.route_count - i - 1) * sizeof(*entry));
303 configuration.route_count--;
304 } else {
305 if (i == DPPD_IPV4_ROUTE_LIMIT)
306 return -ENOSPC;
307 configuration.routes[i] = *entry;
308 if (i == configuration.route_count)
309 configuration.route_count++;
310 }
311 } else {
312 const struct dppd_ipv4_neighbor *entry = &request->payload.neighbor.entry;
313 const uint8_t zero[6] = {0};
314 if (remove && memcmp(entry->mac, zero, sizeof(zero)) != 0)
315 return -EINVAL;
316 uint32_t i;
317 for (i = 0; i < configuration.neighbor_count; ++i)
318 if (configuration.neighbors[i].port_id == entry->port_id &&
319 configuration.neighbors[i].address_be == entry->address_be)
320 break;
321 if (remove) {
322 if (i == configuration.neighbor_count)
323 return -ENOENT;
324 memmove(configuration.neighbors + i, configuration.neighbors + i + 1,
325 (configuration.neighbor_count - i - 1) * sizeof(*entry));
326 configuration.neighbor_count--;
327 } else {
328 if (i == DPPD_IPV4_NEIGHBOR_LIMIT)
329 return -ENOSPC;
330 configuration.neighbors[i] = *entry;
331 if (i == configuration.neighbor_count)
332 configuration.neighbor_count++;
333 }
334 }
335 return dppd_routing_replace(service, expected, &configuration);
336}

338static void build_rule_summary(const struct dppd_rule *rule, struct dppd_management_rule_summary *summary)
339{
340 uint16_t i;
341
342 summary->rule_id = rule->id;
343 summary->generation = rule->generation;
344 summary->group = rule->group;
345 summary->priority = rule->priority;
346 summary->nb_matches = rule->nb_matches;
347 summary->nb_actions = rule->nb_actions;
348 summary->domain = (uint8_t)rule->domain;
349 summary->fallback = (uint8_t)rule->fallback;
350 summary->install_port_id = rule->install_port_id;
351 /*
352 * mask 仅表达“包含哪些类型”，不表达顺序和具体参数；完整 canonical rule
353 * 仍必须通过 get 读取。enum 范围已由 repository apply 前的 rule 校验保证
354 */
355 for (i = 0; i < rule->nb_matches; ++i)
356 summary->match_mask |= 1U << rule->matches[i].type;
357 for (i = 0; i < rule->nb_actions; ++i)
358 summary->action_mask |= 1U << rule->actions[i].type;
359}

361/**
362 * 校验已接收管理请求的协议头，按 operation 调用控制层，并将业务结果写入 response
363 * control 提供规则管理，devices 和 runtime 供设备相关操作使用，request/response 是请求与输出区
364 * 普通 RULE_APPLY：management_poll → 本函数 → control.c/dppd_control_apply → control_apply_impl
365 * 返回 0 表示已形成响应，业务成功与否读取 response->status；必要指针无效返回 -EINVAL
366 */
367int dppd_management_handle(struct dppd_control_service *control, const struct dppd_device_set *devices,
368 struct dppd_runtime *runtime, const struct dppd_management_request *request,
369 struct dppd_management_response *response)
370{
371 int rc;
372
373 if (control == NULL || request == NULL || response == NULL)
374 return -EINVAL;
375 response_header(request, response);
376 /*
377 * size 必须精确匹配而不是仅检查“至少多大”。当前 payload 是本地 C ABI
378 * 接受更大或更小的结构都会让 union 字段偏移与 daemon 的解释不一致
379 */
380 if (request->version != DPPD_MANAGEMENT_VERSION || request->size != sizeof(*request)) {
381 response->status = -EPROTO;
382 return 0;
383 }
384 /**
385 * 回滚失败时 repository 从未发布，但 backend 可能还有实际 flow
386 * 只允许恢复状态、重试、只读健康、能力画像和失败/耗时历史，防止空账本被误认为可以正常运行
387 */
388 /** 同一轮 poll 可以处理多条请求，每条分发前再检查事件，不能沿用轮首的设备状态 */
389 if (runtime != NULL)
390 (void)dppd_runtime_device_poll(runtime);
391 if (control->recovery_state != DPPD_CONTROL_RECOVERY_READY &&
392 request->operation != DPPD_MANAGEMENT_RECOVERY_STATUS && request->operation != DPPD_MANAGEMENT_RECOVERY_RETRY &&
393 request->operation != DPPD_MANAGEMENT_HEALTH_GET && request->operation != DPPD_MANAGEMENT_CAPABILITY_GET &&
394 request->operation != DPPD_MANAGEMENT_RULE_METRICS && request->operation != DPPD_MANAGEMENT_RULE_LATENCY &&
395 request->operation != DPPD_MANAGEMENT_RULE_HISTORY && request->operation != DPPD_MANAGEMENT_ROUTING_STATUS &&
396 request->operation != DPPD_MANAGEMENT_ROUTING_GET && request->operation != DPPD_MANAGEMENT_CONNTRACK_STATUS &&
397 request->operation != DPPD_MANAGEMENT_CONNTRACK_LIST && request->operation != DPPD_MANAGEMENT_NAT_STATUS &&
398 request->operation != DPPD_MANAGEMENT_DNAT_LIST && request->operation != DPPD_MANAGEMENT_ND6_STATUS &&
399 request->operation != DPPD_MANAGEMENT_ND6_LIST && request->operation != DPPD_MANAGEMENT_TRANSFER_PLAN &&
400 request->operation != DPPD_MANAGEMENT_AGE_STATUS && request->operation != DPPD_MANAGEMENT_SHARED_COUNT_GET &&
401 request->operation != DPPD_MANAGEMENT_SHARED_COUNT_LIST &&
402 request->operation != DPPD_MANAGEMENT_DEVICE_STATUS && request->operation != DPPD_MANAGEMENT_DEVICE_RESET) {
403 response->status = -EUCLEAN;
404 return 0;
405 }
406
407 switch (request->operation) {
408 case DPPD_MANAGEMENT_DEVICE_STATUS:
409 dppd_runtime_device_status(runtime, &response->payload.device);
410 rc = runtime != NULL && response->payload.device.bound ? 0 : -ENOTSUP;
411 break;
412 case DPPD_MANAGEMENT_DEVICE_RESET:
413 rc = dppd_runtime_device_reset(runtime, request->payload.device_reset.expected_sequence);
414 dppd_runtime_device_status(runtime, &response->payload.device);
415 break;
416 case DPPD_MANAGEMENT_SHARED_COUNT_CREATE:
417 rc = dppd_control_shared_count_create(control, &request->payload.shared_count_create,
418 &response->payload.shared_count);
419 break;
420 case DPPD_MANAGEMENT_SHARED_COUNT_GET:
421 rc = dppd_control_shared_count_get(control, request->payload.shared_count.id,
422 request->payload.shared_count.expected, &response->payload.shared_count);
423 break;
424 case DPPD_MANAGEMENT_SHARED_COUNT_QUERY:
425 rc = dppd_control_shared_count_query(control, request->payload.shared_count.id,
426 request->payload.shared_count.expected, &response->payload.shared_count_result);
427 break;
428 case DPPD_MANAGEMENT_SHARED_COUNT_DELETE:
429 rc = dppd_control_shared_count_delete(control, request->payload.shared_count.id,
430 request->payload.shared_count.expected, &response->payload.shared_count_deleted.revision);
431 if (rc == 0) {
432 response->payload.shared_count_deleted.id = request->payload.shared_count.id;
433 response->payload.shared_count_deleted.generation = request->payload.shared_count.expected;
434 }
435 break;
436 case DPPD_MANAGEMENT_SHARED_COUNT_LIST:
437 rc = dppd_control_shared_count_list(control, request->payload.shared_count_list.after_id,
438 request->payload.shared_count_list.expected_revision, &response->payload.shared_count_page);
439 break;
440 case DPPD_MANAGEMENT_AGE_STATUS:
441 rc = dppd_control_age_status(control, &response->payload.age_status);
442 break;
443 case DPPD_MANAGEMENT_RULE_AGE_QUERY:
444 rc = dppd_control_query_age(control, request->payload.count_query.rule_id,
445 request->payload.count_query.expected_generation, &response->payload.age_result);
446 break;
447 case DPPD_MANAGEMENT_TRANSFER_PLAN:
448 case DPPD_MANAGEMENT_TRANSFER_APPLY:
449 rc = dppd_control_transfer(control, &request->payload.transfer,
450 request->operation == DPPD_MANAGEMENT_TRANSFER_APPLY, &response->payload.transfer.graph,
451 response->payload.transfer.results, response->payload.transfer.message,
452 sizeof(response->payload.transfer.message));
453 break;
454 case DPPD_MANAGEMENT_RULE_HISTORY:
455 /** 协议查询只复制完成态记录，错误参数或过期版本不产生新的失败事件 */
456 rc = dppd_control_rule_history(control, request->payload.rule_history.after_event_id,
457 request->payload.rule_history.expected_revision, &response->payload.rule_history);
458 break;
459 case DPPD_MANAGEMENT_RULE_LATENCY:
460 /** 耗时历史独立于当前账本，已删除或被撤销的成功提交仍然保留 */
461 dppd_control_rule_latency(control, &response->payload.rule_latency);
462 rc = 0;
463 break;
464 case DPPD_MANAGEMENT_RULE_METRICS:
465 /** 完成态的值复制，不发起事务或驱动访问，也不清除失败历史 */
466 dppd_control_rule_metrics(control, &response->payload.rule_metrics);
467 rc = 0;
468 break;
469 case DPPD_MANAGEMENT_HEALTH_GET:
470 /** 状态查询成功与已就绪是两件事，未就绪原因放在 payload 而不是伪造协议错误 */
471 build_health(control, runtime, &response->payload.health);
472 rc = 0;
473 break;
474 case DPPD_MANAGEMENT_STATS_QUERY:
475 if (request->payload.stats_query.reserved != 0)
476 rc = -EINVAL;
477 else if (runtime == NULL)
478 rc = -ENODEV;
479 else
480 rc = dppd_runtime_stats_query(runtime, request->payload.stats_query.port_id,
481 request->payload.stats_query.queue_id, &response->payload.stats);
482 break;
483 case DPPD_MANAGEMENT_PING:
484 /* ping 不触碰硬件，仅返回 desired state 的轻量健康信息 */
485 response->payload.pong.repository_generation = dppd_rule_repository_generation(&control->rules);
486 response->payload.pong.rule_count = dppd_rule_repository_count(&control->rules);
487 rc = 0;
488 break;
489 case DPPD_MANAGEMENT_RULE_APPLY:
490 /**
491 * CLI 已将参数转成 request 中的安装端口、完整规则和预期版本，这里传给控制层
492 * dppd_control_apply → control_apply_impl → 规划、事务安装、内存记录更新和保存
493 * 此分支只分发请求，rc 在本函数末尾写入 response->status，交回 poll 发送给客户端
494 */
495 rc = dppd_control_apply(control, request->payload.apply.install_port_id, &request->payload.apply.rule,
496 request->payload.apply.expected_generation, &response->payload.apply);
497 break;
498 case DPPD_MANAGEMENT_RULE_CREATE_BATCH:
499 /*
500 * 防止本地 ABI 数组越界；这里不逐条验证规则，也不尝试在协议层拼装部分结果
501 * 规则语义、预检和 actual/desired 的原子边界都属于 control 层，任何错误统一
502 * 写入 response.status，客户端以整个请求为失败处理
503 */
504 if (request->payload.create_batch.count == 0 ||
505 request->payload.create_batch.count > DPPD_MANAGEMENT_BATCH_CREATE_MAX) {
506 rc = -EINVAL;
507 break;
508 }
509 /*
510 * 预先回显 count 只定义成功响应的数组长度；若 control 返回错误，客户端必须先
511 * 检查 status，不能据此读取任何单项结果。这样未来扩展 per-item 诊断也不会
512 * 改变“批量请求只有一个提交结论”的语义
513 */
514 response->payload.create_batch.count = request->payload.create_batch.count;
515 rc = dppd_control_create_batch(control, request->payload.create_batch.rules,
516 request->payload.create_batch.count, response->payload.create_batch.rules);
517 break;
518 case DPPD_MANAGEMENT_RULE_UPDATE_BATCH: {
519 /**
520 * 固定数组最多容纳四条规则，读取数组前先限制 count 的范围
521 * reserved 是留给将来协议扩展的位置，当前版本只接受全零，避免含糊解释请求
522 * 规则内容、版本冲突和事务回滚交给控制层统一处理，协议层不逐条执行更新
523 */
524 static const uint8_t zero_reserved[6] = {0};
525
526 if (request->payload.update_batch.count < 2 ||
527 request->payload.update_batch.count > DPPD_MANAGEMENT_BATCH_UPDATE_MAX ||
528 memcmp(request->payload.update_batch.reserved, zero_reserved, sizeof(zero_reserved)) != 0) {
529 rc = -EINVAL;
530 break;
531 }
532 rc = dppd_control_update_batch(control, request->payload.update_batch.rules,
533 request->payload.update_batch.count, response->payload.update_batch.rules);
534 /**
535 * 只在整批成功时返回结果数量，失败时清空整个结果区域
536 * 防止客户端把失败过程中留下的某个 generation 误认为已经成功的单条回执
537 * 是否已经生效但保存失败，要通过持久化或恢复状态接口进一步确认
538 */
539 if (rc == 0)
540 response->payload.update_batch.count = request->payload.update_batch.count;
541 else
542 memset(&response->payload.update_batch, 0, sizeof(response->payload.update_batch));
543 break;
544 }
545 case DPPD_MANAGEMENT_RULE_GET:
546 /* get 只读取 desired repository，不通过 rte_flow 反查硬件对象 */
547 rc = dppd_rule_repository_get(&control->rules, request->payload.get.rule_id, &response->payload.rule);
548 break;
549 case DPPD_MANAGEMENT_RULE_DELETE:
550 /* 删除同样携带 expected_generation，避免旧客户端误删已更新规则 */
551 rc = dppd_control_remove(control, request->payload.delete_rule.rule_id,
552 request->payload.delete_rule.expected_generation, &response->payload.delete_rule.removed,
553 &response->payload.delete_rule.generation);
554 break;
555 case DPPD_MANAGEMENT_RULE_DELETE_BATCH:
556 /*
557 * 与 create_batch 一样，协议层只防御固定数组边界。控制层负责精确 generation
558 * 预检、actual 删除补偿与 recovery 隔离；不能在此把单条 delete 循环成伪批量
559 */
560 if (request->payload.delete_batch.count < 2 ||
561 request->payload.delete_batch.count > DPPD_MANAGEMENT_BATCH_REMOVE_MAX) {
562 rc = -EINVAL;
563 break;
564 }
565 response->payload.delete_batch.count = request->payload.delete_batch.count;
566 rc = dppd_control_remove_batch(control, request->payload.delete_batch.rules,
567 request->payload.delete_batch.count, response->payload.delete_batch.rules);
568 break;
569 case DPPD_MANAGEMENT_PORT_GET:
570 /*
571 * 这是启动时能力快照，不在每次查询时调用 PMD；既避免主循环中的设备探测
572 * 开销，也保证客户端看到的是 dppd 实际完成配置后的稳定状态
573 */
574 rc = build_port_info(devices, request->payload.port_get.port_id, &response->payload.port);
575 break;
576 case DPPD_MANAGEMENT_CAPABILITY_GET:
577 case DPPD_MANAGEMENT_CAPABILITY_CLEAR: {
578 const uint8_t empty[6] = {0};
579
580 if (memcmp(request->payload.capability.reserved, empty, sizeof(empty)) != 0) {
581 rc = -EINVAL;
582 break;
583 }
584 rc = build_capability_profile(control, devices, request->payload.capability.port_id,
585 &response->payload.capability);
586 if (rc == 0 && request->operation == DPPD_MANAGEMENT_CAPABILITY_CLEAR) {
587 rc = dppd_flow_probe_cache_clear(control->rte_flow.probes, request->payload.capability.port_id);
588 if (rc == 0)
589 rc = build_capability_profile(control, devices, request->payload.capability.port_id,
590 &response->payload.capability);
591 }
592 break;
593 }
594 case DPPD_MANAGEMENT_CAPABILITY_PROBE: {
595 const uint8_t empty[5] = {0};
596 const struct dppd_port *port =
597 devices == NULL ? NULL : dppd_devices_find(devices, request->payload.probe.install_port_id);
598
599 if (memcmp(request->payload.probe.reserved, empty, sizeof(empty)) != 0)
600 rc = -EINVAL;
601 else if (port == NULL)
602 rc = -ENOENT;
603 /** 已知停止或移除的设备不得再发起驱动探测，画像查询仍只读缓存 */
604 else if (!port->started || dppd_devices_removal_requested(devices) ||
605 atomic_load_explicit(&port->removed, memory_order_acquire))
606 rc = -ENODEV;
607 else
608 rc = dppd_control_probe_rule(control, request->payload.probe.install_port_id, &request->payload.probe.rule,
609 request->payload.probe.refresh, &response->payload.probe);
610 break;
611 }
612 case DPPD_MANAGEMENT_RULE_STATUS:
613 /** 只复制控制面安装记录，隔离检查沿用普通规则查询的入口限制 */
614 rc = dppd_control_rule_status(control, request->payload.rule_status.rule_id,
615 request->payload.rule_status.expected_generation, &response->payload.rule_status);
616 break;
617 case DPPD_MANAGEMENT_RULE_COUNT_QUERY:
618 /*
619 * COUNT 查询不改变 repository generation，也不重置 PMD counter
620 * hits/bytes 是否可用最终由目标 PMD 的 rte_flow_query() 决定
621 */
622 rc = dppd_control_query_count(control, request->payload.count_query.rule_id,
623 request->payload.count_query.expected_generation, &response->payload.count);
624 break;
625 case DPPD_MANAGEMENT_RULE_LIST: {
626 struct dppd_rule rules[DPPD_MANAGEMENT_RULE_PAGE_SIZE];
627 struct dppd_management_rule_page *page = &response->payload.rule_page;
628 uint32_t count = 0;
629 bool has_more = false;
630 uint32_t i;
631
632 rc = dppd_rule_repository_list(&control->rules, request->payload.list.after_rule_id,
633 request->payload.list.expected_repository_generation, rules, DPPD_MANAGEMENT_RULE_PAGE_SIZE, &count,
634 &has_more, &page->repository_generation);
635 if (rc != 0)
636 break;
637 page->total_count = dppd_rule_repository_count(&control->rules);
638 page->count = (uint16_t)count;
639 page->has_more = has_more;
640 page->next_after_rule_id = request->payload.list.after_rule_id;
641 for (i = 0; i < count; ++i) {
642 build_rule_summary(&rules[i], &page->rules[i]);
643 page->next_after_rule_id = rules[i].id;
644 }
645 break;
646 }
647 case DPPD_MANAGEMENT_PERSISTENCE_STATUS:
648 /*
649 * 状态查询只读取 control service 中已经发布的字段，不触发磁盘 I/O
650 * last_error 保留最近一次落盘失败的负 errno，便于 CLI 给出直接故障原因
651 */
652 dppd_control_persistence_status(control, &response->payload.persistence);
653 rc = 0;
654 break;
655 case DPPD_MANAGEMENT_PERSISTENCE_FLUSH:
656 /*
657 * flush 不修改硬件或 repository，只把当前完整 desired state 重新落盘
658 * 无论成功与否都填充状态 payload；成功响应可直接展示最新 generation
659 * 失败时 response.status 仍保留底层存储的负 errno
660 */
661 rc = dppd_control_persistence_flush(control);
662 dppd_control_persistence_status(control, &response->payload.persistence);
663 break;
664 case DPPD_MANAGEMENT_RECOVERY_STATUS:
665 /* 只读快照，不触碰任何 backend 对象，隔离模式和正常模式均可查询 */
666 dppd_control_recovery_status(control, &response->payload.recovery);
667 rc = 0;
668 break;
669 case DPPD_MANAGEMENT_RECOVERY_RETRY:
670 /*
671 * retry 只尝试删除本进程仍持有 handle 的遗留对象；成功后由主循环退出
672 * 后续重启重新执行完整 snapshot 恢复，而不在此处发布空 repository
673 */
674 rc = dppd_control_reconciliation_retry(control);
675 dppd_control_recovery_status(control, &response->payload.recovery);
676 break;
677 case DPPD_MANAGEMENT_ROUTING_GET:
678 rc = dppd_routing_get(control->routing, &response->payload.routing);
679 break;
680 case DPPD_MANAGEMENT_CONNTRACK_STATUS:
681 case DPPD_MANAGEMENT_NAT_STATUS:
682 dppd_conntrack_status(control->conntrack, dppd_monotonic_ms(), &response->payload.conntrack);
683 rc = 0;
684 break;
685 case DPPD_MANAGEMENT_ND6_STATUS:
686 dppd_nd6_status(runtime == NULL ? NULL : runtime->nd6, &response->payload.nd6);
687 rc = 0;
688 break;
689 case DPPD_MANAGEMENT_ND6_LIST:
690 rc = dppd_nd6_list(runtime == NULL ? NULL : runtime->nd6, request->payload.conntrack.after_id,
691 request->payload.conntrack.expected_revision, &response->payload.dynamic_neighbors);
692 break;
693 case DPPD_MANAGEMENT_ND6_FLUSH:
694 rc = dppd_nd6_flush(runtime == NULL ? NULL : runtime->nd6, request->payload.conntrack.expected_revision);
695 break;
696 case DPPD_MANAGEMENT_DNAT_LIST:
697 rc = dppd_conntrack_dnat_list(control->conntrack, request->payload.conntrack.after_id, &response->payload.dnat);
698 break;
699 case DPPD_MANAGEMENT_CONNTRACK_LIST:
700 rc = dppd_conntrack_list(control->conntrack, dppd_monotonic_ms(), request->payload.conntrack.after_id,
701 request->payload.conntrack.expected_revision, &response->payload.connections);
702 break;
703 case DPPD_MANAGEMENT_CONNTRACK_FLUSH:
704 rc = dppd_conntrack_flush(control->conntrack, request->payload.conntrack.expected_revision);
705 dppd_conntrack_status(control->conntrack, dppd_monotonic_ms(), &response->payload.conntrack);
706 break;
707 case DPPD_MANAGEMENT_ROUTING_STATUS:
708 dppd_routing_status(control->routing, &response->payload.routing_status);
709 rc = 0;
710 break;
711 case DPPD_MANAGEMENT_ROUTING_REPLACE:
712 rc = dppd_routing_replace(control->routing, request->payload.routing_replace.expected_generation,
713 &request->payload.routing_replace.configuration);
714 dppd_routing_status(control->routing, &response->payload.routing_status);
715 break;
716 case DPPD_MANAGEMENT_ROUTE_SET:
717 case DPPD_MANAGEMENT_ROUTE_DELETE:
718 case DPPD_MANAGEMENT_NEIGHBOR_SET:
719 case DPPD_MANAGEMENT_NEIGHBOR_DELETE:
720 rc = routing_change(control->routing, request);
721 dppd_routing_status(control->routing, &response->payload.routing_status);
722 break;
723 case DPPD_MANAGEMENT_ROUTING_FLUSH:
724 rc = dppd_routing_flush(control->routing);
725 dppd_routing_status(control->routing, &response->payload.routing_status);
726 break;
727 case DPPD_MANAGEMENT_ROUTE6_SET:
728 case DPPD_MANAGEMENT_ROUTE6_DELETE:
729 case DPPD_MANAGEMENT_NEIGHBOR6_SET:
730 case DPPD_MANAGEMENT_NEIGHBOR6_DELETE:
731 rc = routing_change6(control->routing, request);
732 dppd_routing_status(control->routing, &response->payload.routing_status);
733 break;
734 default:
735 rc = -ENOTSUP;
736 break;
737 }
738 /* transport 已经成功解析请求，因此业务错误放入响应而不作为本函数返回值 */
739 response->status = rc;
740 return 0;
741}

743static int start_server(struct dppd_management_server *server, struct dppd_control_service *control,
744 struct dppd_flow_management *flows, const struct dppd_device_set *devices, struct dppd_runtime *runtime,
745 const char *socket_path)
746{
747 struct sockaddr_un address;
748 size_t path_length;
749 int fd;
750
751 if (server == NULL || (control == NULL) == (flows == NULL) || devices == NULL || socket_path == NULL)
752 return -EINVAL;
753 path_length = strlen(socket_path);
754 if (path_length == 0 || path_length > DPPD_MANAGEMENT_SOCKET_PATH_MAX)
755 return -ENAMETOOLONG;
756 /* 先建立明确的未启动状态，保证后续任一步失败都不会留下可误关闭的 fd */
757 memset(server, 0, sizeof(*server));
758 server->socket_fd = -1;
759 server->runtime = runtime;
760
761 /*
762 * SOCK_SEQPACKET 让一次 send 对应一次 recv，避免自行设计流式 framing；
763 * NONBLOCK 防止管理客户端卡住主循环；CLOEXEC 防止未来启动子进程时泄漏 fd
764 */
765 fd = socket(AF_UNIX, SOCK_SEQPACKET | SOCK_NONBLOCK | SOCK_CLOEXEC, 0);
766 if (fd < 0)
767 return -errno;
768 memset(&address, 0, sizeof(address));
769 address.sun_family = AF_UNIX;
770 memcpy(address.sun_path, socket_path, path_length + 1U);
771 /*
772 * 不在 bind 前无条件 unlink：配置错误或另一个存活 daemon 使用同一路径时
773 * 擅自删除会绕过单实例保护。异常退出留下的路径需由部署层确认后清理
774 */
775 if (bind(fd, (const struct sockaddr *)&address, sizeof(address)) != 0) {
776 const int error = -errno;
777 close(fd);
778 return error;
779 }
780 /* socket 默认仅允许启动 dppd 的用户访问，避免未授权规则变更 */
781 if (chmod(socket_path, S_IRUSR | S_IWUSR) != 0 || listen(fd, 16) != 0) {
782 const int error = -errno;
783 close(fd);
784 unlink(socket_path);
785 return error;
786 }
787 /* 只有 bind/chmod/listen 全部成功后才发布 started 状态 */
788 server->socket_fd = fd;
789 server->control = control;
790 server->flows = flows;
791 server->devices = devices;
792 memcpy(server->socket_path, socket_path, path_length + 1U);
793 server->started = true;
794 printf("[dppd] management socket=%s\n", socket_path);
795 return 0;
796}

798int dppd_management_start(struct dppd_management_server *server, struct dppd_control_service *control,
799 const struct dppd_device_set *devices, struct dppd_runtime *runtime, const char *socket_path)
800{
801 return start_server(server, control, NULL, devices, runtime, socket_path);
802}

810/**
811 * 由 dppd 主循环调用，在本机 Unix socket 上受理有预算的请求并发送响应
812 * server 必须已经启动，成功返回 0，参数或连接处理错误返回负错误码
813 * dppctl/exchange 发送请求 → accept4 接受连接 → recv 读取固定结构体
814 * → 普通控制模式调用 management_handle → send 响应 → 关闭本次连接
815 * 若 server->flows 非空则走 flow_management_handle，属于另一种配置模式
816 */
817int dppd_management_poll(struct dppd_management_server *server)
818{
819 uint32_t handled = 0;
820
821 if (server == NULL || !server->started)
822 return -EINVAL;
823
824 /* 每轮设置预算，避免持续连接让统计、信号和退出处理长期得不到调度 */
825 while (handled < DPPD_MANAGEMENT_POLL_BUDGET) {
826 struct dppd_management_request request;
827 struct dppd_management_response response;
828 ssize_t received;
829 struct pollfd ready;
830 int client;
831 int poll_rc;
832
833 /* 客户端 fd 同样保持非阻塞并禁止跨 exec 继承 */
834 client = accept4(server->socket_fd, NULL, NULL, SOCK_NONBLOCK | SOCK_CLOEXEC);
835 if (client < 0) {
836 if (errno == EAGAIN || errno == EWOULDBLOCK)
837 return 0;
838 if (errno == EINTR)
839 continue;
840 return -errno;
841 }
842 handled++;
843
844 /*
845 * 监听 socket 可读只表示连接已排队，不保证请求数据已经到达。这里最多等待
846 * 20ms，既消除 connect/send 竞态，也避免异常客户端卡住 daemon 主循环
847 */
848 ready.fd = client;
849 ready.events = POLLIN;
850 ready.revents = 0;
851 do {
852 poll_rc = poll(&ready, 1, DPPD_MANAGEMENT_CLIENT_WAIT_MS);
853 } while (poll_rc < 0 && errno == EINTR);
854 if (poll_rc <= 0 || (ready.revents & POLLIN) == 0) {
855 close(client);
856 if (poll_rc < 0)
857 return -errno;
858 continue;
859 }
860
861 /**
862 * 每个连接只收一个固定大小请求，长度不同就拒绝，不尝试解释不完整的结构体
863 * MSG_TRUNC 会返回消息原始长度，因此过长请求也不会被误当成大小正确的请求
864 */
865 memset(&request, 0, sizeof(request));
866 received = recv(client, &request, sizeof(request), MSG_TRUNC);
867 if (received == (ssize_t)sizeof(request)) {
868 if (server->flows != NULL)
869 (void)dppd_flow_management_handle(server->flows, server->runtime, &request, &response);
870 else
871 (void)dppd_management_handle(server->control, server->devices, server->runtime, &request, &response);
872 } else {
873 memset(&response, 0, sizeof(response));
874 response.version = DPPD_MANAGEMENT_VERSION;
875 response.size = sizeof(response);
876 response.status = received < 0 ? -errno : -EMSGSIZE;
877 }
878 /*
879 * 客户端提前退出属于连接级事件，不应终止 daemon；MSG_NOSIGNAL 避免
880 * 对已关闭 peer 发送时触发 SIGPIPE。响应很小，SEQPACKET 要么整包发送要么失败
881 */
882 if (send(client, &response, sizeof(response), MSG_NOSIGNAL) < 0 && errno != EPIPE && errno != ECONNRESET) {
883 const int error = -errno;
884 close(client);
885 return error;
886 }
887 close(client);
888 /**
889 * 发现恢复状态后立即结束本轮请求处理，让主循环先停止软件工作线程
890 * 如果继续处理已排队的清理请求，就可能在线程仍转发报文时提前删除残留对象
891 * 下一轮轮询才接收恢复查询或重试请求，确保生命周期顺序清楚
892 */
893 if (server->control != NULL && server->control->recovery_state != DPPD_CONTROL_RECOVERY_READY)
894 break;
895 if (server->flows != NULL) {
896 struct dppd_flow_store_status status;
897 dppd_flow_store_status(server->flows->store, &status);
898 if (!status.ready)
899 break;
900 }
901 }
902 return 0;
903}

905void dppd_management_stop(struct dppd_management_server *server)
906{
907 if (server == NULL || !server->started)
908 return;
909 /*
910 * 先关闭监听 fd，再删除路径，新的客户端不会在退出窗口建立有效连接
911 * started 检查使该函数可安全用于正常退出和多层 goto 清理路径
912 */
913 close(server->socket_fd);
914 unlink(server->socket_path);
915 memset(server, 0, sizeof(*server));
916 server->socket_fd = -1;
917}
