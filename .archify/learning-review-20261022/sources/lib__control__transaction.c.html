6/** 根据规划结果选择软件或硬件的操作接口，让事务流程不用分别编写两套安装逻辑 */
7static const struct dppd_transaction_backend *backend_for(const struct dppd_transaction_backends *backends,
8 enum dppd_plan_backend backend)
9{
10 switch (backend) {
11 case DPPD_PLAN_BACKEND_SOFTWARE:
12 return &backends->software;
13 case DPPD_PLAN_BACKEND_RTE_FLOW:
14 return &backends->rte_flow;
15 }
16 return NULL;
17}

19/**
20 * 校验、准备、提交和回滚是一个后端必须提供的四项能力
21 * finalize 只用于释放额外的临时资源，没有这类资源的后端可以不提供它
22 */
23static int backend_valid(const struct dppd_transaction_backend *backend)
24{
25 return backend != NULL && backend->validate != NULL && backend->prepare != NULL && backend->commit != NULL &&
26 backend->rollback != NULL;
27}

29/**
30 * 尽力撤销本次事务已经占用的资源，不能因为某一条撤销失败就放弃其余条目
31 * commit_attempts 表示有多少条规则已经调用过 commit，包括调用了但返回错误的那一条
32 * 这个信息帮助后端区分“只预留了资源”和“可能已经产生安装副作用”
33 *
34 * 软件规则示例：R1、R2、R3 均已 prepare，R1 提交成功，R2 提交失败，R3 尚未提交
35 * 逆序清理：R3 → published 为 false → 释放准备记录
36 * R2 → published 为 false → 释放准备记录，失败的 commit 已清理未发布副本
37 * R1 → published 为 true → 删除已安装版本，再释放准备记录
38 * 本例 commit_attempts = 2，R1、R2 的 commit_was_attempted 为 true，R3 为 false
39 * 任一回滚失败仍继续清理其余条目，首个回滚错误保存在 rollback_code
40 */
41static void rollback_items(struct dppd_transaction *transaction, const struct dppd_transaction_backends *backends,
42 uint32_t commit_attempts)
43{
44 uint32_t i = transaction->nb_items;
45
46 /** 按最后准备或提交的对象优先撤销，避免先释放了后续对象仍可能依赖的资源 */
47 transaction->state = DPPD_TRANSACTION_ROLLING_BACK;
48 while (i > 0) {
49 struct dppd_transaction_item *item;
50 const struct dppd_transaction_backend *backend;
51 bool commit_was_attempted;
52 int rc;
53
54 --i;
55 item = &transaction->items[i];
56 /**
57 * 只经过校验的条目没有占用资源，不需要回滚
58 * 失败条目若留下非零 token，说明仍有需要清理的资源，不能仅因为 FAILED 就跳过
59 */
60 if (item->state != DPPD_TRANSACTION_ITEM_PREPARED && item->state != DPPD_TRANSACTION_ITEM_COMMITTED &&
61 item->state != DPPD_TRANSACTION_ITEM_FAILED)
62 continue;
63 if (item->state == DPPD_TRANSACTION_ITEM_FAILED && item->backend_token == 0)
64 continue;
65 backend = backend_for(backends, item->plan.backend);
66 commit_was_attempted = i < commit_attempts;
67 rc = backend->rollback(backend->context, item, item->backend_token, commit_was_attempted);
68 if (rc == 0)
69 item->state = DPPD_TRANSACTION_ITEM_ROLLED_BACK;
70 /** 保留第一个清理错误，同时继续尝试清理其余条目，便于上层判断是否必须隔离 */
71 else if (transaction->rollback_code == 0) {
72 transaction->rollback_code = rc;
73 transaction->rollback_item = i;
74 }
75 }
76 transaction->state = transaction->rollback_code == 0 ? DPPD_TRANSACTION_ROLLED_BACK : DPPD_TRANSACTION_FAILED;
77}

79/**
80 * 为一批已完成规划的规则建立事务状态，不复制规则数组，也不安装任何对象
81 * 调用方持有 items 的内存，必须让它存活到事务结束
82 * 这里会清空旧 token，因此不能用 init 代替尚未完成的事务清理
83 */
84int dppd_transaction_init(struct dppd_transaction *transaction, uint64_t transaction_id,
85 struct dppd_transaction_item *items, uint32_t nb_items)
86{
87 uint32_t i;
88
89 if (transaction == NULL || transaction_id == 0 || items == NULL || nb_items == 0)
90 return -EINVAL;
91 transaction->id = transaction_id;
92 transaction->state = DPPD_TRANSACTION_NEW;
93 transaction->items = items;
94 transaction->nb_items = nb_items;
95 transaction->failure_code = 0;
96 transaction->rollback_code = 0;
97 transaction->failure_stage = DPPD_RULE_STAGE_NONE;
98 transaction->failure_item = UINT32_MAX;
99 transaction->rollback_item = UINT32_MAX;
100 for (i = 0; i < nb_items; ++i) {
101 items[i].state = DPPD_TRANSACTION_ITEM_PENDING;
102 items[i].backend_token = 0;
103 items[i].error_code = 0;
104 }
105 return 0;
106}

108/**
109 * 按“全部校验、全部准备、逐条提交”的顺序安装规则
110 * 校验阶段失败直接结束，准备或提交失败则自动回滚已占用的资源
111 * 返回值说明原操作为什么失败，rollback_code 另行说明清理是否也失败
112 * 返回成功仅表示后端对象已安装，规则账本发布和快照保存仍由控制层负责
113 *
114 * validate 全部成功 → prepare 全部成功 → commit 逐条成功 → COMMITTED
115 * validate 失败 → 尚未占用后端资源，直接返回错误
116 * prepare / commit 失败 → rollback_items 逆序清理 → 返回原始错误
117 * rollback_code = 0 → 清理全部成功，事务为 ROLLED_BACK
118 * rollback_code 非 0 → 至少一项清理失败，事务为 FAILED，上层继续处理
119 * 软件 commit 每条都会发布快照，整批尚未完成时，先提交的规则也可能已被 worker 使用
120 * 全部提交成功后，由控制层决定 finalize 收尾，或调用 rollback_committed 撤销
121 */
122int dppd_transaction_run(struct dppd_transaction *transaction, const struct dppd_transaction_backends *backends)
123{
124 uint32_t i;
125 int rc;
126
127 if (transaction == NULL || backends == NULL || transaction->state != DPPD_TRANSACTION_NEW)
128 return -EINVAL;
129
130 transaction->state = DPPD_TRANSACTION_VALIDATING;
131 /**
132 * 第一阶段只询问“能不能做”，不允许后端预留槽位或安装规则
133 * 同时核对规划的 ID 和版本确实对应当前规则，避免拿错计划安装另一个版本
134 */
135 for (i = 0; i < transaction->nb_items; ++i) {
136 struct dppd_transaction_item *item = &transaction->items[i];
137 const struct dppd_transaction_backend *backend = backend_for(backends, item->plan.backend);
138
139 if (item->rule.id == 0 || item->plan.rule_id != item->rule.id ||
140 item->plan.rule_generation != item->rule.generation)
141 rc = -EINVAL;
142 else if (!backend_valid(backend))
143 rc = -ENOTSUP;
144 else
145 rc = backend->validate(backend->context, item);
146 if (rc != 0) {
147 item->state = DPPD_TRANSACTION_ITEM_FAILED;
148 item->error_code = rc;
149 transaction->failure_code = rc;
150 transaction->failure_stage = DPPD_RULE_STAGE_VALIDATE;
151 transaction->failure_item = i;
152 transaction->state = DPPD_TRANSACTION_FAILED;
153 return rc;
154 }
155 item->state = DPPD_TRANSACTION_ITEM_VALIDATED;
156 }
157
158 transaction->state = DPPD_TRANSACTION_PREPARING;
159 /**
160 * 第二阶段为每条规则准备可撤销的资源，例如空槽位或软件发布需要的临时对象
161 * 后一条准备失败时，前面已经准备好的资源也必须归还
162 * 传入零次 commit 尝试，告诉回滚逻辑这一阶段还没有正式安装对象
163 */
164 for (i = 0; i < transaction->nb_items; ++i) {
165 struct dppd_transaction_item *item = &transaction->items[i];
166 const struct dppd_transaction_backend *backend = backend_for(backends, item->plan.backend);
167
168 rc = backend->prepare(backend->context, item, &item->backend_token);
169 if (rc != 0) {
170 item->state = DPPD_TRANSACTION_ITEM_FAILED;
171 item->error_code = rc;
172 transaction->failure_code = rc;
173 transaction->failure_stage = DPPD_RULE_STAGE_PREPARE;
174 transaction->failure_item = i;
175 rollback_items(transaction, backends, 0);
176 return rc;
177 }
178 item->state = DPPD_TRANSACTION_ITEM_PREPARED;
179 }
180
181 transaction->state = DPPD_TRANSACTION_COMMITTING;
182 /**
183 * 第三阶段才真正安装对象，每次提交都使用对应 prepare 返回的 token
184 * 提交报错不一定代表完全没有副作用，因此失败条目本身也要参加回滚
185 */
186 for (i = 0; i < transaction->nb_items; ++i) {
187 struct dppd_transaction_item *item = &transaction->items[i];
188 const struct dppd_transaction_backend *backend = backend_for(backends, item->plan.backend);
189
190 rc = backend->commit(backend->context, item, item->backend_token);
191 if (rc != 0) {
192 item->state = DPPD_TRANSACTION_ITEM_FAILED;
193 item->error_code = rc;
194 transaction->failure_code = rc;
195 transaction->failure_stage = DPPD_RULE_STAGE_COMMIT;
196 transaction->failure_item = i;
197 rollback_items(transaction, backends, i + 1U);
198 return rc;
199 }
200 item->state = DPPD_TRANSACTION_ITEM_COMMITTED;
201 }
202
203 transaction->state = DPPD_TRANSACTION_COMMITTED;
204 return 0;
205}

207/**
208 * 撤销一个已经安装成功、但尚未最终确认的事务
209 * 例如批量更新的新版本全部装好后，旧版本删除失败，就需要撤销整批新版本
210 * 已经 finalize 的事务不再拥有回滚资源，因此不允许通过这个入口继续撤销
211 */
212int dppd_transaction_rollback_committed(struct dppd_transaction *transaction,
213 const struct dppd_transaction_backends *backends)
214{
215 if (transaction == NULL || backends == NULL || transaction->state != DPPD_TRANSACTION_COMMITTED)
216 return -EINVAL;
217 rollback_items(transaction, backends, transaction->nb_items);
218 return transaction->rollback_code;
219}

221/**
222 * 结束本次事务的临时资源生命周期，不删除已经安装的规则
223 * 正常流程在规则账本发布后调用；异常流程决定隔离且不再回滚时也可用于收尾
224 * 逐项清零 token，表示调用方已经放弃再次提交或回滚本次事务的可能
225 * finalize 回调不能报错，因此后端必须把它设计成单纯的临时资源释放动作
226 */
227int dppd_transaction_finalize(struct dppd_transaction *transaction, const struct dppd_transaction_backends *backends)
228{
229 uint32_t i;
230
231 if (transaction == NULL || backends == NULL || transaction->state != DPPD_TRANSACTION_COMMITTED)
232 return -EINVAL;
233 for (i = 0; i < transaction->nb_items; ++i) {
234 struct dppd_transaction_item *item = &transaction->items[i];
235 const struct dppd_transaction_backend *backend = backend_for(backends, item->plan.backend);
236
237 if (item->state != DPPD_TRANSACTION_ITEM_COMMITTED || !backend_valid(backend))
238 return -EUCLEAN;
239 /** 后端没有额外清理回调时也要清空 token，并让整个事务进入不可回滚的终态 */
240 if (backend->finalize != NULL)
241 backend->finalize(backend->context, item, item->backend_token);
242 item->backend_token = 0;
243 }
244 transaction->state = DPPD_TRANSACTION_FINALIZED;
245 return 0;
