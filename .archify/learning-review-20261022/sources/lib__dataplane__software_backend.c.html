20/**
21 * 将计数器与规则数组分开保存，让同一规则版本在复制快照后继续累计 COUNT
22 * 安装新的 generation 会创建新计数器，因此这里的连续统计不表示跨版本继承
23 * references 记录引用它的快照数量，包括尚未发布的副本，不是工作线程数量
24 * hits 和 bytes 使用原子操作，避免多个工作线程同时累计时覆盖彼此的结果
25 */
26struct dppd_software_rule_metrics {
27 atomic_uint_fast64_t references;
28 atomic_uint_fast64_t hits;
29 atomic_uint_fast64_t bytes;
30 /**
31 * 同一版本复制快照时共享安装记录，修改其他规则不会覆盖此版本的原始耗时
32 * 以下字段只由控制面在 writer_lock 内访问，工作线程只使用上面的原子计数器
33 */
34 uint64_t install_duration_ns;
35 uint32_t commit_rule_count;
36 bool timing_available;
37 struct dppd_flow_shared_counts *counts;
38 struct dppd_flow_shared_count_lease shared_lease;
39 struct dppd_software_shared_counter *shared;
40};

48/**
49 * snapshot 表示内存中的一整版软件规则表，让 worker 读取完整、一致的规则，同时允许控制面更新
50 *
51 * snapshot
52 * ├── 有效规则数量 count
53 * ├── 规则数组 objects
54 * │ ├── 匹配条件、动作等规则内容
55 * │ └── metrics 指针 → COUNT 计数器
56 * ├── 精确匹配表
57 * └── ACL 匹配表
58 *
59 * 内存布局：[count][capacity 个 objects 槽位][精确匹配表][ACL]，尾部两张表通过辅助函数定位
60 * worker 读取 active 取得一个快照，再用同一快照完成候选匹配、动作选择和计数器定位
61 *
62 * 旧快照 A：worker 正在读取，active 指向 A
63 * │
64 * └── 复制成 B → 修改 B → 重建 B 的匹配表
65 * │
66 * 完成后将 active 切换到 B
67 * ↓
68 * 后续查询使用新快照 B
69 *
70 * active 是当前已发布快照的指针，retired 链表保存暂时不能释放的旧快照
71 * 已经取得 A 的 worker 继续使用 A，QSBR 确认旧表无人再用后，控制面才回收 A
72 * 这样不会在原表上交错修改匹配条件和动作，worker 不会读到更新一半的规则表
73 * 发布后的规则内容和匹配索引不再原地修改，metrics 指向的 COUNT 计数器仍可持续更新
74 * 此结构保存运行时规则版本，不保存原始报文，也不负责磁盘持久化
75 */
76struct dppd_software_classifier_snapshot {
77 /** 当前有效规则数，槽位容量由 backend->capacity 决定，有效规则不一定连续排列 */
78 uint32_t count;
79 /** 柔性数组，分配时预留 capacity 个槽位，每项保存规则内容及 metrics 指针 */
80 struct dppd_software_rule_object objects[];
81};

83/** 一个待回收的旧规则表，仍可能被 worker 读取，因此从 active 摘下后先保留 */
84struct dppd_software_retired_snapshot {

88 uint64_t token;
89 /** 下一个待回收节点，多次更新可能同时留下多份尚不能释放的旧表 */
90 struct dppd_software_retired_snapshot *next;
91};

118/** 归还一个快照持有的计数器引用，最后一个引用归还时解除共享计数绑定并释放对象，NULL 无操作 */
119static void metrics_put(struct dppd_software_rule_metrics *metrics)
120{
121 /**
122 * references 统计持有该对象的快照数量，atomic_fetch_sub 返回减之前的值
123 * 例：新旧表 A、B 共用计数器，销毁 A 时 2 → 1，返回 2，不进入释放分支
124 * 最后销毁 B 时 1 → 0，返回 1，才释放；这里减的是引用数，不是规则命中次数
125 */
126 if (metrics != NULL && atomic_fetch_sub_explicit(&metrics->references, 1, memory_order_acq_rel) == 1) {
127 if (metrics->shared_lease.owner != 0)
128 (void)dppd_flow_shared_counts_unbind(metrics->counts, &metrics->shared_lease);
129 free(metrics);
130 }
131}

133/**
134 * 销毁 snapshot：逐槽归还计数器引用，再释放包含规则、精确匹配表和 ACL 的整块快照内存
135 * capacity 是分配的槽位数，有效规则可能分散其中，因此不能只遍历 count 个槽位
136 * 未发布副本可直接销毁；已发布快照须由调用方先通过 QSBR 检查，或等待所有 worker 退出
137 * 本函数不等待 worker，也不强制释放仍被其他快照引用的计数器；snapshot 为 NULL 时直接返回
138 */
139static void snapshot_destroy(struct dppd_software_classifier_snapshot *snapshot, uint32_t capacity)
140{
141 uint32_t i;
142
143 if (snapshot == NULL)
144 return;
145 /** 只归还当前表持有的引用，计数器仍被新表使用时保留原来的 hits 和 bytes */
146 for (i = 0; i < capacity; ++i)
147 metrics_put(snapshot->objects[i].metrics);
148 free(snapshot);
149}

151/**
152 * 创建供控制面修改的新规则快照，复制已有规则并共享其计数器，保持未变规则的 COUNT 连续
153 * source 为源快照，NULL 表示创建空表；capacity 为槽位容量，复制已有表时须与其容量一致
154 * 成功返回尚未发布的副本，分配失败返回 NULL；此处不修改 backend->active
155 * ACL 和精确匹配表只初始化为空，调用方完成规则修改后，由 publish_locked 重新构建并发布
156 * 副本从创建时就持有计数器引用，后续操作失败也必须通过 snapshot_destroy 归还
157 */
158static struct dppd_software_classifier_snapshot *snapshot_clone(const struct dppd_software_classifier_snapshot *source,
159 uint32_t capacity)
160{
161 struct dppd_software_classifier_snapshot *copy;
162 uint32_t i;
163
164 /** 一次分配快照头、规则槽位、精确匹配表和 ACL 的空间，初始全部清零 */
165 copy = calloc(1, snapshot_size(capacity));
166 if (copy == NULL)
167 return NULL;
168 dppd_acl_init(snapshot_acl(copy, capacity), capacity);
169 dppd_exact_init(snapshot_exact(copy, capacity), capacity);
170 if (source == NULL)
171 return copy;
172 copy->count = source->count;
173 for (i = 0; i < capacity; ++i) {
174 /**
175 * 结构体赋值复制 rule 内容和 metrics 指针，不复制指针指向的计数器对象
176 * 例：A.objects[i].metrics 和新表 B.objects[i].metrics 都指向同一个 M
177 *
178 * 旧表 A.objects[i] 新表 B.objects[i]
179 ┌──────────────┐ ┌──────────────┐
180 │ rule 内容 │ ──复制内容─────→ │ rule 内容 │
181 │ metrics 指针 │ │ metrics 指针 │
182 └──────┬───────┘ └──────┬───────┘
183 └──────────→ M ←─────────────────┘
184 同一个计数器
185 */
186 copy->objects[i] = source->objects[i];
187 /** B 也持有 M，引用数例如从 1 增为 2；hits 和 bytes 不清零，也不随复制增加 */
188 if (copy->objects[i].metrics != NULL)
189 atomic_fetch_add_explicit(&copy->objects[i].metrics->references, 1, memory_order_relaxed);
190 }
191 return copy;
192}

208/**
209 * 检查 retired 链表，销毁已通过 QSBR 安全检查的旧规则表及其链表节点
210 *
211 * backend 提供待回收链表和读者进度，调用方必须持有 writer_lock，防止控制面同时修改链表
212 * check 返回 0 → 保留该旧表，继续检查下一个节点；返回非零 → 从链表摘下并销毁
213 * false 表示不等待读者，尚不安全的旧表留待后续发布、查询或显式回收时再检查
214 * 无返回值，不修改 active 指向的新表，也不处理报文 mbuf
215 */
216static void reclaim_locked(struct dppd_software_backend *backend)
217{
218 /**
219 * link 指向保存当前节点地址的位置，*link 才是当前节点指针
220 * 初始指向 backend->retired；若保留某节点，再改为指向该节点的 next 字段
221 * 因此给 *link 赋值，可以直接修改链表头或前一个节点的 next
222 */
223 struct dppd_software_retired_snapshot **link = &backend->retired;
224
225 while (*link != NULL) {
226 struct dppd_software_retired_snapshot *retired = *link;
227
228 /** token 标识此旧表的回收进度，返回 0 表示仍有相关在线读者未通过对应安全点 */
229 if (rte_rcu_qsbr_check(backend->qsbr, retired->token, false) == 0) {
230 /** 当前节点继续留在链上，转而检查它的后继，不在这里等读者 */
231 link = &retired->next;
232 continue;
233 }
234 /**
235 * 安全后让前面的指针绕过当前节点，例如 链表头 → A → Z 改为 链表头 → Z
236 * link 所指的位置仍然有效，下一轮从这个位置检查后继，不能再访问即将释放的 A
237 */
238 *link = retired->next;
239 /** 销毁旧规则表并归还它持有的计数器引用，再释放保存 snapshot/token/next 的链表节点 */
240 snapshot_destroy(retired->snapshot, backend->capacity);
241 free(retired);
242 }
243}

245/**
246 * 为 next 中的最终规则构建匹配表，发布新快照，并将旧快照交给 QSBR 回收流程
247 * backend 提供容量、active 和回收链表，next 是调用方准备好的私有副本，调用者须持有 writer_lock
248 * 返回 0 表示发布成功；负错误码表示发布前失败，active 保持原指向，副本由调用方销毁
249 *
250 * 副本 B 的规则内容已准备好
251 * → 清空 B 的 ACL / exact 索引 → 遍历有效规则，逐条加入合适的匹配表
252 * → 预先分配旧表回收节点 → active 从 A 切换到 B
253 * → 保存 A 和新取得的 QSBR token → 尝试回收已安全的旧表
254 */
255static int publish_locked(struct dppd_software_backend *backend, struct dppd_software_classifier_snapshot *next)
256{
257 struct dppd_software_classifier_snapshot *previous;
258 struct dppd_software_retired_snapshot *retired;
259
260 /** 重建的是 B 内部的匹配表，旧表 A 仍供 worker 使用；索引须反映调用方增删改后的最终规则 */
261 struct dppd_acl *acl = dppd_acl_init(snapshot_acl(next, backend->capacity), backend->capacity);
262 struct dppd_exact *exact = dppd_exact_init(snapshot_exact(next, backend->capacity), backend->capacity);
263 for (uint32_t i = 0; i < backend->capacity; ++i) {
264 if (next->objects[i].metrics == NULL)
265 continue;
266 /** 快照自己借用共享对象，实际 flow 已删除时，旧 worker 仍可完成最后一次累计 */
267 struct dppd_software_rule_metrics *metrics = next->objects[i].metrics;
268 const struct dppd_rule *rule = &next->objects[i].rule;
269 if (backend->shared_counts != NULL && metrics->shared_lease.owner == 0) {
270 for (uint16_t a = 0; a < rule->nb_actions; ++a) {
271 if (rule->actions[a].type != DPPD_ACTION_SHARED_COUNT)
272 continue;
273 struct dppd_flow_binding binding = {0};
274 int bound =
275 dppd_flow_shared_counts_bind(backend->shared_counts, rule, &binding, &metrics->shared_lease);
276 if (bound != 0)
277 return bound;
278 metrics->counts = backend->shared_counts;
279 metrics->shared = (void *)binding.actions[a];
280 break;
281 }
282 }
283 /**
284 * 每条有效规则先尝试精确匹配表：
285 * exact_add → 0 → 已加入 exact，处理下一条规则
286 * -ENOTSUP → 改用 acl_add，支持掩码、端口范围等匹配条件
287 * 其他错误 → 返回错误，终止本次发布
288 * acl_add → 0 → 已加入 ACL，处理下一条规则
289 * 非 0 → 返回错误，终止本次发布
290 * 这是安装规则时的分类，worker 收包时使用 classify 查表，不逐包重新添加规则
291 *
292 * i + 1 是规则槽位的 tag，0 留给未命中；tag 不是规则 ID，也不是匹配表内部行号
293 * 例：B.objects[2] → 加入 exact 或 ACL 时保存 tag = 3
294 * 报文匹配返回 3 → B.objects[3 - 1] → 取得该规则的动作和计数器
295 */
296 int rc = dppd_exact_add(exact, &next->objects[i].rule, i + 1);
297 if (rc == -ENOTSUP)
298 rc = dppd_acl_add(acl, &next->objects[i].rule, i + 1);
299 if (rc != 0)
300 return rc;
301 }
302 retired = calloc(1, sizeof(*retired));
303 if (retired == NULL) {
304 /** 在发布前保证能够记录旧 snapshot，避免内存紧张时丢失回收所有权 */
305 return -ENOMEM;
306 }
307 /**
308 * 原子交换 active 指针：写入 next，同时把交换前的指针返回给 previous
309 * 交换前：backend->active → A，next → B，B 的规则和匹配表均已准备好
310 * 交换后：backend->active → B，previous → A，next 仍指向 B
311 * 本操作只换指针，不复制快照，也不释放 A
312 *
313 * release 与 worker 读取 active 时的 acquire 配合，使读到 B 的 worker 能看到 B 的初始化结果
314 * 已持有 A 的 worker → 继续使用 A → 报告安全点 → QSBR 确认后才能释放 A
315 * 从 active 取得 B 的 worker → 使用 B 的规则和匹配表
316 */
317 previous = atomic_exchange_explicit(&backend->active, next, memory_order_release);
318 if (previous == NULL) {
319 /** 没有旧快照需要回收，只释放预分配的回收节点，新快照 B 已经发布 */
320 free(retired);
321 return 0;
322 }
323 /**
324 * 用回收节点 R 保存旧快照 A 的地址，A 本身仍留在内存中供已有读者使用
325 * R.snapshot → A；backend->active → B，两者分别管理待回收旧表和当前规则表
326 */
327 retired->snapshot = previous;
328 /**
329 * A 退出 active 后再取得 QSBR 回收标记，后续据此确认相关读者已结束此前的访问
330 * token 是 QSBR 进度标记，不是时间戳或规则版本；start 不等待，也不释放 A
331 */
332 retired->token = rte_rcu_qsbr_start(backend->qsbr);
333 /**
334 * 将回收节点 R 插到链表头，假设原链头为旧节点 X
335 * 插入前：backend->retired → X → ...；R.snapshot → A
336 * 第一句：R.next → X，接住原来的回收链
337 * 第二句：backend->retired → R → X → ...，R 成为新链头
338 * 原链为空时 R.next 为 NULL；节点的 next 串起回收记录，snapshot 指向各自的旧快照
339 */
340 retired->next = backend->retired;
341 backend->retired = retired;
342 /**
343 * 检查整条回收链，各节点分别用自己的 token 判断是否安全，不在这里等待读者
344 * 以 R 为例：检查未通过 → 保留 R 和 A，留待以后再检查
345 * 检查通过 → 摘下 R → 销毁 A、归还计数器引用 → 释放 R
346 * 无论本次是否回收 A，B 都已发布成功，下面返回 0
347 */
348 reclaim_locked(backend);
349 return 0;
350}

352/**
353 * 检查已经通过通用规则校验的规则能否由软件执行，不替代数组长度和字段合法性检查
354 * 软件只处理入口域的有限动作，不能把不支持的硬件动作静默改成普通转发
355 */
356bool dppd_software_backend_rule_supported(const struct dppd_rule *rule)
357{
358 uint16_t i;
359
360 if (rule == NULL || rule->domain != DPPD_RULE_DOMAIN_INGRESS)
361 return false;
362 for (i = 0; i < rule->nb_matches; ++i) {
363 if (rule->matches[i].type < DPPD_MATCH_ETH || rule->matches[i].type > DPPD_MATCH_IPV6 ||
364 rule->matches[i].type == DPPD_MATCH_REPRESENTED_PORT)
365 return false;
366 }
367 for (i = 0; i < rule->nb_actions; ++i) {
368 switch (rule->actions[i].type) {
369 case DPPD_ACTION_DROP:
370 case DPPD_ACTION_MARK:
371 case DPPD_ACTION_COUNT:
372 case DPPD_ACTION_ALLOW:
373 break;
374 case DPPD_ACTION_QUEUE:
375 case DPPD_ACTION_REPRESENTED_PORT:
376 default:
377 return false;
378 }
379 }
380 return true;
381}

400int dppd_software_backend_init(struct dppd_software_backend *backend, uint32_t capacity)
401{
402 struct dppd_software_classifier_snapshot *initial;
403 size_t qsbr_size;
404
405 if (backend == NULL || capacity == 0)
406 return -EINVAL;
407 memset(backend, 0, sizeof(*backend));
408 initial = snapshot_clone(NULL, capacity);
409 if (initial == NULL)
410 return -ENOMEM;
411 if (pthread_mutex_init(&backend->writer_lock, NULL) != 0) {
412 snapshot_destroy(initial, capacity);
413 return -EIO;
414 }
415 backend->lock_initialized = true;
416 /** reader 上限与 runtime 的 queue/worker 编号空间一致，避免动态编号造成越界 */
417 qsbr_size = rte_rcu_qsbr_get_memsize(DPPD_MAX_WORKERS);
418 /**
419 * QSBR 变量只要求 cache-line 对齐；这里刻意不用 rte_zmalloc，使纯控制面单测
420 * 无需先启动 EAL。真实 dppd 进程仍在 EAL 初始化后正常使用同一块内存
421 */
422 qsbr_size = (qsbr_size + RTE_CACHE_LINE_SIZE - 1U) & ~(size_t)(RTE_CACHE_LINE_SIZE - 1U);
423 backend->qsbr = aligned_alloc(RTE_CACHE_LINE_SIZE, qsbr_size);
424 if (backend->qsbr == NULL || rte_rcu_qsbr_init(backend->qsbr, DPPD_MAX_WORKERS) != 0) {
425 free(backend->qsbr);
426 (void)pthread_mutex_destroy(&backend->writer_lock);
427 snapshot_destroy(initial, capacity);
428 memset(backend, 0, sizeof(*backend));
429 return -ENOMEM;
430 }
431 backend->capacity = capacity;
432 backend->qsbr_initialized = true;
433 atomic_init(&backend->active, initial);
434 return 0;
435}

437void dppd_software_backend_fini(struct dppd_software_backend *backend)
438{
439 struct dppd_software_retired_snapshot *retired;
440 struct dppd_software_classifier_snapshot *active;
441
442 /** fini 不等待 QSBR；调用契约要求 worker 已全部停止，因此此时可直接释放 retired */
443 if (backend == NULL)
444 return;
445 if (backend->lock_initialized)
446 (void)pthread_mutex_lock(&backend->writer_lock);
447 active = atomic_load_explicit(&backend->active, memory_order_relaxed);
448 snapshot_destroy(active, backend->capacity);
449 retired = backend->retired;
450 while (retired != NULL) {
451 struct dppd_software_retired_snapshot *next = retired->next;
452
453 snapshot_destroy(retired->snapshot, backend->capacity);
454 free(retired);
455 retired = next;
456 }
457 if (backend->lock_initialized) {
458 (void)pthread_mutex_unlock(&backend->writer_lock);
459 (void)pthread_mutex_destroy(&backend->writer_lock);
460 }
461 free(backend->qsbr);
462 memset(backend, 0, sizeof(*backend));
463}

465/** 事务的能力检查阶段只判断目标后端和规则能力，不分配资源，也不发布规则 */
466static int transaction_validate(void *context, const struct dppd_transaction_item *item)
467{
468 struct dppd_software_backend *backend = context;
469
470 if (backend == NULL || item == NULL || item->plan.backend != DPPD_PLAN_BACKEND_SOFTWARE ||
471 !supported(backend, &item->rule))
472 return -ENOTSUP;
473 return 0;
474}

476/**
477 * prepare 只分配可回滚的控制面 token，不让未提交规则暴露给 worker。锁内复查容量和
478 * 版本冲突后立即解锁；真正发布留到 commit，以便同一事务的任一 prepare 失败时完全
479 * 不改变数据面可见规则集
480 * 这里不实际预留规则槽位，多个 prepare 成功不代表它们一定能够全部提交
481 * commit 仍需检查当时的容量，批量操作的整体容量预检由上层控制面负责
482 */
483static int transaction_prepare(void *context, const struct dppd_transaction_item *item, uintptr_t *token)
484{
485 struct dppd_software_backend *backend = context;
486 struct dppd_software_pending_rule *pending;
487 struct dppd_software_classifier_snapshot *active;
488 int found;
489
490 if (backend == NULL || item == NULL || token == NULL || item->rule.generation == 0)
491 return -EINVAL;
492 pending = calloc(1, sizeof(*pending));
493 if (pending == NULL)
494 return -ENOMEM;
495 (void)pthread_mutex_lock(&backend->writer_lock);
496 active = atomic_load_explicit(&backend->active, memory_order_acquire);
497 found = snapshot_find(active, backend->capacity, item->rule.id, item->rule.generation);
498 if (found >= 0 || active->count >= backend->capacity) {
499 (void)pthread_mutex_unlock(&backend->writer_lock);
500 free(pending);
501 return found >= 0 ? -EEXIST : -ENOSPC;
502 }
503 (void)pthread_mutex_unlock(&backend->writer_lock);
504 pending->rule = item->rule;
505 *token = (uintptr_t)pending;
506 return 0;
507}

509/**
510 * 控制面提交一条准备好的软件规则：复查当前表，复制快照，加入规则及独立计数器，再发布新表
511 * context 是软件后端，token 是 prepare 返回的 pending 地址，规则内容从 pending->rule 取得
512 * 此处 token 是事务准备记录，不是 QSBR 回收进度标记；item 在本实现中不再读取
513 * 成功返回 0 并设置 pending->published，失败返回负错误码；此函数不会释放 pending
514 * 本次切换只发布这一条规则的安装结果，不代表整个跨规则事务同时对报文生效
515 * 本函数在安装规则时调用，worker 逐包匹配只读取已发布结果，不会每包复制规则表
516 */
517static int transaction_commit(void *context, const struct dppd_transaction_item *item, uintptr_t token)
518{
519 struct dppd_software_backend *backend = context;
520 struct dppd_software_pending_rule *pending = (struct dppd_software_pending_rule *)token;
521 struct dppd_software_classifier_snapshot *active;
522 struct dppd_software_classifier_snapshot *next;
523 struct dppd_software_rule_metrics *metrics;
524 uint32_t i;
525 struct dppd_install_timer timer;
526
527 /** commit 必须再次检查：prepare 与 commit 之间可能已有另一控制请求完成发布 */
528 (void)item;
529 if (backend == NULL || pending == NULL)
530 return -EINVAL;
531 (void)pthread_mutex_lock(&backend->writer_lock);
532 active = atomic_load_explicit(&backend->active, memory_order_acquire);
533 if (snapshot_find(active, backend->capacity, pending->rule.id, pending->rule.generation) >= 0 ||
534 active->count >= backend->capacity) {
535 (void)pthread_mutex_unlock(&backend->writer_lock);
536 return -EEXIST;
537 }
538 dppd_install_timer_start(&timer);
539 /**
540 * 例：旧表只有 R1，本次准备新增 R2，先复制整张表，不修改已发布的 A
541 *
542 * 复制前：backend->active → A：[R1]，pending->rule 保存 R2
543 * 复制后：backend->active → A：[R1]
544 * next → B：[R1] 尚未发布，供控制面修改
545 */
546 next = snapshot_clone(active, backend->capacity);
547 if (next == NULL) {
548 (void)pthread_mutex_unlock(&backend->writer_lock);
549 return -ENOMEM;
550 }
551 /**
552 * 例：A 原有 R1，副本 B 已与 A 共享 R1 的计数器；本次新增 R2，需要另建计数器
553 * 新安装版本从 hits = 0、bytes = 0 开始，不继承同 ID 旧版本的统计
554 */
555 metrics = calloc(1, sizeof(*metrics));
556 if (metrics == NULL) {
557 snapshot_destroy(next, backend->capacity);
558 (void)pthread_mutex_unlock(&backend->writer_lock);
559 return -ENOMEM;
560 }
561 /**
562 * R2 新计数器的初始状态：
563 * references = 1 → 只有副本 B 持有，不表示已经命中一个包
564 * hits = 0 → 尚未匹配过报文
565 * bytes = 0 → 尚未累计报文字节
566 */
567 atomic_init(&metrics->references, 1);
568 atomic_init(&metrics->hits, 0);
569 atomic_init(&metrics->bytes, 0);
570 /** metrics 为 NULL 表示空槽位，将 R2 内容和新计数器一同放入 B，再增加有效规则数 */
571 for (i = 0; i < backend->capacity; ++i) {
572 if (next->objects[i].metrics == NULL) {
573 next->objects[i].rule = pending->rule;
574 next->objects[i].metrics = metrics;
575 next->count++;
576 break;
577 }
578 }
579 /**
580 * 加入 R2 后，规则与计数器的关系：
581 *
582 * A：[R1] ────────────→ R1 原计数器
583 * ↑
584 * B：[R1] ─────────────────┘
585 * [R2] ────────────→ R2 新计数器
586 *
587 * 此时 R1 的计数器由 A、B 共同持有，引用数为 2；R2 的计数器只由 B 持有，引用数为 1
588 * 发布前：backend->active → A，worker 还看不到 B 中新增的 R2
589 * publish_locked 返回
590 * 非 0 → active 仍指向 A → 销毁未发布的 B → 解锁 → 返回发布错误
591 * R1 计数器引用 2 → 1，A 继续使用；R2 计数器引用 1 → 0，随之释放
592 * 为 0 → active 已指向 B，A 已交给 QSBR 回收流程
593 * → 记录安装耗时 → 标记 pending 已发布 → 解锁 → 返回成功
594 * 失败时 B 从未交给 worker，可直接销毁；成功时 B 已由 active 持有，不能在此销毁
595 */
596 int publish_error = publish_locked(backend, next);
597 if (publish_error != 0) {
598 snapshot_destroy(next, backend->capacity);
599 (void)pthread_mutex_unlock(&backend->writer_lock);
600 return publish_error;
601 }
602 /**
603 * 成功发布后完成计时，耗时包括副本分配、指针发布和当次可完成的退役回收
604 * 查询同样持有 writer_lock，所以读不到尚未填完的记录，逐包匹配不读取这些字段
605 */
606 metrics->timing_available = dppd_install_timer_finish(&timer, &metrics->install_duration_ns);
607 metrics->commit_rule_count = 1;
608 /** 历史在已有控制面锁内更新，零纳秒是有效测量，时钟未知单独计数 */
609 dppd_rule_latency_record(&backend->latency_single, metrics->timing_available, metrics->install_duration_ns, 1);
610 /**
611 * published 记录本条规则已经安装，供后续回滚判断是否需要删除该规则版本
612 * 真正发布发生在前面的 active 指针交换，此标志不负责让 worker 切换规则
613 * pending 留给事务的 finalize 或 rollback 释放，本函数只解锁并返回
614 */
615 pending->published = true;
616 (void)pthread_mutex_unlock(&backend->writer_lock);
617 return 0;
618}

620/**
621 * 撤销一条软件规则的安装，token 是 prepare 创建的 pending 地址，返回删除结果或参数错误
622 * pending->published
623 * false → 规则未发布 → 只释放 pending → 返回 0
624 * true → 按 id + generation 删除已发布版本 → 释放 pending → 返回删除结果
625 * commit_was_attempted 只说明调用过 commit，不能说明发布成功，本后端以 published 为准
626 * 删除也要创建并发布新快照，可能因内存不足失败；即使删除失败也会释放 pending
627 * 上层须检查返回值，不能把准备记录已释放当作规则已撤回
628 */
629static int transaction_rollback(void *context, const struct dppd_transaction_item *item, uintptr_t token,
630 bool commit_was_attempted)
631{
632 struct dppd_software_backend *backend = context;
633 struct dppd_software_pending_rule *pending = (struct dppd_software_pending_rule *)token;
634 int rc = 0;
635
636 (void)item;
637 (void)commit_was_attempted;
638 if (backend == NULL || pending == NULL)
639 return -EINVAL;
640 /** commit 尚未发布时只需释放 token；已发布时删除同一 ID+generation 的精确版本 */
641 if (pending->published)
642 rc = dppd_software_backend_remove_version(backend, pending->rule.id, pending->rule.generation);
643 free(pending);
644 return rc;
645}
646
647/**
648 * 释放事务准备记录，不修改已经发布的规则，也不释放规则的计数器
649 * prepare 创建 pending → commit 将规则内容复制进快照 → finalize 释放 pending
650 * 此后 active → 已发布快照 → 规则和计数器仍在，worker 继续使用
651 * 上层 dppd_transaction_finalize 随后清零条目 token，并将事务标记为 FINALIZED
652 * 正常提交完成后使用此入口；进入隔离并放弃继续回滚时也可用它清理准备记录
653 * 因此调用 finalize 本身不能作为业务事务成功的判断依据
654 */
655static void transaction_finalize(void *context, const struct dppd_transaction_item *item, uintptr_t token)
656{
657 (void)context;
658 (void)item;
659 free((struct dppd_software_pending_rule *)token);
660}

662/**
663 * 返回软件后端的事务回调表，context 保存后端地址，供上层调用各阶段函数时传回
664 * transaction.c：backend->commit(context, item, token) → 本文件 transaction_commit
665 * 同样通过 prepare、rollback、finalize 回调完成准备、撤销和临时资源清理
666 */
667struct dppd_transaction_backend dppd_software_transaction_backend(struct dppd_software_backend *backend)
668{
669 const struct dppd_transaction_backend operations = {
670 /**
671 * .成员名 = 初始值 是 C 的指定初始化语法，左侧名称来自 dppd_transaction_backend 的定义
672 * .context = backend → 将后端对象指针存入 operations.context
673 * .commit = transaction_commit → 将函数地址存入 operations.commit，此处不会执行函数
674 * 调用 operations.commit(...) 时，才会执行 transaction_commit(...)
675 * 按成员名指定初值，无须依照结构体成员顺序；未指定的成员按零初始化规则处理
676 */
677 .context = backend,
678 .validate = transaction_validate,
679 .prepare = transaction_prepare,
680 .commit = transaction_commit,
681 .rollback = transaction_rollback,
682 .finalize = transaction_finalize,
683 };
684 return operations;
685}

687/**
688 * 整批替换在私有副本内完成，规则槽位复用，未更新规则保留原计数器
689 * 全部检查和分配成功后仅切换一次指针，失败时丢弃副本而不改变活跃表
690 */
691int dppd_software_backend_update_batch(struct dppd_software_backend *backend, const struct dppd_rule *rules,
692 const uint64_t *expected_generations, uint32_t count)
693{
694 struct dppd_software_classifier_snapshot *active, *next = NULL;
695 uint32_t i;
696 int rc = 0;
697 struct dppd_install_timer timer;
698
699 if (backend == NULL || !backend->lock_initialized || rules == NULL || expected_generations == NULL || count == 0 ||
700 count > backend->capacity)
701 return -EINVAL;
702 (void)pthread_mutex_lock(&backend->writer_lock);
703 active = atomic_load_explicit(&backend->active, memory_order_acquire);
704 for (i = 0; i < count; ++i) {
705 char error[128];
706 uint32_t j;
707
708 if (rules[i].id == 0 || rules[i].generation == 0 || rules[i].generation == UINT64_MAX ||
709 dppd_rule_validate(&rules[i], error, sizeof(error)) != 0) {
710 rc = -EINVAL;
711 goto out;
712 }
713 if (!supported(backend, &rules[i])) {
714 rc = -ENOTSUP;
715 goto out;
716 }
717 for (j = 0; j < i; ++j) {
718 if (rules[j].id == rules[i].id) {
719 rc = -EEXIST;
720 goto out;
721 }
722 }
723 if (expected_generations[i] == 0 || expected_generations[i] == UINT64_MAX ||
724 rules[i].generation <= expected_generations[i] ||
725 snapshot_find(active, backend->capacity, rules[i].id, expected_generations[i]) < 0) {
726 rc = -ESTALE;
727 goto out;
728 }
729 if (snapshot_find(active, backend->capacity, rules[i].id, rules[i].generation) >= 0) {
730 rc = -EEXIST;
731 goto out;
732 }
733 }
734 /** 全批条件检查结束才开始计时，不把无效请求的检查耗时记录为安装耗时 */
735 dppd_install_timer_start(&timer);
736 next = snapshot_clone(active, backend->capacity);
737 if (next == NULL) {
738 rc = -ENOMEM;
739 goto out;
740 }
741 for (i = 0; i < count; ++i) {
742 int index = snapshot_find(next, backend->capacity, rules[i].id, expected_generations[i]);
743 struct dppd_software_rule_metrics *metrics = calloc(1, sizeof(*metrics));
744
745 if (metrics == NULL) {
746 rc = -ENOMEM;
747 goto out;
748 }
749 atomic_init(&metrics->references, 1);
750 atomic_init(&metrics->hits, 0);
751 atomic_init(&metrics->bytes, 0);
752 metrics_put(next->objects[index].metrics);
753 next->objects[index].rule = rules[i];
754 next->objects[index].metrics = metrics;
755 }
756 rc = publish_locked(backend, next);
757 if (rc == 0) {
758 uint64_t duration_ns;
759 bool timing_available = dppd_install_timer_finish(&timer, &duration_ns);
760
761 /** 整批只有一次指针发布，因此只增加一个样本，同时记下这一批涉及的规则数 */
762 dppd_rule_latency_record(&backend->latency_batch, timing_available, duration_ns, count);
763
764 /**
765 * 一次发布同时替换整个批次，因此各新版本共享整批耗时，不能伪装成各自的单条耗时
766 * 这里只修改新分配的记录，未更新版本仍共享原来的记录，失败路径完全不改旧表
767 */
768 for (i = 0; i < count; ++i) {
769 int index = snapshot_find(next, backend->capacity, rules[i].id, rules[i].generation);
770 struct dppd_software_rule_metrics *metrics = next->objects[index].metrics;
771
772 metrics->install_duration_ns = duration_ns;
773 metrics->timing_available = timing_available;
774 metrics->commit_rule_count = count;
775 }
776 next = NULL;
777 }
778out:
779 snapshot_destroy(next, backend->capacity);
780 (void)pthread_mutex_unlock(&backend->writer_lock);
781 return rc;
782}

784/** 先修改私有副本中的整个批次，任何分配或编译失败都不会改变活跃规则及计数 */
785int dppd_software_backend_change_batch(struct dppd_software_backend *backend,
786 const struct dppd_software_change *changes, uint32_t count)
787{
788 struct dppd_software_classifier_snapshot *active, *next = NULL;
789 struct dppd_install_timer timer;
790 uint64_t final_count;
791 int slots[DPPD_SOFTWARE_CHANGE_BATCH_MAX];
792 uint32_t i;
793 int rc = 0;
794
795 if (backend == NULL || !backend->lock_initialized || changes == NULL || count == 0 ||
796 count > DPPD_SOFTWARE_CHANGE_BATCH_MAX)
797 return -EINVAL;
798 pthread_mutex_lock(&backend->writer_lock);
799 active = atomic_load_explicit(&backend->active, memory_order_acquire);
800 final_count = active->count;
801 for (i = 0; i < count; ++i) {
802 const struct dppd_software_change *change = &changes[i];
803 const struct dppd_rule *rule = change->replacement;
804 char error[128];
805 unsigned int found = 0;
806 slots[i] = -1;
807 if (change->id == 0 || change->id == UINT64_MAX || change->expected_generation == UINT64_MAX ||
808 (rule == NULL && change->expected_generation == 0)) {
809 rc = -EINVAL;
810 goto done;
811 }
812 for (uint32_t previous = 0; previous < i; ++previous)
813 if (changes[previous].id == change->id) {
814 rc = -EEXIST;
815 goto done;
816 }
817 for (uint32_t slot = 0; slot < backend->capacity; ++slot)
818 if (active->objects[slot].metrics != NULL && active->objects[slot].rule.id == change->id) {
819 ++found;
820 slots[i] = (int)slot;
821 }
822 if (found > 1) {
823 rc = -EUCLEAN;
824 goto done;
825 }
826 if (change->expected_generation == 0 && found != 0) {
827 rc = -EEXIST;
828 goto done;
829 }
830 if (change->expected_generation != 0 &&
831 (found == 0 || active->objects[slots[i]].rule.generation != change->expected_generation)) {
832 rc = -ESTALE;
833 goto done;
834 }
835 if (rule != NULL) {
836 if (rule->id != change->id || rule->generation == UINT64_MAX ||
837 rule->generation <= change->expected_generation ||
838 dppd_rule_validate(rule, error, sizeof(error)) != 0) {
839 rc = -EINVAL;
840 goto done;
841 }
842 if (!supported(backend, rule)) {
843 rc = -ENOTSUP;
844 goto done;
845 }
846 }
847 if (change->expected_generation == 0)
848 ++final_count;
849 else if (rule == NULL)
850 --final_count;
851 }
852 if (final_count > backend->capacity) {
853 rc = -ENOSPC;
854 goto done;
855 }
856 dppd_install_timer_start(&timer);
857 next = snapshot_clone(active, backend->capacity);
858 if (next == NULL) {
859 rc = -ENOMEM;
860 goto done;
861 }
862 /** 先在副本清除所有被替换或删除的槽位，满表也能用本批腾出的槽位创建其他 ID */
863 for (i = 0; i < count; ++i)
864 if (slots[i] >= 0) {
865 struct dppd_software_rule_object *object = &next->objects[slots[i]];
866 metrics_put(object->metrics);
867 memset(object, 0, sizeof(*object));
868 --next->count;
869 }
870 for (i = 0; i < count; ++i) {
871 struct dppd_software_rule_metrics *metrics;
872 if (changes[i].replacement == NULL)
873 continue;
874 metrics = calloc(1, sizeof(*metrics));
875 if (metrics == NULL) {
876 rc = -ENOMEM;
877 goto done;
878 }
879 atomic_init(&metrics->references, 1);
880 atomic_init(&metrics->hits, 0);
881 atomic_init(&metrics->bytes, 0);
882 for (uint32_t slot = 0; slot < backend->capacity; ++slot)
883 if (next->objects[slot].metrics == NULL) {
884 next->objects[slot].rule = *changes[i].replacement;
885 next->objects[slot].metrics = metrics;
886 slots[i] = (int)slot;
887 ++next->count;
888 break;
889 }
890 }
891 rc = publish_locked(backend, next);
892 if (rc == 0) {
893 uint64_t duration = 0;
894 bool available = dppd_install_timer_finish(&timer, &duration);
895 dppd_rule_latency_record(&backend->latency_batch, available, duration, count);
896 for (i = 0; i < count; ++i)
897 if (changes[i].replacement != NULL) {
898 struct dppd_software_rule_metrics *metrics = next->objects[slots[i]].metrics;
899 metrics->install_duration_ns = duration;
900 metrics->timing_available = available;
901 metrics->commit_rule_count = count;
902 }
903 next = NULL;
904 }
905done:
906 snapshot_destroy(next, backend->capacity);
907 pthread_mutex_unlock(&backend->writer_lock);
908 return rc;
909}

911/** 恢复先准备完整副本，退出先准备空副本，都只编译并发布一次，避免按小批重复编译大表 */
912static int replace_snapshot(struct dppd_software_backend *backend, const struct dppd_rule *rules, uint32_t count,
913 bool require_empty)
914{
915 struct dppd_software_classifier_snapshot *next = NULL;
916 struct dppd_install_timer timer;
917 int rc = 0;
918 if (backend == NULL || !backend->lock_initialized || count > backend->capacity || (count != 0 && rules == NULL))
919 return -EINVAL;
920 pthread_mutex_lock(&backend->writer_lock);
921 struct dppd_software_classifier_snapshot *active = atomic_load_explicit(&backend->active, memory_order_acquire);
922 if (require_empty && active->count != 0) {
923 rc = -EBUSY;
924 goto done;
925 }
926 if (count == 0 && active->count == 0)
927 goto done;
928 for (uint32_t i = 0; i < count; ++i) {
929 char error[128];
930 if (rules[i].id == 0 || rules[i].id == UINT64_MAX || rules[i].generation == 0 ||
931 rules[i].generation == UINT64_MAX || (i != 0 && rules[i - 1U].id >= rules[i].id) ||
932 dppd_rule_validate(&rules[i], error, sizeof(error)) != 0) {
933 rc = -EINVAL;
934 goto done;
935 }
936 if (!supported(backend, &rules[i])) {
937 rc = -ENOTSUP;
938 goto done;
939 }
940 }
941 dppd_install_timer_start(&timer);
942 next = snapshot_clone(NULL, backend->capacity);
943 if (next == NULL) {
944 rc = -ENOMEM;
945 goto done;
946 }
947 for (uint32_t i = 0; i < count; ++i) {
948 struct dppd_software_rule_metrics *metrics = calloc(1, sizeof(*metrics));
949 if (metrics == NULL) {
950 rc = -ENOMEM;
951 goto done;
952 }
953 atomic_init(&metrics->references, 1);
954 atomic_init(&metrics->hits, 0);
955 atomic_init(&metrics->bytes, 0);
956 next->objects[i] = (struct dppd_software_rule_object){rules[i], metrics};
957 ++next->count;
958 }
959 rc = publish_locked(backend, next);
960 if (rc == 0) {
961 uint64_t duration = 0;
962 bool available = dppd_install_timer_finish(&timer, &duration);
963 if (count != 0) {
964 dppd_rule_latency_record(&backend->latency_batch, available, duration, count);
965 for (uint32_t i = 0; i < count; ++i) {
966 next->objects[i].metrics->install_duration_ns = duration;
967 next->objects[i].metrics->timing_available = available;
968 next->objects[i].metrics->commit_rule_count = count;
969 }
970 }
971 next = NULL;
972 }
973done:
974 snapshot_destroy(next, backend->capacity);
975 pthread_mutex_unlock(&backend->writer_lock);
976 return rc;
977}

1005/**
1006 * 按 rule_id 和 generation 删除一个规则版本，返回 0 表示新快照发布成功，负值表示失败
1007 * backend 提供当前快照和写锁；版本必须同时匹配，避免删除同 ID 的其他版本
1008 *
1009 * 例：当前 B 含 R1、R2，本次撤销 R2
1010 * B：[R1, R2] → 复制出 C：[R1, R2] → 清空 C 中 R2 的槽位并归还其计数器引用
1011 * 发布成功：active → C：[R1]，旧 B 交给 QSBR，已有读者仍可完成对 B 的访问
1012 * 发布失败：销毁 C，active 仍指向 B：[R1, R2]，调用方收到删除错误
1013 * 回滚通过发布新版本撤销规则，不把 active 直接切回某个历史快照
1014 */
1015int dppd_software_backend_remove_version(struct dppd_software_backend *backend, uint64_t rule_id, uint64_t generation)
1016{
1017 struct dppd_software_classifier_snapshot *active;
1018 struct dppd_software_classifier_snapshot *next;
1019 int index;
1020
1021 if (backend == NULL || rule_id == 0 || generation == 0)
1022 return -EINVAL;
1023 (void)pthread_mutex_lock(&backend->writer_lock);
1024 active = atomic_load_explicit(&backend->active, memory_order_acquire);
1025 index = snapshot_find(active, backend->capacity, rule_id, generation);
1026 if (index < 0) {
1027 (void)pthread_mutex_unlock(&backend->writer_lock);
1028 return -ENOENT;
1029 }
1030 /** 删除也经由新 snapshot 发布；旧 snapshot 仍保留被删除规则直到 QSBR 放行 */
1031 next = snapshot_clone(active, backend->capacity);
1032 if (next == NULL) {
1033 (void)pthread_mutex_unlock(&backend->writer_lock);
1034 return -ENOMEM;
1035 }
1036 /** 只放弃副本持有的计数器引用，旧快照仍持有自己的引用，不能在此强行释放 */
1037 metrics_put(next->objects[index].metrics);
1038 memset(&next->objects[index], 0, sizeof(next->objects[index]));
1039 next->count--;
1040 int publish_error = publish_locked(backend, next);
1041 if (publish_error != 0) {
1042 snapshot_destroy(next, backend->capacity);
1043 (void)pthread_mutex_unlock(&backend->writer_lock);
1044 return publish_error;
1045 }
1046 (void)pthread_mutex_unlock(&backend->writer_lock);
1047 return 0;
1048}

1199/**
1200 * 比较候选规则优先顺序，返回 true 表示 a 排在 b 前面
1201 * group → priority → id → tag：前项相等才比较下一项，数值小的优先
1202 * 精确规则与 ACL 规则使用同一排序，匹配方式不决定优先级
1203 */
1204static bool rule_before(const struct dppd_rule *a, uint32_t a_tag, const struct dppd_rule *b, uint32_t b_tag)
1205{
1206 if (a->group != b->group)
1207 return a->group < b->group;
1208 if (a->priority != b->priority)
1209 return a->priority < b->priority;
1210 if (a->id != b->id)
1211 return a->id < b->id;
1212 return a_tag < b_tag;
1213}

1215/** 将批次中的次数和字节增量原子加到各规则统计，清空有效项数供后续复用 */
1216void dppd_software_count_batch_flush(struct dppd_software_count_batch *batch)
1217{
1218 if (batch == NULL)
1219 return;
1220 for (uint16_t i = 0; i < batch->count; ++i) {
1221 atomic_fetch_add_explicit(&batch->entries[i].metrics->hits, batch->entries[i].hits, memory_order_relaxed);
1222 atomic_fetch_add_explicit(&batch->entries[i].metrics->bytes, batch->entries[i].bytes, memory_order_relaxed);
1223 }
1224 batch->count = 0;
1225}

1227/**
1228 * 记录一次规则命中和本包字节数，无返回值
1229 * batch 为空时直接原子累加，否则按 metrics 对象合并到批次，减少共享统计写入
1230 * metrics 指针区分规则版本，避免把相同 ID 的新旧版本统计混在一起
1231 * 示例：同一对象先后命中 60 和 100 字节 → 有效项 1，hits 为 2，bytes 为 160
1232 */
1233static void count_packet(struct dppd_software_rule_metrics *metrics, uint32_t bytes,
1234 struct dppd_software_count_batch *batch)
1235{
1236 if (batch == NULL) {
1237 atomic_fetch_add_explicit(&metrics->hits, 1, memory_order_relaxed);
1238 atomic_fetch_add_explicit(&metrics->bytes, bytes, memory_order_relaxed);
1239 return;
1240 }
1241 /** count 是批次内有效统计对象数，同一对象重复命中只增加该项的次数和字节数 */
1242 for (uint16_t i = 0; i < batch->count; ++i)
1243 if (batch->entries[i].metrics == metrics) {
1244 batch->entries[i].hits++;
1245 batch->entries[i].bytes += bytes;
1246 return;
1247 }
1248 /** 新增对象前若没有空位则先提交，32 是对象容量，不要求凑满 32 个包 */
1249 if (batch->count == DPPD_SOFTWARE_COUNT_BATCH_SIZE)
1250 dppd_software_count_batch_flush(batch);
1251 batch->entries[batch->count].metrics = metrics;
1252 batch->entries[batch->count].hits = 1;
1253 batch->entries[batch->count++].bytes = bytes;
1254}

1256/**
1257 * 匹配当前报文，生成软件规则结果并按需累计 COUNT 统计
1258 * 读取规则表 → ACL/精确匹配 → 选定一条规则
1259 * 未命中 → matched = false，返回
1260 * 命中 → DROP/MARK 写入 decision，COUNT 累计命中次数和字节数
1261 */
1262void dppd_software_backend_decide_counted(struct dppd_software_backend *backend, uint16_t ingress_port,
1263 const struct dppd_packet *packet, struct dppd_software_decision *decision, struct dppd_software_count_batch *batch)
1264{
1265 struct dppd_software_classifier_snapshot *active;
1266 struct dppd_software_rule_object *selected = NULL;
1267 memset(decision, 0, sizeof(*decision));
1268 if (backend == NULL || packet == NULL)
1269 return;
1270 /**
1271 * backend->active → 本包的规则快照 active → ACL/精确匹配各自返回候选，0 表示未命中
1272 *
1273 * 读到这里可继续追查：active 中的规则是谁提前准备并发布进去的
1274 * 报文处理：process_batch → routed_decide → pipeline → 本函数读取 active，匹配当前包
1275 * 规则配置：control.c/dppd_control_apply → control_apply_impl → transaction_run
1276 * → 软件 transaction_commit → publish_locked 构建匹配表并更新 backend->active
1277 * 两条流程通过同一个 backend->active 衔接；转去读配置入口是在追查规则来源，并非本函数调用它
1278 * 回到本函数后，继续用取得的同一快照查询下面的 ACL 和 exact 表
1279 *
1280 * 每包只取一次 active，两类匹配共用同一版本，不获取控制面写锁
1281 * 调用线程须已注册上线 QSBR，防止读取期间旧规则表被回收
1282 * acquire 与发布端 release 配对，保证读到新指针时规则内容也已可见
1283 */
1284 active = atomic_load_explicit(&backend->active, memory_order_acquire);
1285 /**
1286 * ACL 支持掩码、端口范围等条件；exact 对完整地址和端口组合做哈希查找
1287 * 各自返回本表优先命中的规则 tag，0 表示未命中，选定候选后再执行动作
1288 */
1289 uint32_t tag = dppd_acl_classify(snapshot_acl(active, backend->capacity), ingress_port, packet);
1290 uint32_t exact_tag = dppd_exact_classify(snapshot_exact(active, backend->capacity), ingress_port, packet);
1291 /**
1292 * ACL tag + exact_tag → 选定一个 tag
1293 * 单个命中 → 直接选用；两个都命中 → 按 rule_before 排序
1294 * tag = 0 → 无规则；tag > 0 → active->objects[tag - 1]，数组下标从 0 开始
1295 */
1296 if (exact_tag != 0 &&
1297 (tag == 0 || rule_before(&active->objects[exact_tag - 1].rule, exact_tag, &active->objects[tag - 1].rule, tag)))
1298 tag = exact_tag;
1299 if (tag != 0)
1300 selected = &active->objects[tag - 1];
1301 if (selected != NULL) {
1302 uint16_t action;
1303
1304 /**
1305 * 只执行选中规则的动作，DROP/MARK 写入软件决策，COUNT 累计命中统计
1306 * R1 示例：[DROP, COUNT] → matched = true、drop = true，并累计本包次数和字节数
1307 * DROP 不退出循环，也不释放报文；释放由 worker.c/process_batch 按最终决策执行
1308 */
1309 decision->matched = true;
1310 for (action = 0; action < selected->rule.nb_actions; ++action) {
1311 const struct dppd_action *source = &selected->rule.actions[action];
1312
1313 if (source->type == DPPD_ACTION_DROP)
1314 decision->drop = true;
1315 else if (source->type == DPPD_ACTION_MARK) {
1316 decision->has_mark = true;
1317 decision->mark_id = source->conf.mark_id;
1318 } else if (source->type == DPPD_ACTION_SHARED_COUNT && !packet->neighbor_retry) {
1319 atomic_fetch_add_explicit(&selected->metrics->shared->hits, 1, memory_order_relaxed);
1320 atomic_fetch_add_explicit(&selected->metrics->shared->bytes, packet->packet_len, memory_order_relaxed);
1321 } else if (source->type == DPPD_ACTION_COUNT && !packet->neighbor_retry)
1322 count_packet(selected->metrics, packet->packet_len, batch);
1323 }
1324 }
1325}
