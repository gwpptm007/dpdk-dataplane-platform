25int dppd_rule_repository_init(struct dppd_rule_repository *repository, uint32_t capacity)
26{
27 if (repository == NULL || capacity == 0)
28 return -EINVAL;
29 memset(repository, 0, sizeof(*repository));
30 repository->records = calloc(capacity, sizeof(*repository->records));
31 if (repository->records == NULL)
32 return -ENOMEM;
33 int rc = dppd_slot_index_init(&repository->index, capacity);
34 if (rc != 0) {
35 free(repository->records);
36 memset(repository, 0, sizeof(*repository));
37 return rc;
38 }
39 repository->capacity = capacity;
40 return 0;
41}

76int dppd_rule_repository_list(const struct dppd_rule_repository *repository, uint64_t after_rule_id,
77 uint64_t expected_repository_generation, struct dppd_rule *rules, uint32_t limit, uint32_t *count, bool *has_more,
78 uint64_t *repository_generation)
79{
80 if (repository == NULL || repository->records == NULL || rules == NULL || limit == 0 || count == NULL ||
81 has_more == NULL || repository_generation == NULL)
82 return -EINVAL;
83 if (expected_repository_generation != DPPD_RULE_GENERATION_ANY &&
84 expected_repository_generation != repository->generation)
85 return -ESTALE;
86
87 /** 对外仍按规则编号分页，内部只二分一次游标，再顺序复制本页对象 */
88 uint32_t position = dppd_slot_index_upper_bound(&repository->index, after_rule_id, UINT64_MAX);
89 uint32_t available = repository->index.count - position;
90 uint32_t output_count = limit < available ? limit : available;
91 for (uint32_t i = 0; i < output_count; ++i) {
92 uint32_t slot = dppd_slot_index_at(&repository->index, position + i);
93 rules[i] = repository->records[slot].rule;
94 }
95 *has_more = output_count < available;
96 *count = output_count;
97 *repository_generation = repository->generation;
98 return 0;
99}

101/**
102 * 将 rule 新增或更新到内存账本，检查预期版本并分配新版本，不安装后端规则或写文件
103 * expected_generation 是调用方预期的旧版本，0 用于期望新建，ANY 表示跳过版本冲突检查
104 * 成功返回 0，result 给出 CREATED、UPDATED 或 UNCHANGED 及规则版本；失败返回负错误码
105 * 调用方须串行访问账本，本函数不加锁
106 *
107 * 按 rule.id 查找记录
108 * 已存在 → 内容相同 → UNCHANGED，返回已有版本，不检查 expected_generation
109 * → 内容不同 → 检查预期旧版本 → 覆盖内容并分配新版本 → UPDATED
110 * 不存在 → 预期为 0 或 ANY → 分配槽位、写入内容及新版本 → CREATED
111 * → 预期为其他版本 → -ESTALE，调用方预期的记录已不存在
112 */
113int dppd_rule_repository_apply(struct dppd_rule_repository *repository, const struct dppd_rule *rule,
114 uint64_t expected_generation, struct dppd_rule_apply_result *result)
115{
116 struct dppd_rule_record *record;
117 char validation_error[128];
118
119 if (repository == NULL || repository->records == NULL || rule == NULL || rule->id == 0 || result == NULL)
120 return -EINVAL;
121 if (dppd_rule_validate(rule, validation_error, sizeof(validation_error)) != 0)
122 return -EINVAL;
123
124 record = find_record(repository, rule->id);
125 if (record != NULL) {
126 /** equal 比较规则内容，不比较规则自身的 generation；相同内容重复提交不推进版本 */
127 if (dppd_rule_equal(&record->rule, rule)) {
128 result->status = DPPD_RULE_UNCHANGED;
129 result->generation = record->rule.generation;
130 return 0;
131 }
132 /**
133 * 例：调用方读到 R1 版本 7，其他请求已将其改成版本 9
134 * 本次携带 expected_generation = 7 且内容不同 → -ESTALE，拒绝覆盖版本 9
135 * 比较的是 R1 当前版本，不是整个仓库的 generation；ANY 明确跳过此检查
136 */
137 if (expected_generation != DPPD_RULE_GENERATION_ANY && expected_generation != record->rule.generation)
138 return -ESTALE;
139 if (repository->generation >= UINT64_MAX - 1U)
140 return -EOVERFLOW;
141 record->rule = *rule;
142 /**
143 * 新版本取仓库递增序列，不采用输入 rule->generation，也不是旧规则版本加 1
144 * 例：仓库版本 12，R1 版本 7 → 用预期版本 7 更新成功 → 仓库和 R1 均变为 13
145 * 其他规则保留各自的版本，更新已有记录不增加 count
146 */
147 record->rule.generation = ++repository->generation;
148 result->status = DPPD_RULE_UPDATED;
149 result->generation = record->rule.generation;
150 return 0;
151 }
152
153 if (expected_generation != DPPD_RULE_GENERATION_ANY && expected_generation != 0)
154 return -ESTALE;
155 if (repository->generation >= UINT64_MAX - 1U)
156 return -EOVERFLOW;
157 uint32_t slot;
158 /** 索引记录 (rule.id, 0) → 槽位，真正的规则版本保存在 record->rule.generation */
159 int rc = dppd_slot_index_insert(&repository->index, rule->id, 0, &slot);
160 if (rc != 0)
161 return rc;
162 record = &repository->records[slot];
163 record->occupied = true;
164 record->rule = *rule;
165 /** 新建也从仓库序列分配版本，例如仓库 13 → 新建 R2 → 仓库和 R2 均为 14，R1 仍为 13 */
166 record->rule.generation = ++repository->generation;
167 repository->count++;
168 result->status = DPPD_RULE_CREATED;
169 result->generation = record->rule.generation;
170 return 0;
171}

173/**
174 * 整批创建尚不存在的规则账本记录，不执行任何硬件或软件安装操作
175 *
176 * 第一轮确认规则合法、ID 不重复且版本与预分配序列相同，再一次建立全部索引
177 * 索引全部成功后才写入记录，所以后一条输入或容量错误不会留下已发布前缀
178 * 该保证依赖调用方串行访问仓库，不等于提供了多线程并发写入保护
179 */
180int dppd_rule_repository_create_batch(struct dppd_rule_repository *repository, const struct dppd_rule *rules,
181 uint32_t count)
182{
183 uint32_t slots[64], inserted = 0;
184 char error[128];
185 int rc;
186 if (repository == NULL || repository->records == NULL || rules == NULL || count == 0 || count > 64)
187 return -EINVAL;
188 if (count > repository->capacity - repository->count)
189 return -ENOSPC;
190 if (repository->generation >= UINT64_MAX - count)
191 return -EOVERFLOW;
192 for (uint32_t i = 0; i < count; ++i) {
193 if (rules[i].id == 0 || rules[i].id == UINT64_MAX || rules[i].generation != repository->generation + i + 1U ||
194 dppd_rule_validate(&rules[i], error, sizeof(error)) != 0)
195 return -EINVAL;
196 if (find_record(repository, rules[i].id) != NULL)
197 return -EEXIST;
198 for (uint32_t j = 0; j < i; ++j)
199 if (rules[j].id == rules[i].id)
200 return -EEXIST;
201 }
202 for (; inserted < count; ++inserted) {
203 rc = dppd_slot_index_insert(&repository->index, rules[inserted].id, 0, &slots[inserted]);
204 if (rc != 0) {
205 while (inserted != 0)
206 dppd_slot_index_erase(&repository->index, rules[--inserted].id, 0);
207 return -EUCLEAN;
208 }
209 }
210 for (uint32_t i = 0; i < count; ++i) {
211 repository->records[slots[i]].rule = rules[i];
212 repository->records[slots[i]].occupied = true;
213 }
214 repository->count += count;
215 repository->generation += count;
216 return 0;
217}

219/** 精确版本、重复编号及版本空间先整批检查，正常删除阶段没有分配或外部调用 */
220static int remove_batch(struct dppd_rule_repository *repository, const struct dppd_rule *rules, uint32_t count,
221 bool advance)
222{
223 uint32_t slots[64];
224 if (repository == NULL || repository->records == NULL || rules == NULL || count == 0 || count > 64)
225 return -EINVAL;
226 if (advance && repository->generation >= UINT64_MAX - count)
227 return -EOVERFLOW;
228 for (uint32_t i = 0; i < count; ++i) {
229 slots[i] = dppd_slot_index_find(&repository->index, rules[i].id, 0);
230 if (slots[i] == DPPD_SLOT_NONE || rules[i].generation == 0 || rules[i].generation == UINT64_MAX ||
231 repository->records[slots[i]].rule.generation != rules[i].generation)
232 return -ESTALE;
233 for (uint32_t j = 0; j < i; ++j)
234 if (rules[j].id == rules[i].id)
235 return -EEXIST;
236 }
237 for (uint32_t i = 0; i < count; ++i)
238 if (dppd_slot_index_erase(&repository->index, rules[i].id, 0) != 0)
239 return -EUCLEAN;
240 for (uint32_t i = 0; i < count; ++i)
241 memset(&repository->records[slots[i]], 0, sizeof(repository->records[slots[i]]));
242 repository->count -= count;
243 if (advance)
244 repository->generation += count;
245 return 0;
246}

260/** 替换前完整校验，版本由同一控制线程连续分配，预检完成后不再分配内存 */
261int dppd_rule_repository_replay_batch(struct dppd_rule_repository *repository, const struct dppd_rule *rules,
262 uint32_t count)
263{
264 char error[128];
265 if (repository == NULL || repository->records == NULL || rules == NULL || count == 0 || count > 64 ||
266 count > repository->capacity - repository->count || repository->generation == UINT64_MAX)
267 return -EINVAL;
268 for (uint32_t i = 0; i < count; ++i) {
269 if (rules[i].id == 0 || rules[i].id == UINT64_MAX || rules[i].generation == 0 ||
270 rules[i].generation > repository->generation || dppd_rule_validate(&rules[i], error, sizeof(error)) != 0)
271 return -EINVAL;
272 if (dppd_slot_index_find(&repository->index, rules[i].id, 0) != DPPD_SLOT_NONE)
273 return -EEXIST;
274 for (uint32_t j = 0; j < i; ++j)
275 if (rules[i].id == rules[j].id)
276 return -EEXIST;
277 }
278 for (uint32_t i = 0; i < count; ++i) {
279 uint32_t slot;
280 int rc = dppd_slot_index_insert(&repository->index, rules[i].id, 0, &slot);
281 if (rc != 0) {
282 for (uint32_t j = 0; j < i; ++j)
283 dppd_slot_index_erase(&repository->index, rules[j].id, 0);
284 return -EUCLEAN;
285 }
286 }
287 for (uint32_t i = 0; i < count; ++i) {
288 uint32_t slot = dppd_slot_index_find(&repository->index, rules[i].id, 0);
289 repository->records[slot].occupied = true;
290 repository->records[slot].rule = rules[i];
291 }
292 repository->count += count;
293 return 0;
294}

296int dppd_rule_repository_update_batch(struct dppd_rule_repository *repository, const struct dppd_rule *rules,
297 const uint64_t *expected_generations, uint32_t count)
298{
299 uint32_t i;
300
301 if (repository == NULL || repository->records == NULL || rules == NULL || expected_generations == NULL ||
302 count == 0)
303 return -EINVAL;
304 if (count > repository->count)
305 return -ENOENT;
306 /**
307 * UINT64_MAX 被用来表示“不检查版本”的 ANY 条件，不能分配给真实规则
308 * 这里一次检查整批需要的版本空间，避免写到中途才发现版本号用尽
309 */
310 if (repository->generation >= UINT64_MAX - count)
311 return -EOVERFLOW;
312 for (i = 0; i < count; ++i) {
313 struct dppd_rule_record *record;
314 char error[128];
315 uint32_t j;
316
317 if (rules[i].id == 0 || dppd_rule_validate(&rules[i], error, sizeof(error)) != 0)
318 return -EINVAL;
319 for (j = 0; j < i; ++j) {
320 if (rules[j].id == rules[i].id)
321 return -EEXIST;
322 }
323 record = find_record(repository, rules[i].id);
324 if (record == NULL)
325 return -ENOENT;
326 if (expected_generations[i] == 0 || expected_generations[i] == DPPD_RULE_GENERATION_ANY ||
327 expected_generations[i] != record->rule.generation)
328 return -ESTALE;
329 }
330 /**
331 * 全部检查通过后才开始写入，这一阶段不申请内存、不调用后端，也不删除记录
332 * 单条 apply 遇到相同内容会返回 UNCHANGED，无法满足批量更新连续分配新版本的约定
333 * 因此这里直接替换完整规则内容，并按请求顺序推进仓库的全局版本号
334 */
335 for (i = 0; i < count; ++i) {
336 struct dppd_rule_record *record = find_record(repository, rules[i].id);
337
338 /** 同一控制线程中没有增删记录，上面已确认存在的 ID 在这里一定能找到 */
339 record->rule = rules[i];
340 record->rule.generation = ++repository->generation;
341 }
342 return 0;
343}

345/**
346 * 删除内存账本中的规则，校验预期版本后撤销索引、清空记录并推进仓库修订号
347 * 成功返回 0，removed 表示是否实际删除，generation 返回操作后的仓库版本；失败返回负错误码
348 * 后端规则须由控制层先删除，本函数不操作 worker 快照或硬件对象
349 * rule_id → 找到 record → slot_index_erase 归还索引槽位 → 清空 record → count 减 1
350 */
351int dppd_rule_repository_remove(struct dppd_rule_repository *repository, uint64_t rule_id, uint64_t expected_generation,
352 bool *removed, uint64_t *generation)
353{
354 struct dppd_rule_record *record;
355
356 if (repository == NULL || repository->records == NULL || rule_id == 0 || removed == NULL || generation == NULL)
357 return -EINVAL;
358 record = find_record(repository, rule_id);
359 if (record == NULL) {
360 if (expected_generation != DPPD_RULE_GENERATION_ANY && expected_generation != 0)
361 return -ESTALE;
362 *removed = false;
363 *generation = repository->generation;
364 return 0;
365 }
366 if (expected_generation != DPPD_RULE_GENERATION_ANY && expected_generation != record->rule.generation)
367 return -ESTALE;
368
369 /** 删除同样需要一个新的有效版本，不能先清空记录再发现版本号已耗尽 */
370 if (repository->generation >= UINT64_MAX - 1U)
371 return -EOVERFLOW;
372 if (dppd_slot_index_erase(&repository->index, rule_id, 0) != 0)
373 return -EUCLEAN;
374 memset(record, 0, sizeof(*record));
375 repository->count--;
376 *removed = true;
377 /* 删除同样推进 generation，避免重放旧版本时出现 ABA 问题 */
378 *generation = ++repository->generation;
379 return 0;
380}

382int dppd_rule_repository_restore(struct dppd_rule_repository *repository, const struct dppd_rule *rules, uint32_t count,
383 uint64_t repository_generation)
384{
385 uint32_t i;
386 char validation_error[128];
387
388 if (repository == NULL || repository->records == NULL || (count != 0 && rules == NULL) ||
389 count > repository->capacity)
390 return -EINVAL;
391 /* 启动恢复只允许发布到全新空仓库，避免覆盖仍有硬件对象对应的在线状态 */
392 if (repository->count != 0 || repository->generation != 0 || repository->index.count != 0)
393 return -EBUSY;
394
395 for (i = 0; i < count; ++i) {
396 if (rules[i].id == 0 || rules[i].generation == 0 || rules[i].generation > repository_generation ||
397 (i != 0 && rules[i - 1U].id >= rules[i].id) ||
398 dppd_rule_validate(&rules[i], validation_error, sizeof(validation_error)) != 0)
399 return -EINVAL;
400 }
401
402 /*
403 * records 已在 init 时一次性分配；完成上面的全量预检后，下面的发布路径不再
404 * 分配内存也不调用外部组件，因此不会出现只写入部分规则的失败点
405 */
406 for (i = 0; i < count; ++i) {
407 uint32_t slot;
408 if (dppd_slot_index_insert(&repository->index, rules[i].id, 0, &slot) != 0) {
409 dppd_slot_index_clear(&repository->index);
410 return -EUCLEAN;
411 }
412 }
413 for (i = 0; i < count; ++i) {
414 uint32_t slot = dppd_slot_index_find(&repository->index, rules[i].id, 0);
415 repository->records[slot].occupied = true;
416 repository->records[slot].rule = rules[i];
417 }
418 repository->count = count;
419 repository->generation = repository_generation;
420 return 0;
