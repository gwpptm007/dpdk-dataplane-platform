6/** 键逐字段写入固定字节数组，避免结构体填充或主机字节序影响哈希结果 */
7#define EXACT_KEY_SIZE 40U
8
9struct exact_row {
10 uint8_t key[EXACT_KEY_SIZE];
11 uint64_t id;
12 uint32_t group, priority, tag, next;
13};
14
15struct dppd_exact {
16 uint32_t capacity, count;
17 struct exact_row rows[];
18};

20/** 桶数组放在行数组之后，每个桶保存行号加一，零表示没有候选 */
21static uint32_t *buckets(const struct dppd_exact *table)
22{
23 return (uint32_t *)(table->rows + table->capacity);
24}
25
26size_t dppd_exact_size(uint32_t capacity)
27{
28 size_t bytes = sizeof(struct dppd_exact) + (size_t)capacity * (sizeof(struct exact_row) + sizeof(uint32_t));
29 return (bytes + 7U) & ~(size_t)7U;
30}

32struct dppd_exact *dppd_exact_init(void *memory, uint32_t capacity)
33{
34 struct dppd_exact *table = memory;
35 memset(memory, 0, dppd_exact_size(capacity));
36 table->capacity = capacity;
37 return table;
38}

40/**
41 * 用 FNV-1a 算法将 40 字节 key 汇总为 32 位哈希值，调用方再取模选择桶
42 * 2166136261 是初始值，16777619 是逐字节异或后使用的乘数，均为算法常量
43 * 不同 key 可能得到相同桶下标，查询时仍须比较完整 key
44 */
45static uint32_t hash_key(const uint8_t *key)
46{
47 uint32_t hash = 2166136261U;
48 for (unsigned int i = 0; i < EXACT_KEY_SIZE; ++i)
49 hash = (hash ^ key[i]) * 16777619U;
50 return hash;
51}

53static bool before(const struct exact_row *a, const struct exact_row *b)
54{
55 if (a->group != b->group)
56 return a->group < b->group;
57 if (a->priority != b->priority)
58 return a->priority < b->priority;
59 if (a->id != b->id)
60 return a->id < b->id;
61 return a->tag < b->tag;
62}

64/** 入口保留两个字节，地址族和传输层各占一个字节，地址区统一预留 IPv6 宽度 */
65static void key_prefix(uint8_t *key, uint16_t ingress, uint8_t family, uint8_t transport)
66{
67 key[0] = (uint8_t)(ingress >> 8);
68 key[1] = (uint8_t)ingress;
69 key[2] = family;
70 key[3] = transport;
71}

73/**
74 * 将完整源/目的 IP 和 TCP/UDP 端口组成精确匹配键，把规则索引加入 table 的哈希桶
75 * rule 提供匹配条件和优先顺序，tag 是调用方规则槽位下标加 1，供命中后查回动作和计数器
76 * 返回 0 表示入表成功；-EINVAL 参数或规则无效，-ENOTSUP 匹配条件不支持，-ENOSPC 表已满
77 *
78 * 完整地址和端口、无额外不支持的条件 → exact 建键入表
79 * 地址掩码、端口通配或范围等条件 → -ENOTSUP → publish_locked 尝试加入 ACL
80 * 本函数在规则安装时调用，只建立匹配索引，不执行 DROP、MARK 或 COUNT 动作
81 */
82int dppd_exact_add(struct dppd_exact *table, const struct dppd_rule *rule, uint32_t tag)
83{
84 char error[128];
85 if (table == NULL || rule == NULL || tag == 0 || rule->domain != DPPD_RULE_DOMAIN_INGRESS ||
86 dppd_rule_validate(rule, error, sizeof(error)) != 0)
87 return -EINVAL;
88 /**
89 * 将规则的匹配字段拼成 40 字节 key，下面的偏移相对 key 起点，不是报文偏移
90 * 偏移 0～1 2 3 4～19 20～35 36～37 38～39
91 * 内容 入口 地址族 传输层 源地址 目的地址 源端口 目的端口
92 * 地址区各留 16 字节以兼容 IPv6；IPv4 只写前 4 字节，后 12 字节保持为 0
93 * 本行初始化未列出的成员也会清零，因此 key 的未使用字节和 next 初始均为 0
94 *
95 * 例：入口 2，IPv4 / UDP，192.0.2.1:1234 → 198.51.100.2:9000
96 * key[0～3] = 00 02、DPPD_L3_IPV4、DPPD_L4_UDP
97 * key[4～19] = c0 00 02 01 + 12 个零字节
98 * key[20～35] = c6 33 64 02 + 12 个零字节
99 * key[36～39] = 04 d2 23 28，即网络字节序的 1234 和 9000
100 * 地址族、传输层使用项目枚举；地址和端口从已有网络字节序字段直接复制
101 *
102 * 安装规则 → 从 rule 拼 key ─┐
103 * 收到报文 → 从 packet 拼 key ┴→ 相同字段得到相同字节，供哈希定位和完整比较
104 * id、group、priority、tag 不放入 key，命中后再用于选规则及查回动作
105 */
106 struct exact_row row = {.id = rule->id, .group = rule->group, .priority = rule->priority, .tag = tag};
107 uint8_t family = 0, transport = 0;
108 for (uint16_t i = 0; i < rule->nb_matches; ++i) {
109 const struct dppd_match *match = &rule->matches[i];
110 switch (match->type) {
111 case DPPD_MATCH_ETH:
112 break;
113 case DPPD_MATCH_IPV4:
114 /**
115 * 掩码中的 1 表示该位必须匹配，exact 要求源、目的地址的全部 32 位都参与比较
116 * 255.255.255.255 = 0xffffffff → 精确到一个 IPv4 地址，允许继续建键
117 * 255.255.255.0 = 0xffffff00 → 最后 8 位不限，表示一个网段，交给 ACL
118 */
119 if (match->spec.ipv4.src_mask_be != UINT32_MAX || match->spec.ipv4.dst_mask_be != UINT32_MAX)
120 return -ENOTSUP;
121 family = DPPD_L3_IPV4;
122 memcpy(row.key + 4, &match->spec.ipv4.src_be, 4);
123 memcpy(row.key + 20, &match->spec.ipv4.dst_be, 4);
124 break;
125 case DPPD_MATCH_IPV6:
126 /** IPv6 同样要求完整 128 位匹配，因此源、目的掩码各 16 个字节都必须为 0xff */
127 for (unsigned int byte = 0; byte < 16; ++byte)
128 if (match->spec.ipv6.src_mask[byte] != 255 || match->spec.ipv6.dst_mask[byte] != 255)
129 return -ENOTSUP;
130 family = DPPD_L3_IPV6;
131 memcpy(row.key + 4, match->spec.ipv6.src, 16);
132 memcpy(row.key + 20, match->spec.ipv6.dst, 16);
133 break;
134 case DPPD_MATCH_TCP:
135 case DPPD_MATCH_UDP:
136 /**
137 * 源、目的端口掩码都须为 0xffff，表示完整比较各自的 16 位端口值
138 * 例：UDP 源端口任意、目的端口 9000 → 源掩码为 0 → exact 不支持，改用 ACL
139 */
140 if (match->spec.l4.src_mask_be != UINT16_MAX || match->spec.l4.dst_mask_be != UINT16_MAX)
141 return -ENOTSUP;
142 transport = match->type == DPPD_MATCH_TCP ? DPPD_L4_TCP : DPPD_L4_UDP;
143 memcpy(row.key + 36, &match->spec.l4.src_be, 2);
144 memcpy(row.key + 38, &match->spec.l4.dst_be, 2);
145 break;
146 default:
147 return -ENOTSUP;
148 }
149 }
150 /** 缺少 IP 地址族或 TCP/UDP 条件就不能组成完整键，例如仅指定 IPv4 地址的规则交给 ACL */
151 if (family == 0 || transport == 0)
152 return -ENOTSUP;
153 if (table->count == table->capacity)
154 return -ENOSPC;
155 /** 前面的循环已填入地址和端口，这里补齐 key[0～3] 的入口、地址族和传输层类型 */
156 key_prefix(row.key, rule->install_port_id, family, transport);
157 /**
158 * 哈希值 % capacity 选出桶下标，新规则存入 rows[count]，并接在该桶链表头部
159 * 桶值和 next 保存行下标加 1，保留 0 表示空链；它们与查回规则动作的 tag 无关
160 *
161 * 例：capacity = 8，count = 3，本次取模得到桶 2，原桶值为 1
162 * 插入前：桶 2 → 编号 1 → rows[0] → 0，结束
163 * row.next = 1 → 新行接住原链头
164 * rows[3] = row → 将局部 row 的内容复制到表内空位
165 * 桶值 = ++count = 4 → 总行数变为 4，桶头改指向编号 4，即 rows[3]
166 * 插入后：桶 2 → 编号 4 → rows[3] → 编号 1 → rows[0] → 0，结束
167 * 若原桶为空，row.next 就是 0；头插只决定遍历顺序，规则优先级由查询时比较
168 */
169 uint32_t bucket = hash_key(row.key) % table->capacity;
170 row.next = buckets(table)[bucket];
171 table->rows[table->count] = row;
172 buckets(table)[bucket] = ++table->count;
173 return 0;
174}

176/**
177 * 用入口、地址族、TCP/UDP 类型及完整源/目的地址和端口组成精确匹配键
178 * 哈希定位桶后比较完整键，同键多条规则按优先顺序选出一条，返回其 tag
179 * 未命中、报文类型不适用或参数无效返回 0，仅查找候选，不执行规则动作
180 */
181uint32_t dppd_exact_classify(const struct dppd_exact *table, uint16_t ingress, const struct dppd_packet *packet)
182{
183 if (table == NULL || table->capacity == 0 || packet == NULL ||
184 (packet->l3_type != DPPD_L3_IPV4 && packet->l3_type != DPPD_L3_IPV6) ||
185 (packet->l4_type != DPPD_L4_TCP && packet->l4_type != DPPD_L4_UDP))
186 return 0;
187 /**
188 * 将报文字段组合成 40 字节查找键，下列偏移均相对 key 起点
189 * 偏移 0～1 2 3 4～19 20～35 36～37 38～39
190 * 内容 入口 地址族 传输层 源地址 目的地址 源端口 目的端口
191 * 地址族和传输层使用项目枚举，地址和端口保留网络字节序
192 * IPv4 地址各写 4 字节，其余地址空间保留为 0；IPv6 地址各写满 16 字节
193 * 规则入表与报文查询使用同一布局，清零保证未使用字节也能一致比较
194 */
195 uint8_t key[EXACT_KEY_SIZE] = {0};
196 key_prefix(key, ingress, (uint8_t)packet->l3_type, (uint8_t)packet->l4_type);
197 if (packet->l3_type == DPPD_L3_IPV4) {
198 memcpy(key + 4, &packet->ipv4_src_be, 4);
199 memcpy(key + 20, &packet->ipv4_dst_be, 4);
200 } else {
201 memcpy(key + 4, packet->ipv6_src, 16);
202 memcpy(key + 20, packet->ipv6_dst, 16);
203 }
204 memcpy(key + 36, &packet->l4_src_port_be, 2);
205 memcpy(key + 38, &packet->l4_dst_port_be, 2);
206 /**
207 * 桶值和 next 均为行下标加一，0 表示链结束，与规则 tag 含义不同
208 * 不同键可能落入同一桶，须比较完整 40 字节，不能仅凭哈希认定命中
209 *
210 * hash_key(key) → 算出哈希值
211 * 哈希值 % table->capacity → 得到桶下标
212 * buckets(table)[桶下标] → 取得桶内第一条规则的行编号
213 * 沿桶值 4 → rows[3] → next 为 1 → rows[0] → next 为 0，结束
214 */
215 const struct exact_row *selected = NULL;
216 for (uint32_t index = buckets(table)[hash_key(key) % table->capacity]; index != 0;) {
217 const struct exact_row *row = &table->rows[index - 1];
218 /** 同键规则可能有多条，保留优先的一条，不能遇到首次命中就返回 */
219 if (memcmp(key, row->key, sizeof(key)) == 0 && (selected == NULL || before(row, selected)))
220 selected = row;
221 index = row->next;
222 }
223 /** 返回本精确表选出的候选 tag，无命中返回 0，交由软件后端与 ACL 候选比较 */
224 return selected == NULL ? 0 : selected->tag;
